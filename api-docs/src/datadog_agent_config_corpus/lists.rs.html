1//! Allowed names in saved Agent configuration records, represented as Rust enums.
2//!
3//! The corpus reader uses these lists to reject unknown names instead of silently accepting a
4//! recording it cannot interpret. [`Source`] identifies where a setting came from, [`Getter`]
5//! names the Agent method called, [`Group`] describes a test case's coverage, and [`Level`]
6//! identifies a recorded warning's severity.
7//!
8//! Agent names were manually reviewed at [`REVIEWED_AT_AGENT_COMMIT`]; the recording format
9//! defines the case groups and permitted reads. When the schema pin changes, review the lists
10//! against their cited definitions before updating that constant.
11

17/// Declares a string enum with its full list and its string forms.
18macro_rules! string_enum {
19    ($(#[$meta:meta])* $name:ident, $all:ident { $($(#[$vmeta:meta])* $variant:ident => $text:literal,)* }) => {
20        $(#[$meta])*
21        #[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
22        pub enum $name {
23            $($(#[$vmeta])* $variant,)*
24        }
25
26        impl $name {
27            /// Every variant, in declaration order.
28            pub const $all: &'static [$name] = &[$($name::$variant,)*];
29
30            /// The name as the corpus writes it.
31            pub fn as_str(self) -> &'static str {
32                match self {
33                    $($name::$variant => $text,)*
34                }
35            }
36
37            /// Parses the name as the corpus writes it.
38            pub fn parse(s: &str) -> Option<Self> {
39                match s {
40                    $($text => Some($name::$variant),)*
41                    _ => None,
42                }
43            }
44        }
45
46        impl fmt::Display for $name {
47            fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
48                f.write_str(self.as_str())
49            }
50        }
51    };
52}

150    Group, ALL {
