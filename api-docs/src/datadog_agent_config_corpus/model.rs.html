1//! Rust types for saved Agent configuration inputs and observations, returned by [`crate::read`].
2//!
3//! A [`Corpus`] holds recording metadata and test [`Case`]s. Each case pairs its inputs with either
4//! a startup error or the Agent's stream settings and configuration method results. Stream settings
5//! describe what ADP receives; method results describe what Agent components read.
6//!
7//! The reader reconstructs omitted fields according to `config-recorder/docs/record.md`, so callers
8//! do not need to understand the file's space-saving rules. [`Corpus::first_snapshot`] combines the
9//! recorded defaults with a case's changes to recover its starting settings. [`Violation`] and
10//! [`Rule`] describe format errors rather than differences between Agent and ADP behavior.
11

56/// One case: its case line and its key lines (record.md §3, §5).

267/// Which family of format rules a violation breaks.
268#[derive(Clone, Copy, Debug, PartialEq, Eq)]
269pub enum Rule {
270    /// Encoding and line structure (record.md §1).
271    File,
272    /// Canonical JSON: whitespace, member order, escaping (record.md §1).
273    Canonical,
274    /// Line order (record.md §1).
275    Order,
276    /// A member's presence, type, `null`-ness or allowed values.
277    Shape,
278    /// The consistency rules between members and lines (record.md §3.3).
279    Consistency,
280    /// The rules for members a writer omits (record.md §3.1, §5.2, §5.3).
281    Reconstruction,
282    /// A getter result that does not match its getter's encoding (getter-map.md §3).
283    GetterResult,
284}
285
286impl fmt::Display for Rule {
287    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
288        f.write_str(match self {
289            Rule::File => "file",
290            Rule::Canonical => "canonical form",
291            Rule::Order => "order",
292            Rule::Shape => "shape",
293            Rule::Consistency => "consistency",
294            Rule::Reconstruction => "reconstruction",
295            Rule::GetterResult => "getter result",
296        })
297    }
298}

300/// A broken format rule.
301#[derive(Clone, Debug, PartialEq, Eq)]
302#[non_exhaustive]
303pub struct Violation {
304    /// The 1-based line, or 0 for the file as a whole.
305    pub line: usize,
306    /// The rule family.
307    pub rule: Rule,
308    /// What is wrong.
309    pub message: String,
310}
311
312impl fmt::Display for Violation {
313    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
314        if self.line == 0 {
315            write!(f, "file: {}: {}", self.rule, self.message)
316        } else {
317            write!(f, "line {}: {}: {}", self.line, self.rule, self.message)
318        }
319    }
320}

322impl Case {
323    /// The stable origin of the case's checks: what produced them, stripped of the naming a
324    /// schema bump reshuffles (case.md §3.2, §3.2.1).
325    ///
326    /// The `baseline`, `behavior` and `unknown` groups keep the case name. A `depth` case drops
327    /// the `--<key>` suffix of a bisection part, leaving `depth-<variant>`. The batched groups
328    /// (`breadth`, `unsupported`, `excluded`) drop the section batch and any `--<key>` suffix,
329    /// leaving `<group>-<source>`, since a recorded read's own `key` member already names the
330    /// exact setting. This is an assertion identity for replay expectations, not a runtime case
331    /// filter or a category grouping.
332    pub fn check_origin(&self) -> &str {
333        match self.group {
334            Group::Baseline | Group::Behavior | Group::Unknown => &self.name,
335            Group::Depth => match self.name.split_once("--") {
336                Some((origin, _)) => origin,
337                None => &self.name,
338            },
339            Group::Breadth | Group::Unsupported | Group::Excluded => {
340                let mut segments = self.name.splitn(3, '-');
341                let len = match (segments.next(), segments.next()) {
342                    (Some(group), Some(source)) => group.len() + 1 + source.len(),
343                    _ => return &self.name,
344                };
345                self.name.get(..len).unwrap_or(&self.name)
346            }
347        }
348    }
349}

351impl Corpus {
352    /// The case with this name.
353    pub fn case(&self, name: &str) -> Option<&Case> {
354        self.cases
355            .binary_search_by(|c| c.name.as_bytes().cmp(name.as_bytes()))
356            .ok()
357            .map(|i| &self.cases[i])
358    }
359
360    /// The keys of the [`BASELINE_CASE`] key lines: every key ADP modeled when the corpus was recorded.
361    ///
362    /// The set is fixed at recording time. It changes only when the corpus is re-recorded, never
363    /// when ADP's support for a key changes afterwards. Keys are lowercase, as the reader requires.
364    ///
365    /// Returns `None` when the baseline case did not start, so a missing baseline cannot pass as an
366    /// empty set. [`crate::read`] rejects a corpus without a baseline case, but not one whose
367    /// baseline hit a startup error.
368    pub fn surface(&self) -> Option<BTreeSet<&str>> {
369        let Outcome::Started(baseline) = &self.case(BASELINE_CASE)?.outcome else {
370            return None;
371        };
372        Some(baseline.keys.iter().map(|k| k.key.as_str()).collect())
373    }
374
375    /// Rebuilds a case's first snapshot, key to setting, by the three layers of record.md §3.2:
376    /// the `baseline-default` case's key-line snapshots, then the case's `side_effects` (an absent
377    /// element removes the key), then the case's own key-line snapshots (`null` removes the key).
378    ///
379    /// Keys the corpus neither models nor saw change are not in the result. Returns `None` when the
380    /// case does not exist or did not start. Without a started baseline case, layer 1 is empty.
381    pub fn first_snapshot(&self, case: &str) -> Option<BTreeMap<&str, &Setting>> {
382        let Outcome::Started(started) = &self.case(case)?.outcome else {
383            return None;
384        };
385        let mut snapshot = BTreeMap::new();
386        if let Some(Case {
387            outcome: Outcome::Started(baseline),
388            ..
389        }) = self.case(BASELINE_CASE)
390        {
391            for k in &baseline.keys {
392                if let Some(s) = &k.snapshot {
393                    snapshot.insert(k.key.as_str(), s);
394                }
395            }
396        }
397        for e in &started.side_effects {
398            match &e.setting {
399                Some(s) => snapshot.insert(e.key.as_str(), s),
400                None => snapshot.remove(e.key.as_str()),
401            };
402        }
403        for k in &started.keys {
404            match &k.snapshot {
405                Some(s) => snapshot.insert(k.key.as_str(), s),
406                None => snapshot.remove(k.key.as_str()),
407            };
408        }
409        Some(snapshot)
410    }
