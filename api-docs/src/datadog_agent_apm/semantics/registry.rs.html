1//! A `Registry` is the parsed, validated in-memory form of one complete semantic-mappings document for traces.
2//!
3//! The document comes from either:
4//!
5//! - the embedded `mappings.json`; or
6//! - one `APM_SEMANTIC_CORE_DD` Remote Configuration file.
7//!
8//! A `Registry` contains:
9//!
10//! - `version` and `metadata.content_hash` labels;
11//! - a local fingerprint of the original JSON bytes; and
12//! - a map from each supported `Concept` to its ordered fallback attributes.
13

27/// Provenance of an attribute convention.
28#[derive(Debug, Clone, Copy, Deserialize, PartialEq, Eq)]
29#[serde(rename_all = "lowercase")]
30pub enum Provider {
31    /// Datadog attribute conventions.
32    Datadog,
33    /// OpenTelemetry attribute conventions.
34    Otel,
35}
36
37/// The expected type of an attribute's value at lookup time.
38///
39/// The lookup is type-strict: an attribute registered as `Int64` is only read
40/// via the accessor's `get_int64` method. String-typed registrations allow the
41/// lookup functions to parse the string into the requested numeric type.
42#[derive(Debug, Clone, Copy, Deserialize, PartialEq, Eq)]
43#[serde(rename_all = "lowercase")]
44pub enum ValueType {
45    /// String-valued attributes.
46    String,
47    /// Signed integer attributes.
48    Int64,
49    /// Floating-point attributes.
50    Float64,
51}
52
53/// A predicate that must match before a fallback tag can be used.
54///
55/// A condition reads an attribute by exact key—no fallback chaining—and
56/// holds up to two predicate fields, all of which must hold (logical AND). A
57/// condition with no predicates set always matches.
58#[derive(Debug, Clone, Default, Deserialize)]
59pub struct Condition {
60    /// Exact attribute name to test, without looking up other names for the same concept.
61    ///
62    /// An absent or `null` JSON field becomes an empty name, matching upstream decoding.
63    #[serde(default, deserialize_with = "null_as_default")]
64    pub attribute: String,
65    /// When set, requires the attribute's presence (or absence) to match.
66    #[serde(default)]
67    pub present: Option<bool>,
68    /// When set, requires the attribute's string value to equal this.
69    #[serde(default)]
70    pub eq: Option<String>,
71}
72
73/// One entry in a concept's fallback precedence list.
74#[derive(Debug, Clone, Deserialize)]
75pub struct TagInfo {
76    /// Attribute name to read. An absent or `null` JSON field becomes an empty name.
77    #[serde(default, deserialize_with = "null_as_default")]
78    pub name: String,
79    /// The source of this attribute convention.
80    pub provider: Provider,
81    /// The convention version, or an empty string when unspecified.
82    #[serde(default, deserialize_with = "null_as_default")]
83    pub version: String,
84    /// The stored attribute type.
85    #[serde(rename = "type")]
86    pub value_type: ValueType,
87    /// Conditions that must all match before this attribute can supply the concept's value.
88    ///
89    /// An absent or `null` list is empty; a `null` element has no predicates and always matches.
90    #[serde(default, deserialize_with = "null_elements_as_default")]
91    pub when: Vec<Condition>,
92}

94/// Treats JSON `null` as the Rust type's default, matching upstream decoding of zero-valued fields.
95fn null_as_default<'de, D, T>(deserializer: D) -> Result<T, D::Error>
96where
97    D: Deserializer<'de>,
98    T: Deserialize<'de> + Default,
99{
100    Option::<T>::deserialize(deserializer).map(Option::unwrap_or_default)
101}
102
103/// Treats a JSON `null` list as empty and `null` elements as defaults, matching upstream decoding.
104fn null_elements_as_default<'de, D, T>(deserializer: D) -> Result<Vec<T>, D::Error>
105where
106    D: Deserializer<'de>,
107    T: Deserialize<'de> + Default,
108{
109    let elements = Option::<Vec<Option<T>>>::deserialize(deserializer)?;
110    Ok(elements.into_iter().flatten().map(Option::unwrap_or_default).collect())
111}

113/// Maps semantic concepts to ordered attribute names, types, and conditions for lookups.
114///
115/// For example, the HTTP status-code concept can read either `http.response.status_code` or `http.status_code`.
116/// A registry holds one complete set of mappings; remote registries replace rather than extend the embedded set.
117#[derive(Clone)]
118pub struct Registry {
119    version: String,
120    content_hash: String,
121    fingerprint: u64,
122    mappings: FastHashMap<Concept, Vec<TagInfo>>,
123}
124
125impl Registry {
126    /// Parses a registry JSON string, rejecting malformed or unsupported entries.
127    ///
128    /// # Errors
129    ///
130    /// Uses the same validation as [`Self::from_slice`]. Errors may quote payload keys and must not be sent to the
131    /// remote endpoint.
132    pub fn from_json(json: &str) -> Result<Self, GenericError> {
133        Self::from_slice(json.as_bytes())
134    }
135
136    /// Parses registry JSON, rejecting unsupported entries.
137    ///
138    /// Use this for embedded mappings, where unsupported entries indicate a build defect. Remote Configuration
139    /// decoding skips unsupported entries instead.
140    ///
141    /// # Errors
142    ///
143    /// Returns an error for invalid JSON, a missing or empty `concepts` object, or a missing or empty
144    /// `metadata.content_hash` string. Also rejects unknown concepts, malformed mappings, and fallbacks with invalid
145    /// fields, including unknown providers or value types. Errors may quote payload keys; do not send them to the
146    /// remote endpoint.
147    pub fn from_slice(json: &[u8]) -> Result<Self, GenericError> {
148        let (registry, skipped) = Self::from_slice_permissive(json)?;
149        if !skipped.is_empty() {
150            return Err(generic_error!(
151                "Registry JSON has parts the registry cannot represent: {}",
152                skipped
153            ));
154        }
155        Ok(registry)
156    }
157
158    /// Parses remote registry JSON while retaining the entries this binary understands.
159    ///
160    /// Remote mappings may contain concepts or conventions added after this binary was built. Unknown concepts,
161    /// malformed mappings, and fallbacks with invalid fields (including unknown providers or value types) are skipped
162    /// and recorded in [`SkipReport`]. The result is accepted even if every entry was skipped; missing mappings are
163    /// not filled from the embedded registry. The report is for local logging only.
164    ///
165    /// A `null` mapping or absent or `null` `fallbacks` gives an empty fallback list. See [`TagInfo`] and [`Condition`]
166    /// for field defaults, and [`Registry::version`] for version handling.
167    ///
168    /// The [fingerprint](Registry::fingerprint) includes skipped entries.
169    ///
170    /// # Errors
171    ///
172    /// Returns an error for invalid JSON, a missing or empty `concepts` object, or a missing or empty
173    /// `metadata.content_hash` string. These requirements apply before skipping entries. Errors never quote the
174    /// payload, so they can be reported to the remote endpoint.
175    pub(crate) fn from_slice_permissive(json: &[u8]) -> Result<(Self, SkipReport), GenericError> {
176        let document: Value = serde_json::from_slice(json).map_err(malformed_error)?;
177        let concepts = document.get("concepts").and_then(Value::as_object);
178        let content_hash = document
179            .pointer("/metadata/content_hash")
180            .and_then(Value::as_str)
181            .unwrap_or_default();
182        check_required(concepts.is_some_and(|c| !c.is_empty()), content_hash)?;
183
184        let mut skipped = SkipReport::default();
185        let mut mappings = FastHashMap::default();
186        for (key, mapping) in concepts.into_iter().flatten() {
187            let Some(concept) = Concept::from_name(key) else {
188                skipped.unknown_concepts.push(key.clone());
189                continue;
190            };
191            let entries: &[Value] = match mapping {
192                // Go decodes a `null` mapping to an empty fallback list.
193                Value::Null => &[],
194                Value::Object(map) => match map.get("fallbacks") {
195                    None | Some(Value::Null) => &[],
196                    Some(Value::Array(entries)) => entries.as_slice(),
197                    _ => {
198                        skipped.malformed_concepts.push(key.clone());
199                        continue;
200                    }
201                },
202                _ => {
203                    skipped.malformed_concepts.push(key.clone());
204                    continue;
205                }
206            };
207            let mut fallbacks = Vec::with_capacity(entries.len());
208            for (position, entry) in entries.iter().enumerate() {
209                match TagInfo::deserialize(entry) {
210                    Ok(tag) => fallbacks.push(tag),
211                    Err(_) => skipped.fallbacks.push((key.clone(), position)),
212                }
213            }
214            mappings.insert(concept, fallbacks);
215        }
216
217        let registry = Self {
218            version: document
219                .get("version")
220                .and_then(Value::as_str)
221                .unwrap_or_default()
222                .to_owned(),
223            content_hash: content_hash.to_owned(),
224            fingerprint: fingerprint_for(json),
225            mappings,
226        };
227        Ok((registry, skipped))
228    }
229
230    /// Returns the registry's `metadata.content_hash` label for logs and status.
231    ///
232    /// This value is not computed or verified here; different mappings can carry the same label. Use
233    /// [`Self::fingerprint`] to detect changes.
234    pub fn content_hash(&self) -> &str {
235        &self.content_hash
236    }
237
238    /// Returns a non-cryptographic hash of the original JSON bytes for change detection within this process.
239    ///
240    /// Compare fingerprints to decide whether to rebuild derived data, such as peer tag keys for APM stats.
241    /// Formatting, metadata, and skipped entries also affect the hash. A collision can leave derived data stale;
242    /// this is not an integrity check.
243    pub fn fingerprint(&self) -> u64 {
244        self.fingerprint
245    }
246
247    /// Returns the ordered list of attribute fallbacks for a concept, or
248    /// `None` if the concept has no registered mappings.
249    pub fn get_attribute_precedence(&self, concept: Concept) -> Option<&[TagInfo]> {
250        self.mappings.get(&concept).map(Vec::as_slice)
251    }
252
253    /// Returns the document's `version` label, or `""` if absent, `null`, or not a string.
254    ///
255    /// This label is informational; it does not control parsing or update selection.
256    pub fn version(&self) -> &str {
257        &self.version
258    }
259}

261/// Records the concept keys and fallback positions skipped by [`Registry::from_slice_permissive`].
262///
263/// Its [`Display`](fmt::Display) reports counts and a bounded list of escaped, shortened keys.
264/// It quotes payload keys: log it locally, but never report it to the remote endpoint.
265#[derive(Debug, Default, PartialEq, Eq)]
266pub(crate) struct SkipReport {
267    /// Concept keys that don't correspond to a known [`Concept`].
268    unknown_concepts: Vec<String>,
269
270    /// Keys of concepts with malformed mappings or `fallbacks`.
271    malformed_concepts: Vec<String>,
272
273    /// Unsupported fallbacks, as (concept key, zero-based position).
274    fallbacks: Vec<(String, usize)>,
275}
276
277impl SkipReport {
278    /// Returns `true` if nothing was skipped.
279    pub(crate) fn is_empty(&self) -> bool {
280        self.unknown_concepts.is_empty() && self.malformed_concepts.is_empty() && self.fallbacks.is_empty()
281    }
282}
283
284impl fmt::Display for SkipReport {
285    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
286        if self.is_empty() {
287            return f.write_str("nothing skipped");
288        }
289        let fallbacks: Vec<String> = self
290            .fallbacks
291            .iter()
292            .take(MAX_LISTED_KEYS)
293            .map(|(key, position)| format!("{:?}[{position}]", shorten_key(key)))
294            .collect();
295        let quoted = |keys: &[String]| -> Vec<String> {
296            keys.iter()
297                .take(MAX_LISTED_KEYS)
298                .map(|key| format!("{:?}", shorten_key(key)))
299                .collect()
300        };
301        let sections: [(&str, usize, Vec<String>); 3] = [
302            (
303                "unknown concepts",
304                self.unknown_concepts.len(),
305                quoted(&self.unknown_concepts),
306            ),
307            (
308                "malformed concepts",
309                self.malformed_concepts.len(),
310                quoted(&self.malformed_concepts),
311            ),
312            ("unrepresentable fallbacks", self.fallbacks.len(), fallbacks),
313        ];
314        let mut separator = "";
315        for (label, count, listed) in sections.into_iter().filter(|(_, count, _)| *count > 0) {
316            let more = if count > listed.len() { ", ..." } else { "" };
317            write!(f, "{separator}{label}: {count} ({}{more})", listed.join(", "))?;
318            separator = "; ";
319        }
320        Ok(())
321    }
322}
323
324/// Caps the keys listed in each skip-report section.
325const MAX_LISTED_KEYS: usize = 10;
326
327/// Caps the characters shown from each key in a skip report.
328const MAX_KEY_CHARS: usize = 64;
329
330fn shorten_key(key: &str) -> Cow<'_, str> {
331    if key.chars().count() <= MAX_KEY_CHARS {
332        return Cow::Borrowed(key);
333    }
334    let mut shortened: String = key.chars().take(MAX_KEY_CHARS).collect();
335    shortened.push('…');
336    Cow::Owned(shortened)
337}

339/// Reports a JSON error by position without quoting the payload.
340fn malformed_error(e: serde_json::Error) -> GenericError {
341    generic_error!(
342        "Registry JSON is malformed at line {}, column {}.",
343        e.line(),
344        e.column()
345    )
346}

348/// Rejects documents without concepts or a content-hash label, matching upstream registry validation.
349fn check_required(has_concepts: bool, content_hash: &str) -> Result<(), GenericError> {
350    if !has_concepts {
351        return Err(generic_error!("Registry JSON contains no concepts."));
352    }
353    if content_hash.is_empty() {
354        return Err(generic_error!("Registry JSON is missing `metadata.content_hash`."));
355    }
356    Ok(())
357}

361/// Hashes a raw registry document for change detection.
362fn fingerprint_for(json: &[u8]) -> u64 {
363    use std::hash::Hasher as _;
364
365    let mut hasher = std::collections::hash_map::DefaultHasher::new();
366    hasher.write(json);
367    hasher.finish()
368}
