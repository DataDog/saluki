36/// Linux Control Groups-specific configuration.
37///
38/// Provides environment-specific paths to both "procfs" and "cgroupfs" filesystems, necessary for querying the Linux
39/// Control Groups v2 unified hierarchy.
40pub struct CgroupsConfiguration {
41    procfs_root: PathBuf,
42    cgroupfs_root: PathBuf,
43}

45impl CgroupsConfiguration {
46    /// Creates a new `CgroupsConfiguration` from the given configuration.
47    ///
48    /// # Errors
49    ///
50    /// If any of the paths in the configuration aren't valid, an error will be returned. This doesn't include,
51    /// however, if any of the configured paths don't _exist_.
52    pub fn from_configuration(
53        config: &GenericConfiguration, feature_detector: FeatureDetector,
54    ) -> Result<Self, GenericError> {
55        let procfs_root = match config.try_get_typed::<PathBuf>("container_proc_root")? {
56            Some(path) => path,
57            None => {
58                if feature_detector.is_feature_available(Feature::HostMappedProcfs) {
59                    PathBuf::from(DEFAULT_HOST_MAPPED_PROCFS_ROOT)
60                } else {
61                    PathBuf::from(DEFAULT_PROCFS_ROOT)
62                }
63            }
64        };
65
66        let cgroupfs_root = match config.try_get_typed::<PathBuf>("container_cgroup_root")? {
67            Some(path) => path,
68            None => {
69                if feature_detector.is_feature_available(Feature::HostMappedProcfs) {
70                    PathBuf::from(DEFAULT_HOST_MAPPED_CGROUPFS_ROOT)
71                } else {
72                    // TODO: Consider if we need to do anything specific for Amazon Linux [1] or does the referenced code only
73                    // matter for cgroups v1?
74                    //
75                    // [1]: https://github.com/DataDog/datadog-agent/blob/fe75b815c2f135f0d2ea85d7a57a8fc8cbf56bd9/pkg/config/setup/config.go#L1172-L1173
76                    PathBuf::from(DEFAULT_CGROUPFS_ROOT)
77                }
78            }
79        };
80
81        Ok(Self {
82            procfs_root,
83            cgroupfs_root,
84        })
85    }

90    }

98/// Reader for querying control groups being used for containerization.
99///
100/// This reader is capable of querying both cgroups v1 and v2 hierarchies, and can be used to find cgroups -- either
101/// within the entire hierarchy, or for a specific process ID -- that are mapped specifically to containers. A simple
102/// naming heuristic is used to both identify and extract container IDs from cgroup names.
103#[derive(Clone)]
104pub struct CgroupsReader {
105    procfs_path: PathBuf,
106    hierarchy_reader: HierarchyReader,
107    interner: GenericMapInterner,
108}
109
110impl CgroupsReader {
111    /// Creates a new `CgroupsReader` from the given configuration and interner.
112    ///
113    /// If either a valid cgroups v1 or v2 hierarchy is found, `Ok(Some)` is returned with the reader. Otherwise,
114    /// `Ok(None)` is returned.
115    ///
116    /// The provided interner will be used exclusively for handling container IDs.
117    ///
118    /// # Errors
119    ///
120    /// If there is an I/O error while attempting to query the current cgroups hierarchy, an error will be returned.
121    pub fn try_from_config(
122        config: &CgroupsConfiguration, interner: GenericMapInterner,
123    ) -> Result<Option<Self>, GenericError> {
124        let hierarchy_reader = HierarchyReader::try_from_config(config)?;
125        Ok(hierarchy_reader.map(|hierarchy_reader| Self {
126            procfs_path: config.procfs_path().to_path_buf(),
127            hierarchy_reader,
128            interner,
129        }))
130    }

132    fn try_cgroup_from_path(&self, cgroup_path: &Path) -> Option<Cgroup> {
133        let container_id = extract_container_id_from_path(cgroup_path, &self.interner)?;
134
135        let metadata = match cgroup_path.metadata() {
136            Ok(metadata) => metadata,
137            Err(e) => {
138                trace!(error = %e, cgroup_controller_path = %cgroup_path.display(), "Failed to get metadata for possible cgroup controller path.");
139                return None;
140            }
141        };
142
143        // A reserved inode can't be attributed to this controller specifically, so we have nothing usable to key an
144        // alias on. Drop the cgroup rather than reporting it with an inode that would resolve the wrong workload -- or
145        // none at all.
146        let controller_inode = metadata.ino();
147        if !is_usable_controller_inode(controller_inode) {
148            debug!(
149                controller_inode,
150                %container_id,
151                cgroup_controller_path = %cgroup_path.display(),
152                "Ignoring cgroup controller with reserved inode.",
153            );
154            return None;
155        }
156
157        trace!(
158            controller_inode,
159            %container_id,
160            cgroup_controller_path = %cgroup_path.display(),
161            "Found valid cgroups controller for container.",
162        );
163
164        Some(Cgroup {
165            ino: Some(controller_inode),
166            container_id,
167        })
168    }

170    /// Gets a cgroup for the given process ID.
171    ///
172    /// This method will attempt to find the cgroup for the given process ID by looking at the `/proc/<pid>/cgroup`
173    /// file. If the process ID doesn't exist or isn't attached to a cgroup, `None` will be returned.
174    pub fn get_cgroup_by_pid(&self, pid: u32) -> Option<Cgroup> {
175        // See if the given process ID exists in the proc filesystem _and_ if there's a cgroup path for it.
176        let proc_pid_cgroup_path = self.procfs_path.join(pid.to_string()).join("cgroup");
177        let lines = match read_lines(&proc_pid_cgroup_path) {
178            Ok(lines) => lines,
179            Err(e) => match e.kind() {
180                io::ErrorKind::NotFound => {
181                    debug!(pid, cgroup_lookup_path = %proc_pid_cgroup_path.display(), "Process does not exist or is not attached to a cgroup.");
182                    return None;
183                }
184                _ => {
185                    debug!(error = %e, pid, cgroup_lookup_path = %proc_pid_cgroup_path.display(), "Failed to read cgroup file for process.");
186                    return None;
187                }
188            },
189        };
190
191        let base_controller_name = self.hierarchy_reader.base_controller();
192
193        // We're looking for the first line that matches our base controller name, and then we'll see if it's attached
194        // to the container based on the name, and if so, return it.
195        for entry in lines.iter().filter_map(|s| CgroupControllerEntry::try_from_str(s)) {
196            if entry.name == base_controller_name {
197                // We explicitly try to extract the container ID from the reported cgroup controller path, rather than
198                // trying to stick it on the end of our configured root cgroups path. This is because unless we're in the
199                // host's cgroup namespace, the path we get here will be the leaf directory -- the part with the
200                // container ID in it -- but it will be relative in a way that doesn't allow it to be appended to the
201                // root cgroups path, and so trying to query it to get the controller inode, and all of that, will fail.
202                //
203                // The names in that path are all we need: matching them doesn't touch the filesystem, so it works just
204                // as well for a relative path as an absolute one.
205                if let Some(container_id) = extract_container_id_from_path(entry.path, &self.interner) {
206                    return Some(Cgroup {
207                        ino: None,
208                        container_id,
209                    });
210                }
211            } else {
212                debug!(pid, cgroup_lookup_path = %proc_pid_cgroup_path.display(), base_controller_name, "Found cgroup controller for process, but it doesn't match the base controller.");
213            }
214        }
215
216        debug!(pid, cgroup_lookup_path = %proc_pid_cgroup_path.display(), base_controller_name, "Could not find matching base cgroup controller for process.");
217
218        None
219    }

221    /// Gets all child cgroups in the current cgroups hierarchy.
222    ///
223    /// Individual paths that can't be traversed -- most commonly because a container exited and its cgroup was removed
224    /// while we were walking the hierarchy -- are skipped rather than aborting the traversal. If any of those skips
225    /// could have hidden a cgroup that still exists, the returned traversal is marked as incomplete. See
226    /// [`TraversalResult::is_complete`] for why that distinction matters.
227    pub fn get_child_cgroups(&self) -> TraversalResult {
228        // Walk the cgroups hierarchy and collect all cgroups that we can find that are related to containers..
229        let root_path = self.hierarchy_reader.root_path();
230
231        match visit_subdirectories(root_path, |path| self.try_cgroup_from_path(path)) {
232            Ok(traversal) => traversal,
233            Err(e) => {
234                // We only get here if the hierarchy root itself couldn't be read, which generally points at a
235                // misconfigured cgroupfs path rather than a transient condition.
236                warn!(error = %e, cgroups_root = %root_path.display(), "Failed to visit cgroups hierarchy.");
237
238                TraversalResult::unreadable()
239            }
240        }
241    }
242}

244/// The result of traversing the cgroups hierarchy.
245///
246/// This accumulates as the traversal runs: [`visit_subdirectories`] creates one, records each cgroup it's handed and
247/// each path it couldn't read, and returns it.
248#[derive(Default)]
249pub struct TraversalResult {
250    cgroups: Vec<Cgroup>,
251    skipped: usize,
252    obscured: usize,
253}
254
255impl TraversalResult {
256    /// Creates a traversal representing a hierarchy that couldn't be read at all.
257    ///
258    /// The result is empty and not [complete][Self::is_complete], since failing to read the root hides everything
259    /// beneath it.
260    fn unreadable() -> Self {
261        Self {
262            cgroups: Vec::new(),
263            skipped: 0,
264            obscured: 1,
265        }
266    }
267
268    /// Records a container cgroup found during the traversal.
269    fn record_cgroup(&mut self, cgroup: Cgroup) {
270        self.cgroups.push(cgroup);
271    }
272
273    /// Records a path that couldn't be read, classifying whether skipping it may have hidden existing cgroups.
274    fn record_skip(&mut self, e: &io::Error, path: &Path) {
275        self.skipped += 1;
276
277        match e.kind() {
278            // The directory is gone, so everything beneath it is gone too. There's nothing left for the skip to hide,
279            // and callers tracking those cgroups are right to consider them removed.
280            //
281            // This is routine rather than exceptional: cgroups are removed as the workloads attached to them exit, and
282            // we have no way to hold the hierarchy still while we walk it.
283            io::ErrorKind::NotFound => {
284                trace!(error = %e, path = %path.display(), "Path disappeared during traversal. Skipping.");
285            }
286
287            // We can't read this subtree, so we've never reported anything from it, so there's nothing a caller could
288            // be tracking for us to hide from them.
289            //
290            // This assumes the permissions aren't changing underneath us: a directory that was readable and becomes
291            // unreadable would be misclassified here. That's rare enough to accept, and the alternative -- treating
292            // every permission error as obscuring -- would permanently mark traversals unreliable whenever some part
293            // of the tree is simply not ours to read.
294            io::ErrorKind::PermissionDenied => {
295                debug!(error = %e, path = %path.display(), "Path is not readable. Skipping.");
296            }
297
298            // The subtree is still there and we just failed to read it this time, so anything beneath it is now
299            // invisible to us despite still existing.
300            _ => {
301                self.obscured += 1;
302                debug!(error = %e, path = %path.display(), "Failed to traverse path. Skipping.");
303            }
304        }
305    }
306
307    /// Returns whether absence from this traversal can be taken to mean a cgroup no longer exists.
308    ///
309    /// When this is `false`, part of the hierarchy that may still hold live cgroups couldn't be read, so the cgroups
310    /// reported are not exhaustive. The entries present are still valid, and callers can safely treat them as live, but
311    /// callers **MUST NOT** infer that a previously known cgroup was removed simply because it's absent here.
312    ///
313    /// Note that this can be `true` even when [`skipped`][Self::skipped] is non-zero: a skipped path that couldn't have
314    /// hidden a live cgroup -- because the path is gone, or because we've never been able to read it -- doesn't make
315    /// the set unreliable.
316    pub fn is_complete(&self) -> bool {
317        self.obscured == 0
318    }
319
320    /// Returns the number of paths skipped due to recoverable errors during the traversal.
321    ///
322    /// This counts every skip, including those that don't affect [`is_complete`][Self::is_complete], and is intended
323    /// for telemetry rather than for deciding how much to trust the result.
324    pub fn skipped(&self) -> usize {
325        self.skipped
326    }
327
328    /// Consumes `self` and returns the container cgroups found during the traversal.
329    pub fn into_cgroups(self) -> Vec<Cgroup> {
330        self.cgroups
331    }
332}

334#[derive(Clone)]
335enum HierarchyReader {
336    V1 {
337        base_controller_path: PathBuf,
338        controllers: HashMap<String, PathBuf>,
339    },
340
341    V2 {
342        root: PathBuf,
343        controllers: Vec<String>,
344    },
345}
346
347impl HierarchyReader {
348    fn try_from_config(config: &CgroupsConfiguration) -> Result<Option<Self>, GenericError> {
349        // Open the mount file from procfs to scan through and find any cgroups subsystems.
350        let mounts_path = config.procfs_path().join("mounts");
351        let mount_entries = read_lines(&mounts_path)
352            .with_error_context(|| format!("Failed to read mount entries from procfs ({})", mounts_path.display()))?;
353
354        let mut controllers = HashMap::new();
355        let mut maybe_cgroups_v2 = None;
356
357        // For each mount line, check if its of the `cgroup` or `cgroup2` type. Skip everything else.
358        for mount_entry in mount_entries {
359            // Split the line into fields, and take the second and third values. We always expect at least three fields
360            // in a line if it's a line that might possibly be a cgroup mount.
361            let mut fields = mount_entry.split_whitespace();
362            let maybe_cgroup_path = fields.nth(1);
363            let maybe_fs_type = fields.nth(0);
364
365            if let (Some(raw_cgroup_path), Some(fs_type)) = (maybe_cgroup_path, maybe_fs_type) {
366                let cgroup_path = Path::new(raw_cgroup_path);
367
368                // Make sure this path is rooted within our configured cgroupfs path.
369                //
370                // When we're inside a container that has a host-mapped cgroupfs path, the `mounts` file might end up
371                // having duplicate entries (like one set as `/sys/fs/cgroup` and another set as `/host/sys/fs/cgroup`,
372                // etc)... and we want to use the one that matches our configured cgroupfs path as that's the one that
373                // will actually have the cgroups we care about.
374                if !cgroup_path.starts_with(config.cgroupfs_path()) {
375                    continue;
376                }
377
378                match fs_type {
379                    // For cgroups v1, we have to go through all mounts we see to build a full list of enabled controlled.
380                    "cgroup" => process_cgroupv1_mount_entry(cgroup_path, &mut controllers),
381                    // For cgroups v2, we only need to find the unified root mountpoint, and then we can create our reader.
382                    "cgroup2" => maybe_cgroups_v2 = process_cgroupv2_mount_entry(cgroup_path)?,
383                    _ => {}
384                }
385            }
386        }
387
388        // If we didn't find any cgroups v1 controllers, then we potentially return the cgroups v2 hierarchy if found...
389        // otherwise, this will just return `None`.
390        if controllers.is_empty() {
391            if maybe_cgroups_v2.is_some() {
392                debug!("Using cgroups v2 hierarchy.");
393            }
394
395            return Ok(maybe_cgroups_v2);
396        }
397
398        // If we're here, we potentially have a cgroups v1 hierarchy.  Find our base controller -- the memory controller
399        // -- and once we do that, we can create our reader.
400        let base_controller_path = controllers
401            .get(CGROUPS_V1_BASE_CONTROLLER_NAME)
402            .cloned()
403            .ok_or_else(|| {
404                generic_error!(
405                    "Failed to find base controller ({}) in cgroups v1 hierarchy.",
406                    CGROUPS_V1_BASE_CONTROLLER_NAME
407                )
408            })?;
409
410        debug!(root = %base_controller_path.display(), controllers_len = controllers.len(), "Using cgroups v1 hierarchy.");
411
412        Ok(Some(HierarchyReader::V1 {
413            base_controller_path,
414            controllers,
415        }))
416    }
417
418    fn base_controller(&self) -> Option<&'static str> {
419        match self {
420            Self::V1 { .. } => Some(CGROUPS_V1_BASE_CONTROLLER_NAME),
421
422            // Since cgroups v2 is "unified", there's no base controller path.
423            Self::V2 { .. } => None,
424        }
425    }
426
427    fn root_path(&self) -> &Path {
428        match self {
429            Self::V1 {
430                base_controller_path, ..
431            } => base_controller_path.as_path(),
432            Self::V2 { root, .. } => root.as_path(),
433        }
434    }
435}

455struct CgroupControllerEntry<'a> {
456    id: usize,
457    name: Option<&'a str>,
458    path: &'a Path,
459}
460
461impl<'a> CgroupControllerEntry<'a> {
462    fn try_from_str(line: &'a str) -> Option<Self> {
463        let mut fields = line.splitn(3, ':');
464
465        let id = fields.next()?.parse::<usize>().ok()?;
466        let name = fields.next().map(|s| if s.is_empty() { None } else { Some(s) })?;
467        let path = fields.next()?;
468
469        if path.is_empty() {
470            return None;
471        }
472
473        Some(Self {
474            id,
475            name,
476            path: Path::new(path),
477        })
478    }
479}

481fn process_cgroupv1_mount_entry(cgroup_path: &Path, controllers: &mut HashMap<String, PathBuf>) {
482    // Split the cgroup path, since there can be multiple controllers mounted at the same path.
483    let path_controllers = cgroup_path
484        .file_name()
485        .and_then(|s| s.to_str().map(|s| s.split(',')))
486        .into_iter()
487        .flatten();
488    for path_controller in path_controllers {
489        // If we have an existing path mapping for this controller, keep whichever one is the
490        // shortest, as we want the more generic path.
491        if let Some(existing_path) = controllers.get(path_controller) {
492            if existing_path.as_os_str().len() < cgroup_path.as_os_str().len() {
493                continue;
494            }
495        }
496
497        controllers.insert(path_controller.to_string(), PathBuf::from(cgroup_path));
498    }
499}

501fn process_cgroupv2_mount_entry(cgroup_path: &Path) -> Result<Option<HierarchyReader>, GenericError> {
502    // Read and get the list of active/enabled controllers.
503    let controllers_path = cgroup_path.join(CGROUPS_V2_CONTROLLERS_FILE);
504    let controllers = read_lines(&controllers_path)
505        .with_error_context(|| {
506            format!(
507                "Failed to read controllers from cgroups v2 hierarchy ({}).",
508                controllers_path.display()
509            )
510        })?
511        .into_iter()
512        .flat_map(|s| s.split_whitespace().map(|s| s.to_string()).collect::<Vec<_>>())
513        .collect::<Vec<_>>();
514
515    Ok(Some(HierarchyReader::V2 {
516        root: cgroup_path.to_path_buf(),
517        controllers,
518    }))
519}

521fn read_lines(path: &Path) -> io::Result<Vec<String>> {
522    let file = OpenOptions::new().read(true).open(path)?;
523
524    let reader = BufReader::new(file).lines();
525
526    let mut lines = Vec::new();
527    for line in reader {
528        lines.push(line?);
529    }
530
531    Ok(lines)
532}

534/// Visits every subdirectory beneath the given path, collecting the cgroups that `visit` identifies.
535///
536/// Subdirectories that can't be read are skipped, along with everything beneath them, and recorded in the returned
537/// [`TraversalResult`]. Callers that need to distinguish "this subdirectory is gone" from "we couldn't see this
538/// subdirectory" **MUST** check [`TraversalResult::is_complete`].
539///
540/// # Errors
541///
542/// If the given path itself can't be queried or listed, an error is returned: nothing was seen, so there's no result
543/// worth reporting. Failures below the given path are never fatal.
544fn visit_subdirectories<P, F>(path: P, mut visit: F) -> Result<TraversalResult, GenericError>
545where
546    P: AsRef<Path>,
547    F: FnMut(&Path) -> Option<Cgroup>,
548{
549    let root = path.as_ref();
550
551    // We can only visit directories, so if the initial path we're given isn't a directory, then we can't do anything.
552    let metadata = fs::metadata(root)
553        .with_error_context(|| format!("Failed to query metadata for traversal root ({}).", root.display()))?;
554    if !metadata.is_dir() {
555        return Ok(TraversalResult::default());
556    }
557
558    let mut traversal = TraversalResult::default();
559
560    // Do an initial pass on our path to get all of its subdirectories, which we'll visit, and then also use as the seed
561    // for further visiting.
562    let mut stack = vec![root.to_path_buf()];
563    while let Some(path) = stack.pop() {
564        // A directory can be removed between the point where we discovered it and the point where we pop it off the
565        // stack to read it, so failing here costs us that subtree but shouldn't stop us from walking the rest.
566        let dir_reader = match fs::read_dir(&path) {
567            Ok(dir_reader) => dir_reader,
568            Err(e) => {
569                // Failing on the root is fatal, unlike failing anywhere below it. Every other path costs us one
570                // subtree, but if we can't list the root then we haven't seen anything at all -- and an empty result
571                // that claims to be complete tells callers every cgroup they know about has gone away.
572                //
573                // Note that this is reachable even though we successfully stat'd the root above: listing a directory
574                // needs read permission, while stat'ing it only needs to traverse its parent.
575                if path.as_path() == root {
576                    return Err(e)
577                        .with_error_context(|| format!("Failed to read traversal root ({}).", root.display()));
578                }
579
580                traversal.record_skip(&e, &path);
581                continue;
582            }
583        };
584
585        for entry in dir_reader {
586            let entry = match entry {
587                Ok(entry) => entry,
588                Err(e) => {
589                    traversal.record_skip(&e, &path);
590                    continue;
591                }
592            };
593
594            let entry_path = entry.path();
595            let file_type = match entry.file_type() {
596                Ok(file_type) => file_type,
597                Err(e) => {
598                    traversal.record_skip(&e, &entry_path);
599                    continue;
600                }
601            };
602
603            if file_type.is_dir() {
604                if let Some(cgroup) = visit(&entry_path) {
605                    traversal.record_cgroup(cgroup);
606                }
607
608                stack.push(entry_path);
609            }
610        }
611    }
612
613    Ok(traversal)
614}

616/// Gets the current process's container ID from its local cgroup membership.
617///
618/// This intentionally reads the process namespace's `/proc/self/cgroup` instead of a configured procfs root, which may

625fn get_container_id_from_cgroup_lines(lines: &[String], interner: &GenericMapInterner) -> Option<MetaString> {
626    lines
627        .iter()
628        .filter_map(|line| CgroupControllerEntry::try_from_str(line))
629        .filter_map(|entry| entry.path.file_name().and_then(|name| name.to_str()))

633/// Returns `true` if the given inode can identify a specific cgroup controller.
634///
635/// Reserved inodes -- see [`MAX_RESERVED_INODE`] -- are reported by some filesystems for paths that aren't a distinct
636/// object, so they can't be used to tell one controller apart from another.
637fn is_usable_controller_inode(inode: u64) -> bool {
638    inode > MAX_RESERVED_INODE
639}

641/// Matches a container ID anywhere within a cgroup name.
642///
643/// This regular expression is meant to capture:
644/// - 64 character hexadecimal strings (standard format for container IDs almost everywhere)
645/// - 32 character hexadecimal strings followed by a dash and a number (used by AWS ECS)
646/// - 8 character hexadecimal strings followed by up to four groups of 4 character hexadecimal strings separated by
647///   dashes (essentially a UUID, used by Pivotal Cloud Foundry's Garden technology)
648static CONTAINER_REGEX: LazyLock<Regex> =
649    LazyLock::new(|| Regex::new("([0-9a-f]{64})|([0-9a-f]{32}-\\d+)|([0-9a-f]{8}(-[0-9a-f]{4}){4}$)").unwrap());
650
651fn extract_container_id(cgroup_name: &str, interner: &GenericMapInterner) -> Option<MetaString> {
652    match match_container_id(cgroup_name, interner) {
653        ContainerIdMatch::Container(container_id) => Some(container_id),
654        ContainerIdMatch::Excluded | ContainerIdMatch::Uninternable | ContainerIdMatch::NoMatch => None,
655    }
656}
657
658/// What a single cgroup name turned out to be.
659enum ContainerIdMatch {
660    /// The cgroup belongs to a container, with the given ID.
661    Container(MetaString),
662
663    /// The cgroup is named after a container but doesn't represent one.
664    Excluded,
665
666    /// The cgroup belongs to a container, but interning its ID failed.
667    ///
668    /// We know which container this is and simply can't name it, which is different from not knowing: a caller walking
669    /// a path **MUST NOT** keep searching outwards, since the answer it found would be a different container.
670    Uninternable,
671
672    /// The cgroup isn't named after a container at all.
673    NoMatch,
674}
675
676/// Matches a single cgroup name against the container ID heuristic.
677///
678/// [`ContainerIdMatch::Excluded`] is reported separately from [`ContainerIdMatch::NoMatch`] because the two mean
679/// different things to a caller walking a path: a name that isn't a container tells you nothing about its ancestors,
680/// but a name that is deliberately excluded is a definitive answer for that cgroup.
681fn match_container_id(cgroup_name: &str, interner: &GenericMapInterner) -> ContainerIdMatch {
682    let container_id = match CONTAINER_REGEX.find(cgroup_name) {
683        Some(container_id) => container_id,
684        None => return ContainerIdMatch::NoMatch,
685    };
686
687    // Note that this is checked against the full cgroup name, not against the ID we just matched out of it: the match
688    // is a bare hexadecimal string, which can never carry any of these prefixes or suffixes.
689    if is_container_named_but_not_a_container(cgroup_name) {
690        return ContainerIdMatch::Excluded;
691    }
692
693    match interner.try_intern(container_id.as_str()) {
694        Some(interned) => ContainerIdMatch::Container(MetaString::from(interned)),
695        None => {
696            error!(container_id = %container_id.as_str(), "Failed to intern container ID.");
697            ContainerIdMatch::Uninternable
698        }
699    }
700}
701
702/// Resolves the container ID for a cgroup path, falling back to the path's ancestors.
703///
704/// A container's workload can sit in a cgroup nested below the one named for the container, in which case the leaf
705/// doesn't carry the ID but one of its ancestors does. The deepest ancestor that names a container wins, so the most
706/// specific enclosing container is the one reported.
707///
708/// The search stops early, returning `None`, in two cases where continuing outwards would answer with some *other*
709/// container:
710///
711/// - The leaf is named after a container but isn't one. Such a cgroup isn't part of the container's workload at all.
712/// - A container is identified but its ID can't be interned. We know which container it is and just can't name it,
713///   which is not the same as not knowing.
714fn extract_container_id_from_path(cgroup_path: &Path, interner: &GenericMapInterner) -> Option<MetaString> {
715    let leaf_name = cgroup_path.file_name().and_then(|s| s.to_str())?;
716
717    match match_container_id(leaf_name, interner) {
718        ContainerIdMatch::Container(container_id) => return Some(container_id),
719        ContainerIdMatch::Excluded | ContainerIdMatch::Uninternable => return None,
720        ContainerIdMatch::NoMatch => {}
721    }
722
723    // `ancestors` yields the path itself first, which we've already checked, so skip it.
724    for ancestor in cgroup_path.ancestors().skip(1) {
725        let ancestor_name = match ancestor.file_name().and_then(|s| s.to_str()) {
726            Some(ancestor_name) => ancestor_name,
727            None => continue,
728        };
729
730        match match_container_id(ancestor_name, interner) {
731            ContainerIdMatch::Container(container_id) => return Some(container_id),
732
733            // An excluded ancestor is a statement about that cgroup, not about the one we're resolving, so the
734            // container enclosing it can still be the right answer.
735            ContainerIdMatch::Excluded | ContainerIdMatch::NoMatch => {}
736
737            // Reporting the next container out would attribute this cgroup to the wrong one, so stop here.
738            ContainerIdMatch::Uninternable => return None,
739        }
740    }
741
742    None
743}

745/// Returns `true` if a cgroup is named after a container but doesn't represent that container's workload.
746fn is_container_named_but_not_a_container(cgroup_name: &str) -> bool {
747    // With the systemd cgroup driver, a `.mount` cgroup can sit alongside a container's own cgroup. It exists, but no
748    // process is ever attached to it, so it holds no stats.
749    //
750    // The `conmon` cgroups belong to the CRI-O/Podman monitor process supervising a container, rather than to the
751    // container itself.
752    cgroup_name.ends_with(".mount")
753        || cgroup_name.starts_with("crio-conmon-")
754        || cgroup_name.starts_with("libpod-conmon-")
755}

971    /// Builds an interner that can still resolve `held_id` but has no room left for `blocked_id`.
972    ///
973    /// The returned handles have to be kept alive for as long as the interner is used: an entry is reclaimed as soon
974    /// as its last handle drops, so dropping them would un-fill the interner.
975    fn interner_holding_but_blocking(held_id: &str, blocked_id: &str) -> (GenericMapInterner, Vec<InternedString>) {
976        let interner = GenericMapInterner::new(NonZeroUsize::new(1024).unwrap());
977        let mut held = vec![interner.try_intern(held_id).expect("interner starts out empty")];
978
979        // The interner is sharded, so "full" is per-shard: we have to keep adding distinct strings until the shard
980        // that `blocked_id` hashes to is the one that fills up. Probing with `blocked_id` itself is harmless, since
981        // dropping the handle immediately gives the entry back.
982        for filler in 0.. {
983            if interner.try_intern(blocked_id).is_none() {
984                break;
985            }
986
987            assert!(filler < 100_000, "interner never filled up");
988
989            if let Some(interned) = interner.try_intern(&format!("filler-{:060}", filler)) {
990                held.push(interned);
991            }
992        }
993
994        // `held_id` has to still be resolvable, otherwise the tests below would pass for the wrong reason: they need
995        // the ancestor fallback to be *capable* of succeeding, so that declining to take it means something.
996        assert!(interner.try_intern(held_id).is_some());
997
998        (interner, held)
999    }

1168    #[test]
1169    fn visit_subdirectories_skips_directories_removed_mid_traversal() {
1170        let root = tempdir().unwrap();
1171        for name in ["a", "b", "c"] {
1172            fs::create_dir(root.path().join(name)).unwrap();
1173        }
1174
1175        // Every subdirectory of `root` is visited and pushed onto the traversal stack before any of them is read back,
1176        // so removing one that was already visited guarantees that reading it later fails with `ENOENT`. That's the
1177        // same race we lose in production when a container exits mid-traversal, but without depending on any timing.
1178        let mut visited = Vec::new();
1179        let traversal = visit_subdirectories(root.path(), |path| {
1180            visited.push(path.to_path_buf());
1181            if visited.len() == 2 {
1182                fs::remove_dir(&visited[0]).unwrap();
1183            }
1184            None
1185        })
1186        .unwrap();
1187
1188        // The removed directory was still visited -- we saw it before it went away -- but reading it was skipped
1189        // rather than aborting the traversal, so its two siblings were still read.
1190        assert_eq!(traversal.skipped, 1);
1191
1192        // A directory that's gone can't be hiding anything, so the traversal is still trustworthy.
1193        assert_eq!(traversal.obscured, 0);
1194        assert!(traversal.is_complete());
1195        assert_eq!(visited_names(root.path(), &visited), names(&["a", "b", "c"]));
1196    }
