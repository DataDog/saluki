1//! Structured `DogStatsD` event generation.
2
3use rand::{Rng, RngExt};
4
5use super::common;
6
7/// Optional-field prefixes: hostname, aggregation key, priority, source type, alert type. Bad
8/// priority and alert values are logged and defaulted by the Agent, never dropped, so any content
9/// forwards.
10const OPT_PREFIXES: &[&[u8]] = &[b"h:", b"k:", b"p:", b"s:", b"t:"];
11
12/// Optional-field counts: mostly none, with a boundary tail.
13const OPT_COUNTS: &[usize] = &[0, 0, 0, 0, 1, 1, 2, 3, 127, 255];
14
15/// A structured event: `_e{title_len,text_len}:title|text[|opt...]`.
16#[derive(Clone, Debug)]
17pub(crate) struct Event {
18    /// Title content, required non-empty.
19    title: Vec<u8>,
20    /// Text content, may be empty.
21    text: Vec<u8>,
22    /// `key:value` tags.
23    tags: Vec<Vec<u8>>,
24    /// Optional chunks, each carrying its own prefix.
25    options: Vec<Vec<u8>>,
26}
27
28impl Event {
29    /// Sample an event with content over the full forwarded alphabet. The body is length-delimited,
30    /// so title and text carry any bytes including delimiters.
31    pub(crate) fn generate(rng: &mut (impl Rng + ?Sized), budget: usize) -> Option<Self> {
32        // `_e{N,M}:title|text` is the skeleton. The header digits grow with the field lengths, so a
33        // conservative four bytes per length keeps the estimate above what serialization will write for
34        // any field this generator can build.
35        const HEADER: usize = "_e{".len() + 4 + 1 + 4 + "}:".len() + 1;
36        let title_room = budget.checked_sub(HEADER)?;
37        let title = common::identifier_within(rng, title_room);
38        if title.is_empty() {
39            return None;
40        }
41        let spent = HEADER + title.len();
42        let text = common::optional_text_within(rng, budget.saturating_sub(spent));
43        let spent = spent + text.len();
44        let tags = common::tags_within(rng, budget.saturating_sub(spent));
45        let spent = spent + common::tags_len(&tags);
46        Some(Self {
47            title,
48            text,
49            tags,
50            options: options(rng, budget.saturating_sub(spent)),
51        })
52    }
53
54    /// Serialize the event to datagram bytes, without the trailing `\n`. The header lengths are the
55    /// true byte lengths of the title and text, so the Agent never rejects on a length mismatch.
56    pub(crate) fn serialize(&self, out: &mut Vec<u8>) {
57        let mut itoa = itoa::Buffer::new();
58        out.extend_from_slice(b"_e{");
59        out.extend_from_slice(itoa.format(self.title.len()).as_bytes());
60        out.push(b',');
61        out.extend_from_slice(itoa.format(self.text.len()).as_bytes());
62        out.extend_from_slice(b"}:");
63        out.extend_from_slice(&self.title);
64        out.push(b'|');
65        out.extend_from_slice(&self.text);
66        common::serialize_tags(&self.tags, out);
67        for opt in &self.options {
68            out.push(b'|');
69            out.extend_from_slice(opt);
70        }
71    }
72}
73
74/// Sample a run of optional chunks.
75fn options(rng: &mut (impl Rng + ?Sized), budget: usize) -> Vec<Vec<u8>> {
76    let count = OPT_COUNTS[rng.random_range(0..OPT_COUNTS.len())];
77    let mut out = Vec::new();
78    let mut room = budget;
79    for _ in 0..count {
80        let prefix = OPT_PREFIXES[rng.random_range(0..OPT_PREFIXES.len())];
81        // Each chunk carries a leading `|` on the wire.
82        let Some(body_room) = room.checked_sub(1 + prefix.len()) else {
83            break;
84        };
85        let mut chunk = prefix.to_vec();
86        chunk.extend_from_slice(&common::optional_text_within(rng, body_room));
87        room -= 1 + chunk.len();
88        out.push(chunk);
89    }
90    out
