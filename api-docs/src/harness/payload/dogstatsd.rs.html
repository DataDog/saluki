1//! `DogStatsD` payload generation.
2//!
3//! Generation builds three structured message types, metric, event, and service check, then
4//! serializes each to datagram bytes. Every message is checked with [`crate::dogstatsd::is_malformed`]
5//! on its serialized bytes and repaired until the Datadog Agent would forward it. There is no
6//! clean/feral/mixed configuration and no per-line vibe. The legal space is exactly the set of
7//! payloads `is_malformed` accepts. Content fields range over the full forwarded alphabet including
8//! delimiter bytes, and the predicate sorts the delimiter-bearing cases the Agent keeps from the ones
9//! it drops.
10//!
11//! ```text
12//! metric:        <NAME>:<VALUE>(:<VALUE>)*|<TYPE>[|@<RATE>][|#<TAGS>][|c:..][|e:..][|card:..]
13//! event:         _e{<TITLE_LEN>,<TEXT_LEN>}:<TITLE>|<TEXT>[|h:..][|k:..][|p:..][|s:..][|t:..][|#<TAGS>]
14//! service check: _sc|<NAME>|<STATUS>[|h:..][|m:..][|#<TAGS>]
15//! ```
16

26/// How many times to re-sample a message whose serialized bytes the Agent would drop before falling
27/// back to a guaranteed-well-formed line. Construction is mostly-valid, so the fallback is rare.
28const REPAIR_TRIES: usize = 16;
29
30/// A guaranteed-well-formed metric, used only if repair is exhausted, so the pack loop always
31/// terminates.
32const FALLBACK_LINE: &[u8] = b"harness.fallback:1|c";
33
34/// Ceiling on a generated datagram, the Datadog Agent's default `dogstatsd_buffer_size`. A run caps
35/// each datagram to the smaller of this and the SUT's sampled receive buffer, so a packed datagram
36/// always fits one read and the SUT never truncates a line mid-token.
37pub const PAYLOAD_BYTE_LIMIT: usize = 8_192;
38
39/// What a generated payload holds, for anchoring assertions.
40#[derive(Clone, Copy, Debug, Default)]
41pub struct Payload {
42    /// Lines packed into the buffer.
43    pub lines: usize,
44    /// Largest packed multi-value run among those lines. Zero when none.
45    pub max_packed: usize,
46}
47
48/// One of the three `DogStatsD` message types, structured.
49enum Message {
50    Metric(metrics::Metric),
51    Event(events::Event),
52    ServiceCheck(service_checks::ServiceCheck),
53}
54
55impl Message {
56    /// Sample a message type. The mix is heavily metric-weighted: 98% metric, 1% event, 1% service
57    /// check. Metrics drive the aggregate context and sketch paths, so the bulk of load goes there
58    /// while events and service checks still fire often enough to keep their anchors non-vacuous.
59    fn generate(rng: &mut (impl Rng + ?Sized), budget: usize) -> Option<Self> {
60        match rng.random_range(0..100u32) {
61            0 => events::Event::generate(rng, budget).map(Message::Event),
62            1 => service_checks::ServiceCheck::generate(rng, budget).map(Message::ServiceCheck),
63            _ => metrics::Metric::generate(rng, budget).map(Message::Metric),
64        }
65    }
66
67    /// Serialize to datagram bytes, without the trailing `\n`.
68    fn serialize(&self, out: &mut Vec<u8>) {
69        match self {
70            Message::Metric(m) => m.serialize(out),
71            Message::Event(e) => e.serialize(out),
72            Message::ServiceCheck(s) => s.serialize(out),
73        }
74    }
75
76    /// The packed multi-value run length this message contributes, or zero.
77    fn packed(&self) -> usize {
78        match self {
79            Message::Metric(m) => m.packed(),
80            Message::Event(_) | Message::ServiceCheck(_) => 0,
81        }
82    }
83}
84
85/// Sample one message and serialize it to bytes the Agent forwards, within `budget` bytes including
86/// the trailing `\n`. Re-samples up to [`REPAIR_TRIES`] times a message the Agent would drop or one
87/// that overruns the budget, then falls back to [`FALLBACK_LINE`]. Returns `None` only when even the
88/// fallback does not fit.
89///
90/// The budget is a construction input, not a filter applied afterwards: a sample that does not fit is
91/// another sample to take, never a reason to end the payload. Ending it would leave the datagram
92/// short, or empty when the very first sample overran, and a short datagram is workload the SUT never
93/// sees while the driver still spends one of its configured sends.
94fn wellformed_line(rng: &mut (impl Rng + ?Sized), budget: usize) -> Option<(Vec<u8>, usize)> {
95    for _ in 0..REPAIR_TRIES {
96        // The message is built against the budget, so it fits by construction. The retry is only for
97        // the Agent's own drop rules, which are about content rather than size.
98        let Some(message) = Message::generate(rng, budget.saturating_sub(1)) else {
99            break;
100        };
101        let mut bytes = Vec::new();
102        message.serialize(&mut bytes);
103        if bytes.len() < budget && is_malformed(&bytes).is_ok() {
104            return Some((bytes, message.packed()));
105        }
106    }
107    (FALLBACK_LINE.len() < budget).then(|| (FALLBACK_LINE.to_vec(), 0))
108}
109
110/// Pack whole `\n`-terminated lines into `buf` until the budget cannot hold another line. Each line is
111/// sampled against the budget still free, so the datagram fills rather than ending on the first
112/// oversized sample, never exceeds `limit_bytes`, and holds only whole lines every one of which the
113/// Agent forwards. Clears `buf` first.
114pub fn write_payload(rng: &mut (impl Rng + ?Sized), buf: &mut Vec<u8>, limit_bytes: usize) -> Payload {
115    buf.clear();
116    let mut payload = Payload::default();
117    loop {
118        // The budget left, `\n` included. Every accepted line consumes at least one byte, so the
119        // remaining budget strictly shrinks and the loop ends once nothing fits.
120        let Some((line, packed)) = wellformed_line(rng, limit_bytes - buf.len()) else {
121            break;
122        };
123        buf.extend_from_slice(&line);
124        buf.push(b'\n');
125        payload.lines += 1;
126        payload.max_packed = payload.max_packed.max(packed);
127    }
128    payload
129}

131#[cfg(test)]
132mod test {
133    use proptest::prelude::*;
134    use rand::rngs::SmallRng;
135    use rand::SeedableRng;
136
137    use super::{write_payload, FALLBACK_LINE, PAYLOAD_BYTE_LIMIT};
138    use crate::dogstatsd::is_malformed;
139
140    /// Lines carry no interior newline and each is `\n`-terminated, so the line count equals the
141    /// newline count.
142    #[allow(clippy::naive_bytecount)]
143    fn newline_count(buf: &[u8]) -> usize {
144        buf.iter().filter(|&&b| b == b'\n').count()
145    }
146
147    proptest! {
148        /// Every datagram the generator emits is one the Agent forwards. A packed datagram must be
149        /// entirely well-formed.
150        #[test]
151        fn property_test_every_payload_is_well_formed(seed: u64) {
152            let mut rng = SmallRng::seed_from_u64(seed);
153            let mut buf = Vec::new();
154            for _ in 0..8 {
155                write_payload(&mut rng, &mut buf, PAYLOAD_BYTE_LIMIT);
156                prop_assert_eq!(is_malformed(&buf), Ok(()), "emitted a droppable datagram: {:?}", String::from_utf8_lossy(&buf));
157            }
158        }
159
160        #[test]
161        fn property_test_payload_stays_within_its_limit(seed: u64, limit_bytes: u16) {
162            let mut rng = SmallRng::seed_from_u64(seed);
163            let limit_bytes = usize::from(limit_bytes);
164            let mut buf = Vec::new();
165            let payload = write_payload(&mut rng, &mut buf, limit_bytes);
166
167            prop_assert!(buf.len() <= limit_bytes);
168            prop_assert_eq!(newline_count(&buf), payload.lines);
169            if !buf.is_empty() {
170                prop_assert_eq!(buf[buf.len() - 1], b'\n');
171            }
172        }
173
174        /// A budget that admits any line at all yields load. An empty datagram burns one of the
175        /// driver's configured sends without reaching the SUT, so the generator must fill the budget
176        /// it was given rather than give up on the first oversized sample.
177        #[test]
178        fn property_test_payload_is_never_empty_when_a_line_fits(seed: u64, limit_bytes in (FALLBACK_LINE.len() + 1)..4096usize) {
179            let mut rng = SmallRng::seed_from_u64(seed);
180            let mut buf = Vec::new();
181            let payload = write_payload(&mut rng, &mut buf, limit_bytes);
182
183            prop_assert!(!buf.is_empty(), "empty datagram at limit {}", limit_bytes);
184            prop_assert!(payload.lines > 0);
185        }
186    }
