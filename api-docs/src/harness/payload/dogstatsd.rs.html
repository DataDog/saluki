1//! `DogStatsD` payload generation from a pooled context working set.
2//!
3//! A driver no longer samples a fresh identity per line. It fetches a working set of [`Context`]s
4//! from the shared intake pool, [`crate::contexts`], and renders a fresh per-occurrence payload
5//! against a sampled context each line, so identities recur while their load varies. Every rendered
6//! line is repaired by [`Context::render_wellformed_within`] to one the Datadog Agent forwards, so a packed
7//! datagram holds only whole lines the Agent keeps. There is no clean/feral/mixed configuration. The
8//! legal space is exactly the set of payloads [`crate::dogstatsd::is_malformed`] accepts.
9//!
10//! ```text
11//! metric:        <NAME>:<VALUE>(:<VALUE>)*|<TYPE>[|@<RATE>][|#<TAGS>][|c:..][|e:..][|card:..]
12//! event:         _e{<TITLE_LEN>,<TEXT_LEN>}:<TITLE>|<TEXT>[|opt...]|d:<TS>[|#<TAGS>]
13//! service check: _sc|<NAME>|<STATUS>[|opt...]|d:<TS>[|#<TAGS>]|m:<MESSAGE>
14//! ```

22/// Ceiling on a generated datagram, the Datadog Agent's default `dogstatsd_buffer_size`. A run caps
23/// each datagram to the smaller of this and the SUT's sampled receive buffer, so a packed datagram
24/// always fits one read and the SUT never truncates a line mid-token.
25pub const PAYLOAD_BYTE_LIMIT: usize = 8_192;
26
27/// What a generated payload holds, for anchoring assertions.
28#[derive(Clone, Copy, Debug, Default)]
29pub struct Payload {
30    /// Lines packed into the buffer.
31    pub lines: usize,
32    /// Largest packed multi-value run among those lines. Zero when none.
33    pub max_packed: usize,
34}
35
36/// How many affordable contexts a line tries before the datagram ends. A context whose occurrence
37/// payloads keep landing on the drop side is one to move past, not a reason to stop packing.
38const RENDER_PICKS: usize = 4;
39
40/// A driver's working set, ordered by how many bytes each identity forces on every render.
41///
42/// The order is computed once per fetch. Packing a line then finds the affordable contexts by
43/// partition point rather than filtering the whole set, which matters because one invocation packs
44/// thousands of datagrams from a set of up to a thousand identities.
45#[derive(Clone, Debug)]
46pub struct WorkingSet {
47    /// Contexts by ascending floor.
48    contexts: Vec<Context>,
49    /// `contexts[i].floor()`, same order, so the affordable contexts are a prefix.
50    floors: Vec<usize>,
51}
52
53impl WorkingSet {
54    /// Order `contexts` by floor and record each one.
55    #[must_use]
56    pub fn new(mut contexts: Vec<Context>) -> Self {
57        contexts.sort_by_key(Context::floor);
58        let floors = contexts.iter().map(Context::floor).collect();
59        Self { contexts, floors }
60    }
61
62    /// Whether the set holds nothing.
63    #[must_use]
64    pub fn is_empty(&self) -> bool {
65        self.contexts.is_empty()
66    }
67
68    /// The smallest floor in the set, or `None` when the set is empty.
69    #[must_use]
70    pub fn smallest_floor(&self) -> Option<usize> {
71        self.floors.first().copied()
72    }
73
74    /// Pick uniformly among the contexts whose identity fits `budget`, or `None` when none does.
75    fn pick(&self, rng: &mut (impl Rng + ?Sized), budget: usize) -> Option<&Context> {
76        let affordable = self.floors.partition_point(|&floor| floor <= budget);
77        (affordable > 0).then(|| &self.contexts[rng.random_range(0..affordable)])
78    }
79}
80
81/// Pack whole `\n`-terminated lines into `buf`, each a fresh render of a context that fits the room
82/// left under `limit_bytes`. A context is picked from those the remaining budget can hold and rendered
83/// against that budget, so the datagram fills instead of ending on the first oversized render and
84/// nothing is built and thrown away. A context whose occurrence payloads all land on the drop side is
85/// passed over for another, up to [`RENDER_PICKS`], rather than ending the datagram. Clears `buf`
86/// first. An empty working set, or a budget too small for any identity, yields an empty payload.
87pub fn write_payload(
88    rng: &mut (impl Rng + ?Sized), contexts: &WorkingSet, buf: &mut Vec<u8>, limit_bytes: usize,
89) -> Payload {
90    buf.clear();
91    let mut payload = Payload::default();
92    let mut line = Vec::new();
93    loop {
94        // `\n` is part of what a line costs.
95        let Some(budget) = limit_bytes.checked_sub(buf.len() + 1) else {
96            break;
97        };
98        let mut rendered = None;
99        for _ in 0..RENDER_PICKS {
100            let Some(context) = contexts.pick(rng, budget) else {
101                break;
102            };
103            line.clear();
104            if let Some(packed) = context.render_wellformed_within(rng, &mut line, budget) {
105                rendered = Some(packed);
106                break;
107            }
108        }
109        let Some(packed) = rendered else {
110            break;
111        };
112        buf.extend_from_slice(&line);
113        buf.push(b'\n');
114        payload.lines += 1;
115        payload.max_packed = payload.max_packed.max(packed);
116    }
117    payload
118}

120#[cfg(test)]
121mod test {
122    use proptest::prelude::*;
123    use rand::rngs::SmallRng;
124    use rand::SeedableRng;
125
126    use super::{write_payload, WorkingSet, PAYLOAD_BYTE_LIMIT};
127    use crate::contexts::{Context, Kind};
128    use crate::dogstatsd::is_malformed;
129
130    /// A working set of minted contexts across all three kinds.
131    fn pool(rng: &mut SmallRng, n: usize) -> WorkingSet {
132        WorkingSet::new(
133            (0..n)
134                .filter_map(|_| Context::mint_within(Kind::sample(rng), rng, PAYLOAD_BYTE_LIMIT))
135                .collect(),
136        )
137    }
138
139    /// Lines carry no interior newline and each is `\n`-terminated, so the line count equals the
140    /// newline count.
141    #[allow(clippy::naive_bytecount)]
142    fn newline_count(buf: &[u8]) -> usize {
143        buf.iter().filter(|&&b| b == b'\n').count()
144    }
145
146    proptest! {
147        /// A rendered metric carries exactly the tag set its identity holds. The Agent assigns the tag
148        /// set from every `#`-prefixed optional field it sees, last one winning, so a second such field
149        /// would swap the identity's tags for whatever an occurrence body happened to contain. That
150        /// would make one pooled identity render as several, putting cardinality past its cap and one
151        /// point in each of many series. Occurrence bodies do carry `|` and `#`, but segments are
152        /// always joined by a separator, so the two never land adjacent. This pins that.
153        #[test]
154        fn property_test_metric_carries_one_tag_field(seed: u64) {
155            let mut rng = SmallRng::seed_from_u64(seed);
156            let contexts = pool(&mut rng, 8);
157            let mut buf = Vec::new();
158            for _ in 0..4 {
159                write_payload(&mut rng, &contexts, &mut buf, PAYLOAD_BYTE_LIMIT);
160                for line in buf.split(|&b| b == b'\n') {
161                    if line.is_empty() || line.starts_with(b"_e{") || line.starts_with(b"_sc") {
162                        continue;
163                    }
164                    // Field 0 is `name:value` and field 1 the type. The Agent reads a tag set only from
165                    // the optional fields after those, so a `#` opening either of the first two is not
166                    // one.
167                    let tag_fields = line
168                        .split(|&b| b == b'|')
169                        .skip(2)
170                        .filter(|field| field.first() == Some(&b'#'))
171                        .count();
172                    prop_assert!(
173                        tag_fields <= 1,
174                        "{tag_fields} tag fields in {:?}",
175                        String::from_utf8_lossy(line)
176                    );
177                }
178            }
179        }
180
181        /// Every datagram the driver packs from pooled contexts is one the Agent forwards. A packed
182        /// datagram must be entirely well-formed.
183        #[test]
184        fn property_test_every_payload_is_well_formed(seed: u64) {
185            let mut rng = SmallRng::seed_from_u64(seed);
186            let contexts = pool(&mut rng, 8);
187            let mut buf = Vec::new();
188            for _ in 0..8 {
189                write_payload(&mut rng, &contexts, &mut buf, PAYLOAD_BYTE_LIMIT);
190                prop_assert_eq!(is_malformed(&buf), Ok(()), "emitted a droppable datagram: {:?}", String::from_utf8_lossy(&buf));
191            }
192        }
193
194        /// A budget that can hold some context's identity yields load. An empty datagram spends one of
195        /// the driver's configured sends without reaching the SUT, so the packer must fill the budget it
196        /// was given rather than give up on the first context that does not fit. The fallback line is
197        /// the floor below which a render that keeps landing on the drop side has nowhere to go.
198        #[test]
199        fn property_test_payload_is_never_empty_when_a_context_fits(seed: u64, limit_bytes in 1..8_192usize) {
200            let mut rng = SmallRng::seed_from_u64(seed);
201            let contexts = pool(&mut rng, 8);
202            let mut buf = Vec::new();
203            let payload = write_payload(&mut rng, &contexts, &mut buf, limit_bytes);
204
205            let smallest = contexts.smallest_floor().expect("pool is non-empty");
206            if limit_bytes > smallest {
207                prop_assert!(!buf.is_empty(), "empty datagram at limit {limit_bytes}, smallest floor {smallest}");
208                prop_assert!(payload.lines > 0);
209            }
210        }
211
212        #[test]
213        fn property_test_payload_stays_within_its_limit(seed: u64, limit_bytes: u16) {
214            let mut rng = SmallRng::seed_from_u64(seed);
215            let contexts = pool(&mut rng, 8);
216            let limit_bytes = usize::from(limit_bytes);
217            let mut buf = Vec::new();
218            let payload = write_payload(&mut rng, &contexts, &mut buf, limit_bytes);
219
220            prop_assert!(buf.len() <= limit_bytes);
221            prop_assert_eq!(newline_count(&buf), payload.lines);
222            if !buf.is_empty() {
223                prop_assert_eq!(buf[buf.len() - 1], b'\n');
224            }
225        }
226
227        /// An empty working set yields no load, not a panic.
228        #[test]
229        fn property_test_empty_pool_yields_empty_payload(seed: u64) {
230            let mut rng = SmallRng::seed_from_u64(seed);
231            let mut buf = Vec::new();
232            let payload = write_payload(&mut rng, &WorkingSet::new(Vec::new()), &mut buf, PAYLOAD_BYTE_LIMIT);
233            prop_assert_eq!(payload.lines, 0);
234            prop_assert!(buf.is_empty());
235        }
236    }
