1//! The context protocol: reusable `DogStatsD` identities a driver renders load against.
2//!
3//! A [`Context`] is a per-type stable identity — a metric's `(kind, name, tags)`, an event's title +
4//! tags + option fields, a service check's name + tags + host. It is minted over the free generator's
5//! full content alphabet (`payload/dogstatsd/common.rs`) and accepted only when a probe render is
6//! `!is_malformed` (see [`crate::dogstatsd`]) — conforming to the Agent parser and nothing stricter.
7//! The driver varies the per-occurrence payload (value/text/status, extensions, timestamp) each
8//! render, so a pooled identity recurs while its load varies.
9//!
10//! A shared intake pool mints identities up to a per-kind cap then recurs them, and serves them to
11//! drivers over the length-prefixed binary codec here ([`encode_response`] / [`decode_response`]),
12//! which carries non-UTF-8 names and tags that JSON could not.
13

22/// How many times to re-mint an identity whose probe render the Agent would drop before falling back
23/// to a guaranteed-sound identity. Mint is mostly-valid, so the fallback is rare.
24const REMINT_TRIES: usize = 16;
25
26/// How many times to re-render a context whose per-occurrence payload the Agent would drop before
27/// falling back to a guaranteed-well-formed line.
28const RENDER_TRIES: usize = 8;
29
30/// Digits allowed for a rendered length field, generous so a floor is never an underestimate.
31pub(crate) const LEN_DIGITS: usize = 5;
32
33/// Digits allowed for a rendered timestamp, likewise generous.
34pub(crate) const TS_DIGITS: usize = 20;

36/// The three context kinds.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum Kind {
39    /// A metric context.
40    Metric,
41    /// An event context.
42    Event,
43    /// A service-check context.
44    ServiceCheck,
45}
46
47impl Kind {
48    /// Sample a kind by the message-type weight — 98% metric, 1% event, 1% service check.
49    #[must_use]
50    pub fn sample(rng: &mut (impl Rng + ?Sized)) -> Kind {
51        match rng.random_range(0..100u32) {
52            0 => Kind::Event,
53            1 => Kind::ServiceCheck,
54            _ => Kind::Metric,
55        }
56    }
57}
58
59/// A reusable `DogStatsD` identity of one of the three kinds.
60#[derive(Clone, Debug, PartialEq, Eq, Hash)]
61pub enum Context {
62    /// A metric identity.
63    Metric(metric::MetricContext),
64    /// An event identity.
65    Event(event::EventContext),
66    /// A service-check identity.
67    ServiceCheck(service_check::ServiceCheckContext),
68}
69
70impl Context {
71    /// Mint a context of `kind` whose renders fit `budget` and that the Agent forwards, or `None` when
72    /// no such identity is available.
73    ///
74    /// The identity is built against the budget, so it fits by construction and nothing is minted then
75    /// measured. The remaining re-mint loop is about content alone: the alphabet carries protocol
76    /// delimiters and some combinations land on the drop side, which a probe render is what detects.
77    /// An exhausted loop yields `None` rather than an identity of another kind, so the caller never
78    /// stores a metric in the event or service-check working set.
79    #[must_use]
80    pub fn mint_within(kind: Kind, rng: &mut (impl Rng + ?Sized), budget: usize) -> Option<Context> {
81        for _ in 0..REMINT_TRIES {
82            let context = match kind {
83                Kind::Metric => Context::Metric(metric::MetricContext::mint_within(rng, budget)?),
84                Kind::Event => Context::Event(event::EventContext::mint_within(rng, budget)?),
85                Kind::ServiceCheck => {
86                    Context::ServiceCheck(service_check::ServiceCheckContext::mint_within(rng, budget)?)
87                }
88            };
89            let mut probe = Vec::new();
90            context.render(rng, &mut probe);
91            if is_malformed(&probe).is_ok() {
92                return Some(context);
93            }
94        }
95        None
96    }
97
98    /// Render one datagram line (no trailing `\n`) for this identity with a fresh per-occurrence
99    /// payload. Returns the packed multi-value run length, or zero.
100    pub fn render(&self, rng: &mut (impl Rng + ?Sized), out: &mut Vec<u8>) -> usize {
101        match self {
102            Context::Metric(c) => c.render(rng, out),
103            Context::Event(c) => c.render(rng, out),
104            Context::ServiceCheck(c) => c.render(rng, out),
105        }
106    }
107
108    /// Bytes every render of this identity must spend, whatever the per-occurrence payload.
109    #[must_use]
110    pub fn floor(&self) -> usize {
111        match self {
112            Context::Metric(c) => c.floor(),
113            Context::Event(c) => c.floor(),
114            Context::ServiceCheck(c) => c.floor(),
115        }
116    }
117
118    /// Render one line within `budget`, or `None` when the budget cannot hold this identity.
119    fn render_within(&self, rng: &mut (impl Rng + ?Sized), out: &mut Vec<u8>, budget: usize) -> Option<usize> {
120        match self {
121            Context::Metric(c) => c.render_within(rng, out, budget),
122            Context::Event(c) => c.render_within(rng, out, budget),
123            Context::ServiceCheck(c) => c.render_within(rng, out, budget),
124        }
125    }
126
127    /// Render one datagram line the Agent forwards within `budget`, or `None` when this context has no
128    /// forwardable rendering that fits. The budget is a construction input to each attempt rather than
129    /// a filter on the result, and an exhausted attempt count yields nothing rather than a line
130    /// carrying an identity the pool never issued.
131    pub fn render_wellformed_within(
132        &self, rng: &mut (impl Rng + ?Sized), out: &mut Vec<u8>, budget: usize,
133    ) -> Option<usize> {
134        for _ in 0..RENDER_TRIES {
135            let start = out.len();
136            let packed = self.render_within(rng, out, budget)?;
137            if is_malformed(&out[start..]).is_ok() {
138                return Some(packed);
139            }
140            out.truncate(start);
141        }
142        None
143    }
144
145    /// Append this context's tagged, length-prefixed encoding.
146    pub fn encode(&self, out: &mut Vec<u8>) {
147        match self {
148            Context::Metric(c) => {
149                put_u8(out, 0);
150                c.encode(out);
151            }
152            Context::Event(c) => {
153                put_u8(out, 1);
154                c.encode(out);
155            }
156            Context::ServiceCheck(c) => {
157                put_u8(out, 2);
158                c.encode(out);
159            }
160        }
161    }
162
163    /// Decode one context, advancing `*pos`. Returns `None` on truncation or an unknown tag.
164    fn decode(buf: &[u8], pos: &mut usize) -> Option<Context> {
165        Some(match get_u8(buf, pos)? {
166            0 => Context::Metric(metric::MetricContext::decode(buf, pos)?),
167            1 => Context::Event(event::EventContext::decode(buf, pos)?),
168            2 => Context::ServiceCheck(service_check::ServiceCheckContext::decode(buf, pos)?),
169            _ => return None,
170        })
171    }
172}

174/// Encode a `GET /contexts` response body: a `u32` count then each context.
175#[must_use]
176pub fn encode_response(contexts: &[Context]) -> Vec<u8> {
177    let mut out = Vec::new();
178    // A response holds the N contexts a driver asked for, far below u32::MAX.
179    let count = u32::try_from(contexts.len()).unwrap_or(u32::MAX);
180    out.extend_from_slice(&count.to_le_bytes());
181    for context in contexts {
182        context.encode(&mut out);
183    }
184    out
185}
186
187/// Decode a `GET /contexts` response body. Returns `None` on any truncation or malformed field, so a
188/// partial or corrupt body is an error, not a panic. Never pre-sizes from the wire count.
189#[must_use]
190pub fn decode_response(buf: &[u8]) -> Option<Vec<Context>> {
191    let mut pos = 0;
192    let count = get_u32(buf, &mut pos)?;
193    let mut contexts = Vec::new();
194    for _ in 0..count {
195        contexts.push(Context::decode(buf, &mut pos)?);
196    }
197    Some(contexts)
198}

200/// A fresh per-occurrence Unix timestamp for a `d:` field. Any positive integer forwards.
201pub(crate) fn fresh_timestamp(rng: &mut (impl Rng + ?Sized)) -> u64 {
202    rng.random_range(1..=2_000_000_000u64)
203}

205// --- shared length-prefixed binary codec ---
206
207/// Append one byte.
208pub(crate) fn put_u8(out: &mut Vec<u8>, b: u8) {
209    out.push(b);
210}
211
212/// Append `len` as a little-endian `u16`, saturating an over-long field. Minted fields are bounded
213/// well under `u16::MAX`, so saturation never fires in practice.
214fn put_u16(out: &mut Vec<u8>, len: usize) {
215    let len = u16::try_from(len).unwrap_or(u16::MAX);
216    out.extend_from_slice(&len.to_le_bytes());
217}
218
219/// Append a `u16` length prefix then the bytes.
220pub(crate) fn put_bytes(out: &mut Vec<u8>, bytes: &[u8]) {
221    put_u16(out, bytes.len());
222    out.extend_from_slice(bytes);
223}
224
225/// Append a `u16` count then each byte run.
226pub(crate) fn put_tags(out: &mut Vec<u8>, tags: &[Vec<u8>]) {
227    put_u16(out, tags.len());
228    for tag in tags {
229        put_bytes(out, tag);
230    }
231}

233/// Read one byte, advancing `*pos`.
234pub(crate) fn get_u8(buf: &[u8], pos: &mut usize) -> Option<u8> {
235    let byte = *buf.get(*pos)?;
236    *pos += 1;
237    Some(byte)
238}
239
240/// Read a little-endian `u16` as a `usize`, advancing `*pos`.
241fn get_u16(buf: &[u8], pos: &mut usize) -> Option<usize> {
242    let end = pos.checked_add(2)?;
243    let slice = buf.get(*pos..end)?;
244    *pos = end;
245    Some(u16::from_le_bytes([slice[0], slice[1]]) as usize)
246}
247
248/// Read a little-endian `u32` as a `usize`, advancing `*pos`.
249fn get_u32(buf: &[u8], pos: &mut usize) -> Option<usize> {
250    let end = pos.checked_add(4)?;
251    let slice = buf.get(*pos..end)?;
252    *pos = end;
253    Some(u32::from_le_bytes([slice[0], slice[1], slice[2], slice[3]]) as usize)
254}
255
256/// Read a `u16`-prefixed byte run, advancing `*pos`.
257pub(crate) fn get_bytes<'a>(buf: &'a [u8], pos: &mut usize) -> Option<&'a [u8]> {
258    let len = get_u16(buf, pos)?;
259    let end = pos.checked_add(len)?;
260    let slice = buf.get(*pos..end)?;
261    *pos = end;
262    Some(slice)
263}
264
265/// Read a `u16`-counted list of byte runs, advancing `*pos`. Does not pre-size from the wire count.
266pub(crate) fn get_tags(buf: &[u8], pos: &mut usize) -> Option<Vec<Vec<u8>>> {
267    let count = get_u16(buf, pos)?;
268    let mut tags = Vec::new();
269    for _ in 0..count {
270        tags.push(get_bytes(buf, pos)?.to_vec());
271    }
272    Some(tags)
273}

275#[cfg(test)]
276mod tests {
277    use proptest::prelude::*;
278    use rand::rngs::SmallRng;
279    use rand::SeedableRng;
280
281    use super::{decode_response, encode_response, Context, Kind};
282    use crate::dogstatsd::is_malformed;
283
284    fn any_kind() -> impl Strategy<Value = Kind> {
285        prop_oneof![Just(Kind::Metric), Just(Kind::Event), Just(Kind::ServiceCheck)]
286    }
287
288    proptest! {
289        /// A minted context conforms to is_malformed. Content carries delimiters, so a raw render may
290        /// land on the drop side. The repair loop is the sorter, and any line it does yield forwards.
291        #[test]
292        fn property_test_render_wellformed_always_forwards(seed: u64, kind in any_kind()) {
293            let mut rng = SmallRng::seed_from_u64(seed);
294            let Some(context) = Context::mint_within(kind, &mut rng, 8_192) else { return Ok(()) };
295            for _ in 0..8 {
296                let mut line = Vec::new();
297                if context.render_wellformed_within(&mut rng, &mut line, 8_192).is_some() {
298                    prop_assert_eq!(is_malformed(&line), Ok(()), "a rendered line was droppable");
299                }
300            }
301        }
302
303        /// A response of minted contexts round-trips through the binary codec, non-UTF-8 and all.
304        #[test]
305        fn property_test_response_round_trips(seed: u64) {
306            let mut rng = SmallRng::seed_from_u64(seed);
307            let contexts: Vec<Context> = (0..8)
308                .filter_map(|_| Context::mint_within(Kind::sample(&mut rng), &mut rng, 8_192))
309                .collect();
310            let wire = encode_response(&contexts);
311            let decoded = decode_response(&wire);
312            prop_assert_eq!(decoded.as_deref(), Some(contexts.as_slice()));
313        }
314
315        /// Decode never panics and rejects every truncated prefix of a valid body.
316        #[test]
317        fn property_test_decode_rejects_truncation(seed: u64) {
318            let mut rng = SmallRng::seed_from_u64(seed);
319            let contexts: Vec<Context> = (0..4)
320                .filter_map(|_| Context::mint_within(Kind::sample(&mut rng), &mut rng, 8_192))
321                .collect();
322            let wire = encode_response(&contexts);
323            for cut in 0..wire.len() {
324                let _ = decode_response(&wire[..cut]);
325            }
326            prop_assert_eq!(decode_response(&wire).map(|c| c.len()), Some(contexts.len()));
327        }
328    }
