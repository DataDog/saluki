22static COARSE_TIME_INITIALIZED: Once = Once::new();
23static COARSE_TIME: AtomicU64 = AtomicU64::new(0);
24const COARSE_TIME_UPDATE_INTERVAL: Duration = Duration::from_millis(250);

26/// Get the current Unix timestamp, in seconds.
27///
28/// This function is accurate, as it always retrieves the current time for each call. In scenarios where this function
29/// is being called frequently, it may pose an unacceptable performance overhead. In such cases, consider using
30/// `get_coarse_unix_timestamp`, which provides a cached value that's updated periodically.
31pub fn get_unix_timestamp() -> u64 {
32    let since_unix_epoch = SystemTime::now()
33        .duration_since(SystemTime::UNIX_EPOCH)
34        .unwrap_or_default();
35    since_unix_epoch.as_secs()
36}
37
38/// Get the current Unix timestamp, in nanoseconds.
39///
40/// Like `get_unix_timestamp`, this function always retrieves the current time for each call.
41pub fn get_unix_timestamp_nanos() -> u64 {
42    let since_unix_epoch = SystemTime::now()
43        .duration_since(SystemTime::UNIX_EPOCH)
44        .unwrap_or_default();
45    since_unix_epoch.as_nanos() as u64
46}

48/// Get the current coarse Unix timestamp, in seconds.
49///
50/// In scenarios where the current Unix timestamp is needed frequently, this function provides a cached value that's
51/// updated periodically. As the precision of the timestamp is one second, this function allows trading off accuracy
52/// (see below) for reduced overhead. The resulting value is considered "coarse," because it might be off by significant
53/// percentage of the overall precision.
54///
55/// # Accuracy
56///
57/// The underlying coarse time is updated roughly every 250 milliseconds, so the value returned by this function may be
58/// behind by up to 250 milliseconds. This means that if calling the function at true time `t` (where `t` is in
59/// seconds), the value returned may be `t-1` _or_ `t`, but will never be behind by more than 250 milliseconds, and
60/// never _ahead_ of `t`.
61pub fn get_coarse_unix_timestamp() -> u64 {
62    // Initialize a background thread to update the coarse time if it hasn't been initialized yet.
63    COARSE_TIME_INITIALIZED.call_once(|| {
64        // Initialize the coarse time with the current Unix timestamp.
65        COARSE_TIME.store(get_unix_timestamp(), Relaxed);
66
67        thread::spawn(|| {
68            loop {
69                // Sleep for 250 milliseconds.
70                thread::sleep(COARSE_TIME_UPDATE_INTERVAL);
71
72                // Update the coarse time with the current Unix timestamp.
73                COARSE_TIME.store(get_unix_timestamp(), Relaxed);
74            }
75        });
76    });
77
78    COARSE_TIME.load(Relaxed)
79}

81/// A point in time, with nanosecond precision.
82///
83/// Write it with [`Timestamp::format`], or parse it from RFC 3339 with [`str::parse`]. It can represent any time from
84/// the year -9999 to 9999-12-30T22:00:00Z.
85#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
86pub struct Timestamp(jiff::Timestamp);
87
88impl Timestamp {
89    /// The Unix epoch, 1970-01-01T00:00:00Z.
90    pub const UNIX_EPOCH: Self = Self(jiff::Timestamp::UNIX_EPOCH);
91
92    /// Gets the current time.
93    ///
94    /// # Errors
95    ///
96    /// If the system clock is outside of the range that `Timestamp` supports, an error is returned.
97    pub fn now() -> Result<Self, GenericError> {
98        jiff::Timestamp::try_from(SystemTime::now())
99            .map(Self)
100            .map_err(|e| generic_error!("System clock is outside of the supported range: {}", e))
101    }
102
103    /// Creates a `Timestamp` from a number of seconds since the Unix epoch, or `None` if it's out of range.
104    pub fn from_unix_seconds(seconds: i64) -> Option<Self> {
105        jiff::Timestamp::from_second(seconds).ok().map(Self)
106    }
107
108    /// Creates a `Timestamp` from a number of milliseconds since the Unix epoch, or `None` if it's out of range.
109    pub fn from_unix_millis(millis: i64) -> Option<Self> {
110        jiff::Timestamp::from_millisecond(millis).ok().map(Self)
111    }
112
113    /// Creates a `Timestamp` from a number of nanoseconds since the Unix epoch, or `None` if it's out of range.
114    pub fn from_unix_nanos(nanos: i128) -> Option<Self> {
115        jiff::Timestamp::from_nanosecond(nanos).ok().map(Self)
116    }
117
118    /// Returns the number of whole milliseconds since the Unix epoch, rounded down to the earlier millisecond.
119    pub fn as_unix_millis(self) -> i64 {
120        // jiff truncates toward zero, so a time before the epoch with a fractional millisecond needs one less.
121        let millis = self.0.as_millisecond();
122        if self.0.subsec_nanosecond() % 1_000_000 < 0 {
123            millis - 1
124        } else {
125            millis
126        }
127    }
128
129    /// Returns the number of nanoseconds since the Unix epoch.
130    pub fn as_unix_nanos(self) -> i128 {
131        self.0.as_nanosecond()
132    }
133
134    /// Returns the number of whole seconds from `earlier` to this timestamp, rounded toward zero.
135    ///
136    /// The result is negative if `earlier` is actually later, such as when the system clock went backwards.
137    pub fn whole_seconds_since(self, earlier: Self) -> i64 {
138        self.0.duration_since(earlier.0).as_secs()
139    }
140
141    /// Returns a value that writes this timestamp in the given format.
142    pub fn format(self, format: TimestampFormat<'_>) -> FormattedTimestamp<'_> {
143        FormattedTimestamp {
144            timestamp: self,
145            format,
146        }
147    }
148}

152
153    fn from_str(input: &str) -> Result<Self, Self::Err> {
154        input
155            .parse()
156            .map(Self)
157            .map_err(|e| generic_error!("Invalid RFC 3339 timestamp '{}': {}", input, e))
158    }
159}

161/// A time zone, for writing a [`Timestamp`] as local time.
162#[derive(Clone, Debug)]
163pub struct TimeZone(jiff::tz::TimeZone);
164
165impl TimeZone {
166    /// Gets UTC.
167    pub fn utc() -> &'static Self {
168        static UTC: TimeZone = TimeZone(jiff::tz::TimeZone::UTC);
169        &UTC
170    }
171
172    /// Gets the local time zone of the system.
173    ///
174    /// The time zone is detected on the first call and reused for the life of the process. If the `TZ` environment
175    /// variable is set, it alone decides the time zone: a value that can't be resolved gives UTC, without looking
176    /// anywhere else. Otherwise, the time zone comes from `/etc/localtime` on Unix or the system settings on Windows,
177    /// falling back to UTC if no time zone can be found.
178    ///
179    /// The first call may read the system's time zone database from disk, so make it during startup rather than on a
180    /// hot path.
181    pub fn system() -> &'static Self {
182        static SYSTEM: OnceLock<TimeZone> = OnceLock::new();
183        SYSTEM.get_or_init(|| Self(jiff::tz::TimeZone::try_system().unwrap_or(jiff::tz::TimeZone::UTC)))
184    }
185
186    /// Creates a time zone from a POSIX `TZ` rule, such as `EST5EDT,M3.2.0,M11.1.0`.
187    ///
188    /// # Errors
189    ///
190    /// If the rule isn't valid, an error is returned.
191    pub fn posix(rule: &str) -> Result<Self, GenericError> {
192        jiff::tz::TimeZone::posix(rule)
193            .map(Self)
194            .map_err(|e| generic_error!("Invalid POSIX time zone rule '{}': {}", rule, e))
195    }
196}

198/// A format for writing a [`Timestamp`].
199///
200/// The examples are all 2024-12-31T23:59:59.123456789Z, written in `America/New_York` where the format takes a time
201/// zone. Formats in whole seconds or milliseconds truncate the rest, toward the earlier time.
202#[derive(Clone, Copy, Debug)]
203pub enum TimestampFormat<'a> {
204    /// RFC 3339 with a numeric offset: `2024-12-31T18:59:59.123456789-05:00`.
205    ///
206    /// Fractional seconds use the fewest of 0, 3, 6, or 9 digits that represent the timestamp exactly, and UTC is
207    /// written as `+00:00` rather than `Z`.
208    ///
209    /// RFC 3339 can only write offsets in whole minutes and under 24 hours. An offset with seconds, such as `-01:00:30`
210    /// from `TZ=TST1:00:30`, is rounded to the nearest minute, and the local time shifts by up to 30 seconds to match,
211    /// so that the timestamp still names the same instant. The last example in section 5.8 of RFC 3339 does the same.
212    /// An offset that rounds to 24 hours or more is replaced by UTC.
213    Rfc3339(&'a TimeZone),
214
215    /// RFC 3339 in UTC, in whole seconds: `2024-12-31T23:59:59Z`.
216    Rfc3339UtcSeconds,
217
218    /// RFC 3339 in UTC, in milliseconds: `2024-12-31T23:59:59.123Z`.
219    Rfc3339UtcMillis,
220
221    /// Date and time, in whole seconds: `2024-12-31 18:59:59`.
222    DateTime(&'a TimeZone),
223
224    /// Date and time, in whole seconds, followed by the time zone's abbreviation: `2024-12-31 18:59:59 EST`.
225    DateTimeWithAbbreviation(&'a TimeZone),
226
227    /// Go's default layout for a time in UTC, in whole seconds: `2024-12-31 23:59:59 +0000 UTC`.
228    GoUtc,
229
230    /// Date and time without separators except between them, in whole seconds: `20241231-185959`.
231    CompactDateTime(&'a TimeZone),
232
233    /// Time of day, in whole seconds: `18:59:59`.
234    TimeOfDay(&'a TimeZone),
235
236    /// Date and time in UTC, without separators and with nine fractional digits: `20241231235959123456789`.
237    ///
238    /// Parse it with [`parse_compact_utc_nanos`].
239    CompactUtcNanos,
240}

242/// A [`Timestamp`] to be written in a [`TimestampFormat`], returned by [`Timestamp::format`].
243#[derive(Clone, Copy, Debug)]
244pub struct FormattedTimestamp<'a> {
245    timestamp: Timestamp,
246    format: TimestampFormat<'a>,
247}
248
249impl fmt::Display for FormattedTimestamp<'_> {
250    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
251        let timestamp = self.timestamp.0;
252        match self.format {
253            TimestampFormat::Rfc3339(tz) => write_rfc3339(f, timestamp, rfc3339_offset(tz.0.to_offset(timestamp))),
254            TimestampFormat::Rfc3339UtcSeconds => write_rfc3339_utc(f, timestamp, 0),
255            TimestampFormat::Rfc3339UtcMillis => write_rfc3339_utc(f, timestamp, 3),
256            TimestampFormat::DateTime(tz) => write_date_time(f, &tz.0.to_datetime(timestamp)),
257            TimestampFormat::DateTimeWithAbbreviation(tz) => {
258                let info = tz.0.to_offset_info(timestamp);
259                write_date_time(f, &info.offset().to_datetime(timestamp))?;
260                write!(f, " {}", info.abbreviation())
261            }
262            TimestampFormat::GoUtc => {
263                write_date_time(f, &Offset::UTC.to_datetime(timestamp))?;
264                f.write_str(" +0000 UTC")
265            }
266            TimestampFormat::CompactDateTime(tz) => {
267                let dt = tz.0.to_datetime(timestamp);
268                write_year(f, dt.year())?;
269                write!(
270                    f,
271                    "{:02}{:02}-{:02}{:02}{:02}",
272                    dt.month(),
273                    dt.day(),
274                    dt.hour(),
275                    dt.minute(),
276                    dt.second()
277                )
278            }
279            TimestampFormat::TimeOfDay(tz) => {
280                let dt = tz.0.to_datetime(timestamp);
281                write!(f, "{:02}:{:02}:{:02}", dt.hour(), dt.minute(), dt.second())
282            }
283            TimestampFormat::CompactUtcNanos => {
284                let dt = Offset::UTC.to_datetime(timestamp);
285                write_year(f, dt.year())?;
286                write!(
287                    f,
288                    "{:02}{:02}{:02}{:02}{:02}{:09}",
289                    dt.month(),
290                    dt.day(),
291                    dt.hour(),
292                    dt.minute(),
293                    dt.second(),
294                    dt.subsec_nanosecond()
295                )
296            }
297        }
298    }
299}

301/// Parses a timestamp written in [`TimestampFormat::CompactUtcNanos`] into nanoseconds since the Unix epoch.
302///
303/// Returns `None` unless the input is exactly 23 digits that make a valid date and time. Unlike [`Timestamp`], this
304/// works up to the end of the year 9999.
305pub fn parse_compact_utc_nanos(input: &str) -> Option<i128> {
306    let digits = input.as_bytes();
307    if digits.len() != 23 || !digits.iter().all(u8::is_ascii_digit) {
308        return None;
309    }
310
311    let field = |range: Range<usize>| {
312        digits[range]
313            .iter()
314            .fold(0i32, |value, digit| value * 10 + i32::from(digit - b'0'))
315    };
316    let datetime = jiff::civil::DateTime::new(
317        i16::try_from(field(0..4)).ok()?,
318        i8::try_from(field(4..6)).ok()?,
319        i8::try_from(field(6..8)).ok()?,
320        i8::try_from(field(8..10)).ok()?,
321        i8::try_from(field(10..12)).ok()?,
322        i8::try_from(field(12..14)).ok()?,
323        field(14..23),
324    )
325    .ok()?;
326
327    // Subtracting civil datetimes, rather than converting to a `Timestamp`, works up to the end of the year 9999.
328    let since_epoch = datetime.duration_since(jiff::civil::DateTime::constant(1970, 1, 1, 0, 0, 0, 0));
329    Some(since_epoch.as_nanos())
330}

332/// Returns the offset to write in RFC 3339 for `offset`: rounded to the nearest minute, or UTC if that's 24 hours or
333/// more.
334fn rfc3339_offset(offset: Offset) -> Offset {
335    // Rounding only fails when the result is outside of jiff's offset range, which is also 24 hours or more.
336    match offset.round(Unit::Minute) {
337        Ok(rounded) if rounded.seconds().abs() < 24 * 60 * 60 => rounded,
338        _ => Offset::UTC,
339    }
340}
341
342fn write_rfc3339(f: &mut fmt::Formatter<'_>, timestamp: jiff::Timestamp, offset: Offset) -> fmt::Result {
343    let nanos = offset.to_datetime(timestamp).subsec_nanosecond();
344    let precision = if nanos == 0 {
345        0
346    } else if nanos % 1_000_000 == 0 {
347        3
348    } else if nanos % 1_000 == 0 {
349        6
350    } else {
351        9
352    };
353
354    DateTimePrinter::new()
355        .precision(Some(precision))
356        .print_timestamp_with_offset(&timestamp, offset, StdFmtWrite(f))
357        .map_err(|_| fmt::Error)
358}
359
360fn write_rfc3339_utc(f: &mut fmt::Formatter<'_>, timestamp: jiff::Timestamp, precision: u8) -> fmt::Result {
361    DateTimePrinter::new()
362        .precision(Some(precision))
363        .print_timestamp(&timestamp, StdFmtWrite(f))
364        .map_err(|_| fmt::Error)
365}
366
367fn write_date_time(f: &mut fmt::Formatter<'_>, dt: &jiff::civil::DateTime) -> fmt::Result {
368    write_year(f, dt.year())?;
369    write!(
370        f,
371        "-{:02}-{:02} {:02}:{:02}:{:02}",
372        dt.month(),
373        dt.day(),
374        dt.hour(),
375        dt.minute(),
376        dt.second()
377    )
378}

380/// Writes a year with at least four digits, and a sign only when it's negative.
381fn write_year(f: &mut fmt::Formatter<'_>, year: i16) -> fmt::Result {
382    if year < 0 {
383        f.write_char('-')?;
384    }
385    write!(f, "{:04}", year.unsigned_abs())
386}

731    fn coarse_timestamp_never_exceeds_accurate_timestamp() {
732        // The documented contract is that the coarse timestamp is never _ahead_ of the true time:
733        // it is only ever equal to it or lagging behind it, because it is a cached snapshot of a
734        // previous `get_unix_timestamp()` call. Sample repeatedly to make an accidental "ahead"
735        // read unlikely to slip through.
736        for _ in 0..1_000 {
737            let coarse = get_coarse_unix_timestamp();
738            let accurate = get_unix_timestamp();
739            assert!(
740                coarse <= accurate,
741                "coarse timestamp {coarse} must never be ahead of accurate timestamp {accurate}"
742            );
743        }
744    }

746    #[test]
747    fn coarse_timestamp_is_monotonically_non_decreasing() {
748        // The coarse timestamp is only ever replaced with a newer `get_unix_timestamp()` value, so
749        // successive reads must never go backwards, including across an update-interval boundary.
750        let first = get_coarse_unix_timestamp();
751
752        let mut previous = first;
753        for _ in 0..1_000 {
754            let current = get_coarse_unix_timestamp();
755            assert!(
756                current >= previous,
757                "coarse timestamp went backwards: {previous} -> {current}"
758            );
759            previous = current;
760        }
761
762        // Sleep past a couple of update intervals so the background updater runs, then confirm the
763        // value has not regressed.
764        sleep(COARSE_TIME_UPDATE_INTERVAL * 3);
765        let after_sleep = get_coarse_unix_timestamp();
766        assert!(
767            after_sleep >= previous,
768            "coarse timestamp regressed after update: {previous} -> {after_sleep}"
769        );
770    }

772    #[test]
773    fn coarse_timestamp_tracks_accurate_within_staleness_bound() {
774        // The documented staleness bound is that the coarse value may lag true time by at most
775        // ~250ms, i.e. in whole seconds it is either `t` or `t-1`. Once the background updater is
776        // running, the difference against the accurate timestamp should therefore be no more than
777        // one second. Retry across a short window to absorb one-off scheduling jitter on busy CI.
778        get_coarse_unix_timestamp();
779        sleep(COARSE_TIME_UPDATE_INTERVAL * 2);
780
781        let mut within_bound = false;
782        for _ in 0..20 {
783            let accurate = get_unix_timestamp();
784            let coarse = get_coarse_unix_timestamp();
785            if accurate - coarse <= 1 {
786                within_bound = true;
787                break;
788            }
789            sleep(Duration::from_millis(100));
790        }
791
792        assert!(
793            within_bound,
794            "coarse timestamp never came within one second of the accurate timestamp"
795        );
796    }
