24/// A DogStatsD metric packet.
25///
26/// See the [DogStatsD datagram format][datagram] reference for the wire format and the protocol versions that
27/// introduced each field.
28///
29/// [datagram]: https://docs.datadoghq.com/extend/dogstatsd/datagram_shell/?tab=metrics
30pub struct MetricPacket<'a> {
31    /// Name of the metric.
32    pub metric_name: &'a str,
33
34    /// Tags attached to the metric.
35    pub tags: RawTags<'a>,
36
37    /// The metric kind (counter, gauge, rate, etc.) and its sample points.
38    pub values: MetricValues,
39
40    /// Number of sample points represented by `values`.
41    pub num_points: u64,
42
43    /// Optional Unix timestamp for the sample, in seconds (protocol v1.3).
44    pub timestamp: Option<u64>,
45
46    /// Local Data attached to the metric, carried in the `c:` field (protocol v1.2, extended in v1.4).
47    ///
48    /// Identifies the workload that emitted the metric. Carries a container ID (`ci-<id>`) or, when unavailable, a
49    /// cgroup node inode (`in-<inode>`).
50    pub local_data: Option<&'a str>,
51
52    /// External Data attached to the metric, carried in the `e:` field (protocol v1.5).
53    ///
54    /// Used to convey a richer blob of workload identity data resolved by the receiver.
55    pub external_data: Option<&'a str>,
56
57    /// Cardinality hint for origin tag enrichment, carried in the `card:` field (protocol v1.6).
58    ///
59    /// Specifies which origin tags the receiver should attach to the metric.
60    pub cardinality: Option<OriginTagCardinality>,
61
62    /// Unit for this metric, if any.
63    pub unit: Option<&'static str>,
64}

66#[inline]
67pub fn parse_dogstatsd_metric<'a>(
68    input: &'a [u8], config: &DogStatsDCodecConfiguration,
69) -> IResult<&'a [u8], MetricPacket<'a>> {
70    // We always parse the metric name and value(s) first, where value is both the kind (counter, gauge, etc) and the
71    // actual value itself.
72    let metric_name_parser = if config.permissive {
73        permissive_metric_name
74    } else {
75        ascii_alphanum_and_seps
76    };
77    let (remaining, (metric_name, (metric_type, raw_metric_values))) =
78        separated_pair(metric_name_parser, tag(":"), raw_metric_values).parse(input)?;
79
80    // At this point, we may have some of this additional data, and if so, we also then would have a pipe separator at
81    // the very front, which we'd want to consume before going further.
82    //
83    // After that, we simply split the remaining bytes by the pipe separator, and then try and parse each chunk to see
84    // if it's any of the protocol extensions we know of.
85    let mut maybe_sample_rate = None;
86    let mut maybe_tags = None;
87    let mut maybe_local_data = None;
88    let mut maybe_timestamp = None;
89    let mut maybe_external_data = None;
90    let mut maybe_cardinality = None;
91
92    let remaining = if !remaining.is_empty() {
93        let (mut remaining, _) = tag("|")(remaining)?;
94
95        while let Some((chunk, tail)) = split_at_delimiter(remaining, b'|') {
96            // An empty field (for example, `metric:1|c||#tags`) is a structural defect. In permissive mode, we skip it
97            // and keep parsing the fields after it, matching the reference DogStatsD server. Otherwise, we reject the
98            // metric.
99            if chunk.is_empty() {
100                if !config.permissive {
101                    return Err(nom::Err::Error(Error::new(remaining, ErrorKind::Verify)));
102                }
103                remaining = tail;
104                continue;
105            }
106
107            match chunk[0] {
108                // Sample rate: indicates client-side sampling of this metric which will need to be "reinflated" at some
109                // point downstream to calculate the true metric value.
110                b'@' => {
111                    let (_, raw_sample_rate) = all_consuming(preceded(tag("@"), double)).parse(chunk)?;
112
113                    // The reference DogStatsD server treats a sample rate of zero on a distribution, histogram or timer
114                    // as unsampled. Clamping it to the minimum sample rate instead would count each point hundreds of
115                    // millions of times.
116                    maybe_sample_rate = if raw_sample_rate == 0.0
117                        && matches!(
118                            metric_type,
119                            MetricType::Distribution | MetricType::Histogram | MetricType::Timer
120                        ) {
121                        None
122                    } else {
123                        let sample_rate = sample_rate(metric_name, config)(raw_sample_rate)
124                            .map_err(|_| nom::Err::Error(Error::new(chunk, ErrorKind::MapRes)))?;
125                        Some(sample_rate)
126                    };
127                }
128                // Tags: additional tags to be added to the metric.
129                b'#' => {
130                    let (_, tags) = all_consuming(preceded(tag("#"), tags(config))).parse(chunk)?;
131                    maybe_tags = Some(tags);
132                }
133                // Local Data: client-provided data used for resolving the entity ID that this metric originated from.
134                //
135                // In permissive mode, a malformed value is ignored rather than rejecting the whole metric. The
136                // reference DogStatsD server doesn't validate this field and just fails to resolve an origin from it,
137                // so either way, the metric gets no origin tags from it.
138                b'c' if chunk.len() > 1 && chunk[1] == b':' && config.client_origin_detection => {
139                    match all_consuming(preceded(tag("c:"), local_data)).parse(chunk) {
140                        Ok((_, local_data)) => maybe_local_data = Some(local_data),
141                        Err(_) if config.permissive => {
142                            debug!(metric_name, "Ignoring malformed Local Data field in DogStatsD metric.")
143                        }
144                        Err(e) => return Err(e),
145                    }
146                }
147                // Timestamp: client-provided timestamp for the metric, relative to the Unix epoch, in seconds.
148                //
149                // Like the reference DogStatsD server, a timestamp of zero is rejected: it isn't a usable point in
150                // time, and treating it as "timestamped" would drop the sample rate without actually setting a
151                // timestamp.
152                b'T' if config.timestamps => {
153                    let (_, timestamp) = all_consuming(preceded(tag("T"), unix_timestamp)).parse(chunk)?;
154                    if timestamp == 0 {
155                        return Err(nom::Err::Error(Error::new(chunk, ErrorKind::Verify)));
156                    }
157                    maybe_timestamp = Some(timestamp);
158                }
159                // External Data: client-provided data used for resolving the entity ID that this metric originated from.
160                //
161                // In permissive mode, a malformed value is ignored rather than rejecting the whole metric. The
162                // reference DogStatsD server doesn't validate this field and just fails to resolve an origin from it,
163                // so either way, the metric gets no origin tags from it.
164                b'e' if chunk.len() > 1 && chunk[1] == b':' && config.client_origin_detection => {
165                    match all_consuming(preceded(tag("e:"), external_data)).parse(chunk) {
166                        Ok((_, external_data)) => maybe_external_data = Some(external_data),
167                        Err(_) if config.permissive => {
168                            debug!(
169                                metric_name,
170                                "Ignoring malformed External Data field in DogStatsD metric."
171                            )
172                        }
173                        Err(e) => return Err(e),
174                    }
175                }
176                // Cardinality: client-provided cardinality for the metric.
177                b'c' if chunk.starts_with(CARDINALITY_PREFIX) && config.client_origin_detection => {
178                    let (_, cardinality) = cardinality(chunk)?;
179                    maybe_cardinality = cardinality;
180                }
181                _ => {
182                    // We don't know what this is, so we just skip it.
183                    //
184                    // TODO: Should we throw an error, warn, or be silently permissive?
185                }
186            }
187
188            remaining = tail;
189        }
190
191        // TODO: Similarly to the above comment, should having any remaining data here cause us to throw an error, warn,
192        // or be silently permissive?
193
194        remaining
195    } else {
196        remaining
197    };
198
199    // Capture the unit from the metric type before it is erased into a MetricValues variant.
200    // Only timing metrics carry an implicit unit; all other types have no unit.
201    let maybe_unit = if matches!(metric_type, MetricType::Timer) {
202        Some("millisecond")
203    } else {
204        None
205    };
206
207    let effective_sample_rate = if matches!(&metric_type, MetricType::Count) && maybe_timestamp.is_some() {
208        // Match the Datadog Agent no-aggregation pipeline: timestamped DogStatsD counts are forwarded as
209        // pre-aggregated points, so their sample rate is not used to reinflate the count value.
210        None
211    } else {
212        maybe_sample_rate
213    };
214
215    let (num_points, mut metric_values) =
216        metric_values_from_raw(raw_metric_values, metric_type, effective_sample_rate, config.permissive)?;
217
218    // If we got a timestamp, apply it to all metric values.
219    if let Some(timestamp) = maybe_timestamp {
220        metric_values.set_timestamp(timestamp);
221    }
222
223    let tags = maybe_tags.unwrap_or_else(RawTags::empty);
224
225    Ok((
226        remaining,
227        MetricPacket {
228            metric_name,
229            tags,
230            values: metric_values,
231            num_points,
232            timestamp: maybe_timestamp,
233            local_data: maybe_local_data,
234            external_data: maybe_external_data,
235            cardinality: maybe_cardinality,
236            unit: maybe_unit,
237        },
238    ))
239}

241#[inline]
242fn permissive_metric_name(input: &[u8]) -> IResult<&[u8], &str> {
243    // Like the reference DogStatsD server, we allow any characters in the metric name other than the `:` value
244    // delimiter and the `|` field delimiter. The name must still be valid UTF-8, since we can't safely handle it
245    // otherwise.
246    let (remaining, raw_name) = take_while1(|c: u8| c != b':' && c != b'|').parse(input)?;
247    match simdutf8::basic::from_utf8(raw_name) {
248        Ok(name) => Ok((remaining, name)),
249        Err(_) => Err(nom::Err::Error(Error::new(raw_name, ErrorKind::Verify))),
250    }
251}

253#[inline]
254fn sample_rate<'a>(
255    metric_name: &'a str, config: &'a DogStatsDCodecConfiguration,
256) -> impl Fn(f64) -> Result<SampleRate, &'static str> + 'a {
257    let minimum_sample_rate = config.minimum_sample_rate;
258
259    move |mut raw_sample_rate| {
260        if raw_sample_rate < minimum_sample_rate {
261            raw_sample_rate = minimum_sample_rate;
262            warn!(
263                "Sample rate for metric '{}' is below minimum of {}. Clamping to minimum.",
264                metric_name, minimum_sample_rate
265            );
266        }
267        SampleRate::try_from(raw_sample_rate)
268    }
269}

271#[inline]
272fn raw_metric_values(input: &[u8]) -> IResult<&[u8], (MetricType, &[u8])> {
273    let (remaining, raw_values) = terminated(take_while1(|b| b != b'|'), tag("|")).parse(input)?;
274    let (remaining, raw_kind) = alt((tag("g"), tag("c"), tag("ms"), tag("h"), tag("s"), tag("d"))).parse(remaining)?;
275
276    let metric_type = match raw_kind {
277        b"c" => MetricType::Count,
278        b"g" => MetricType::Gauge,
279        b"s" => MetricType::Set,
280        b"ms" => MetricType::Timer,
281        b"h" => MetricType::Histogram,
282        b"d" => MetricType::Distribution,
283        _ => unreachable!("should be constrained by alt parser"),
284    };
285
286    // Make sure the raw value(s) are valid UTF-8 before we use them later on.
287    if simdutf8::basic::from_utf8(raw_values).is_err() {
288        return Err(nom::Err::Error(Error::new(raw_values, ErrorKind::Verify)));
289    }
290
291    // Make sure a numeric metric has at least one value. In permissive mode, empty values in a multi-value payload (for
292    // example, `1::2`) are skipped when parsing, but a payload made up of only empty values (for example, `:` or `::`)
293    // has no values at all, which the reference DogStatsD server rejects.
294    //
295    // A set's value isn't split on `:`, so `::` is a valid set member.
296    if !matches!(metric_type, MetricType::Set) && raw_values.iter().all(|&b| b == b':') {
297        return Err(nom::Err::Error(Error::new(raw_values, ErrorKind::Verify)));
298    }
299
300    Ok((remaining, (metric_type, raw_values)))
301}

303#[inline]
304fn metric_values_from_raw(
305    input: &[u8], metric_type: MetricType, sample_rate: Option<SampleRate>, skip_empty_values: bool,
306) -> Result<(u64, MetricValues), NomParserError<'_>> {
307    let mut num_points = 0;
308    let floats = FloatIter::new(input, skip_empty_values).inspect(|_| num_points += 1);
309
310    let values = match metric_type {
311        MetricType::Count => MetricValues::counter_sampled_fallible(floats, sample_rate)?,
312        MetricType::Gauge => MetricValues::gauge_fallible(floats)?,
313        MetricType::Set => {
314            num_points = 1;
315
316            // SAFETY: We've already checked above that `input` is valid UTF-8.
317            let value = unsafe { std::str::from_utf8_unchecked(input) };
318            MetricValues::set(value.to_string())
319        }
320        MetricType::Timer | MetricType::Histogram => MetricValues::histogram_sampled_fallible(floats, sample_rate)?,
321        MetricType::Distribution => MetricValues::distribution_sampled_fallible(floats, sample_rate)?,
322    };
323
324    Ok((num_points, values))
325}

327struct FloatIter<'a> {
328    raw_values: &'a [u8],
329    skip_empty_values: bool,
330}
331
332impl<'a> FloatIter<'a> {
333    fn new(raw_values: &'a [u8], skip_empty_values: bool) -> Self {
334        Self {
335            raw_values,
336            skip_empty_values,
337        }
338    }
339}
340
341impl<'a> Iterator for FloatIter<'a> {
342    type Item = Result<f64, NomParserError<'a>>;
343
344    fn next(&mut self) -> Option<Self::Item> {
345        loop {
346            if self.raw_values.is_empty() {
347                return None;
348            }
349
350            let (raw_value, tail) = split_at_delimiter(self.raw_values, b':')?;
351            self.raw_values = tail;
352
353            // An empty value in a multi-value payload (for example, `1::2`) is a structural defect. In permissive mode,
354            // we skip it, matching the reference DogStatsD server. Otherwise, it fails to parse as a float below,
355            // rejecting the metric.
356            if raw_value.is_empty() && self.skip_empty_values {
357                continue;
358            }
359
360            // SAFETY: The caller that creates `ValueIter` is responsible for ensuring that the entire byte slice is valid
361            // UTF-8.
362            let value_s = unsafe { std::str::from_utf8_unchecked(raw_value) };
363            match value_s.parse::<f64>() {
364                Ok(value) if value.is_finite() => return Some(Ok(value)),
365                Ok(_) => {
366                    debug!(value = value_s, "Dropping non-finite DogStatsD metric value.");
367                }
368                Err(_) => return Some(Err(nom::Err::Error(Error::new(raw_value, ErrorKind::Float)))),
369            }
370        }
371    }
372}

566    #[test]
567    fn set_values_made_of_colons_are_accepted() {
568        // A set's value isn't split on `:`, so a value such as `::` (the IPv6 unspecified address) is a valid set
