8/// A standard HTTP retry lifecycle that emits contextual information about HTTP requests and responses..
9///
10/// This lifecycle emits user-friendly logs about retry attempts, including the request URI and response code. It
11/// provides additional destructuring/introspection of errors to surface contextual information such as requests failing
12/// due to DNS, connection errors, TLS, and so on.
13#[derive(Clone)]
14pub struct StandardHttpRetryLifecycle;
15
16impl<B, B2, E> RetryLifecycle<http::Request<B>, http::Response<B2>, E> for StandardHttpRetryLifecycle
17where
18    E: DynError,
19{
20    fn before_retry(
21        &self, req: &http::Request<B>, res: &Result<http::Response<B2>, E>, retry_backoff: Duration, error_count: u32,
22    ) {
23        let request_uri = SanitizedRequestUri(req.uri());
24        let categorized_error = CategorizedError::try_categorize(res);
25
26        // The HTTP/2 fields are only present when the failure was an HTTP/2 error: `None` values are not recorded.
27        let http2_details = categorized_error.http2_details();
28
29        warn!(
30            error_count,
31            %request_uri,
32            http2.error_kind = http2_details.map(|details| details.kind.as_str()),
33            http2.reason_code = http2_details.and_then(|details| details.reason).map(u32::from),
34            http2.received_from_remote = http2_details.map(|details| details.received_from_remote),
35            "{}. Retrying after {:?}.", categorized_error, retry_backoff
36        );
37    }
38
39    fn after_success(&self, req: &http::Request<B>, _: &Result<http::Response<B2>, E>) {
40        let request_uri = SanitizedRequestUri(req.uri());
41        debug!(%request_uri, "Request succeeded.");
42    }
43}

45struct SanitizedRequestUri<'a>(&'a http::Uri);

47impl fmt::Display for SanitizedRequestUri<'_> {
48    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
49        // We _should_ always have a scheme and a host, but we'll just make sure they exist first, to be safe. We'll
50        // require needing both to be present to print either of them.
51        let maybe_scheme = self.0.scheme_str();
52        let maybe_host = self.0.host();
53
54        if let (Some(scheme), Some(host)) = (maybe_scheme, maybe_host) {
55            write!(f, "{}://{}", scheme, host)?;
56        }
57
58        let maybe_port = self.0.port_u16();
59        if let Some(port) = maybe_port {
60            write!(f, ":{}", port)?;
61        }
62
63        // Now print the request path, which is always present.
64        write!(f, "{}", self.0.path())
65    }
66}

68/// Maximum number of errors visited when walking a source chain.
69///
70/// Chains are short in practice, so the limit only exists to bound the walk if an error ever reports itself, directly or
71/// indirectly, as its own source.
72const MAX_SOURCE_CHAIN_DEPTH: usize = 32;
73
74enum CategorizedError {
75    Client(String),
76    Tls(String),
77    Http2(Http2ErrorDetails),
78    Http(StatusCode),
79    Other(String),
80}
81
82impl CategorizedError {
83    fn try_categorize<B, E>(res: &Result<http::Response<B>, E>) -> Self
84    where
85        E: DynError,
86    {
87        match res {
88            Ok(resp) => Self::Http(resp.status()),
89            Err(e) => Self::extract_nested(e.as_dyn_error()),
90        }
91    }
92
93    fn extract_nested(error: &(dyn std::error::Error + 'static)) -> Self {
94        // Walk the source chain looking for an error we can say something specific about. Wrappers along the way, such
95        // as `hyper-util`'s client error, get no handling of their own: we only care about what they wrap.
96        let mut current = error;
97        for _ in 0..MAX_SOURCE_CHAIN_DEPTH {
98            if let Some(rustls_error) = current.downcast_ref::<rustls::Error>() {
99                return Self::from_rustls(rustls_error);
100            }
101
102            if let Some(http2_error) = current.downcast_ref::<h2::Error>() {
103                return Self::Http2(Http2ErrorDetails::from_http2(http2_error));
104            }
105
106            match next_source(current) {
107                Some(source) => current = source,
108                None => break,
109            }
110        }
111
112        // Nothing in the chain was recognized, so we report the deepest error we reached, since that's the one closest
113        // to the actual failure.
114        if let Some(client_error) = current.downcast_ref::<hyper_util::client::legacy::Error>() {
115            return Self::Client(client_error.to_string());
116        }
117
118        Self::Other(current.to_string())
119    }
120
121    fn http2_details(&self) -> Option<&Http2ErrorDetails> {
122        match self {
123            Self::Http2(details) => Some(details),
124            _ => None,
125        }
126    }
127
128    fn from_rustls(error: &rustls::Error) -> Self {
129        // We're really just specializing a few known types of errors to generate a better error message, but otherwise
130        // we'll fallback on the description given by the error itself.
131        let reason = match error {
132            rustls::Error::InvalidCertificate(cert_error) => format!(
133                "peer certificate is invalid: {}",
134                rustls_cert_error_to_string(cert_error)
135            ),
136            _ => error.to_string(),
137        };
138
139        Self::Tls(reason)
140    }
141}
142
143impl fmt::Display for CategorizedError {
144    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
145        match self {
146            CategorizedError::Client(reason) => write!(f, "Request failed due to a client error: {}", reason),
147            CategorizedError::Tls(reason) => write!(f, "Request failed due to a TLS error: {}", reason),
148            CategorizedError::Http2(details) => write!(f, "Request failed due to {}", details),
149            CategorizedError::Http(status_code) => write!(
150                f,
151                "Server responded with non-success status code {}.",
152                status_code.as_str()
153            ),
154            CategorizedError::Other(reason) => write!(f, "Request failed: {}", reason),
155        }
156    }
157}

159/// Returns the next error in a source chain, if any.
160fn next_source<'a>(error: &'a (dyn std::error::Error + 'static)) -> Option<&'a (dyn std::error::Error + 'static)> {
161    // `std::io::Error::source` skips the error that the I/O error itself wraps, and returns that error's source instead,
162    // so we have to ask for the wrapped error directly.
163    if let Some(io_error) = error.downcast_ref::<std::io::Error>() {
164        if let Some(inner) = io_error.get_ref() {
165            return Some(inner);
166        }
167    }
168
169    error.source()
170}

175    Reset,

189/// The details of an `h2::Error` that we report.
190///
191/// We snapshot a small set of fields instead of holding the error, so that we never log its `Debug` output or the debug
192/// data carried by a GOAWAY frame.
193struct Http2ErrorDetails {
194    kind: Http2ErrorKind,
195    reason: Option<h2::Reason>,
196    received_from_remote: bool,
197
198    /// The error's own text, used only when there is no reason code to describe the failure.
199    fallback: Option<String>,
200}
201
202impl Http2ErrorDetails {
203    fn from_http2(error: &h2::Error) -> Self {
204        let kind = if error.is_go_away() {
205            Http2ErrorKind::GoAway
206        } else if error.is_reset() {
207            Http2ErrorKind::Reset
208        } else {
209            Http2ErrorKind::Other
210        };
211
212        // The error's text includes GOAWAY debug data, so we only take it when we have nothing better.
213        let reason = error.reason();
214        let fallback = reason.is_none().then(|| error.to_string());
215
216        Self {
217            kind,
218            reason,
219            received_from_remote: error.is_remote(),
220            fallback,
221        }
222    }
223}
224
225impl fmt::Display for Http2ErrorDetails {
226    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
227        let description = match self.kind {
228            Http2ErrorKind::GoAway => "an HTTP/2 GOAWAY",
229            Http2ErrorKind::Reset => "an HTTP/2 stream reset",
230            Http2ErrorKind::Other => "an HTTP/2 error",
231        };
232
233        write!(f, "{}", description)?;
234
235        if self.received_from_remote {
236            write!(f, " received from the remote peer")?;
237        }
238
239        match self.reason {
240            Some(reason) => write!(f, " (reason: {}, code {})", reason.description(), u32::from(reason)),
241            None => match &self.fallback {
242                Some(fallback) => write!(f, ": {}", fallback),
243                None => Ok(()),
244            },
245        }
246    }
247}

249fn rustls_cert_error_to_string(cert_error: &rustls::CertificateError) -> Cow<'static, str> {
250    match cert_error {
251        rustls::CertificateError::BadEncoding => "certificate incorrectly encoded".into(),
252        rustls::CertificateError::Expired => "certificate expired (current time is after notAfter time)".into(),
253        rustls::CertificateError::NotValidYet => {
254            "certificate not valid yet (current time is before notBefore time)".into()
255        }
256        rustls::CertificateError::Revoked => "certificate has been revoked".into(),
257        rustls::CertificateError::UnhandledCriticalExtension => {
258            "certificate contains an extension marked critical, but it was not processed by the certificate validator"
259                .into()
260        }
261        rustls::CertificateError::UnknownIssuer => "certificate chain is not issued by a known root certificate".into(),
262        rustls::CertificateError::UnknownRevocationStatus => {
263            "certificate's revocation status could not be determined".into()
264        }
265        rustls::CertificateError::ExpiredRevocationList => {
266            "certificate's revocation status could not be determined due to an expired CRL".into()
267        }
268        rustls::CertificateError::BadSignature => {
269            "certificate is not signed correctly by the key of its alleged issuer".into()
270        }
271        rustls::CertificateError::NotValidForName => "certificate is not valid for the given entity name".into(),
272        rustls::CertificateError::InvalidPurpose => "certificate is not valid for the requested purpose".into(),
273        rustls::CertificateError::ApplicationVerificationFailure => {
274            "certificate is valid overall, but the handshake was rejected".into()
275        }
276
277        // This one could be a generic error that doesn't fit the above, returned by `rustls`, or it could be coming from
278        // a custom certificate verifier which we don't know about, or can't reasonably know about to compensate for
279        // here... so we'll just return it as-is.
280        rustls::CertificateError::Other(other) => format!("generic error: {}", other).into(),
281        other => format!("generic unhandled error: {:?}", other).into(),
282    }
283}

344    /// Runs a request over an in-memory HTTP/2 connection and returns the error the client sees.
345    ///
346    /// The server side completes the handshake, waits until the client has sent its request, and then runs `reject`,
347    /// which is expected to fail either the stream or the connection. `reject` hands the connection back, since the
348    /// server has to keep polling it to flush what `reject` queued, and has to hold it open so that closing the socket
349    /// doesn't race the client's read.
350    async fn failed_http2_request<F, Fut>(reject: F) -> h2::Error
351    where
352        F: FnOnce(h2::server::Connection<tokio::io::DuplexStream, Bytes>) -> Fut + Send + 'static,
353        Fut: Future<Output = h2::server::Connection<tokio::io::DuplexStream, Bytes>> + Send,
354    {
355        let (client_io, server_io) = tokio::io::duplex(4096);
356        let (request_sent_tx, request_sent_rx) = tokio::sync::oneshot::channel();
357
358        tokio::spawn(async move {
359            let connection = h2::server::handshake(server_io).await.unwrap();
360            request_sent_rx.await.unwrap();
361
362            let mut connection = reject(connection).await;
363            let _ = std::future::poll_fn(|cx| connection.poll_closed(cx)).await;
364            std::future::pending::<()>().await;
365        });
366
367        let (send_request, connection) = h2::client::handshake(client_io).await.unwrap();
368        let driver = tokio::spawn(connection);
369
370        let mut send_request = send_request.ready().await.unwrap();
371        let request = http::Request::get("https://localhost/api/v2/series").body(()).unwrap();
372        let (response, _send_stream) = send_request.send_request(request, true).unwrap();
373        request_sent_tx.send(()).unwrap();
374
375        let error = response.await.expect_err("request should have failed");
376        driver.abort();
377
378        error
379    }
