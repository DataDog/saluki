39/// The type-erased body type used internally by [`HttpClient`].
40///
41/// All request bodies are converted to this type before being sent over the wire, which ensures a single
42/// monomorphization of the underlying HTTP/2 and TLS stacks regardless of the caller's body type.
43pub type ClientBody = BoxBody<Bytes, Box<dyn std::error::Error + Send + Sync>>;
44
45#[pin_project]
46struct ClientBodyAdapter<B> {
47    #[pin]
48    inner: B,
49    size_hint: SizeHint,
50}
51
52impl<B> Body for ClientBodyAdapter<B>
53where
54    B: Body,
55    B::Data: Buf,
56    B::Error: Into<Box<dyn std::error::Error + Send + Sync>>,
57{
58    type Data = Bytes;
59    type Error = Box<dyn std::error::Error + Send + Sync>;
60
61    fn poll_frame(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Result<Frame<Self::Data>, Self::Error>>> {
62        self.project().inner.poll_frame(cx).map(|maybe_frame| {
63            maybe_frame.map(|result| {
64                result
65                    .map(|frame| frame.map_data(|mut data| data.copy_to_bytes(data.remaining())))
66                    .map_err(Into::into)
67            })
68        })
69    }
70
71    fn size_hint(&self) -> SizeHint {
72        self.size_hint
73    }
74}

88    /// Sends a request to the server, and waits for a response.
89    ///
90    /// The request body is type-erased internally, so callers can use any body type that implements
91    /// [`Body`] with `Data` types that implement [`Buf`].
92    ///
93    /// # Errors
94    ///
95    /// If there was an error sending the request, an error will be returned.
96    pub async fn send<B>(&mut self, req: Request<B>) -> Result<Response<Incoming>, GenericError>
97    where
98        B: Body + Send + Sync + 'static,
99        B::Data: Buf + Send,
100        B::Error: Into<Box<dyn std::error::Error + Send + Sync>>,
101    {
102        let mut req = req.map(into_client_body);
103        let captured_conn = capture_connection(&mut req);
104        let result = self
105            .inner
106            .ready()
107            .await
108            .map_err(GenericError::from_boxed)?
109            .call(req)
110            .await;
111
112        check_connection_state(captured_conn);
113
114        result.map_err(GenericError::from_boxed)
115    }
116}

118impl Service<Request<ClientBody>> for HttpClient {
119    type Response = Response<Incoming>;
120    type Error = BoxError;
121    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
122
123    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
124        self.inner.poll_ready(cx)
125    }
126
127    fn call(&mut self, mut req: Request<ClientBody>) -> Self::Future {
128        let captured_conn = capture_connection(&mut req);
129        let fut = self.inner.call(req);
130
131        Box::pin(async move {
132            let result = fut.await;
133
134            check_connection_state(captured_conn);
135
136            result
137        })
138    }
139}

142///
143/// This uses `Buf::copy_to_bytes` for the data conversion, which is zero-copy when the underlying

155/// An HTTP client builder.
156///
157/// Provides an ergonomic builder API for configuring an HTTP client.
158///
159/// # Defaults
160///
161/// A number of sensible defaults are provided:
162///
163/// - support for both HTTP and HTTPS (uses platform's root certificates for server certificate validation)
164/// - support for both HTTP/1.1 and HTTP/2 (automatically negotiated via ALPN)
165/// - non-infinite timeouts for various stages of the request lifecycle (30 second connect timeout, 60 second per-request timeout)
166/// - connection pool for reusing connections (45 second idle connection timeout, and a maximum of 5 idle connections
167///   per host)
168/// - support for FIPS-compliant cryptography if the `fips` feature is enabled in the `saluki-tls` crate
169pub struct HttpClientBuilder {
170    connector_builder: HttpsCapableConnectorBuilder,
171    hyper_builder: Builder,
172    tls_builder: ClientTLSConfigBuilder,
173    client_tls_config: Option<ClientConfig>,
174    request_timeout: Option<Duration>,
175    endpoint_telemetry: Option<EndpointTelemetryLayer>,
176    proxies: Option<Vec<Proxy>>,
177}

371    /// Builds the `HttpClient`.
372    ///
373    /// # Errors
374    ///
375    /// If there was an error building the TLS configuration for the client, or if a supplied complete TLS
376    /// configuration fails FIPS validation in a FIPS build, an error will be returned.
377    pub fn build(self) -> Result<HttpClient, GenericError> {
378        let http_protocol = self.connector_builder.http_protocol;
379        let tls_config = match self.client_tls_config {
380            Some(mut config) => {
381                ensure_client_config_fips_compliant(&config)?;
382                config.alpn_protocols.clear();
383                config
384            }
385            None => self.tls_builder.build()?,
386        };
387        let connector = self.connector_builder.build(tls_config)?;
388        let tls_handshake_timeout = connector.tls_handshake_timeout();
389        // The inner connector verifies HTTPS proxies; this separate config verifies destinations through CONNECT.
390        // Keep native roots and verification enabled here even if the inner connector uses an insecure local config.
391        let mut proxy_tls_config = ClientTLSConfigBuilder::new()
392            .with_root_cert_store(load_platform_root_certificates_inner()?)
393            .build()?;
394        if http_protocol == HttpProtocol::Http2 {
395            // The proxy connector can't report a negotiated `h2`. Advertise `h2` only when the client requires it.
396            proxy_tls_config.alpn_protocols = http_protocol_alpns(http_protocol);
397        }
398        let mut proxy_connector = hyper_http_proxy::ProxyConnector::unsecured(connector);
399        proxy_connector.set_tls(Some(TlsConnector::from(Arc::new(proxy_tls_config))));
400        // A zero timeout means the handshake deadline is disabled, matching `HttpsCapableConnector`'s own
401        // handling of `Duration::ZERO` for direct connections.
402        let proxy_tls_handshake_timeout = (!tls_handshake_timeout.is_zero()).then_some(tls_handshake_timeout);
403        proxy_connector.set_tls_handshake_timeout(proxy_tls_handshake_timeout);
404        if let Some(proxies) = &self.proxies {
405            for proxy in proxies {
406                proxy_connector.add_proxy(proxy.to_owned());
407            }
408        }
409        let mut hyper_builder = self.hyper_builder;
410        if http_protocol == HttpProtocol::Http2 {
411            hyper_builder.http2_only(true);
412        }
413        let client = hyper_builder.build(proxy_connector);
414
415        let inner = ServiceBuilder::new()
416            .option_layer(self.request_timeout.map(TimeoutLayer::new))
417            .option_layer(self.endpoint_telemetry)
418            .service(client.map_err(BoxError::from))
419            .boxed_clone();
420
421        Ok(HttpClient { inner })
422    }
423}

807    /// Answers each request on `io` with a 200 response, speaking only HTTP/2.
808    async fn serve_h2<T>(io: T, seen_versions: mpsc::UnboundedSender<Version>)
809    where
810        T: AsyncRead + AsyncWrite + Unpin + Send + 'static,
811    {
812        let service = service_fn(move |request: Request<Incoming>| {
813            let _ = seen_versions.send(request.version());
814            async { Ok::<_, Infallible>(Response::new(Full::new(Bytes::from_static(b"ok")))) }
815        });
816
817        // A client that doesn't speak HTTP/2 fails the connection. Each test asserts on the client side.
818        let _ = http2::Builder::new(TokioExecutor::new())
819            .serve_connection(TokioIo::new(io), service)
820            .await;
821    }

858    async fn send_request_to_tls_server(
859        builder: HttpClientBuilder, server_config: ServerConfig,
860    ) -> Result<Option<Vec<u8>>, GenericError> {
861        let listener = TcpListener::bind("127.0.0.1:0").await?;
862        let port = listener.local_addr()?.port();
863        let server_task = tokio::spawn(async move {
864            let (stream, _) = listener.accept().await.expect("server should accept a connection");
865            let mut stream = TlsAcceptor::from(Arc::new(server_config))
866                .accept(stream)
867                .await
868                .expect("TLS handshake should succeed");
869            let negotiated_alpn = stream.get_ref().1.alpn_protocol().map(ToOwned::to_owned);
870            let mut request = [0; 4096];
871            let bytes_read = stream.read(&mut request).await.expect("server should read the request");
872            assert!(bytes_read > 0, "server should receive an HTTP request");
873            stream
874                .write_all(b"HTTP/1.1 200 OK\r\ncontent-length: 0\r\nconnection: close\r\n\r\n")
875                .await
876                .expect("server should write the response");
877            negotiated_alpn
878        });
879
880        let mut client = builder.with_http_protocol(HttpProtocol::Http1).build()?;
881        let request = Request::get(format!("https://localhost:{port}/"))
882            .body(Empty::<Bytes>::new())
883            .expect("request should build");
884        let response = timeout(Duration::from_secs(5), client.send(request))
885            .await
886            .map_err(|_| GenericError::msg("HTTP request timed out"))??;
887        assert_eq!(response.status(), http::StatusCode::OK);
888        Ok(timeout(Duration::from_secs(5), server_task)
889            .await
890            .expect("TLS server should finish")
891            .expect("TLS server task should not panic"))
892    }
