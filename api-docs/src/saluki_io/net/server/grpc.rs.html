45/// Resolved keepalive parameters for a gRPC server.
46///
47/// Defaults are 2 h interval, 20 s timeout, and no connection age limit.
48#[derive(Clone, Debug)]
49pub struct GrpcKeepalive {
50    /// Interval between HTTP/2 keepalive PING frames.
51    pub http2_keepalive_interval: Duration,
52
53    /// Timeout for receiving a PONG after a keepalive PING before closing the connection.
54    pub http2_keepalive_timeout: Duration,
55
56    /// Maximum duration a connection may exist before the server sends GOAWAY. A zero duration
57    /// means no limit.
58    pub max_connection_age: Duration,
59
60    /// Grace period after `max_connection_age` before the connection is forcibly closed. A zero
61    /// duration means no limit.
62    pub max_connection_age_grace: Duration,
63}
64
65impl Default for GrpcKeepalive {
66    fn default() -> Self {
67        Self {
68            http2_keepalive_interval: Duration::from_secs(2 * 60 * 60),
69            http2_keepalive_timeout: Duration::from_secs(20),
70            max_connection_age: Duration::ZERO,
71            max_connection_age_grace: Duration::ZERO,
72        }
73    }
74}

76/// A gRPC server.
77///
78/// Allows serving multiple gRPC services from a single endpoint.
79///
80/// This type is a thin wrapper over helper types from `tonic` and `axum`, and principally is meant to provide an opaque
81/// gRPC server implementation that operates correctly when run under supervision. As such, this type can't be manually
82/// served: it is only usable by adding it to a supervisor.
83///
84/// # Supervision
85///
86/// The listen address is bound during initialization, so a failure to bind is raised before the supervised worker
87/// starts running, and a restart rebinds.
88///
89/// The server will attempt to gracefully shutdown existing connections when the parent supervisor signals shutdown.
90/// This will cause the worker to utilize the maximum allowable grace period during shutdown: it will attempt to take as
91/// long as necessary to gracefully shutdown existing connections, bounded only by the parent supervisor.
92pub struct GrpcServer {
93    listen_addr: ListenAddress,
94    routes: Option<Routes>,
95    graceful_shutdown_timeout: Option<Duration>,
96    keepalive: GrpcKeepalive,
97    tls_config: Option<ServerConfig>,
98    bound_address_id: Option<Identifier>,
99}
100
101impl GrpcServer {
102    /// Creates an empty server with no attached services, configured to listen on the given address.
103    pub fn new(listen_addr: ListenAddress) -> Self {
104        Self {
105            listen_addr,
106            routes: None,
107            graceful_shutdown_timeout: None,
108            keepalive: GrpcKeepalive::default(),
109            tls_config: None,
110            bound_address_id: None,
111        }
112    }
113
114    /// Sets the identifier used to publish the bound TCP address.
115    ///
116    /// Defaults to not publishing the address. Unix listen addresses never publish an address.
117    pub fn with_bound_address_id(mut self, id: impl Into<Identifier>) -> Self {
118        self.bound_address_id = Some(id.into());
119        self
120    }
121
122    /// Sets the graceful shutdown timeout for this server.
123    ///
124    /// During shutdown, the server will for all in-flight connections to complete before ultimately completing itself.
125    /// When no timeout is specified, this will lead to the worker taking the maximum allowable time to shutdown if
126    /// connections are blocked or otherwise "stuck." Setting an explicit graceful shutdown timeout will cause the
127    /// worker to bound how long it waits for in-flight connections to shutdown before forcefully completing and moving
128    /// on.
129    ///
130    /// Defaults to no timeout (wait as long as allowed).
131    pub fn with_graceful_shutdown_timeout(mut self, timeout: Duration) -> Self {
132        self.graceful_shutdown_timeout = Some(timeout);
133        self
134    }
135
136    /// Sets the TLS configuration for the server.
137    ///
138    /// This enables TLS, after which the server only accepts connections that are encrypted with TLS.
139    ///
140    /// Defaults to TLS being disabled.
141    pub fn with_tls_config(mut self, config: ServerConfig) -> Self {
142        self.tls_config = Some(config);
143        self
144    }
145
146    fn publish_bound_address<F>(&self, local_addr: F, configured_addr: &SocketAddr) -> Result<(), InitializationError>
147    where
148        F: FnOnce() -> io::Result<SocketAddr>,
149    {
150        let Some(id) = self.bound_address_id.clone() else {
151            return Ok(());
152        };
153
154        let bound_address = local_addr()
155            .with_error_context(|| format!("Failed to query bound address for gRPC server ({}).", configured_addr))?;
156        let dataspace = DataspaceRegistry::try_current()
157            .ok_or_else(|| saluki_error::generic_error!("Dataspace not available for gRPC server."))?;
158        dataspace.assert(BoundServerAddress(bound_address), id);
159
160        Ok(())
161    }
162
163    /// Adds a new service to this server.
164    pub fn add_service<S>(mut self, svc: S) -> Self
165    where
166        S: Service<Request<Body>, Error = Infallible> + NamedService + Clone + Send + Sync + 'static,
167        S::Response: axum::response::IntoResponse,
168        S::Future: Send + 'static,
169    {
170        let routes = self.routes.take().unwrap_or_default().add_service(svc);
171
172        Self {
173            routes: Some(routes),
174            ..self
175        }
176    }
177
178    /// Sets the keepalive parameters for this server.
179    pub fn with_keepalive(mut self, keepalive: GrpcKeepalive) -> Self {
180        self.keepalive = keepalive;
181        self
182    }

185#[async_trait]
186impl Supervisable for GrpcServer {
187    fn name(&self) -> &str {
188        "grpc_server"
189    }
190
191    fn shutdown_strategy(&self) -> ShutdownStrategy {
192        // Utilize the maximum allowable grace period to give connections a chance to gracefully shutdown.
193        ShutdownStrategy::Graceful(Duration::MAX)
194    }
195
196    async fn initialize(&self, process_shutdown: ShutdownHandle) -> Result<SupervisorFuture, InitializationError> {
197        let routes = self.routes.clone().unwrap_or_default();
198        let shutdown_timeout = self.graceful_shutdown_timeout.unwrap_or(Duration::MAX);
199
200        // Build the tonic server with keepalive settings applied.
201        let ka = &self.keepalive;
202        let mut server = Server::default()
203            .http2_keepalive_interval(Some(ka.http2_keepalive_interval))
204            .http2_keepalive_timeout(Some(ka.http2_keepalive_timeout));
205        if ka.max_connection_age != Duration::ZERO {
206            server = server.max_connection_age(ka.max_connection_age);
207        }
208        if ka.max_connection_age_grace != Duration::ZERO {
209            server = server.max_connection_age_grace(ka.max_connection_age_grace);
210        }
211
212        // Prepare TLS config once: both TCP and Unix paths use the same acceptor.
213        let tls_acceptor = if let Some(mut config) = self.tls_config.clone() {
214            ensure_server_config_fips_compliant(&mut config)
215                .error_context("Failed to configure TLS for gRPC server")?;
216
217            // ALPN: advertise h2 so TLS clients negotiate HTTP/2 directly.
218            config.alpn_protocols.push(b"h2".to_vec());
219
220            Some(tokio_rustls::TlsAcceptor::from(Arc::new(config)))
221        } else {
222            None
223        };
224
225        match &self.listen_addr {
226            ListenAddress::Tcp(addr) => {
227                if let Some(acceptor) = tls_acceptor {
228                    let listener = tokio::net::TcpListener::bind(*addr)
229                        .await
230                        .with_error_context(|| format!("Failed to bind listener for gRPC server ({}).", addr))?;
231                    self.publish_bound_address(|| listener.local_addr(), addr)?;
232
233                    // Spawn each TLS handshake concurrently so a stalled client cannot block the accept loop.
234                    let incoming = spawn_tls_handshake_loop(listener, acceptor);
235
236                    Ok(Box::pin(async move {
237                        let (drain_tx, drain_rx) = oneshot::channel();
238                        let serve = server.serve_with_incoming_shutdown(routes, incoming, async move {
239                            let _ = drain_rx.await;
240                        });
241
242                        pin!(serve, process_shutdown);
243
244                        select! {
245                            result = &mut serve => result.error_context("Failed to serve gRPC server."),
246
247                            _ = &mut process_shutdown => {
248                                let _ = drain_tx.send(());
249
250                                match timeout(shutdown_timeout, serve).await {
251                                    Ok(Ok(())) => Ok(()),
252                                    Ok(Err(e)) => Err(e).error_context("Failed to serve gRPC server."),
253                                    Err(_) => {
254                                        warn!("Failed to gracefully drain gRPC connections.");
255                                        Ok(())
256                                    },
257                                }
258                            },
259                        }
260                    }))
261                } else {
262                    let listener = TcpIncoming::bind(*addr)
263                        .with_error_context(|| format!("Failed to bind listener for gRPC server ({}).", addr))?;
264                    self.publish_bound_address(|| listener.local_addr(), addr)?;
265
266                    Ok(Box::pin(async move {
267                        let (drain_tx, drain_rx) = oneshot::channel();
268                        let serve = server.serve_with_incoming_shutdown(routes, listener, async move {
269                            let _ = drain_rx.await;
270                        });
271
272                        pin!(serve, process_shutdown);
273
274                        select! {
275                            result = &mut serve => result.error_context("Failed to serve gRPC server."),
276
277                            _ = &mut process_shutdown => {
278                                let _ = drain_tx.send(());
279
280                                match timeout(shutdown_timeout, serve).await {
281                                    Ok(Ok(())) => Ok(()),
282                                    Ok(Err(e)) => Err(e).error_context("Failed to serve gRPC server."),
283                                    Err(_) => {
284                                        warn!("Failed to gracefully drain gRPC connections.");
285                                        Ok(())
286                                    },
287                                }
288                            },
289                        }
290                    }))
291                }
292            }
293            #[cfg(unix)]
294            ListenAddress::Unix(path) => {
295                let path = path.clone();
296                ensure_unix_socket_free(&path)
297                    .await
298                    .with_error_context(|| format!("Failed to clear gRPC Unix socket '{}'.", path.display()))?;
299                let listener = tokio::net::UnixListener::bind(&path)
300                    .with_error_context(|| format!("Failed to bind gRPC Unix listener on '{}'.", path.display()))?;
301                set_unix_socket_write_only(&path).await.with_error_context(|| {
302                    format!("Failed to set permissions on gRPC Unix socket '{}'.", path.display())
303                })?;
304
305                if let Some(acceptor) = tls_acceptor {
306                    // TLS over Unix sockets: same concurrent handshake pattern as TCP.
307                    let incoming = spawn_tls_handshake_loop(listener, acceptor);
308
309                    Ok(Box::pin(async move {
310                        let (drain_tx, drain_rx) = oneshot::channel();
311                        let serve = server.serve_with_incoming_shutdown(routes, incoming, async move {
312                            let _ = drain_rx.await;
313                        });
314
315                        pin!(serve, process_shutdown);
316
317                        select! {
318                            result = &mut serve => result.error_context("Failed to serve gRPC server."),
319
320                            _ = &mut process_shutdown => {
321                                let _ = drain_tx.send(());
322
323                                match timeout(shutdown_timeout, serve).await {
324                                    Ok(Ok(())) => Ok(()),
325                                    Ok(Err(e)) => Err(e).error_context("Failed to serve gRPC server."),
326                                    Err(_) => {
327                                        warn!("Failed to gracefully drain gRPC connections.");
328                                        Ok(())
329                                    },
330                                }
331                            },
332                        }
333                    }))
334                } else {
335                    let incoming = tokio_stream::wrappers::UnixListenerStream::new(listener);
336
337                    Ok(Box::pin(async move {
338                        let (drain_tx, drain_rx) = oneshot::channel();
339                        let serve = server.serve_with_incoming_shutdown(routes, incoming, async move {
340                            let _ = drain_rx.await;
341                        });
342
343                        pin!(serve, process_shutdown);
344
345                        select! {
346                            result = &mut serve => result.error_context("Failed to serve gRPC server."),
347
348                            _ = &mut process_shutdown => {
349                                let _ = drain_tx.send(());
350
351                                match timeout(shutdown_timeout, serve).await {
352                                    Ok(Ok(())) => Ok(()),
353                                    Ok(Err(e)) => Err(e).error_context("Failed to serve gRPC server."),
354                                    Err(_) => {
355                                        warn!("Failed to gracefully drain gRPC connections.");
356                                        Ok(())
357                                    },
358                                }
359                            },
360                        }
361                    }))
362                }
363            }
364            _ => Err(InitializationError::Failed {
365                source: saluki_error::generic_error!("gRPC endpoint must be a TCP or Unix address."),
366            }),
367        }
368    }
369}

371/// Accepts connections from a listener and performs TLS handshakes concurrently.
372///
373/// Each accepted connection gets its own spawned task for the TLS handshake, so a stalled client that connects
374/// without sending a TLS ClientHello cannot block subsequent connections from being accepted.
375fn spawn_tls_handshake_loop<L, S>(
376    listener: L, acceptor: tokio_rustls::TlsAcceptor,
377) -> ReceiverStream<Result<TlsServerStream<S>, std::io::Error>>
378where
379    L: AcceptConnection<S> + Send + 'static,
380    S: AsyncRead + AsyncWrite + Unpin + Send + 'static,
381{
382    let (tx, rx) = mpsc::channel(128);
383
384    tokio::spawn(async move {
385        loop {
386            match listener.accept().await {
387                Ok(stream) => {
388                    let acceptor = acceptor.clone();
389                    let tx = tx.clone();
390
391                    tokio::spawn(async move {
392                        match acceptor.accept(stream).await {
393                            Ok(tls_stream) => {
394                                let _ = tx.send(Ok(TlsServerStream(tls_stream))).await;
395                            }
396                            Err(e) => {
397                                debug!(error = %e, "gRPC TLS handshake failed; skipping connection.");
398                            }
399                        }
400                    });
401                }
402                Err(e) => {
403                    let _ = tx.send(Err(e)).await;
404                    break;
405                }
406            }
407        }
408    });
409
410    ReceiverStream::new(rx)
411}

413/// Trait abstracting over `TcpListener` and `UnixListener` accept loops.
414trait AcceptConnection<S> {
415    fn accept(&self) -> impl std::future::Future<Output = std::io::Result<S>> + Send;
416}
