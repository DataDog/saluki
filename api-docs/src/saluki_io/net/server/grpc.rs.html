18/// A gRPC server.
19///
20/// Allows serving multiple gRPC services from a single endpoint.
21///
22/// This type is a thin wrapper over helper types from `tonic` and `axum`, and principally is meant to provide an opaque
23/// gRPC server implementation that operates correctly when run under supervision. As such, this type can't be manually
24/// served: it is only usable by adding it to a supervisor.
25///
26/// # Supervision
27///
28/// The listen address is bound during initialization, so a failure to bind is raised before the supervised worker
29/// starts running, and a restart rebinds.
30///
31/// The server will attempt to gracefully shutdown existing connections when the parent supervisor signals shutdown.
32/// This will cause the worker to utilize the maximum allowable grace period during shutdown: it will attempt to take as
33/// long as necessary to gracefully shutdown existing connections, bounded only by the parent supervisor.
34///
35/// # Missing
36///
37/// - Accepting `ListenAddress` to optionally listen on a Unix Domain socket or other connection-oriented transport.
38pub struct GrpcServer {
39    listen_addr: SocketAddr,
40    routes: Option<Routes>,
41    graceful_shutdown_timeout: Option<Duration>,
42}
43
44impl GrpcServer {
45    /// Creates an empty server with no attached services, configured to listen on the given address.
46    pub fn new(listen_addr: SocketAddr) -> Self {
47        Self {
48            listen_addr,
49            routes: None,
50            graceful_shutdown_timeout: None,
51        }
52    }
53
54    /// Sets the graceful shutdown timeout for this server.
55    ///
56    /// During shutdown, the server will for all in-flight connections to complete before ultimately completing itself.
57    /// When no timeout is specified, this will lead to the worker taking the maximum allowable time to shutdown if
58    /// connections are blocked or otherwise "stuck." Setting an explicit graceful shutdown timeout will cause the
59    /// worker to bound how long it waits for in-flight connections to shutdown before forcefully completing and moving
60    /// on.
61    ///
62    /// Defaults to no timeout (wait as long as allowed).
63    pub fn with_graceful_shutdown_timeout(mut self, timeout: Duration) -> Self {
64        self.graceful_shutdown_timeout = Some(timeout);
65        self
66    }
67
68    /// Adds a new service to this server.
69    pub fn add_service<S>(mut self, svc: S) -> Self
70    where
71        S: Service<Request<Body>, Error = Infallible> + NamedService + Clone + Send + Sync + 'static,
72        S::Response: axum::response::IntoResponse,
73        S::Future: Send + 'static,
74    {
75        let routes = self.routes.take().unwrap_or_default().add_service(svc);
76
77        Self {
78            routes: Some(routes),
79            ..self
80        }
81    }
82}

84#[async_trait]
85impl Supervisable for GrpcServer {
86    fn name(&self) -> &str {
87        "grpc_server"
88    }
89
90    fn shutdown_strategy(&self) -> ShutdownStrategy {
91        // Utilize the maximum allowable grace period to give connections a chance to gracefully shutdown.
92        ShutdownStrategy::Graceful(Duration::MAX)
93    }
94
95    async fn initialize(&self, process_shutdown: ShutdownHandle) -> Result<SupervisorFuture, InitializationError> {
96        // Try binding our listen address during initialization to surface issues earlier.
97        let listener = TcpIncoming::bind(self.listen_addr)
98            .with_error_context(|| format!("Failed to bind listener for gRPC server ({}).", self.listen_addr))?;
99
100        let listen_addr = self.listen_addr;
101        let routes = self.routes.clone().unwrap_or_default();
102        let shutdown_timeout = self.graceful_shutdown_timeout.unwrap_or(Duration::MAX);
103
104        Ok(Box::pin(async move {
105            let (drain_tx, drain_rx) = oneshot::channel();
106            let serve = Server::default().serve_with_incoming_shutdown(routes, listener, async move {
107                let _ = drain_rx.await;
108            });
109
110            pin!(serve, process_shutdown);
111
112            select! {
113                result = &mut serve => result.error_context("Failed to serve gRPC server."),
114
115                _ = &mut process_shutdown => {
116                    // Tell the server to shutdown and drain connections, and then continue driving the server
117                    // future to actually let it finish up and wait for all connections to drain/close.
118                    let _ = drain_tx.send(());
119
120                    match timeout(shutdown_timeout, serve).await {
121                        Ok(Ok(())) => Ok(()),
122                        Ok(Err(e)) => Err(e).error_context("Failed to serve gRPC server."),
123                        Err(_) => {
124                            warn!(%listen_addr, "Failed to gracefully drain gRPC connections.");
125                            Ok(())
126                        },
127                    }
128                },
129            }
130        }))
131    }
132}

135mod tests {

145    /// Bound on any server await in these tests, so a hang fails rather than stalling the suite.
146    const TEST_TIMEOUT: Duration = Duration::from_secs(10);
147
148    /// Reserves a loopback port and releases it, yielding an address a server can bind.
149    fn free_local_addr() -> SocketAddr {
150        let listener = StdTcpListener::bind("127.0.0.1:0").expect("should bind an ephemeral port");
151        let addr = listener.local_addr().expect("should have a local address");
152        drop(listener);
153        addr
154    }
155
156    #[tokio::test]
157    async fn binds_during_initialization() {
158        // The listener is bound by `initialize`, so the port is taken before anything serves. That is what makes a bind
159        // failure a non-restartable initialization error rather than a runtime one.
160        let addr = free_local_addr();
161        let run = GrpcServer::new(addr)
162            .initialize(ShutdownHandle::noop())
163            .await
164            .expect("should initialize");
165
166        assert!(
167            StdTcpListener::bind(addr).is_err(),
168            "initialization should have bound {addr} before the worker future ran"
169        );
170
171        drop(run);
172    }
173
174    #[tokio::test]
175    async fn bind_failure_is_an_initialization_error() {
176        let addr = free_local_addr();
177        let _held = StdTcpListener::bind(addr).expect("should hold the address");
178
179        match GrpcServer::new(addr).initialize(ShutdownHandle::noop()).await {
180            Ok(_) => panic!("initialization should have failed to bind {addr}"),
181            Err(e) => {
182                let error = e.to_string();
183                assert!(error.contains("Failed to bind listener"), "unexpected error: {error}");
184            }
185        }
186    }
187
188    #[tokio::test]
189    async fn releases_its_port_once_the_worker_finishes() {
190        let addr = free_local_addr();
191        let mut coordinator = ShutdownCoordinator::default();
192        let run = GrpcServer::new(addr)
193            .initialize(coordinator.register())
194            .await
195            .expect("should initialize");
196
197        coordinator.shutdown();
198        timeout(TEST_TIMEOUT, run)
199            .await
200            .expect("server should stop on shutdown")
201            .expect("server should stop cleanly");
202
203        assert!(
204            StdTcpListener::bind(addr).is_ok(),
205            "the server should have released {addr} when its worker finished"
206        );
207    }

209    #[tokio::test]
210    async fn an_idle_peer_does_not_wedge_the_drain() {
211        // `tonic` waits for every connection to close and imposes no bound of its own, so a peer that connects and then
212        // does nothing would otherwise hold shutdown open indefinitely.
213        let addr = free_local_addr();
214        let mut coordinator = ShutdownCoordinator::default();
215        let run = GrpcServer::new(addr)
216            .with_graceful_shutdown_timeout(Duration::from_secs(1))
217            .initialize(coordinator.register())
218            .await
219            .expect("should initialize");
220        let run = tokio::spawn(run);
221
222        let mut stream = TcpStream::connect(addr).await.expect("should connect");
223        stream.write_all(b"PRI * HTTP/2.0\r\n").await.expect("should write");
224        stream.flush().await.expect("should flush");
225        tokio::time::sleep(Duration::from_millis(100)).await;
226
227        coordinator.shutdown();
228        timeout(TEST_TIMEOUT, run)
229            .await
230            .expect("server should finish draining rather than waiting on an idle peer")
231            .expect("server task should not panic")
232            .expect("server should stop cleanly");
233    }
