35#[derive(Clone)]

37    override_tx: mpsc::Sender<Option<(Duration, Level)>>,
38
39    /// Least verbose level that an override may set.
40    minimum_level: Level,
41}

43/// An API handler for updating metric filtering directives at runtime.
44///
45/// This handler exposes two main routes -- `/metrics/override` and `/metrics/reset` -- which allow for
46/// overriding the default metric filtering directives (configured at startup) at runtime, and then resetting them once
47/// the override is no longer needed.
48///
49/// As this has the potential for incredibly high metrics cardinality at runtime, the override is set with a specific
50/// duration in which it will apply. Once an override has been active for the configured duration, it will automatically
51/// be reset unless the override is refreshed before the duration elapses.
52///
53/// The maximum duration for an override is 60 minutes.
54///
55/// An override that's less verbose than the configured minimum level is rejected, so that an override can't filter out
56/// metrics that the application relies on, such as those that status reporting is built from.
57pub struct MetricsAPIHandler {
58    state: MetricsHandlerState,
59}
60
61impl MetricsAPIHandler {
62    async fn override_handler(
63        State(state): State<MetricsHandlerState>, params: Query<OverrideQueryParams>, body: String,
64    ) -> impl IntoResponse {
65        // Make sure the override length is within the acceptable range.
66        if params.time_secs > MAXIMUM_OVERRIDE_LENGTH_SECS {
67            return (
68                StatusCode::BAD_REQUEST,
69                format!(
70                    "override time cannot be greater than {} seconds",
71                    MAXIMUM_OVERRIDE_LENGTH_SECS
72                ),
73            );
74        }
75
76        // Parse the override duration and create a new filter from the body.
77        let duration = Duration::from_secs(params.time_secs);
78        let new_level = match Level::try_from(body.as_str()) {
79            Ok(level) => level,
80            Err(e) => {
81                return (
82                    StatusCode::BAD_REQUEST,
83                    format!("failed to parse override filter: {}", e),
84                )
85            }
86        };
87        if new_level > state.minimum_level {
88            return (
89                StatusCode::BAD_REQUEST,
90                format!(
91                    "override level `{}` cannot be less verbose than `{}`",
92                    level_name(new_level),
93                    level_name(state.minimum_level)
94                ),
95            );
96        }
97
98        // Instruct the override processor to apply the new metric filtering directives for the given duration.
99        let _ = state.override_tx.send(Some((duration, new_level))).await;
100
101        (StatusCode::OK, "acknowledged".to_string())
102    }
103
104    async fn reset_handler(State(state): State<MetricsHandlerState>) {
105        // Instruct the override processor to immediately reset back to the original metric filtering directives.
106        let _ = state.override_tx.send(None).await;
107    }
108}

110impl APIHandler for MetricsAPIHandler {
111    type State = MetricsHandlerState;
112
113    fn generate_initial_state(&self) -> Self::State {
114        self.state.clone()
115    }
116
117    fn generate_routes(&self) -> Router<Self::State> {
118        Router::new()
119            .route("/metrics/override", post(Self::override_handler))
120            .route("/metrics/reset", post(Self::reset_handler))
121    }
122}

124/// A worker that processes dynamic metric filter override requests.
125///
126/// When running, the worker asserts a set of routes (based on [`MetricsAPIHandler`]) that allow triggering
127/// an override of the current metrics filter directives as well as clearing (resetting) the active override.
128pub struct MetricsOverrideWorker {
129    handler: MetricsAPIHandler,
130    state: Arc<Mutex<MetricsOverrideWorkerState>>,
131}
132
133struct MetricsOverrideWorkerState {
134    filter_handle: FilterHandle,
135    override_rx: mpsc::Receiver<Option<(Duration, Level)>>,
136}
137
138impl MetricsOverrideWorker {
139    /// Creates a new `MetricsOverrideWorker` driving the given filter handle.
140    ///
141    /// Overrides that are less verbose than `minimum_level` are rejected.
142    pub(super) fn new(filter_handle: FilterHandle, minimum_level: Level) -> Self {
143        let (override_tx, override_rx) = mpsc::channel(1);
144        let handler = MetricsAPIHandler {
145            state: MetricsHandlerState {
146                override_tx,
147                minimum_level,
148            },
149        };
150        Self {
151            handler,
152            state: Arc::new(Mutex::new(MetricsOverrideWorkerState {
153                filter_handle,
154                override_rx,
155            })),
156        }
157    }
158}

160#[async_trait]
161impl Supervisable for MetricsOverrideWorker {
162    fn name(&self) -> &str {
163        "level-override-processor"
164    }
165
166    async fn initialize(&self, process_shutdown: ShutdownHandle) -> Result<SupervisorFuture, InitializationError> {
167        let mut state = self.state.clone().lock_owned().await;
168        let metrics_route = DynamicRoute::http(EndpointType::Privileged, &self.handler);
169
170        Ok(Box::pin(async move {
171            DataspaceRegistry::try_current()
172                .ok_or_else(|| generic_error!("Dataspace not available."))?
173                .assert(metrics_route, "metrics-api");
174
175            process_override_requests(&mut state, process_shutdown).await;
176
177            Ok(())
178        }))
179    }
180}
181
182async fn process_override_requests(state: &mut MetricsOverrideWorkerState, process_shutdown: ShutdownHandle) {
183    let mut override_active = false;
184    let override_timeout = sleep(Duration::MAX);
185
186    pin!(override_timeout, process_shutdown);
187
188    loop {
189        select! {
190            _ = &mut process_shutdown => break,
191            maybe_override = state.override_rx.recv() => match maybe_override {
192                Some(Some((duration, new_level))) => {
193                    info!(level = level_name(new_level), "Overriding existing metric filtering directive for {} seconds...", duration.as_secs());
194
195                    state.filter_handle.override_filter(new_level);
196
197                    // Mark ourselves as having an active override and update the override timeout.
198                    override_active = true;
199                    override_timeout.as_mut().reset(tokio::time::Instant::now() + duration);
200                },
201
202                Some(None) => {
203                    // We've been instructed to immediately reset the filter back to the original one, so simply update
204                    // the override timeout to fire as soon as possible.
205                    override_timeout.as_mut().reset(tokio::time::Instant::now());
206                },
207
208                // Our sender has dropped, so there's no more override requests for us to handle.
209                None => break,
210            },
211            _ = &mut override_timeout => {
212                // Our override timeout has fired. If we have an active override, reset it.
213                //
214                // Otherwise, this just means that we've been running for a while without any overrides, so we can just
215                // reset the timeout with a long duration.
216                if override_active {
217                    override_active = false;
218
219                    state.filter_handle.reset_filter();
220
221                    info!("Restored original metric filtering directive.");
222                }
223
224                override_timeout.as_mut().reset(tokio::time::Instant::now() + Duration::from_secs(MAXIMUM_OVERRIDE_LENGTH_SECS));
225            }
226        }
227    }
228}

230// This module mirrors the well-tested override/reset worker in `logging/api.rs`. Unlike the logging worker -- whose
231// effect is observable through a locally-constructed `tracing` reload handle -- the metrics worker's only observable
232// effect is on a `FilterHandle`, which is created by installing the process-global metrics recorder. Because that
233// recorder can only be installed once per process, all of the worker's behaviors are exercised in a single test,
234// observed end-to-end through the real metrics pipeline.

243    // Emits a DEBUG-level counter and immediately evicts it, returning whether the resulting snapshot carried the
244    // counter as an upsert -- i.e. whether a DEBUG-level metric currently passes the runtime filter level.
245    async fn debug_metric_passes_filter(stream: &mut MetricsStream, name: &'static str) -> bool {
246        counter!(level: Level::DEBUG, name).increment(1);
247        assert!(
248            delete_counter(Key::from_name(name)),
249            "the counter should exist before eviction"
250        );
251
252        let snapshot = tokio::time::timeout(Duration::from_secs(1), stream.next())
253            .await
254            .expect("a snapshot should be produced after eviction")
255            .expect("the metrics stream should not close");
256
257        !snapshot.upserts.is_empty()
258    }
259
260    async fn wait_until_filter_passes(stream: &mut MetricsStream, name: &'static str, expected: bool) {
261        let deadline = tokio::time::Instant::now() + Duration::from_secs(2);
262
263        loop {
264            if debug_metric_passes_filter(stream, name).await == expected {
265                return;
266            }
267
268            assert!(
269                tokio::time::Instant::now() < deadline,
270                "filter never reached `passes={expected}`"
271            );
272
273            sleep(Duration::from_millis(10)).await;
274        }
275    }

277    #[tokio::test]
278    async fn override_reset_and_expiration_toggle_the_runtime_filter_level() {
279        let (filter_handle, _flusher) = initialize_metrics("test".to_string(), Level::INFO)
280            .await
281            .expect("metrics subsystem should initialize");
282
283        let (override_tx, override_rx) = mpsc::channel(1);
284        let mut state = MetricsOverrideWorkerState {
285            filter_handle,
286            override_rx,
287        };
288        let processor = tokio::spawn(async move {
289            process_override_requests(&mut state, ShutdownHandle::noop()).await;
290        });
291
292        let mut stream = MetricsStream::register();
293        const PROBE: &str = "override_reset_probe";
294
295        // With the default INFO filter, a DEBUG-level metric is filtered out.
296        assert!(!debug_metric_passes_filter(&mut stream, PROBE).await);
297
298        // Overriding to DEBUG lets the DEBUG-level metric through.
299        override_tx
300            .send(Some((Duration::from_secs(60), Level::DEBUG)))
301            .await
302            .expect("send override");
303        wait_until_filter_passes(&mut stream, PROBE, true).await;
304
305        // Resetting restores the INFO default, filtering the DEBUG-level metric out again.
306        override_tx.send(None).await.expect("send reset");
307        wait_until_filter_passes(&mut stream, PROBE, false).await;
308
309        // A time-boxed override auto-expires and restores the default filter without an explicit reset.
310        override_tx
311            .send(Some((Duration::from_millis(250), Level::DEBUG)))
312            .await
313            .expect("send time-boxed override");
314        wait_until_filter_passes(&mut stream, PROBE, true).await;
315        wait_until_filter_passes(&mut stream, PROBE, false).await;
316
317        // Dropping the controller side makes the worker exit cleanly.
318        drop(override_tx);
319        processor.await.expect("override processor should exit cleanly");
320    }

322    #[tokio::test]
323    async fn override_cannot_be_less_verbose_than_the_minimum_level() {
324        let (override_tx, mut override_rx) = mpsc::channel(1);
325        let state = MetricsHandlerState {
326            override_tx,
327            minimum_level: Level::INFO,
328        };
329        let request = |level: &str| {
330            MetricsAPIHandler::override_handler(
331                State(state.clone()),
332                Query(OverrideQueryParams { time_secs: 60 }),
333                level.to_string(),
334            )
335        };
336
337        // Less verbose levels would hide metrics that the application relies on, so they're rejected without being
338        // forwarded to the override processor.
339        for level in ["warn", "error"] {
340            let response = request(level).await.into_response();
341            assert_eq!(
342                response.status(),
343                StatusCode::BAD_REQUEST,
344                "`{level}` should be rejected"
345            );
346            assert!(override_rx.try_recv().is_err(), "`{level}` shouldn't be applied");
347        }
348
349        // The minimum level itself and more verbose levels are accepted, whatever the configured default level is.
350        for (name, level) in [("info", Level::INFO), ("debug", Level::DEBUG), ("trace", Level::TRACE)] {
351            let response = request(name).await.into_response();
352            assert_eq!(response.status(), StatusCode::OK, "`{name}` should be accepted");
353            assert_eq!(override_rx.recv().await, Some(Some((Duration::from_secs(60), level))));
354        }
355    }
