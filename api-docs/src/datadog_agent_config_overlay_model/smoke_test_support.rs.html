1//! Types necessary to keep the configuration smoke tests working. Intended to be transitional
2//! until we can get Saluki/ADP to stop deserializing directly from Agent configuration. At that
3//! time the smoke tests will be rendered inert and can be removed.
4use serde::{Deserialize, Serialize};
5
6/// A configuration consumer tracked by the legacy smoke-test registry.
7///
8/// Most variants name structs that consume Agent configuration directly. Sentinel variants track
9/// consumers that do not fit the legacy deserialization test model.
10#[derive(Copy, Clone, Debug, Eq, PartialEq, Ord, PartialOrd, Hash, Deserialize, Serialize)]
11pub enum ConfigurationStruct {
12    AggregateConfiguration,
13    ContainerdConfiguration,
14    DatadogEventsConfiguration,
15    DatadogLogsConfiguration,
16    DatadogServiceChecksConfiguration,
17    DogStatsDDebugLogConfiguration,
18    DogStatsDMapperConfiguration,
19    DogStatsDPrefixFilterConfiguration,
20
21    /// Keys consumed through the typed configuration translation system.
22    TypedConfigSystem,
23
24    /// Keys read via `get_typed` / `try_get_typed` rather than struct deserialization.
25    #[serde(rename = "get_typed")]
26    GetTyped,
27
28    /// Sentinel for supported keys that are being promoted during the transition to
29    /// typed-configuration access. These were missing from our inventory when config smoke
30    /// testing was introduced and for various reasons may not be legitimate targets for that
31    /// testing modality. Regardless, these fields will be part of the typed translation system
32    /// and this sentinel can be removed when our testing model moves over to that system.
33    #[serde(rename = "NO_SMOKE")]
34    NoSmoke,
35}
36
37impl ConfigurationStruct {
38    /// Apparently we travel back and forth between string constant representation and the struct
39    /// names themselves. This function recovers the string constant name for a struct.
40    pub fn as_smoke_test_const(&self) -> &'static str {
41        match self {
42            ConfigurationStruct::AggregateConfiguration => "AGGREGATE_CONFIGURATION",
43            ConfigurationStruct::ContainerdConfiguration => "CONTAINERD_CONFIGURATION",
44            ConfigurationStruct::DatadogEventsConfiguration => "DATADOG_EVENTS_CONFIGURATION",
45            ConfigurationStruct::DatadogLogsConfiguration => "DATADOG_LOGS_CONFIGURATION",
46            ConfigurationStruct::DatadogServiceChecksConfiguration => "DATADOG_SERVICE_CHECKS_CONFIGURATION",
47            ConfigurationStruct::DogStatsDDebugLogConfiguration => "DOGSTATSD_DEBUG_LOG_CONFIGURATION",
48            ConfigurationStruct::DogStatsDMapperConfiguration => "DOGSTATSD_MAPPER_CONFIGURATION",
49            ConfigurationStruct::DogStatsDPrefixFilterConfiguration => "DOGSTATSD_PREFIX_FILTER_CONFIGURATION",
50            ConfigurationStruct::TypedConfigSystem => "TYPED_CONFIG_SYSTEM",
51            ConfigurationStruct::GetTyped => "GET_TYPED",
52            ConfigurationStruct::NoSmoke => "NO_SMOKE",
53        }
54    }
55}
