1//! Serde deserialization for list schema fields with multiple source shapes.
2//!
3//! String lists can arrive as sequences or space-separated environment strings. Map values
4//! containing string lists can likewise arrive as scalars or sequences. Free-form object arrays can
5//! arrive as sequences or JSON-encoded strings. These adapters normalize each form at the boundary.
6
7use std::collections::HashMap;
8use std::fmt;
9
10use serde::de::{self, DeserializeOwned, Deserializer, SeqAccess, Visitor};
11use serde::Deserialize;
12
13/// Deserialize a `Vec<String>` from either a sequence or a space-separated string.
14///
15/// A string is split on whitespace (matching the Agent's space-separated env convention); a
16/// sequence is taken element by element; a null is an empty list. Any other JSON shape is a type
17/// error.
18pub(crate) fn deserialize_space_separated_or_seq<'de, D>(deserializer: D) -> Result<Vec<String>, D::Error>
19where
20    D: Deserializer<'de>,
21{
22    struct SpaceSeparatedOrSeq;
23
24    impl<'de> Visitor<'de> for SpaceSeparatedOrSeq {
25        type Value = Vec<String>;
26
27        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
28            f.write_str("a sequence or a space-separated string")
29        }
30
31        fn visit_str<E: de::Error>(self, v: &str) -> Result<Vec<String>, E> {
32            Ok(v.split_whitespace().map(str::to_owned).collect())
33        }
34
35        fn visit_seq<A: SeqAccess<'de>>(self, mut seq: A) -> Result<Vec<String>, A::Error> {
36            let mut values = Vec::new();
37            while let Some(v) = seq.next_element()? {
38                values.push(v);
39            }
40            Ok(values)
41        }
42
43        fn visit_unit<E: de::Error>(self) -> Result<Vec<String>, E> {
44            Ok(Vec::new())
45        }
46
47        fn visit_none<E: de::Error>(self) -> Result<Vec<String>, E> {
48            Ok(Vec::new())
49        }
50    }
51
52    deserializer.deserialize_any(SpaceSeparatedOrSeq)
53}

55/// Deserialize a JSON array from either a sequence or a JSON-encoded string.
56///
57/// The element type is inferred from the field; an element that does not fit it is a type error,
58/// so the schema's item shape is enforced at the boundary. A null array is an empty array, and a
59/// null element is the element type's default: the Agent reads a null in a `[]map[string]string`
60/// as a nil (empty) map.
61pub(crate) fn deserialize_json_array_or_string<'de, D, T>(deserializer: D) -> Result<Vec<T>, D::Error>
62where
63    T: DeserializeOwned + Default,
64    D: Deserializer<'de>,
65{
66    #[derive(Deserialize)]
67    #[serde(untagged)]
68    enum JsonArrayOrString<T> {
69        Array(Vec<Option<T>>),
70        String(String),
71    }
72
73    let values = match Option::<JsonArrayOrString<T>>::deserialize(deserializer)? {
74        None => return Ok(Vec::new()),
75        Some(JsonArrayOrString::Array(values)) => values,
76        Some(JsonArrayOrString::String(value)) => {
77            serde_json::from_str::<Vec<Option<T>>>(&value).map_err(de::Error::custom)?
78        }
79    };
80    Ok(values.into_iter().map(Option::unwrap_or_default).collect())
81}

83/// Deserialize string-list map values from either scalar strings or sequences.
84///
85/// Scalar values are normalized into one-element vectors. Unlike standalone string-list fields,
86/// scalar map values are not split on whitespace because each scalar represents one complete value.
87/// The map may also arrive as a JSON-encoded string (see
88/// [`deserialize_map_or_json_string`](crate::cast_de::deserialize_map_or_json_string)).
89pub(crate) fn deserialize_string_map_scalar_or_seq<'de, D>(
90    deserializer: D,
91) -> Result<HashMap<String, Vec<String>>, D::Error>
92where
93    D: Deserializer<'de>,
94{
95    #[derive(Deserialize)]
96    #[serde(untagged)]
97    enum ScalarOrSeq {
98        Scalar(String),
99        Seq(Vec<String>),
100    }
101
102    let values = crate::cast_de::deserialize_map_or_json_string::<_, ScalarOrSeq>(deserializer)?;
103    Ok(values
104        .into_iter()
105        .map(|(key, value)| {
106            let value = match value {
107                ScalarOrSeq::Scalar(value) => vec![value],
108                ScalarOrSeq::Seq(values) => values,
109            };
110            (key, value)
111        })
112        .collect())
113}

115#[cfg(test)]
116mod tests {
117    use serde_json::Value;
118
119    use super::*;
120
121    #[derive(serde::Deserialize)]
122    struct Holder {
123        #[serde(deserialize_with = "deserialize_space_separated_or_seq")]
124        list: Vec<String>,
125    }
126
127    fn parse(json: &str) -> Vec<String> {
128        serde_json::from_str::<Holder>(json).unwrap().list
129    }
130
131    #[test]
132    fn sequence_passes_through() {
133        assert_eq!(parse(r#"{"list": ["a", "b"]}"#), vec!["a", "b"]);
134        assert_eq!(parse(r#"{"list": []}"#), Vec::<String>::new());
135    }
136
137    #[test]
138    fn space_separated_string_is_split() {
139        assert_eq!(
140            parse(r#"{"list": "env:prod team:core"}"#),
141            vec!["env:prod", "team:core"]
142        );
143        assert_eq!(parse(r#"{"list": "solo"}"#), vec!["solo"]);
144    }
145
146    #[test]
147    fn whitespace_runs_and_padding_are_ignored() {
148        assert_eq!(parse(r#"{"list": "  a   b  "}"#), vec!["a", "b"]);
149        assert_eq!(parse(r#"{"list": ""}"#), Vec::<String>::new());
150    }
151
152    #[test]
153    fn null_is_an_empty_list() {
154        assert_eq!(parse(r#"{"list": null}"#), Vec::<String>::new());
155    }
156
157    #[test]
158    fn wrong_shape_is_rejected() {
159        assert!(serde_json::from_str::<Holder>(r#"{"list": 5}"#).is_err());
160    }
161
162    #[derive(serde::Deserialize)]
163    struct MapHolder {
164        #[serde(deserialize_with = "deserialize_string_map_scalar_or_seq")]
165        map: HashMap<String, Vec<String>>,
166    }
167
168    fn parse_map(json: &str) -> HashMap<String, Vec<String>> {
169        serde_json::from_str::<MapHolder>(json).unwrap().map
170    }
171
172    #[test]
173    fn string_map_accepts_scalar_and_sequence_values() {
174        let parsed = parse_map(r#"{"map":{"one":"api-key","many":["first","second"],"none":[]}}"#);
175        assert_eq!(parsed["one"], ["api-key"]);
176        assert_eq!(parsed["many"], ["first", "second"]);
177        assert!(parsed["none"].is_empty());
178    }
179
180    #[test]
181    fn string_map_reads_null_as_empty() {
182        assert!(parse_map(r#"{"map":null}"#).is_empty());
183        assert!(parse_map(r#"{"map":{}}"#).is_empty());
184    }
185
186    #[test]
187    fn string_map_rejects_non_string_values() {
188        assert!(serde_json::from_str::<MapHolder>(r#"{"map":{"endpoint":5}}"#).is_err());
189    }
190
191    #[test]
192    fn string_map_accepts_a_json_encoded_string() {
193        let parsed = parse_map(r#"{"map":"{\"one\":\"api-key\",\"many\":[\"first\",\"second\"]}"}"#);
194        assert_eq!(parsed["one"], ["api-key"]);
195        assert_eq!(parsed["many"], ["first", "second"]);
196    }
197
198    #[test]
199    fn string_map_rejects_a_string_that_is_not_a_json_map() {
200        for encoded in [r#""not json""#, r#""[\"api-key\"]""#, r#""{\"endpoint\":5}""#] {
201            let json = format!(r#"{{"map":{encoded}}}"#);
202            assert!(serde_json::from_str::<MapHolder>(&json).is_err(), "{json}");
203        }
204    }

206    #[derive(serde::Deserialize)]
207    struct JsonArrayHolder {
208        #[serde(deserialize_with = "deserialize_json_array_or_string")]
209        values: Vec<Value>,
210    }
211
212    #[test]
213    fn json_array_accepts_a_sequence_or_encoded_string() {
214        let sequence: JsonArrayHolder = serde_json::from_str(r#"{"values":[{"name":"one"}]}"#).unwrap();
215        let encoded: JsonArrayHolder = serde_json::from_str(r#"{"values":"[{\"name\":\"one\"}]"}"#).unwrap();
216
217        assert_eq!(sequence.values, encoded.values);
218        assert_eq!(sequence.values, [serde_json::json!({ "name": "one" })]);
219    }
220
221    #[test]
222    fn json_array_reads_null_as_empty() {
223        let holder: JsonArrayHolder = serde_json::from_str(r#"{"values":null}"#).unwrap();
224        assert!(holder.values.is_empty());
225    }
226
227    #[test]
228    fn json_array_keeps_null_free_form_elements() {
229        let holder: JsonArrayHolder = serde_json::from_str(r#"{"values":[null,{"name":"one"}]}"#).unwrap();
230        assert_eq!(holder.values, [Value::Null, serde_json::json!({ "name": "one" })]);
231    }
232
233    #[derive(serde::Deserialize)]
234    struct StringMapArrayHolder {
235        #[serde(deserialize_with = "deserialize_json_array_or_string")]
236        values: Vec<HashMap<String, String>>,
237    }
238
239    #[test]
240    fn string_map_array_reads_null_elements_as_empty_maps() {
241        // The Agent reads a null in a `[]map[string]string` as a nil map.
242        for json in [
243            r#"{"values":[null,{"name":"one"}]}"#,
244            r#"{"values":"[null,{\"name\":\"one\"}]"}"#,
245        ] {
246            let holder: StringMapArrayHolder = serde_json::from_str(json).unwrap();
247            assert_eq!(holder.values.len(), 2, "{json}");
248            assert!(holder.values[0].is_empty(), "{json}");
249            assert_eq!(holder.values[1]["name"], "one", "{json}");
250        }
251    }
252
253    #[test]
254    fn json_array_rejects_a_non_array_encoded_string() {
255        assert!(serde_json::from_str::<JsonArrayHolder>(r#"{"values":"{\"name\":\"one\"}"}"#).is_err());
256    }
