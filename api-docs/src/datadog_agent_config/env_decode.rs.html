1//! Decode a raw environment-variable string into the JSON shape the schema declares for a leaf.
2//!
3//! The Datadog Agent turns an environment string into a typed value in one place
4//! (`insertNodeFromString`): a registered per-key transformer when the schema names an
5//! `env_parser`, otherwise a cast against the key's default type. This module ports both halves so
6//! ADP can build a resolved view of real JSON arrays, objects, numbers, and booleans at each
7//! nested path, which a single ordinary deserialization then reads.
8//!
9//! Supported splitting and delimiter behavior matches the Agent bug-for-bug (see the per-function
10//! notes, which cite the Agent source). Error handling deliberately diverges: the Agent logs a malformed
11//! value and substitutes an empty result, while this module propagates a hard error so the caller
12//! can reject the value instead of silently losing it. [`crate::cast_de`] notes one exception, and
13//! [`EnvDecode::JsonArrayOrRawString`] is another.
14//!
15//! The scalar type fallbacks are the string half of [`crate::cast_de`], which every configuration
16//! source shares, so an environment variable and a file accept the same spellings for a leaf.
17//!
18//! One type is intentionally left as a string: a `Duration` leaf keeps its raw text and is parsed by
19//! `crate::duration_de` at deserialize time, because a duration also arrives from the file (a Go
20//! duration string) and from the Agent stream (integer nanoseconds), so that leaf must stay
21//! shape-tolerant regardless of the environment.
22

27/// How to decode a raw environment string into a JSON value for one leaf.
28///
29/// The named-parser variants mirror the schema's `env_parser`, and the type fallbacks are the casts
30/// the Agent applies when no `env_parser` is declared, keyed by the leaf's declared type. The
31/// remaining variants are for Saluki-only keys, which have no Agent cast to match; the Datadog table
32/// never uses them.
33#[derive(Clone, Copy, Debug, Eq, PartialEq)]
34pub enum EnvDecode {
35    /// `env_parser: json`: parse the string as a JSON document.
36    Json,
37    /// `env_parser: comma_separated`: split on `,`, no trimming.
38    CommaSeparated,
39    /// `env_parser: comma_and_space_separated`: split on `,` or space, dropping empties.
40    CommaAndSpaceSeparated,
41    /// `env_parser: comma_then_space_separated`: split on `,` if present else space, trimmed.
42    CommaThenSpaceSeparated,
43    /// `env_parser: csv_comma_separated`: the first CSV record, with leading spaces trimmed and lazy quotes.
44    CsvCommaSeparated,
45    /// `env_parser: json_list_or_comma_separated`: a `[...]` JSON list, else split on `,`.
46    JsonListOrCommaSeparated,
47    /// `env_parser: json_list_or_space_separated`: a `[...]` JSON list, else split on space.
48    JsonListOrSpaceSeparated,
49    /// `env_parser: traces_span`: `svc|op=rate,...` into a `{name: rate}` object.
50    TracesSpan,
51    /// Type fallback: a plain string, taken verbatim.
52    RawString,
53    /// Type fallback: a boolean, parsed with Go's `strconv.ParseBool` grammar.
54    Bool,
55    /// Type fallback: a signed integer, in Go's `strconv.ParseInt` base-0 grammar (the Agent's `int`
56    /// cast): optional sign, `0x`/`0o`/`0b` prefixes, a leading `0` as octal, and `_` separators.
57    Integer,
58    /// Type fallback: a finite floating-point number, in Go's `strconv.ParseFloat` grammar.
59    Float,
60    /// Type fallback: a string list (the Agent's `[]string` cast). A JSON array of strings or `null`
61    /// is decoded as JSON; anything else is split on whitespace.
62    StringList,
63    /// Type fallback: a duration, carried through as a string for `crate::duration_de`.
64    DurationString,
65    /// Type fallback: a map/object (or any non-scalar), parsed as a JSON document.
66    JsonValue,
67    /// A JSON array, else the raw string.
68    ///
69    /// For a key with a list default whose consumer decodes a string itself. The Agent decodes the value
70    /// of such a key only when it's a JSON array (`decodeJSONEnv`, `pkg/config/nodetreemodel/config.go`).
71    /// It stores any other value, such as invalid JSON or a JSON object, as the string it is, and the
72    /// consumer rejects that string only when the feature that uses the key runs.
73    JsonArrayOrRawString,
74    /// Saluki-only: a signed decimal integer, optionally followed by an all-zero fraction. A leading
75    /// `0` is decimal.
76    DecimalInteger,
77    /// Saluki-only: a decimal floating-point number, with no `_` separators or hexadecimal form.
78    DecimalFloat,
79    /// Saluki-only: a string list split on whitespace, with no JSON form.
80    WhitespaceList,
81}

83/// Decodes `raw` into the JSON value `how` prescribes.
84///
85/// # Errors
86///
87/// Returns a human-readable message when `raw` is malformed for `how` (invalid JSON, a bad boolean
88/// or number, or a `traces_span` token that is not `name=rate`). The caller pairs it with the
89/// environment variable name.
90pub fn decode(raw: &str, how: EnvDecode) -> Result<Value, String> {
91    match how {
92        EnvDecode::Json | EnvDecode::JsonValue => parse_json(raw),
93        EnvDecode::JsonArrayOrRawString => Ok(match parse_json(raw) {
94            Ok(array @ Value::Array(_)) => array,
95            _ => Value::String(raw.to_string()),
96        }),
97        EnvDecode::CommaSeparated => Ok(comma_separated(raw)),
98        EnvDecode::CommaAndSpaceSeparated => Ok(comma_and_space_separated(raw)),
99        EnvDecode::CommaThenSpaceSeparated => Ok(comma_then_space_separated(raw)),
100        EnvDecode::CsvCommaSeparated => Ok(string_array(csv_comma_separated(raw))),
101        EnvDecode::JsonListOrCommaSeparated => json_list_or_split(raw, ','),
102        EnvDecode::JsonListOrSpaceSeparated => json_list_or_split(raw, ' '),
103        EnvDecode::TracesSpan => traces_span(raw),
104        EnvDecode::RawString | EnvDecode::DurationString => Ok(Value::String(raw.to_string())),
105        EnvDecode::Bool => cast_de::parse_bool(raw).map(Value::Bool),
106        EnvDecode::Integer => cast_de::parse_i64(raw).map(|n| Value::Number(n.into())),
107        EnvDecode::Float => cast_de::parse_f64(raw).map(finite_number),
108        EnvDecode::StringList => Ok(string_array(cast_de::parse_string_list(raw))),
109        EnvDecode::DecimalInteger => cast_de::parse_decimal_i64(raw).map(|n| Value::Number(n.into())),
110        EnvDecode::DecimalFloat => cast_de::parse_decimal_f64(raw).map(finite_number),
111        EnvDecode::WhitespaceList => Ok(string_array(raw.split_whitespace())),
112    }
113}

115/// `env_parser: json` and the map/object type fallback (`ParseEnvJSON`, `config.go:734`).
116///
117/// The Agent decodes into the declared type; we parse to a generic `Value` and let the leaf's
118/// own deserializer enforce its shape. A parse error propagates (the Agent logs and yields the zero
119/// value).
120fn parse_json(raw: &str) -> Result<Value, String> {
121    serde_json::from_str(raw).map_err(|e| format!("invalid JSON: {e}"))
122}
123
124/// `env_parser: comma_separated` (`ParseEnvSplitComma`, `config.go:704`).
125///
126/// `strings.Split` on `,` with no trimming; an empty string yields an empty list (the Agent's
127/// explicit special case), not a one-element list containing `""`.
128fn comma_separated(raw: &str) -> Value {
129    if raw.is_empty() {
130        return Value::Array(Vec::new());
131    }
132    string_array(raw.split(','))
133}
134
135/// `env_parser: comma_and_space_separated` (`ParseEnvSplitCommaAndSpace`, `helper/env.go:87`).
136///
137/// `strings.FieldsFunc` on `,` or space: consecutive separators collapse and empty fields are
138/// dropped.
139fn comma_and_space_separated(raw: &str) -> Value {
140    string_array(raw.split([',', ' ']).filter(|s| !s.is_empty()))
141}
142
143/// `env_parser: comma_then_space_separated` (`ParseEnvSplitCommaThenSpace`, `helper/env.go:99`).
144///
145/// If the string contains a comma, split on commas only; otherwise split on spaces. Each element is
146/// trimmed, and empties are kept (`strings.Split` semantics), so `a,,b` yields three elements.
147fn comma_then_space_separated(raw: &str) -> Value {
148    let sep = if raw.contains(',') { ',' } else { ' ' };
149    string_array(raw.split(sep).map(str::trim))
150}

152/// `env_parser: csv_comma_separated` (`ParseEnvCSVSplit`, `helper/env.go:69`).
153///
154/// The first record of `encoding/csv` with `TrimLeadingSpace` and `LazyQuotes` set, ported from
155/// `Reader.readRecord`:
156///
157/// - Lines end at `\n`, and `\r\n` reads as `\n`. Empty lines before the record are skipped, and an
158///   input with no record yields an empty list (the Agent logs `EOF` and yields an empty list).
159/// - Leading whitespace is trimmed from every field. An unquoted field runs to the next `,` or the
160///   end of the line, and keeps any `"` it contains.
161/// - A field starting with `"` is quoted: `""` reads as `"`, the field ends at a `"` followed by
162///   `,` or the end of the line, any other `"` is kept, and the field may span lines. A quoted field
163///   that never closes runs to the end of the input.
164///
165/// With lazy quotes, `encoding/csv` has no other error for a first record, so this can't fail.
166fn csv_comma_separated(raw: &str) -> Vec<String> {
167    // `readLine` reads `\r\n` as `\n`, and drops a `\r` that ends the input.
168    let normalized = raw.replace("\r\n", "\n");
169    let mut lines = normalized
170        .strip_suffix('\r')
171        .unwrap_or(&normalized)
172        .split_inclusive('\n');
173
174    // Skip the empty lines before the record.
175    let Some(mut line) = lines.find(|line| *line != "\n") else {
176        return Vec::new();
177    };
178
179    let mut fields = Vec::new();
180    loop {
181        line = line.trim_start();
182        if !line.starts_with('"') {
183            match line.split_once(',') {
184                Some((field, rest)) => {
185                    fields.push(field.to_string());
186                    line = rest;
187                    continue;
188                }
189                None => {
190                    fields.push(line.strip_suffix('\n').unwrap_or(line).to_string());
191                    return fields;
192                }
193            }
194        }
195
196        // A quoted field.
197        line = &line[1..];
198        let mut field = String::new();
199        loop {
200            if let Some((before, after)) = line.split_once('"') {
201                field.push_str(before);
202                line = after;
203                if let Some(rest) = line.strip_prefix('"') {
204                    field.push('"');
205                    line = rest;
206                } else if let Some(rest) = line.strip_prefix(',') {
207                    fields.push(field);
208                    line = rest;
209                    break;
210                } else if line.is_empty() || line == "\n" {
211                    fields.push(field);
212                    return fields;
213                } else {
214                    field.push('"');
215                }
216            } else if !line.is_empty() {
217                field.push_str(line);
218                line = lines.next().unwrap_or("");
219            } else {
220                fields.push(field);
221                return fields;
222            }
223        }
224    }
225}

227/// `env_parser: json_list_or_{comma,space}_separated` (`jsonOrSplitBy`, `helper/env.go:120`).
228///
229/// A value delimited by `[` and `]` is parsed as a JSON string list, where a `null` element is an
230/// empty string as in Go's `json.Unmarshal`; anything else is split on `sep`. The input is trimmed
231/// before the `[...]` test. A JSON list that fails to parse propagates an error (the Agent logs and
232/// yields nil).
233fn json_list_or_split(raw: &str, sep: char) -> Result<Value, String> {
234    let trimmed = raw.trim();
235    if trimmed.starts_with('[') && trimmed.ends_with(']') {
236        let list: Vec<Option<String>> =
237            serde_json::from_str(trimmed).map_err(|e| format!("invalid JSON string list: {e}"))?;
238        return Ok(string_array(list.into_iter().map(Option::unwrap_or_default)));
239    }
240    Ok(string_array(trimmed.split(sep)))
241}

243/// `env_parser: traces_span` (`parseAnalyzedSpans`, `helper/env.go:35`).
244///
245/// `service|operation=rate,...` into `{ "service|operation": rate }`. An empty string yields an
246/// empty object; a token that is not `name=rate` with a numeric rate propagates an error.
247fn traces_span(raw: &str) -> Result<Value, String> {
248    let mut map = Map::new();
249    if raw.is_empty() {
250        return Ok(Value::Object(map));
251    }
252    for token in raw.split(',') {
253        let (name, rate) = token
254            .split_once('=')
255            .ok_or_else(|| format!("bad traces_span token `{token}`: expected name=rate"))?;
256        // Unlike `parse_f64`, the Agent's `strconv.ParseFloat` here does not trim whitespace.
257        let rate = Some(rate)
258            .filter(|rate| rate.trim() == *rate)
259            .and_then(|rate| cast_de::parse_f64(rate).ok())
260            .ok_or_else(|| format!("bad traces_span rate in `{token}`"))?;
261        map.insert(name.to_string(), finite_number(rate));
262    }
263    Ok(Value::Object(map))
264}

266/// A JSON number for a float the `cast_de` parsers accepted, which they guarantee is finite.
267fn finite_number(parsed: f64) -> Value {
268    Value::Number(Number::from_f64(parsed).expect("a finite float is a JSON number"))
269}

271/// Collects strings into a JSON array of strings.
272fn string_array<S: AsRef<str>, I: IntoIterator<Item = S>>(items: I) -> Value {
273    Value::Array(
274        items
275            .into_iter()
276            .map(|s| Value::String(s.as_ref().to_string()))
277            .collect(),
278    )
279}
