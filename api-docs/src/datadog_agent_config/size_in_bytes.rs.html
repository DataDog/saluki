1//! Parse byte-size settings the way the Agent's `GetSizeInBytes` does.
2//!
3//! The Agent reads `log_file_max_size` and `dogstatsd_log_file_max_size` with
4//! `parseSizeInBytes` (`pkg/config/nodetreemodel/helpers.go`), which accepts a narrower unit grammar
5//! than the `bytesize` crate and reads every unit as a power of two.
6
7/// Parses a size setting into a byte count, as the Agent's `GetSizeInBytes` does.
8///
9/// This ports `parseSizeInBytes` and the string path of `cast.ToInt`:
10///
11/// 1. Surrounding whitespace is trimmed.
12/// 2. If the text is at least three bytes long and ends in `b` or `B`, a preceding `k`, `m`, or `g`
13///    (either case) selects a multiplier of 2^10, 2^20, or 2^30 and both letters are removed. Any
14///    other preceding byte leaves a multiplier of one and removes only the `b`. A two-byte value such
15///    as `1B` is never stripped, so it is not a number. There is no `T`, no bare `K`/`M`/`G`, and no
16///    `iB` spelling.
17/// 3. The remaining text is trimmed again, an all-zero fraction is dropped (`10.0` is `10`), and the
18///    result is read as a Go integer literal with base 0: an optional sign, then decimal digits or a
19///    `0x`, `0o`, `0b`, or leading-`0` (octal) prefix, with `_` digit separators only between digits
20///    or after a prefix. The value must fit in an `i64`.
21/// 4. The count is multiplied by the multiplier and must fit in a `u64`.
22///
23/// Returns `None` when the Agent's parse fails, the value is negative, or the product overflows. The
24/// Agent reads all of these as `0` without a diagnostic, so `None` keeps them apart from a
25/// configured `0`. Otherwise, returns the Agent's byte count.
26pub fn parse_size_in_bytes(raw: &str) -> Option<u64> {
27    let (digits, multiplier) = split_unit(raw.trim());
28    let count = u64::try_from(parse_go_int(trim_zero_decimal(digits))?).ok()?;
29    count.checked_mul(multiplier)
30}

32/// Splits trimmed text into its numeric part and the multiplier its unit suffix selects.
33///
34/// Works on bytes: unit suffixes are ASCII, so removing them preserves UTF-8 character boundaries.
35fn split_unit(text: &str) -> (&str, u64) {
36    let bytes = text.as_bytes();
37    let len = bytes.len();
38    if len < 3 || !matches!(bytes[len - 1], b'b' | b'B') {
39        return (text, 1);
40    }
41
42    let (multiplier, suffix_len) = match bytes[len - 2].to_ascii_lowercase() {
43        b'k' => (1 << 10, 2),
44        b'm' => (1 << 20, 2),
45        b'g' => (1 << 30, 2),
46        _ => (1, 1),
47    };
48    (text[..len - suffix_len].trim(), multiplier)
49}

51/// `cast`'s `trimZeroDecimal`: scans back over `0` and `.` and cuts at the first `.` that has a `0`
52/// after it, so `"10.0"` and `"10.0."` are both `"10"`.
53///
54/// [`crate::cast_de`] keeps a narrower form for integer settings that drops only a single all-zero
55/// fraction.
56fn trim_zero_decimal(text: &str) -> &str {
57    let mut found_zero = false;
58    for (index, byte) in text.bytes().enumerate().rev() {
59        match byte {
60            b'.' if found_zero => return &text[..index],
61            b'.' => {}
62            b'0' => found_zero = true,
63            _ => return text,
64        }
65    }
66    text
67}

69/// `strconv.ParseInt(s, 0, 0)` on a 64-bit platform, returning `None` where Go returns an error.
70fn parse_go_int(text: &str) -> Option<i64> {
71    let (negative, unsigned) = match text.as_bytes().first() {
72        Some(b'+') => (false, &text[1..]),
73        Some(b'-') => (true, &text[1..]),
74        _ => (false, text),
75    };
76
77    let magnitude = i128::from(parse_go_uint(unsigned)?);
78    // `i64::MIN` has a magnitude one larger than `i64::MAX`.
79    i64::try_from(if negative { -magnitude } else { magnitude }).ok()
80}

82/// `strconv.ParseUint(s, 0, 64)`: base prefixes and underscores follow Go's integer literal rules.
83fn parse_go_uint(text: &str) -> Option<u64> {
84    let bytes = text.as_bytes();
85    let &first = bytes.first()?;
86
87    // A prefix needs a digit position after it, so a bare `0x` is an octal `0` followed by `x`.
88    let (base, digits) = if first == b'0' {
89        match bytes.get(1).map(u8::to_ascii_lowercase) {
90            Some(b'b') if bytes.len() >= 3 => (2, &bytes[2..]),
91            Some(b'o') if bytes.len() >= 3 => (8, &bytes[2..]),
92            Some(b'x') if bytes.len() >= 3 => (16, &bytes[2..]),
93            _ => (8, &bytes[1..]),
94        }
95    } else {
96        (10, bytes)
97    };
98
99    let mut value: u64 = 0;
100    let mut saw_underscore = false;
101    for &byte in digits {
102        let digit = match byte {
103            b'_' => {
104                saw_underscore = true;
105                continue;
106            }
107            b'0'..=b'9' => byte - b'0',
108            b'a'..=b'z' | b'A'..=b'Z' => byte.to_ascii_lowercase() - b'a' + 10,
109            _ => return None,
110        };
111        if digit >= base {
112            return None;
113        }
114        value = value.checked_mul(u64::from(base))?.checked_add(u64::from(digit))?;
115    }
116
117    if saw_underscore && !underscores_ok(bytes) {
118        return None;
119    }
120    Some(value)
121}

123/// `strconv.underscoreOK` for text without a sign: an underscore must sit between two digits, or
124/// between a base prefix and a digit.
125fn underscores_ok(bytes: &[u8]) -> bool {
126    /// What the previous byte was.
127    #[derive(PartialEq)]
128    enum Previous {
129        Start,
130        Digit,
131        Underscore,
132        Other,
133    }
134
135    let mut previous = Previous::Start;
136    let mut hex = false;
137    let mut rest = bytes;
138    if let [b'0', prefix, tail @ ..] = bytes {
139        if matches!(prefix.to_ascii_lowercase(), b'b' | b'o' | b'x') {
140            // A base prefix counts as a digit for separator placement.
141            previous = Previous::Digit;
142            hex = prefix.eq_ignore_ascii_case(&b'x');
143            rest = tail;
144        }
145    }
146
147    for &byte in rest {
148        if byte.is_ascii_digit() || (hex && byte.is_ascii_hexdigit()) {
149            previous = Previous::Digit;
150        } else if byte == b'_' {
151            if previous != Previous::Digit {
152                return false;
153            }
154            previous = Previous::Underscore;
155        } else if previous == Previous::Underscore {
156            return false;
157        } else {
158            previous = Previous::Other;
159        }
160    }
161    previous != Previous::Underscore
162}
