1//! The worker's protocol state: its cursor, its cache of assigned files, and the statuses it reports.
2//!
3//! All of this is discarded when the worker restarts, so a restarted worker fetches and decodes everything again.

20/// Apply states, as the Agent numbers them.
21const UNACKNOWLEDGED: u64 = 1;
22const ACKNOWLEDGED: u64 = 2;
23const ERROR: u64 = 3;
24
25pub(crate) const COLLISION: &str =
26    "Several configuration files share this configuration ID, so none of them was applied.";
27
28/// One assigned configuration file.
29struct CachedFile {
30    path: ConfigPath,
31    meta: TargetMeta,
32
33    /// The opaque product-specific payload.
34    ///
35    /// Retained because the Agent omits unchanged files that the client advertises as cached. A decoder is rebuilt from
36    /// these bytes whenever a product's assignment changes, so the client caches payloads rather than decoded values
37    /// and has no decoded state to invalidate.
38    contents: Vec<u8>,
39}
40
41/// What a product was last decoded from, and the verdicts that decoding produced.
42struct Decoded {
43    /// The subscription the snapshot was published to.
44    generation: Generation,
45
46    /// The product's assigned paths and their hashes. The product is decoded again only when these change.
47    inputs: Vec<(String, [u8; 32])>,
48
49    /// Each configuration's verdict from its product's last decoding.
50    verdicts: BTreeMap<ConfigId, Verdict>,
51}
52
53pub(crate) struct Repository {
54    root_version: u64,
55    targets_version: u64,
56    backend_state: Vec<u8>,
57
58    /// Assigned files keyed by their full path.
59    files: BTreeMap<String, CachedFile>,
60
61    /// Subscribed products that have been decoded since their subscription was created.
62    ///
63    /// A subscribed product missing from here makes the next request ask for the full assignment; see
64    /// [`request`](Self::request).
65    decoded: BTreeMap<String, Decoded>,
66
67    /// Why the last poll failed, reported to the Agent on the next one.
68    pub(crate) last_error: Option<String>,
69
70    /// Whether the last response reported the Agent's configuration as expired.
71    expired: bool,
72}

74impl Repository {
75    pub(crate) fn new() -> Self {
76        Self {
77            // The Agent rejects a root version below 1, and does not send the initial root.
78            root_version: 1,
79            targets_version: 0,
80            backend_state: Vec::new(),
81            files: BTreeMap::new(),
82            decoded: BTreeMap::new(),
83            last_error: None,
84            expired: false,
85        }

88    /// Forgets the files and statuses of products that are no longer subscribed, and the decoding of any product that
89    /// has been subscribed again since.
90    pub(crate) fn prune(&mut self, live: &BTreeMap<String, Generation>) {
91        self.files.retain(|_, file| live.contains_key(&file.path.product));
92        self.decoded
93            .retain(|product, decoded| live.get(product) == Some(&decoded.generation));
94    }

96    /// Builds the next poll for the products in `live`, which should have been passed to [`prune`](Self::prune) first.
97    pub(crate) fn request(
98        &self, client_id: &str, config: &RcClientConfiguration, live: &BTreeMap<String, Generation>,
99    ) -> ClientGetConfigsRequest {
100        let targets_version = if self.needs_full_assignment(live) {
101            0
102        } else {
103            self.targets_version
104        };
105
106        let ClientKind::Agent(agent) = &config.kind;
107        ClientGetConfigsRequest {
108            client: Some(Client {
109                state: Some(ClientState {
110                    root_version: self.root_version,
111                    targets_version,
112                    config_states: self.config_states(),
113                    has_error: self.last_error.is_some(),
114                    error: self.last_error.clone().unwrap_or_default(),
115                    backend_client_state: self.backend_state.clone(),
116                }),
117                id: client_id.to_owned(),
118                products: live.keys().cloned().collect(),
119                is_agent: true,
120                client_agent: Some(ClientAgent {
121                    name: agent.name.clone(),
122                    version: agent.version.clone(),
123                    cluster_name: agent.cluster_name.clone().unwrap_or_default(),
124                    cluster_id: agent.cluster_id.clone().unwrap_or_default(),
125                    ..Default::default()
126                }),
127                ..Default::default()
128            }),
129            cached_target_files: self
130                .files
131                .values()
132                // The Agent rejects a whole request that advertises an empty file.
133                .filter(|file| file.meta.length > 0)
134                .map(|file| TargetFileMeta {
135                    path: file.path.raw.clone(),
136                    length: file.meta.length as i64,
137                    hashes: vec![TargetFileHash {
138                        algorithm: "sha256".to_owned(),
139                        hash: faster_hex::hex_string(&file.meta.sha256),
140                    }],
141                })
142                .collect(),
143        }
144    }
145
146    /// Whether the next poll must ask for the full assignment by sending targets version 0.
147    ///
148    /// The Agent answers with nothing at all while our targets version matches its own, even when our product list has
149    /// changed or its configuration has since recovered. Sending version 0 makes it send the full assignment, omitting
150    /// only the files we advertise as cached. That is needed in two cases:
151    ///
152    /// - A live product has not been decoded, so a product subscribed after the first poll is delivered without waiting
153    ///   for an unrelated change.
154    /// - The last response was expired. The client adopted its targets version and forgot every file, so a recovery
155    ///   that leaves the version unchanged would otherwise never deliver the configurations again. While the Agent
156    ///   stays expired it answers with its expiry response before comparing versions, so this costs nothing.
157    fn needs_full_assignment(&self, live: &BTreeMap<String, Generation>) -> bool {
158        self.expired || !live.keys().all(|product| self.decoded.contains_key(product))
159    }
160
161    /// Reports one row per configuration ID, which is the finest identity the protocol can report against.
162    fn config_states(&self) -> Vec<ConfigState> {
163        let mut rows: BTreeMap<(&str, &ConfigId), ConfigState> = BTreeMap::new();
164        for file in self.files.values() {
165            let product = file.path.product.as_str();
166            let id = &file.path.config_id;
167            let row = rows.entry((product, id)).or_insert_with(|| {
168                let verdict = self.decoded.get(product).and_then(|decoded| decoded.verdicts.get(id));
169                let (apply_state, apply_error) = match verdict {
170                    None => (UNACKNOWLEDGED, String::new()),
171                    Some(Verdict::Acknowledged) => (ACKNOWLEDGED, String::new()),
172                    Some(Verdict::DecodeRejected(reason))
173                    | Some(Verdict::BuildRejected(reason))
174                    | Some(Verdict::Collided(reason)) => (ERROR, reason.clone()),
175                    Some(Verdict::Panicked) => (ERROR, PANICKED.to_owned()),
176                };
177                ConfigState {
178                    id: id.to_string(),
179                    version: 0,
180                    product: product.to_owned(),
181                    apply_state,
182                    apply_error,
183                }
184            });
185            // Colliding files share a row; report the newest of their versions.
186            row.version = row.version.max(file.meta.version);
187        }
188        rows.into_values().collect()
189    }

191    /// Applies a response to a poll for the products in `requested`, publishing to every product whose inputs changed.
192    ///
193    /// Returns how the response left the client's assignment, which counts the poll.
194    ///
195    /// # Errors
196    ///
197    /// Returns an error if the response is malformed or a payload does not match its metadata. Nothing is then
198    /// committed or published, so the next poll resumes from the same cursor.
199    pub(crate) fn apply(
200        &mut self, response: ClientGetConfigsResponse, requested: &BTreeMap<String, Generation>, shared: &Shared,
201        metrics: &Metrics,
202    ) -> Result<PollOutcome, GenericError> {
203        let ClientGetConfigsResponse {
204            roots,
205            targets,
206            target_files,
207            client_configs,
208            config_status,
209        } = response;
210
211        // The Agent sends an empty response when nothing has changed. An expired status is a change on its own.
212        let expired = config_status == ConfigStatus::Expired as i32;
213        if !expired && roots.is_empty() && targets.is_empty() && target_files.is_empty() && client_configs.is_empty() {
214            // This leaves `self.expired` set after an expiry, so the next poll asks for the full assignment once more.
215            // That is harmless: the Agent answers it with the files it has.
216            return Ok(PollOutcome::Ok);
217        }
218
219        // Validate everything before changing anything. The Agent sends targets with every expired response, but the
220        // message allows none; such a response assigns nothing and leaves the targets version where it was.
221        let targets = if expired && targets.is_empty() {
222            None
223        } else {
224            Some(Targets::parse(&targets)?)
225        };
226        let root_version = match roots.last() {
227            // TUF root versions only increase, so a lower one never moves the cursor back.
228            Some(root) => root_version(root)?.max(self.root_version),
229            None => self.root_version,
230        };
231        let mut sent: HashMap<String, Vec<u8>> = target_files.into_iter().map(|file| (file.path, file.raw)).collect();
232        let mut assigned = Vec::with_capacity(client_configs.len());
233        let mut seen = HashSet::with_capacity(client_configs.len());
234        for raw in &client_configs {
235            // A later duplicate would pair its metadata with bytes the first occurrence already replaced.
236            if !seen.insert(raw.as_str()) {
237                continue;
238            }
239            let path = ConfigPath::parse(raw)
240                .ok_or_else(|| generic_error!("Assigned configuration path {raw} is malformed."))?;
241            if !requested.contains_key(&path.product) {
242                continue;
243            }
244            let meta = match &targets {
245                Some(targets) => targets.meta(raw)?,
246                None => {
247                    return Err(generic_error!(
248                        "Configuration {raw} is assigned without targets metadata."
249                    ))
250                }
251            };
252            match sent.get(raw) {
253                Some(payload) => meta.verify(raw, payload)?,
254                None => {
255                    let cached = self.files.get(raw);
256                    if !cached
257                        .is_some_and(|cached| cached.meta.length == meta.length && cached.meta.sha256 == meta.sha256)
258                    {
259                        return Err(generic_error!(
260                            "Configuration {raw} is assigned but neither sent nor cached."
261                        ));
262                    }
263                }
264            }
265            assigned.push((path, meta));
266        }
267
268        let mut previous = std::mem::take(&mut self.files);
269        for (path, meta) in assigned {
270            let contents = match sent.remove(&path.raw) {
271                Some(payload) => payload,
272                None => {
273                    previous
274                        .remove(&path.raw)
275                        .expect("an unsent path was validated as cached, and each path is assigned once")
276                        .contents
277                }
278            };
279            self.files.insert(path.raw.clone(), CachedFile { path, meta, contents });
280        }
281        self.root_version = root_version;
282        if let Some(targets) = &targets {
283            self.targets_version = targets.version;
284        }
285        // Responses without targets and targets without a `custom` object both keep the previous backend state.
286        if let Some(backend_state) = targets.and_then(|targets| targets.backend_state) {
287            self.backend_state = backend_state;
288        }
289
290        if expired && !self.expired {
291            warn!("The Agent reports its Remote Configuration as expired; configurations are withdrawn until it recovers.");
292        } else if !expired && self.expired {
293            info!("The Agent's Remote Configuration is no longer expired.");
294        }
295        self.expired = expired;
296
297        for product in requested.keys() {
298            self.decode(product, shared, metrics);
299        }
300        Ok(if expired { PollOutcome::Expired } else { PollOutcome::Ok })
301    }

303    /// Decodes and publishes `product` if its inputs changed since it was last decoded.
304    fn decode(&mut self, product: &str, shared: &Shared, metrics: &Metrics) {
305        let files: Vec<&CachedFile> = self
306            .files
307            .values()
308            .filter(|file| file.path.product == product)
309            .collect();
310        let inputs: Vec<(String, [u8; 32])> = files
311            .iter()
312            .map(|file| (file.path.raw.clone(), file.meta.sha256))
313            .collect();
314        if self
315            .decoded
316            .get(product)
317            .is_some_and(|decoded| decoded.inputs == inputs)
318        {
319            return;
320        }
321
322        let mut by_id: BTreeMap<&ConfigId, Vec<&CachedFile>> = BTreeMap::new();
323        for file in &files {
324            by_id.entry(&file.path.config_id).or_default().push(file);
325        }
326        let mut assignment = Vec::with_capacity(by_id.len());
327        let mut collisions = Vec::new();
328        for (id, group) in by_id {
329            match group.as_slice() {
330                [file] => assignment.push((id.clone(), file.contents.as_slice())),
331                _ => {
332                    warn!(
333                        product,
334                        config_id = %id,
335                        paths = ?group.iter().map(|file| file.path.raw.as_str()).collect::<Vec<_>>(),
336                        "Rejected configurations that share a configuration ID."
337                    );
338                    collisions.push((id.clone(), Verdict::Collided(COLLISION.to_owned())));
339                }
340            }
341        }
342
343        let Some((generation, evaluation)) = shared.assign(product, assignment) else {
344            // Unsubscribed since the poll was sent; the next poll forgets it.
345            self.decoded.remove(product);
346            return;
347        };
348
349        match &evaluation.outcome {
350            Outcome::Accepted(()) => {
351                debug!(product, "Published a configuration snapshot.");
352                metrics.count_snapshot_published(product);
353            }
354            Outcome::Rejected { reason, .. } => warn!(product, reason, "Rejected a configuration snapshot."),
355            Outcome::Panicked => {
356                error!(
357                    product,
358                    "Product decoder panicked; rejected its configuration snapshot."
359                );
360                // Every assigned configuration panicked; an empty assignment still counts its panic once.
361                let panicked = evaluation.verdicts.len().max(1) as u64;
362                metrics.count_rejected(product, RejectionStage::Panic, panicked);
363            }
364        }
365        let mut verdicts = BTreeMap::new();
366        for (id, verdict) in evaluation.verdicts.into_iter().chain(collisions) {
367            // A build or panic rejection was logged once for the whole snapshot above, and a panic was counted there.
368            match &verdict {
369                Verdict::Acknowledged | Verdict::Panicked => {}
370                Verdict::DecodeRejected(reason) => {
371                    warn!(product, config_id = %id, reason, "Rejected a configuration.");
372                    metrics.count_rejected(product, RejectionStage::Decode, 1);
373                }
374                Verdict::BuildRejected(_) => metrics.count_rejected(product, RejectionStage::Build, 1),
375                Verdict::Collided(_) => metrics.count_rejected(product, RejectionStage::Collision, 1),
376            }
377            verdicts.insert(id, verdict);
378        }
379
380        self.decoded.insert(
381            product.to_owned(),
382            Decoded {
383                generation,
384                inputs,
385                verdicts,
386            },
387        );
388    }
