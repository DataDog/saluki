1//! The client's observability telemetry.
2//!
3//! Subscribers see only their snapshots and their own decoders' errors. Failed polls, responses the client cannot
4//! apply, and configurations it rejects before decoding, such as colliding IDs, are visible only through these metrics
5//! and the worker's logs.
6
7use saluki_metrics::MetricsBuilder;
8
9/// Counts completed polls, tagged by how they ended.
10const POLLS_TOTAL: &str = "remote_config_polls_total";
11
12/// Counts rejected configurations, tagged by the product and by which rule rejected them.
13const CONFIGURATIONS_REJECTED_TOTAL: &str = "remote_config_configurations_rejected_total";
14
15/// Counts published snapshots, tagged by the product.
16const SNAPSHOTS_PUBLISHED_TOTAL: &str = "remote_config_snapshots_published_total";
17
18/// Measures how long the client has gone without a successful poll, in seconds.
19///
20/// A poll succeeds when the Agent answers, including an expired response and an answer that Remote Configuration is
21/// disabled on the Agent, so the gauge does not climb while Remote Configuration is deliberately off. The polls
22/// counter's `unimplemented` outcome shows that case.
23const SECONDS_SINCE_SUCCESSFUL_POLL: &str = "remote_config_seconds_since_last_successful_poll";
24
25/// How one completed poll ended.
26///
27/// The variants are the `outcome` tag of the polls counter. Every poll ends in exactly one of them: a response the
28/// client could not apply, including a payload that fails its hash, counts as an invalid response rather than a
29/// success.
30#[derive(Clone, Copy)]
31pub(crate) enum PollOutcome {
32    /// The Agent answered a response the client applied.
33    Ok,
34
35    /// The Agent answered but reported its configuration expired, which withdraws every assignment.
36    ///
37    /// The poll itself succeeded, so it does not count as a failure of the transport.
38    Expired,
39
40    /// The RPC failed, including a poll the Agent did not answer within the request timeout.
41    RpcError,
42
43    /// The Agent does not implement Remote Configuration or has it disabled.
44    Unimplemented,
45
46    /// The Agent answered a response the client could not apply.
47    InvalidResponse,
48}
49
50impl PollOutcome {
51    fn tag(&self) -> &'static str {
52        match self {
53            Self::Ok => "ok",
54            Self::Expired => "expired",
55            Self::RpcError => "rpc_error",
56            Self::Unimplemented => "unimplemented",
57            Self::InvalidResponse => "invalid_response",
58        }
59    }
60}
61
62/// Which of the client's rules rejected a configuration.
63///
64/// The variants are the `stage` tag of the rejected-configurations counter. The hash check that aborts a whole poll is
65/// not a stage: it rejects no configuration, it counts as a poll outcome.
66#[derive(Clone, Copy)]
67pub(crate) enum RejectionStage {
68    /// The product's decoder rejected this configuration's payload.
69    Decode,
70
71    /// The configuration decoded, but the product's snapshot as a whole failed to build.
72    Build,
73
74    /// The product's decoder panicked, which rejects its whole assignment.
75    ///
76    /// A panic over an empty assignment rejects no configuration but still counts once, so every panic is counted.
77    Panic,
78
79    /// This configuration shares its ID with another, so none of them reached a decoder.
80    Collision,
81}
82
83impl RejectionStage {
84    fn tag(&self) -> &'static str {
85        match self {
86            Self::Decode => "decode",
87            Self::Build => "build",
88            Self::Panic => "panic",
89            Self::Collision => "collision",
90        }
91    }
92}
93
94/// The client's metrics.
95///
96/// The `product` tag is bounded to the subscribed products, so the counters it tags cannot grow without bound while
97/// the client runs.
98pub(crate) struct Metrics(MetricsBuilder);
99
100impl Metrics {
101    pub(crate) fn new() -> Self {
102        Self(MetricsBuilder::default())
103    }
104
105    /// Counts one completed poll.
106    pub(crate) fn count_poll(&self, outcome: PollOutcome) {
107        self.0
108            .register_counter_with_tags(POLLS_TOTAL, [("outcome", outcome.tag())])
109            .increment(1);
110    }
111
112    /// Counts `count` configurations rejected at `stage` while evaluating `product`.
113    pub(crate) fn count_rejected(&self, product: &str, stage: RejectionStage, count: u64) {
114        self.0
115            .register_counter_with_tags(
116                CONFIGURATIONS_REJECTED_TOTAL,
117                [("product", product.to_owned()), ("stage", stage.tag().to_owned())],
118            )
119            .increment(count);
120    }
121
122    /// Counts one snapshot published to a product's subscriptions.
123    pub(crate) fn count_snapshot_published(&self, product: &str) {
124        self.0
125            .register_counter_with_tags(SNAPSHOTS_PUBLISHED_TOTAL, [("product", product.to_owned())])
126            .increment(1);
127    }
128
129    /// Records how long the client has gone without a successful poll.
130    ///
131    /// Any definitive answer from the Agent is a success here: an applied response, an expired one, or `Unimplemented`.
132    /// An RPC error or an invalid response is not.
133    ///
134    /// Until the first success, `seconds` measures from when the worker started, which is how long the client has been
135    /// without one.
136    pub(crate) fn set_seconds_since_successful_poll(&self, seconds: u64) {
137        self.0.register_gauge(SECONDS_SINCE_SUCCESSFUL_POLL).set(seconds as f64);
138    }
139}
