1//! The context pool: per-kind bounded, lazily-filled sets the differential drivers draw from so
2//! contexts recur across flushes.
3//!
4//! Every driver invocation is a fresh process; without coordination each would mint its own contexts
5//! and the space would grow without bound and never recur. The pool holds one shared set per kind
6//! behind a hard per-kind cap: it mints a new context of a sampled kind while that kind is under its
7//! cap, then draws an existing one of that kind at random. Once a kind's cumulative requests exceed
8//! its cap the set is exhausted and its contexts recur across flushes, which is what gives the
9//! differential oracle multi-point curves to align.
10//!
11//! The caps are read from `context_source.yaml` on the first [`Pool::serve`] call. The intake starts
12//! before `first_sample_config` samples that file, but the first `/contexts` request only ever comes
13//! from a driver, which runs after `first_sample_config` — so the config is present by then.
14

28/// A per-kind bounded, lazily-filled pool of contexts.
29#[derive(Debug)]
30pub struct Pool {
31    /// Directory holding `context_source.yaml`, read once on the first serve.
32    config_dir: PathBuf,
33    /// The resolved caps and the minted contexts, behind one lock.
34    state: Mutex<PoolState>,
35}
36
37/// The pool's mutable state: the resolved per-kind caps and the minted contexts.
38#[derive(Debug, Default)]
39struct PoolState {
40    /// The per-kind caps, resolved from the config on the first serve.
41    caps: Option<ContextSourceConfig>,
42    /// Minted metric contexts, grown to the metric cap then drawn from.
43    metric: Vec<Context>,
44    /// Minted event contexts.
45    event: Vec<Context>,
46    /// Minted service-check contexts.
47    service_check: Vec<Context>,
48    /// Hashes of every context held, so a duplicate mint does not spend a cap slot. Hashes rather
49    /// than the contexts themselves, since the pool already holds up to a million of them and a
50    /// second copy would double that. A hash collision rejects a distinct identity, which costs one
51    /// remint and nothing else.
52    seen: HashSet<u64>,
53}
54
55/// How many times a duplicate mint is retried before the slot is left for a later request. A small
56/// alphabet under a tight budget collides often, and spinning here would stall the serve.
57const MINT_TRIES: usize = 8;
58
59/// The hash a pooled identity is deduplicated by.
60fn digest(context: &Context) -> u64 {
61    let mut hasher = DefaultHasher::new();
62    context.hash(&mut hasher);
63    hasher.finish()
64}
65
66impl Pool {
67    /// A pool that resolves its caps from `context_source.yaml` in `config_dir` on the first serve.
68    #[must_use]
69    pub fn new(config_dir: PathBuf) -> Self {
70        Self {
71            config_dir,
72            state: Mutex::new(PoolState::default()),
73        }
74    }
75
76    /// Serve `n` contexts: for each slot sample a kind, mint a fresh one while that kind is under its
77    /// cap, else draw an existing one of that kind. The whole operation holds the lock, so concurrent
78    /// requests never race on `rng` or overshoot a cap.
79    ///
80    /// # Errors
81    ///
82    /// Returns an error if `context_source.yaml` cannot be read on the first call.
83    pub fn serve<R: Rng + ?Sized>(&self, n: usize, rng: &mut R) -> anyhow::Result<Vec<Context>> {
84        // A poisoned lock still holds a valid pool; recover it rather than panic.
85        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
86
87        let caps = if let Some(caps) = state.caps {
88            caps
89        } else {
90            let resolved =
91                ContextSourceConfig::read(&self.config_dir).context("read context source config for the pool caps")?;
92            state.caps = Some(resolved);
93            resolved
94        };
95
96        let mut out = Vec::with_capacity(n);
97        let mut served_existing = false;
98        for _ in 0..n {
99            let kind = Kind::sample(rng);
100            let PoolState {
101                metric,
102                event,
103                service_check,
104                seen,
105                ..
106            } = &mut *state;
107            let (pool, cap) = match kind {
108                Kind::Metric => (metric, caps.metric_contexts),
109                Kind::Event => (event, caps.event_contexts),
110                Kind::ServiceCheck => (service_check, caps.service_check_contexts),
111            };
112            // The identity is minted against this timeline's real datagram budget, so every served
113            // context has a rendering the driver can pack. A budget too small for this kind, or an
114            // alphabet that keeps landing on the drop side, yields nothing rather than a context of
115            // another kind, so a kind's working set never fills with identities it did not ask for.
116            // A duplicate identity would spend a cap slot without adding a distinct context, so the
117            // kind would stop minting early and the run would explore fewer identities than configured.
118            // Remint on a duplicate instead of pushing it.
119            let mut minted = None;
120            if pool.len() < cap {
121                for _ in 0..MINT_TRIES {
122                    let Some(context) = Context::mint_within(kind, rng, caps.payload_byte_limit.saturating_sub(1))
123                    else {
124                        break;
125                    };
126                    if seen.insert(digest(&context)) {
127                        minted = Some(context);
128                        break;
129                    }
130                }
131            }
132            if let Some(context) = minted {
133                pool.push(context.clone());
134                out.push(context);
135            } else if !pool.is_empty() {
136                let context = pool[rng.random_range(0..pool.len())].clone();
137                served_existing = true;
138                out.push(context);
139            }
140        }
141        let metric = state.metric.len();
142        let event = state.event.len();
143        let service_check = state.service_check.len();
144        drop(state);
145
146        assert_always!(
147            metric <= caps.metric_contexts
148                && event <= caps.event_contexts
149                && service_check <= caps.service_check_contexts,
150            "context pool never exceeds its per-kind caps",
151            &json!({
152                "metric": metric,
153                "event": event,
154                "service_check": service_check,
155                "caps": { "metric": caps.metric_contexts, "event": caps.event_contexts, "service_check": caps.service_check_contexts }
156            })
157        );
158        assert_sometimes!(served_existing, "context source served an existing context", &json!({}));
159        Ok(out)
160    }
161}

177    /// Write a `context_source.yaml` with the given per-kind caps into a fresh temp dir.
178    fn temp_config(metric: usize, event: usize, service_check: usize) -> PathBuf {
179        static SEQ: AtomicUsize = AtomicUsize::new(0);
180        let dir = std::env::temp_dir().join(format!(
181            "ctxpool-{}-{}",
182            std::process::id(),
183            SEQ.fetch_add(1, Ordering::Relaxed)
184        ));
185        std::fs::create_dir_all(&dir).expect("create temp config dir");
186        let config = ContextSourceConfig {
187            payload_byte_limit: 8_192,
188            metric_contexts: metric,
189            event_contexts: event,
190            service_check_contexts: service_check,
191        };
192        std::fs::write(
193            dir.join("context_source.yaml"),
194            config.to_yaml().expect("render config"),
195        )
196        .expect("write config");
197        dir
198    }

200    #[test]
201    fn fills_to_caps_then_repeats() {
202        let mut rng = SmallRng::seed_from_u64(0);
203        // Small metric cap so metric contexts recur within the run.
204        let pool = Pool::new(temp_config(4, 1_000, 1_000));
205
206        let mut metric_wire = BTreeSet::new();
207        let mut metric_total = 0;
208        for _ in 0..50 {
209            for context in pool.serve(5, &mut rng).expect("serve") {
210                if let Context::Metric(_) = context {
211                    let mut wire = Vec::new();
212                    context.encode(&mut wire);
213                    metric_wire.insert(wire);
214                    metric_total += 1;
215                }
216            }
217        }
218        assert!(
219            metric_wire.len() <= 4,
220            "distinct metric {} exceeds cap 4",
221            metric_wire.len()
222        );
223        assert!(metric_total > metric_wire.len(), "expected metric repeats");
224    }

226    // A cap counts distinct identities. A duplicate mint that consumed a slot would stop the kind
227    // minting early and leave the run exploring fewer identities than configured.
228    #[test]
229    fn pooled_contexts_are_distinct() {
230        let mut rng = SmallRng::seed_from_u64(11);
231        let pool = Pool::new(temp_config(64, 64, 64));
232        let mut all = Vec::new();
233        for _ in 0..16 {
234            all.extend(pool.serve(32, &mut rng).expect("serve"));
235        }
236        let state = pool.state.lock().expect("lock");
237        for held in [&state.metric, &state.event, &state.service_check] {
238            let distinct: HashSet<&Context> = held.iter().collect();
239            assert_eq!(distinct.len(), held.len(), "a kind holds a duplicate identity");
240        }
241    }

243    #[test]
244    fn serves_exactly_n_and_round_trips_the_wire() {
245        let mut rng = SmallRng::seed_from_u64(7);
246        let pool = Pool::new(temp_config(1_000, 1_000, 1_000));
247        let contexts = pool.serve(9, &mut rng).expect("serve");
248        assert_eq!(contexts.len(), 9);
249
250        let wire = encode_response(&contexts);
251        let decoded = decode_response(&wire);
252        assert_eq!(decoded.as_deref(), Some(contexts.as_slice()));
253    }
