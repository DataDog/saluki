1//! UTF-8-tolerant `/api/v2/series` `MetricPayload` and `/api/beta/sketches` `SketchPayload` decode.
2//!
3//! rust-protobuf validates UTF-8 on `string` fields and rejects the entire payload on
4//! any non-UTF-8 byte. The Datadog Agent forwards feral non-UTF-8 `DogStatsD` names and
5//! tags into those fields unchecked, so a strict `parse_from_bytes` drops every legitimate
6//! agent-lane payload that carries one bad byte, along with all the contexts it holds.
7//!
8//! v2 tag leniency is source-gated exactly as the production intake gates it. On a failed strict
9//! `proto.Unmarshal` the backend retries into a tags-as-`bytes` message ONLY when the request's
10//! `User-Agent` classifies as `SourceAgent` (`getUserAgent`, apps/propjoe/util.go:284-342, retry at
11//! api_series_v2_handler.go:225-231). So a non-UTF-8 v2 `tags` value is lossy-converted to U+FFFD for
12//! the `datadog-agent` source and whole-payload-rejected for every other source. Every non-tag v2
13//! string field (metric name, unit, source-type, resource) stays UTF-8-strict for ALL sources, since
14//! those fields remain `string` in the bytes-tag variant too, so a non-UTF-8 name drops the payload
15//! regardless of source, exactly as production does.
16//!
17//! The `/api/beta/sketches` path is source-independent: the backend decodes it with a single
18//! `proto.Unmarshal` and no bytes-tag fallback (common.go:447), so a non-UTF-8 byte in ANY sketch
19//! string field, `tags` included, whole-payload-rejects. The v3 path is source-independent too and
20//! lives in `decode_series_v3`. Genuinely malformed wire always fails.

42/// Why the intake did not decode a `/api/v2/series` body.
43#[derive(Clone, Copy, Debug, Eq, PartialEq)]
44pub(crate) enum Rejection {
45    /// A non-tag string field held non-UTF-8 bytes. A correct Agent does not emit one, so a
46    /// decode-faithful assertion treats this as a failure.
47    NonUtf8StrictField,
48    /// A sketch `tags` value held non-UTF-8 bytes. The sketch intake rejects it, but a correct Agent
49    /// forwarding a feral tag is venerable, so a decode-faithful assertion treats this as a faithful
50    /// reject. Only the sketch path produces this.
51    NonUtf8Tag,
52    /// The bytes are not a well-formed protobuf message. No real producer should emit this.
53    MalformedWire,
54}
55
56impl From<protobuf::Error> for Rejection {
57    fn from(_: protobuf::Error) -> Self {
58        Rejection::MalformedWire
59    }
60}

62/// The submitter source classified from the HTTP `User-Agent`, mirroring the backend's `getUserAgent`
63/// (apps/propjoe/util.go:284-342). Only the `datadog-agent` case is load-bearing for v2 decode: the
64/// backend retries a failed strict `proto.Unmarshal` into the tags-as-`bytes` variant ONLY for
65/// `SourceAgent` (api_series_v2_handler.go:225-231), so a feral v2 tag is U+FFFD-sanitized for the
66/// Agent and whole-payload-rejected for every other source. v3 and sketch decode never consult this.
67#[derive(Clone, Copy, Debug, Eq, PartialEq)]
68pub(crate) enum Source {
69    /// `User-Agent` lowercased begins `datadog-agent`. The one source whose v2 tags are sanitized.
70    DatadogAgent,
71    /// Any other `User-Agent`, present or absent (the backend's `SourceNone`/`SourceUnknown`/other
72    /// categories). A non-UTF-8 v2 tag whole-payload-rejects, matching the skipped bytes-tag retry.
73    Other,
74}
75
76impl Source {
77    /// Classify the submitter from the `User-Agent` value, matching the backend's `SourceAgent` test
78    /// (util.go:286-287): lowercase the header, then check the `datadog-agent` prefix. An absent header
79    /// is `Other` (the backend's `SourceNone`). The prefix is ASCII, so ASCII lowercasing is faithful.
80    pub(crate) fn from_user_agent(user_agent: Option<&str>) -> Self {
81        match user_agent {
82            Some(ua) if ua.to_ascii_lowercase().starts_with("datadog-agent") => Self::DatadogAgent,
83            _ => Self::Other,
84        }
85    }
86}

88/// Replace each contiguous run of invalid UTF-8 bytes with a single U+FFFD, matching Go's
89/// `strings.ToValidUTF8` as production's `sanitizePayload` applies it to Agent tags. Rust's
90/// `from_utf8_lossy` emits one replacement per maximal subpart, which yields a different tag
91/// string for a multi-byte invalid run and so a false divergence. Only tags use this.
92fn to_valid_utf8(bytes: &[u8]) -> String {
93    let mut out = String::with_capacity(bytes.len());
94    let mut prev_invalid = false;
95    for chunk in bytes.utf8_chunks() {
96        let valid = chunk.valid();
97        if !valid.is_empty() {
98            out.push_str(valid);
99            prev_invalid = false;
100        }
101        if !chunk.invalid().is_empty() {
102            if !prev_invalid {
103                out.push('\u{FFFD}');
104            }
105            prev_invalid = true;
106        }
107    }
108    out
109}

111/// Strict UTF-8. Production keeps every non-tag string field UTF-8-validated, so a
112/// non-UTF-8 byte here drops the payload as it would in production.
113fn strict(bytes: Vec<u8>) -> Result<String, Rejection> {
114    String::from_utf8(bytes).map_err(|_| Rejection::NonUtf8StrictField)
115}

117/// Strict UTF-8 for a sketch `tags` value. A feral tag is venerable Agent behavior, so it rejects as
118/// `NonUtf8Tag`, distinct from the `NonUtf8StrictField` a non-tag field returns, letting the sketch
119/// decode-faithful assertion treat it as a faithful reject.
120fn strict_tag(bytes: Vec<u8>) -> Result<String, Rejection> {
121    String::from_utf8(bytes).map_err(|_| Rejection::NonUtf8Tag)
122}

124fn unpack(tag: u32) -> Result<(u32, WireType), Rejection> {
125    let wire = WireType::new(tag & 0x7).ok_or(Rejection::MalformedWire)?;
126    let field = tag >> 3;
127    if field == 0 {
128        return Err(Rejection::MalformedWire);
129    }
130    Ok((field, wire))
131}

133/// Decode a `/api/v2/series` body into a `MetricPayload`. Tag leniency is source-gated: for
134/// `Source::DatadogAgent` a non-UTF-8 `tags` value is lossy-converted, for `Source::Other` it returns
135/// `NonUtf8StrictField` and drops the payload, matching the backend's `SourceAgent`-only bytes-tag
136/// retry. A non-UTF-8 non-tag string field returns `NonUtf8StrictField` for every source, matching
137/// production's reject. Structurally malformed protobuf returns `MalformedWire`.
138pub(crate) fn decode_metric_payload(body: &[u8], source: Source) -> Result<MetricPayload, Rejection> {
139    let mut is = CodedInputStream::from_bytes(body);
140    let mut payload = MetricPayload::new();
141    while let Some(tag) = is.read_raw_tag_or_eof()? {
142        match unpack(tag)? {
143            (1, WireType::LengthDelimited) => payload.series.push(decode_series(&is.read_bytes()?, source)?),
144            (_, wire) => is.skip_field(wire)?,
145        }
146    }
147    Ok(payload)
148}

150fn decode_series(body: &[u8], source: Source) -> Result<MetricSeries, Rejection> {
151    let mut is = CodedInputStream::from_bytes(body);
152    let mut series = MetricSeries::new();
153    while let Some(tag) = is.read_raw_tag_or_eof()? {
154        match unpack(tag)? {
155            (1, WireType::LengthDelimited) => series.resources.push(decode_resource(&is.read_bytes()?)?),
156            (2, WireType::LengthDelimited) => series.metric = strict(is.read_bytes()?)?,
157            // Source-gated: the Agent's feral tag is sanitized; any other source rejects it, since the
158            // backend runs its bytes-tag retry only for `SourceAgent`.
159            (3, WireType::LengthDelimited) => {
160                let bytes = is.read_bytes()?;
161                let tag = match source {
162                    Source::DatadogAgent => to_valid_utf8(&bytes),
163                    Source::Other => strict(bytes)?,
164                };
165                series.tags.push(tag);
166            }
167            (4, WireType::LengthDelimited) => series.points.push(decode_point(&is.read_bytes()?)?),
168            (5, WireType::Varint) => series.type_ = EnumOrUnknown::from_i32(is.read_int32()?),
169            (6, WireType::LengthDelimited) => series.unit = strict(is.read_bytes()?)?,
170            (7, WireType::LengthDelimited) => series.source_type_name = strict(is.read_bytes()?)?,
171            (8, WireType::Varint) => series.interval = is.read_int64()?,
172            // Origin metadata. It carries only numeric fields, so UTF-8 leniency does not apply
173            // and a strict sub-message parse stays production-faithful. Dropping it would leave
174            // series.metadata unset and make the Pyld16 origin check vacuous.
175            (9, WireType::LengthDelimited) => {
176                let bytes = is.read_bytes()?;
177                merge_metadata(&mut series.metadata, &bytes)?;
178            }
179            (_, wire) => is.skip_field(wire)?,
180        }
181    }
182    Ok(series)
183}

185/// Merge one `metadata` occurrence into what earlier occurrences left, the way Go's decoder does
186/// (`google.golang.org/protobuf/proto/decode.go`, `unmarshalSingular`): a submessage merges into the
187/// existing one, a scalar replaces. rust-protobuf instead replaces a nested message field wholesale
188/// (`protobuf::rt::read_singular_message_into_field`), so the descent is by hand to stay faithful to
189/// production rather than to this crate's parser.
190fn merge_metadata(metadata: &mut MessageField<Metadata>, bytes: &[u8]) -> Result<(), Rejection> {
191    let mut is = CodedInputStream::from_bytes(bytes);
192    let target = metadata.mut_or_insert_default();
193    while let Some(tag) = is.read_raw_tag_or_eof()? {
194        match unpack(tag)? {
195            (1, WireType::LengthDelimited) => {
196                let origin = is.read_bytes()?;
197                merge_origin(target.origin.mut_or_insert_default(), &origin)?;
198            }
199            (_, wire) => is.skip_field(wire)?,
200        }
201    }
202    Ok(())
203}

205/// Merge one payload-level `metadata` occurrence, the sketch path's [`merge_metadata`] twin. Every
206/// field is a scalar, so the merge is flat. The strings reject non-UTF-8 as `NonUtf8StrictField`, which
207/// is the reason the whole sketch path reports for a feral non-tag field.
208fn merge_common_metadata(metadata: &mut MessageField<CommonMetadata>, bytes: &[u8]) -> Result<(), Rejection> {
209    let mut is = CodedInputStream::from_bytes(bytes);
210    let target = metadata.mut_or_insert_default();
211    while let Some(tag) = is.read_raw_tag_or_eof()? {
212        match unpack(tag)? {
213            (1, WireType::LengthDelimited) => target.agent_version = strict(is.read_bytes()?)?,
214            (2, WireType::LengthDelimited) => target.timezone = strict(is.read_bytes()?)?,
215            (3, WireType::Fixed64) => target.current_epoch = is.read_double()?,
216            (4, WireType::LengthDelimited) => target.internal_ip = strict(is.read_bytes()?)?,
217            (5, WireType::LengthDelimited) => target.public_ip = strict(is.read_bytes()?)?,
218            (6, WireType::LengthDelimited) => target.api_key = strict(is.read_bytes()?)?,
219            (_, wire) => is.skip_field(wire)?,
220        }
221    }
222    Ok(())
223}

278            (2, WireType::LengthDelimited) => {
279                let bytes = is.read_bytes()?;

281            }
282            (_, wire) => is.skip_field(wire)?,
283        }
284    }

288fn decode_sketch(body: &[u8]) -> Result<Sketch, Rejection> {
289    let mut is = CodedInputStream::from_bytes(body);
290    let mut sketch = Sketch::new();
291    while let Some(tag) = is.read_raw_tag_or_eof()? {
292        match unpack(tag)? {
293            (1, WireType::LengthDelimited) => sketch.metric = strict(is.read_bytes()?)?,
294            (2, WireType::LengthDelimited) => sketch.host = strict(is.read_bytes()?)?,
295            (3, WireType::LengthDelimited) => sketch
296                .distributions
297                .push(Distribution::parse_from_bytes(&is.read_bytes()?)?),
298            // Sketch tags have no bytes-tag retry on this path, so a feral tag whole-payload-rejects.
299            // It rejects as `NonUtf8Tag`, marking the venerable case so decode-faithfulness stays green.
300            (4, WireType::LengthDelimited) => sketch.tags.push(strict_tag(is.read_bytes()?)?),
301            (7, WireType::LengthDelimited) => sketch.dogsketches.push(Dogsketch::parse_from_bytes(&is.read_bytes()?)?),
302            (8, WireType::LengthDelimited) => {
303                let bytes = is.read_bytes()?;
304                merge_metadata(&mut sketch.metadata, &bytes)?;
305            }
306            (_, wire) => is.skip_field(wire)?,
307        }
308    }
309    Ok(sketch)
310}

312/// Running cursors into the flat, cross-metric point columns, mirroring the production reader's
313/// per-column state. They advance for every metric, since production drains a validation-dropped
314/// series' points too.
315#[derive(Default)]
316struct PointCursors {
317    timestamp: usize,
318    f64: usize,
319    f32: usize,
320    sint: usize,
321    numbins: usize,
322    bins: usize,
323}
324
325/// Read one point's worth of the flat columns exactly as the production intake does, advancing `cur`
326/// and materializing the `(wire bucket-start, value)` it carries, or `None` when a column it touches
327/// underruns. A metric keeps its leading points up to the first `None`; the backend truncates the rest,
328/// and `cur` still advances so a dropped series' points drain from the shared columns. `timestamps` is
329/// already delta-decoded to absolute bucket-starts; `bin_keys` is delta-decoded per sketch in place.
330#[allow(
331    clippy::cast_precision_loss,
332    reason = "sketch sum/min/max ride the wire as sint64 and are materialized as f64, matching the Agent's own lossy widening"
333)]
334fn read_point(
335    data: &V3MetricData, is_sketch: bool, value_type: u64, timestamps: &[i64], bin_keys: &mut [i32],
336    cur: &mut PointCursors,
337) -> Option<(u64, BucketValue)> {
338    let ts_idx = cur.timestamp;
339    cur.timestamp += 1;
340    if cur.timestamp > timestamps.len() {
341        return None;
342    }
343
344    if is_sketch {
345        let nb_idx = cur.numbins;
346        cur.numbins += 1;
347        if cur.numbins > data.sketchNumBins.len() {
348            return None;
349        }
350        // A bin count too large for `usize` can only exceed the bin columns, so treat it as a truncated
351        // point and drop the rest of this series, never the whole payload.
352        let Ok(numbins) = usize::try_from(data.sketchNumBins[nb_idx]) else {
353            return None;
354        };
355        let bins_start = cur.bins;
356        cur.bins = cur.bins.saturating_add(numbins);
357        let (f64_pre, f32_pre, sint_pre) = (cur.f64, cur.f32, cur.sint);
358        // The Agent writes a sketch as sum, min, max in the value column, then count in sint64.
359        match value_type {
360            0x30 => {
361                cur.f64 += 3;
362                cur.sint += 1;
363            }
364            0x20 => {
365                cur.f32 += 3;
366                cur.sint += 1;
367            }
368            0x10 => cur.sint += 4,
369            0x00 => cur.sint += 1,
370            // An undefined value-type nibble advances no value column, matching the production reader's
371            // sketch branch, which has no default arm (reader.go:497-508). Pyld39 asserts the nibble is
372            // defined, so an undefined one is an encoder-invariant violation caught RED there. Staying
373            // faithful here keeps that violation a single clean signal rather than a cursor desync that
374            // would also manufacture a spurious differential divergence.
375            _ => {}
376        }
377        if cur.f64 > data.valsFloat64.len()
378            || cur.f32 > data.valsFloat32.len()
379            || cur.sint > data.valsSint64.len()
380            || cur.bins > data.sketchBinKeys.len()
381            || cur.bins > data.sketchBinCnts.len()
382        {
383            return None;
384        }
385        let (sum, min, max, count) = match value_type {
386            0x30 => (
387                data.valsFloat64[f64_pre],
388                data.valsFloat64[f64_pre + 1],
389                data.valsFloat64[f64_pre + 2],
390                data.valsSint64[sint_pre],
391            ),
392            0x20 => (
393                f64::from(data.valsFloat32[f32_pre]),
394                f64::from(data.valsFloat32[f32_pre + 1]),
395                f64::from(data.valsFloat32[f32_pre + 2]),
396                data.valsSint64[sint_pre],
397            ),
398            0x10 => (
399                data.valsSint64[sint_pre] as f64,
400                data.valsSint64[sint_pre + 1] as f64,
401                data.valsSint64[sint_pre + 2] as f64,
402                data.valsSint64[sint_pre + 3],
403            ),
404            0x00 => (0.0, 0.0, 0.0, data.valsSint64[sint_pre]),
405            // Undefined nibble consumed no value column, so read none and emit a zero summary. The
406            // summary is not part of a context's identity, so this cannot change the context set.
407            _ => (0.0, 0.0, 0.0, 0),
408        };
409        let bins_end = bins_start + numbins;
410        delta_decode_i32(&mut bin_keys[bins_start..bins_end]);
411        let bins: Vec<(i32, u32)> = bin_keys[bins_start..bins_end]
412            .iter()
413            .copied()
414            .zip(data.sketchBinCnts[bins_start..bins_end].iter().copied())
415            .collect();
416        let bucket_start = u64::try_from(timestamps[ts_idx]).ok()?;
417        Some((
418            bucket_start,
419            BucketValue::Sketch(SketchValue {
420                count,
421                sum,
422                min,
423                max,
424                bins,
425            }),
426        ))
427    } else {
428        let (f64_pre, f32_pre, sint_pre) = (cur.f64, cur.f32, cur.sint);
429        match value_type {
430            0x30 => cur.f64 += 1,
431            0x20 => cur.f32 += 1,
432            0x10 => cur.sint += 1,
433            _ => {}
434        }
435        if cur.f64 > data.valsFloat64.len() || cur.f32 > data.valsFloat32.len() || cur.sint > data.valsSint64.len() {
436            return None;
437        }
438        let value = match value_type {
439            0x30 => data.valsFloat64[f64_pre],
440            0x20 => f64::from(data.valsFloat32[f32_pre]),
441            0x10 => data.valsSint64[sint_pre] as f64,
442            _ => 0.0,
443        };
444        let bucket_start = u64::try_from(timestamps[ts_idx]).ok()?;
445        Some((bucket_start, BucketValue::Scalar(value)))
446    }
447}

449/// Materialize a metric's kept points, draining `cur` across the full `declared` count. Points after the
450/// first column underrun truncate away, matching the backend; `cur` still advances so later metrics read
451/// from the right offsets. The timestamp column is shared, so once it is spent no later point survives.
452fn materialize_points(
453    data: &V3MetricData, is_sketch: bool, value_type: u64, declared: u64, timestamps: &[i64], bin_keys: &mut [i32],
454    cur: &mut PointCursors,
455) -> Vec<(u64, BucketValue)> {
456    let mut points = Vec::new();
457    let mut truncated = false;
458    for _ in 0..declared {
459        match read_point(data, is_sketch, value_type, timestamps, bin_keys, cur) {
460            Some(point) if !truncated => points.push(point),
461            _ => truncated = true,
462        }
463        if cur.timestamp > timestamps.len() {
464            break;
465        }
466    }
467    points
468}

470/// Delta-decode a column in place: each entry becomes the running sum of itself and all before it. Go
471/// int64 addition wraps, so `wrapping_add` mirrors the production reader.
472fn delta_decode(column: &mut [i64]) {
473    for i in 1..column.len() {
474        column[i] = column[i].wrapping_add(column[i - 1]);
475    }
476}

485/// A v3 series the intake kept: its metric name, its tagset with any `host` resource folded in as a
486/// `host:<name>` tag, and its metric kind. This mirrors what the v2 lane's `Context` carries, so v3

497/// Bounds-check a delta-decoded reference into a dictionary of `len` entries. A negative index or one
498/// at or past `len` is `MalformedWire`, matching the production intake's `ref < 0 || ref >= len`
499/// checks. `try_from` folds the sign check and the 32-bit-target width check into the same rejection.
500fn index(v: i64, len: usize) -> Result<usize, Rejection> {
501    let idx = usize::try_from(v).map_err(|_| Rejection::MalformedWire)?;
502    if idx >= len {
503        return Err(Rejection::MalformedWire);
504    }
505    Ok(idx)
506}

508/// Whether a delta-decoded reference is in `[0, len)`, the sound predicate for the v3 reference-bound
509/// assertions. A negative or 32-bit-overflowing value is out of range.
510fn ref_in_range(v: i64, len: usize) -> bool {
511    usize::try_from(v).is_ok_and(|idx| idx < len)
512}

514/// The point and sketch column lengths a correct v3 encoder emits, derived from each metric's packed
515/// type and point count. A scalar point lands in the column named by its value-type nibble, and a Zero
516/// value lands nowhere. A sketch point spreads its summary across that column plus a count in
517/// `valsSint64` and contributes one `sketchNumBins` entry. This mirrors the encoder at dd-source apiv3
518/// metric.go and sketch.go, so the sizing assertions flag any payload a correct Agent would not produce.
519#[derive(Debug, Default, Eq, PartialEq)]
520struct ExpectedColumnSizes {
521    timestamps: u64,
522    vals_float64: u64,
523    vals_float32: u64,
524    vals_sint64: u64,
525    sketch_num_bins: u64,
526}
527
528fn expected_column_sizes(types: &[u64], num_points: &[u64]) -> ExpectedColumnSizes {
529    let mut e = ExpectedColumnSizes::default();
530    for (packed, points) in types.iter().zip(num_points.iter()) {
531        let points = *points;
532        e.timestamps = e.timestamps.saturating_add(points);
533        let value_type = packed & 0xF0;
534        if packed & 0xF == 4 {
535            e.sketch_num_bins = e.sketch_num_bins.saturating_add(points);
536            match value_type {
537                0x30 => {
538                    e.vals_float64 = e.vals_float64.saturating_add(points.saturating_mul(3));
539                    e.vals_sint64 = e.vals_sint64.saturating_add(points);
540                }
541                0x20 => {
542                    e.vals_float32 = e.vals_float32.saturating_add(points.saturating_mul(3));
543                    e.vals_sint64 = e.vals_sint64.saturating_add(points);
544                }
545                0x10 => e.vals_sint64 = e.vals_sint64.saturating_add(points.saturating_mul(4)),
546                0x00 => e.vals_sint64 = e.vals_sint64.saturating_add(points),
547                _ => {}
548            }
549        } else {
550            match value_type {
551                0x30 => e.vals_float64 = e.vals_float64.saturating_add(points),
552                0x20 => e.vals_float32 = e.vals_float32.saturating_add(points),
553                0x10 => e.vals_sint64 = e.vals_sint64.saturating_add(points),
554                _ => {}
555            }
556        }
557    }
558    e
559}

762/// Natively decode a `/api/intake/metrics/v3/series` body into the series the intake keeps. This is an
763/// independent reimplementation of the production intake's v3 decode and per-series validation, not a
764/// wrapper over the data plane's own decoder, so it catches divergences from production rather than
765/// masking them.
766///
767/// The v3 native series API is dictionary + delta encoded columnar protobuf. Production applies a
768/// two-tier failure model that this mirrors exactly:
769///
770/// 1. Whole-payload abort (return `Err`, keep nothing): any dictionary-decode error, any structural,
771///    bounds, or reference error while reconstructing a metric, and non-UTF-8 bytes in the name, unit,
772///    resource, or source-type string dictionaries. Production aborts the whole payload here too.
773/// 2. Per-series drop-and-continue (drop that series, keep the rest): a per-series validation failure
774///    only. A single bad series never drops the whole payload.
775///
776/// Only the tag dictionary tolerates non-UTF-8, sanitized to match the backend's tag handling, keeping
777/// the v2 names-strict/tags-lenient policy.
778#[allow(
779    clippy::too_many_lines,
780    reason = "one cohesive columnar decode walk, and splitting it would scatter the shared per-column cursor state"
781)]
782pub(crate) fn decode_series_v3(
783    target: Target, established_host: &OnceLock<String>, now_secs: i64, body: &[u8],
784) -> Result<Vec<V3Series>, Rejection> {
785    let payload = V3Payload::parse_from_bytes(body)?;
786    // Production errors on nil metric data; both paths keep nothing, so abort the whole payload.
787    let data = payload.metricData.as_ref().ok_or(Rejection::MalformedWire)?;
788
789    // Encode the columnar self-consistency invariants (D.1/D.2/D.3) before any per-series drop, so a
790    // disorderly Agent payload goes RED even when the tolerant decode below would absorb it.
791    assert_column_invariants(target, data);
792
793    let empty_tags: &[String] = &[];
794    let empty_resources: &[String] = &[];
795    let meta_tags = payload.metadata.as_ref().map_or(empty_tags, |m| &m.tags);
796    let meta_resources = payload.metadata.as_ref().map_or(empty_resources, |m| &m.resources);
797
798    // Decode dictionaries. Only the tag dictionary is UTF-8-lenient; the rest reject non-UTF-8, as
799    // production does, aborting the whole payload.
800    let dict_name = unpack_str_dict(target, &data.dictNameStr, false)?;
801    let dict_tag = unpack_str_dict(target, &data.dictTagStr, true)?;
802    let dict_unit = unpack_str_dict(target, &data.dictUnitStr, false)?;
803    let tagsets = unpack_tagsets(target, &data.dictTagsets, &dict_tag, meta_tags)?;
804    let dict_resource = unpack_str_dict(target, &data.dictResourceStr, false)?;
805    let resources = unpack_resources(
806        target,
807        &data.dictResourceLen,
808        &data.dictResourceType,
809        &data.dictResourceName,
810        &dict_resource,
811        meta_resources,
812    )?;
813    let dict_source_type = unpack_str_dict(target, &data.dictSourceTypeName, false)?;
814    let origin_triples = unpack_origin(target, &data.dictOriginInfo)?;
815
816    // Persistent delta accumulators, matching the production reader's per-column running state. Go
817    // int64 addition wraps, so `wrapping_add` mirrors it and the sign/bounds checks catch the rest.
818    let mut name_ref = 0i64;
819    let mut tagset_ref = 0i64;
820    let mut resources_ref = 0i64;
821    let mut source_type_ref = 0i64;
822    let mut origin_ref = 0i64;
823    let mut unit_ref = 0i64;
824    // unitRefs is sparse: its own cursor advances only for flagged metrics, decoupled from the metric index.
825    let mut unit_idx = 0usize;
826    let mut kept = Vec::new();
827    let mut processed = 0usize;
828    let mut cur = PointCursors::default();
829    // Absolute wire bucket-starts, delta-decoded once for the whole shared column.
830    let mut timestamps = data.timestamps.clone();
831    delta_decode(&mut timestamps);
832    // Sketch bin keys are delta-decoded per sketch during the walk, so keep a mutable copy.
833    let mut bin_keys = data.sketchBinKeys.clone();
834
835    for i in 0..data.types.len() {
836        // Every per-metric column must have an entry for this metric. Any short column aborts the
837        // whole payload, matching the production reader's bounds checks.
838        if i >= data.nameRefs.len()
839            || i >= data.tagsetRefs.len()
840            || i >= data.resourcesRefs.len()
841            || i >= data.sourceTypeNameRefs.len()
842            || i >= data.originInfoRefs.len()
843            || i >= data.intervals.len()
844            || i >= data.numPoints.len()
845        {
846            return Err(Rejection::MalformedWire);
847        }
848        // Pyld26: the payload's metric count is len(types); count metrics processed to check it.
849        processed += 1;
850
851        name_ref = name_ref.wrapping_add(data.nameRefs[i]);
852        // Pyld43: the accumulated nameRef indexes dictNameStr in range.
853        assert_always!(
854            ref_in_range(name_ref, dict_name.len()),
855            "Pyld43.name_ref",
856            &json!({ "lane": target,"ref": name_ref, "len": dict_name.len() })
857        );
858        let name_idx = index(name_ref, dict_name.len())?;
859        tagset_ref = tagset_ref.wrapping_add(data.tagsetRefs[i]);
860        // Pyld44: the accumulated tagsetRef indexes the tagset table in range.
861        assert_always!(
862            ref_in_range(tagset_ref, tagsets.len()),
863            "Pyld44.tagset_ref",
864            &json!({ "lane": target,"ref": tagset_ref, "len": tagsets.len() })
865        );
866        let tagset_idx = index(tagset_ref, tagsets.len())?;
867        // unitRefs is sparse: the production reader consumes an entry only when this metric's type
868        // carries flagHasUnit, via a separate cursor with delta accumulation (reader.go:371). A flagged
869        // metric that outruns the column aborts the whole payload, matching the reader's errUnexpectedEof.
870        if data.types[i] & FLAG_HAS_UNIT != 0 {
871            let Some(&unit_delta) = data.unitRefs.get(unit_idx) else {
872                return Err(Rejection::MalformedWire);
873            };
874            unit_idx += 1;
875            unit_ref = unit_ref.wrapping_add(unit_delta);
876            // Pyld48: each unitRef indexes dictUnitStr in range.
877            assert_always!(
878                ref_in_range(unit_ref, dict_unit.len()),
879                "Pyld48.unit_ref",
880                &json!({ "lane": target,"ref": unit_ref, "len": dict_unit.len() })
881            );
882            index(unit_ref, dict_unit.len())?;
883        }
884        resources_ref = resources_ref.wrapping_add(data.resourcesRefs[i]);
885        // Pyld45: the accumulated resourcesRef indexes the resource table in range.
886        assert_always!(
887            ref_in_range(resources_ref, resources.len()),
888            "Pyld45.resource_ref",
889            &json!({ "lane": target,"ref": resources_ref, "len": resources.len() })
890        );
891        let resources_idx = index(resources_ref, resources.len())?;
892        source_type_ref = source_type_ref.wrapping_add(data.sourceTypeNameRefs[i]);
893        // Pyld46: the accumulated sourceTypeNameRef indexes dictSourceTypeName in range.
894        assert_always!(
895            ref_in_range(source_type_ref, dict_source_type.len()),
896            "Pyld46.source_type_ref",
897            &json!({ "lane": target,"ref": source_type_ref, "len": dict_source_type.len() })
898        );
899        index(source_type_ref, dict_source_type.len())?;
900        origin_ref = origin_ref.wrapping_add(data.originInfoRefs[i]);
901        // Pyld47: the accumulated originInfoRef indexes the origin table in range.
902        assert_always!(
903            ref_in_range(origin_ref, origin_triples.len()),
904            "Pyld47.origin_ref",
905            &json!({ "lane": target,"ref": origin_ref, "len": origin_triples.len() })
906        );
907        let origin_idx = index(origin_ref, origin_triples.len())?;
908
909        let name = &dict_name[name_idx];
910        let tags = &tagsets[tagset_idx];
911        let series_resources = &resources[resources_idx];
912
913        // Pyld16-v3: a referenced origin triple is enum-valid. Index 0 is the empty/default origin.
914        if origin_idx > 0 {
915            let (product, category, service) = origin_triples[origin_idx];
916            let out = origin_triple_out_of_domain(product, category, service);
917            assert_always!(
918                out.is_none(),
919                "Pyld16-v3.origin_in_domain",
920                &json!({ "lane": target,"metric": name, "out_of_domain": out })
921            );
922        }
923
924        // Pyld17: every series resolves a non-empty host resource, and all series on a lane share one.
925        let host_name = series_resources
926            .iter()
927            .find(|(type_, _)| type_ == "host")
928            .map(|(_, host)| host.as_str());
929        let host_ok =
930            host_name.is_some_and(|host| !host.is_empty() && established_host.get_or_init(|| host.to_owned()) == host);
931        assert_always!(
932            host_ok,
933            "Pyld17.host_resource_resolved",
934            &json!({ "lane": target,"metric": name, "host": host_name, "established": established_host.get() })
935        );
936
937        let kind = match data.types[i] & 0xF {
938            1 => MetricKind::Count,
939            2 => MetricKind::Rate,
940            3 => MetricKind::Gauge,
941            4 => MetricKind::Sketch,
942            // An out-of-range nibble: production keeps the series and forwards its type verbatim, so
943            // keep it as Other rather than dropping it and masking a producer bug.
944            _ => MetricKind::Other,
945        };
946
947        // Drain this metric's declared points from the shared flat columns. Run for every metric,
948        // kept or dropped, because production drains a dropped series' points too and the cursors are
949        // shared across metrics. The truncated count itself is not kept, only the draining matters.
950        let is_sketch = (data.types[i] & 0xF) == 4;
951        let value_type = data.types[i] & 0xF0;
952        // Pyld39: the value-type nibble is a defined ValueType. A correct v3 encoder always emits one
953        // of these, so an undefined nibble is an encoder-invariant violation. Encode it as an assertion
954        // rather than silently decoding around it. Decode continues either way. Fired before the
955        // per-series drop so an undefined nibble is caught even on a series that is later dropped.
956        assert_always!(
957            matches!(value_type, 0x00 | 0x10 | 0x20 | 0x30),
958            "Pyld39.value_type_defined",
959            &json!({ "lane": target,"value_type": value_type })
960        );
961        let points = materialize_points(
962            data,
963            is_sketch,
964            value_type,
965            data.numPoints[i],
966            &timestamps,
967            &mut bin_keys,
968            &mut cur,
969        );
970
971        assert_series_shape(
972            target,
973            name,
974            tags,
975            series_resources,
976            data.types[i] & 0xF,

995    }

1040/// Pyld51: every dictionary entry is referenced, directly by a metric or transitively through another
1041/// entry. Reachability is one-level and inclusive: a tag entry counts as reached if any tagset uses it,
1042/// a tagset if any metric or any back-reference uses it, a resource-string entry if any resource group
1043/// uses it. That reaches the encoder's legitimate prefix-tagsets and second-level dictionaries, so only
1044/// an entry referenced by nothing at all -- a misbehaving Agent's leftover -- reddens. Runs after the
1045/// metric loop, so every reference re-accumulated here is already in range.
1046#[allow(
1047    clippy::too_many_arguments,
1048    reason = "each dictionary's length is a distinct reachability domain"
1049)]
1050fn assert_no_orphans(
1051    target: Target, data: &V3MetricData, name_len: usize, tag_len: usize, unit_len: usize, res_len: usize,
1052    source_len: usize, origin_len: usize, tagsets_len: usize, resources_len: usize,
1053) {
1054    let mut name_seen = vec![false; name_len];
1055    let mut tag_seen = vec![false; tag_len];
1056    let mut unit_seen = vec![false; unit_len];
1057    let mut res_seen = vec![false; res_len];
1058    let mut source_seen = vec![false; source_len];
1059    let mut origin_seen = vec![false; origin_len];
1060    let mut tagset_seen = vec![false; tagsets_len];
1061    let mut resources_seen = vec![false; resources_len];
1062
1063    let (mut nr, mut tr, mut rr, mut sr, mut or_, mut ur) = (0i64, 0i64, 0i64, 0i64, 0i64, 0i64);
1064    let mut unit_idx = 0usize;
1065    for i in 0..data.types.len() {
1066        nr = nr.wrapping_add(data.nameRefs[i]);
1067        mark_seen(&mut name_seen, nr);
1068        tr = tr.wrapping_add(data.tagsetRefs[i]);
1069        mark_seen(&mut tagset_seen, tr);
1070        rr = rr.wrapping_add(data.resourcesRefs[i]);
1071        mark_seen(&mut resources_seen, rr);
1072        sr = sr.wrapping_add(data.sourceTypeNameRefs[i]);
1073        mark_seen(&mut source_seen, sr);
1074        or_ = or_.wrapping_add(data.originInfoRefs[i]);
1075        mark_seen(&mut origin_seen, or_);
1076        // unitRefs is sparse: only a flagged metric consumes an entry, matching the decode walk above.
1077        if data.types[i] & FLAG_HAS_UNIT != 0 {
1078            if let Some(&unit_delta) = data.unitRefs.get(unit_idx) {
1079                unit_idx += 1;
1080                ur = ur.wrapping_add(unit_delta);
1081                mark_seen(&mut unit_seen, ur);
1082            }
1083        }
1084    }
1085
1086    // Tagset entries reach their positive tag-dict entries and their negative prefix-tagset back-refs.
1087    let mut j = 0usize;
1088    while j < data.dictTagsets.len() {
1089        let size = usize::try_from(data.dictTagsets[j]).unwrap_or(0);
1090        j += 1;
1091        let mut idx = 0i64;
1092        for offset in 0..size {
1093            let Some(&delta) = data.dictTagsets.get(j + offset) else {
1094                break;
1095            };
1096            idx = idx.wrapping_add(delta);
1097            if idx < 0 {
1098                if let Some(back) = idx.checked_neg() {
1099                    mark_seen(&mut tagset_seen, back);
1100                }
1101            } else {
1102                mark_seen(&mut tag_seen, idx);
1103            }
1104        }
1105        j += size;
1106    }
1107
1108    // Resource groups reach their type and name entries in the resource string dictionary.
1109    let mut start = 0usize;
1110    for &group_len in &data.dictResourceLen {
1111        let size = usize::try_from(group_len).unwrap_or(0);
1112        let end = start
1113            .saturating_add(size)
1114            .min(data.dictResourceType.len())
1115            .min(data.dictResourceName.len());
1116        let (mut t, mut n) = (0i64, 0i64);
1117        for pos in start..end {
1118            t = t.wrapping_add(data.dictResourceType[pos]);
1119            mark_seen(&mut res_seen, t);
1120            n = n.wrapping_add(data.dictResourceName[pos]);
1121            mark_seen(&mut res_seen, n);
1122        }
1123        start = start.saturating_add(size);
1124    }
1125
1126    let orphan = first_orphan(&[
1127        ("name", &name_seen),
1128        ("tag", &tag_seen),
1129        ("unit", &unit_seen),
1130        ("resource", &res_seen),
1131        ("source_type", &source_seen),
1132        ("origin", &origin_seen),
1133        ("tagset", &tagset_seen),
1134        ("resources", &resources_seen),
1135    ]);
1136    assert_always!(
1137        orphan.is_none(),
1138        "Pyld51.no_orphans",
1139        &json!({ "lane": target,"orphan": orphan })
1140    );
1141}

1143/// Apply the production intake's per-series keep checks (in order) and fold the host resource into a tag.
1144/// Returns the series' output tags when kept, or `None` when any check drops it. Lane parity with v2:
1145/// the first `host` resource with a non-empty name becomes a `host:<name>` tag so v3 contexts equal v2
1146/// contexts for equivalent input.
1147///
1148/// The backend's `validateSeries` also drops a series on a `processOrigin` failure
1149/// (api_series_v3_handler.go:538-544), the one source-dependent v3 check. The rig omits it knowingly:
1150/// v3 decode folds origin into numeric-only reference fields that never enter a context's identity, so
1151/// a dropped origin can never change the emitted context set, and modeling it would need the source
1152/// threaded into v3, which must stay source-independent.
1153fn kept_series_tags(name: &str, tags: &[String], resources: &[(String, String)]) -> Option<Vec<String>> {
1154    if tags.len() > MAX_TAG_COUNT || resources.len() > MAX_RESOURCE_COUNT || !metric_name_kept(name) {
1155        return None;
1156    }
1157    let host = resources.iter().find(|(type_, _)| type_ == "host");
1158    if let Some((_, host_name)) = host {
1159        if host_name.len() > MAX_HOST_NAME_LEN {
1160            return None;
1161        }
1162    }
1163    let mut out_tags = tags.to_vec();
1164    if let Some((_, host_name)) = host {
1165        if !host_name.is_empty() {
1166            out_tags.push(format!("host:{host_name}"));
1167        }
1168    }
1169    Some(out_tags)
1170}

1172/// Unpack a v3 varint-length-prefixed string dictionary. Index 0 is the empty string (base-1
1173/// indexing). When `sanitize` is set the tag dictionary's non-UTF-8 entries collapse to `to_valid_utf8`
1174/// (Go's `strings.ToValidUTF8`); otherwise non-UTF-8 aborts the whole payload as production does. A
1175/// length prefix that overruns the buffer is `MalformedWire`.
1176fn unpack_str_dict(target: Target, raw: &[u8], sanitize: bool) -> Result<Vec<String>, Rejection> {
1177    let mut dict = vec![String::new()];
1178    let mut offset = 0;
1179    while offset < raw.len() {
1180        let (len, consumed) = read_uvarint(&raw[offset..])?;
1181        offset += consumed;
1182        let len = usize::try_from(len).map_err(|_| Rejection::MalformedWire)?;
1183        let end = offset.checked_add(len).ok_or(Rejection::MalformedWire)?;
1184        // Pyld49: each string-dict entry's length prefix stays within the blob.
1185        assert_always!(
1186            end <= raw.len(),
1187            "Pyld49.dict_blob_bounds",
1188            &json!({ "lane": target,"end": end, "blob_len": raw.len() })
1189        );
1190        if end > raw.len() {
1191            return Err(Rejection::MalformedWire);
1192        }
1193        let bytes = &raw[offset..end];
1194        let str = if sanitize {
1195            to_valid_utf8(bytes)
1196        } else {
1197            // Pyld50: the non-tag dictionaries are strict UTF-8. Tags are coerced, so this fires only for
1198            // name/unit/resource/source-type dicts, where a non-UTF-8 byte is an Agent fault.
1199            let decoded = strict(bytes.to_vec());
1200            assert_always!(
1201                decoded.is_ok(),
1202                "Pyld50.dict_strict_utf8",
1203                &json!({ "lane": target,"entry_len": bytes.len() })
1204            );
1205            decoded?
1206        };
1207        dict.push(str);
1208        offset = end;
1209    }
1210    Ok(dict)
1211}

1213/// Unpack the tagsets dictionary: a flat `[]i64` of length-prefixed groups. Index 0 is the empty
1214/// tagset. Within a group a delta accumulator resets to 0; a positive index selects
1215/// `dictTagStr[idx]` and a negative index splices the earlier `tagsets[-idx]` back-reference, both
1216/// bounds-checked. Any bounds or reference error aborts the whole payload. Payload metadata tags are
1217/// then unioned into every tagset, series tags first.
1218fn unpack_tagsets(
1219    target: Target, packed: &[i64], tag_dict: &[String], meta_tags: &[String],
1220) -> Result<Vec<Vec<String>>, Rejection> {
1221    let mut tagsets: Vec<Vec<String>> = vec![Vec::new()];
1222    let mut i = 0usize;
1223    while i < packed.len() {
1224        let raw_size = packed[i];
1225        i += 1;
1226        let remaining = packed.len() - i;
1227        // Pyld52: the group-size prefix is in [0, remaining].
1228        assert_always!(
1229            ref_in_range(raw_size, remaining + 1),
1230            "Pyld52.tagset_group_size",
1231            &json!({ "lane": target,"size": raw_size, "remaining": remaining })
1232        );
1233        let size = usize::try_from(raw_size).map_err(|_| Rejection::MalformedWire)?;
1234        if size > remaining {
1235            return Err(Rejection::MalformedWire);
1236        }
1237        let mut tags = Vec::with_capacity(size + meta_tags.len());
1238        let mut idx: i64 = 0;
1239        for offset in 0..size {
1240            idx = idx.wrapping_add(packed[i + offset]);
1241            if idx < 0 {
1242                // A negative index is a `tagsets[-idx]` back-reference. `i64::MIN` has no positive
1243                // negation, so `checked_neg` folds that into the same bounds rejection.
1244                let back = idx.checked_neg().ok_or(Rejection::MalformedWire)?;
1245                // Pyld54: a negative back-reference resolves to an earlier tagset, never self or forward.
1246                assert_always!(
1247                    ref_in_range(back, tagsets.len()),
1248                    "Pyld54.tagset_back_reference",
1249                    &json!({ "lane": target,"back": back, "available": tagsets.len() })
1250                );
1251                tags.extend_from_slice(&tagsets[index(back, tagsets.len())?]);
1252            } else {
1253                // Pyld53: a positive entry indexes `dictTagStr` in range.
1254                assert_always!(
1255                    ref_in_range(idx, tag_dict.len()),
1256                    "Pyld53.tagset_tag_index",
1257                    &json!({ "lane": target,"idx": idx, "len": tag_dict.len() })
1258                );
1259                tags.push(tag_dict[index(idx, tag_dict.len())?].clone());
1260            }
1261        }
1262        i += size;
1263        tagsets.push(tags);
1264    }
1265
1266    union_metadata_tags(&mut tagsets, meta_tags);
1267    Ok(tagsets)
1268}

1270/// Union payload metadata tags into every tagset. An empty tagset takes all metadata tags; otherwise
1271/// only metadata tags not already present are appended, series tags first.
1272fn union_metadata_tags(tagsets: &mut [Vec<String>], meta_tags: &[String]) {
1273    if meta_tags.is_empty() {
1274        return;
1275    }
1276    for tags in tagsets.iter_mut() {
1277        if tags.is_empty() {
1278            tags.extend_from_slice(meta_tags);
1279            continue;
1280        }
1281        for mt in meta_tags {
1282            if !tags.contains(mt) {
1283                tags.push(mt.clone());
1284            }
1285        }
1286    }
1287}

1289/// Unpack the resources dictionary from the parallel length, type, and name columns. Index 0 is the
1290/// empty resource set. Each group has its own delta accumulators for the type and name references,
1291/// both indexing `dictResourceStr`; each entry is a `(Type, Name)` pair. Any bounds or reference
1292/// error aborts the whole payload. Payload metadata resource pairs are appended to every group.
1293fn unpack_resources(
1294    target: Target, len_col: &[i64], type_col: &[i64], name_col: &[i64], res_dict: &[String], meta_resources: &[String],
1295) -> Result<Vec<Vec<(String, String)>>, Rejection> {
1296    // Pyld59: payload metadata resources come in (type, name) pairs, so the column has even length.
1297    assert_always!(
1298        meta_resources.len().is_multiple_of(2),
1299        "Pyld59.metadata_resources_even",
1300        &json!({ "lane": target,"len": meta_resources.len() })
1301    );
1302    let meta_pairs = if meta_resources.is_empty() {
1303        Vec::new()
1304    } else {
1305        if !meta_resources.len().is_multiple_of(2) {
1306            return Err(Rejection::MalformedWire);
1307        }
1308        meta_resources
1309            .chunks_exact(2)
1310            .map(|pair| (pair[0].clone(), pair[1].clone()))
1311            .collect::<Vec<_>>()
1312    };
1313
1314    let mut resources: Vec<Vec<(String, String)>> = vec![Vec::new()];
1315    let mut start = 0usize;
1316    for &raw_size in len_col {
1317        // Pyld56: each resource group-length entry is non-negative.
1318        assert_always!(
1319            raw_size >= 0,
1320            "Pyld56.resource_group_length_non_negative",
1321            &json!({ "lane": target,"size": raw_size })
1322        );
1323        let size = usize::try_from(raw_size).map_err(|_| Rejection::MalformedWire)?;
1324        let end = start.checked_add(size).ok_or(Rejection::MalformedWire)?;
1325        if end > type_col.len() || end > name_col.len() {
1326            return Err(Rejection::MalformedWire);
1327        }
1328        let mut type_ref = 0i64;
1329        let mut name_ref = 0i64;
1330        let mut set = Vec::with_capacity(size + meta_pairs.len());
1331        for pos in start..end {
1332            type_ref = type_ref.wrapping_add(type_col[pos]);
1333            name_ref = name_ref.wrapping_add(name_col[pos]);
1334            // Pyld57: resource type and name refs index `dictResourceStr` in range.
1335            assert_always!(
1336                ref_in_range(type_ref, res_dict.len()) && ref_in_range(name_ref, res_dict.len()),
1337                "Pyld57.resource_field_refs",
1338                &json!({ "lane": target,"type_ref": type_ref, "name_ref": name_ref, "len": res_dict.len() })
1339            );
1340            let type_idx = index(type_ref, res_dict.len())?;
1341            let name_idx = index(name_ref, res_dict.len())?;
1342            set.push((res_dict[type_idx].clone(), res_dict[name_idx].clone()));
1343        }
1344        set.extend(meta_pairs.iter().cloned());
1345        resources.push(set);
1346        start = end;
1347    }
1348    Ok(resources)
1349}

1351/// Unpack the origin-info dictionary into `(product, category, service)` triples. Index 0 is the

1354fn unpack_origin(target: Target, raw: &[i32]) -> Result<Vec<(i32, i32, i32)>, Rejection> {

1371/// Reads a base-128 unsigned varint, matching Go's `encoding/binary.Uvarint` exactly, including its
1372/// overflow rule: a value needing more than 64 bits is rejected, so a 10th byte greater than 1 fails
1373/// as production would rather than silently truncating a mutated length prefix.
1374fn read_uvarint(data: &[u8]) -> Result<(u64, usize), Rejection> {
1375    let mut value = 0u64;
1376    let mut shift = 0u32;
1377    for (i, &byte) in data.iter().enumerate() {
1378        if i == 10 {
1379            return Err(Rejection::MalformedWire);
1380        }
1381        if byte & 0x80 == 0 {
1382            if i == 9 && byte > 1 {
1383                return Err(Rejection::MalformedWire);
1384            }
1385            return Ok((value | (u64::from(byte) << shift), i + 1));
1386        }
1387        value |= u64::from(byte & 0x7f) << shift;
1388        shift += 7;
1389    }
1390    Err(Rejection::MalformedWire)
1391}

1959    // Frame `bytes` as one length-delimited protobuf field. Bodies stay under 128 bytes so the
1960    // length is a single varint byte, which keeps the hand-built wire in these properties simple.
1961    fn framed(field: u8, bytes: &[u8]) -> Vec<u8> {
1962        let len = u8::try_from(bytes.len()).expect("field body under 128 bytes");
1963        assert!(len < 0x80, "field body under 128 bytes");
1964        let mut out = vec![(field << 3) | 2, len];
1965        out.extend_from_slice(bytes);
1966        out
1967    }

1970    // the round-trip equality; leniency on non-UTF-8 tags is exercised separately over raw wire.
1971    fn arb_series() -> impl Strategy<Value = MetricSeries> {
1972        (
1973            prop::collection::vec(".{0,8}", 0..3),
1974            ".{0,8}",
1975            ".{0,8}",
1976            ".{0,8}",
1977            prop::collection::vec((-1e9f64..1e9f64, any::<i64>()), 0..4),
1978            0i32..4,
1979            any::<i64>(),
1980            prop::option::of((any::<u32>(), any::<u32>(), any::<u32>())),
1981            prop::collection::vec((".{0,8}", ".{0,8}"), 0..2),
1982        )
1983            .prop_map(
1984                |(tags, metric, unit, source_type_name, points, ty, interval, origin, resources)| {
1985                    let mut series = MetricSeries::new();
1986                    for tag in tags {
1987                        series.tags.push(tag);
1988                    }
1989                    series.metric = metric;
1990                    series.unit = unit;
1991                    series.source_type_name = source_type_name;
1992                    for (value, timestamp) in points {
1993                        let mut point = MetricPoint::new();
1994                        point.value = value;
1995                        point.timestamp = timestamp;
1996                        series.points.push(point);
1997                    }
1998                    series.type_ = EnumOrUnknown::from_i32(ty);
1999                    series.interval = interval;
2000                    if let Some((product, category, service)) = origin {
2001                        let mut o = Origin::new();
2002                        o.origin_product = product;
2003                        o.origin_category = category;
2004                        o.origin_service = service;
2005                        let mut metadata = Metadata::new();
2006                        metadata.origin = MessageField::some(o);
2007                        series.metadata = MessageField::some(metadata);
2008                    }
2009                    for (type_, name) in resources {
2010                        let mut resource = Resource::new();
2011                        resource.type_ = type_;
2012                        resource.name = name;
2013                        series.resources.push(resource);
2014                    }
2015                    series
2016                },
2017            )
2018    }
