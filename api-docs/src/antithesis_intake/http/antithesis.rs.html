1//! Private HTTP routes used by the Antithesis scenario drivers.
2//!
3//! This module owns the private query/control routes:
4//!
5//! - `GET /antithesis/metrics/{target}`: returns one lane's captured contexts and the intake's
6//!   current time for `agent` or `adp`.
7//! - `GET /contexts?n=N`: serves `N` contexts from the shared pool over the binary codec, so the
8//!   drivers render recurring identities.
9
10use antithesis_sdk::prelude::*;
11use antithesis_sdk::random::AntithesisRng;
12use axum::{
13    extract::{Path, Query, State},
14    http::StatusCode,
15    routing::get,
16    Json, Router,
17};
18use harness::contexts::encode_response;
19use rand::rand_core::UnwrapErr;
20use serde::Deserialize;
21use serde_json::json;
22
23use super::state::AppState;
24use crate::capture;
25
26/// The largest working set a single `/contexts` request may ask for. A request over this is rejected
27/// rather than served, so a malformed `n` cannot make the pool mint an unbounded response.
28const MAX_CONTEXTS_PER_REQUEST: usize = 65_536;
29
30pub(super) fn routes() -> Router<AppState> {
31    Router::new()
32        .route("/antithesis/metrics/{target}", get(metrics))
33        .route("/contexts", get(contexts))
34}
35
36async fn metrics(
37    State(state): State<AppState>, Path(target): Path<String>,
38) -> Result<Json<capture::LaneView>, StatusCode> {
39    let Some(target) = capture::Target::parse(&target) else {
40        return Err(StatusCode::BAD_REQUEST);
41    };
42    let Some(now) = capture::EpochSeconds::now() else {
43        return Err(StatusCode::INTERNAL_SERVER_ERROR);
44    };
45    Ok(Json(capture::LaneView {
46        now,
47        contexts: state.recorder.contexts(target),
48    }))
49}
50
51/// The `n` query parameter of `GET /contexts`.
52#[derive(Debug, Deserialize)]
53struct ContextQuery {
54    /// How many contexts to serve.
55    n: usize,
56}
57
58/// Serve `n` contexts from the shared pool, encoded with the binary codec so non-UTF-8 names and tags
59/// round-trip. Rejects an out-of-range `n` with 400 and a pool read error with 500.
60async fn contexts(State(state): State<AppState>, Query(query): Query<ContextQuery>) -> Result<Vec<u8>, StatusCode> {
61    if query.n == 0 || query.n > MAX_CONTEXTS_PER_REQUEST {
62        return Err(StatusCode::BAD_REQUEST);
63    }
64    let mut rng = UnwrapErr(AntithesisRng);
65    let contexts = state
66        .pool
67        .serve(query.n, &mut rng)
68        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
69    assert_reachable!("context source served a request", &json!({ "n": query.n }));
70    Ok(encode_response(&contexts))
71}
