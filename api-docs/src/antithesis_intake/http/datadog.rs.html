1//! Datadog-compatible HTTP intake routes.
2//!
3//! This module owns the public routes that Datadog Agent and ADP send to:
4//!
5//! - `POST /api/v2/series`: accepts metric series payloads, records payload
6//!   shape assertions, and captures scalar metric contexts.
7//! - `POST /api/intake/metrics/v3/series`: accepts the Datadog Agent's v3 native
8//!   series payloads (dictionary + delta encoded columnar protobuf) and captures
9//!   scalar metric contexts.
10//! - `POST /api/beta/sketches`: accepts distribution sketch payloads and
11//!   captures sketch contexts.
12//! - `POST /api/v1/events_batch`: accepts protobuf event batches.
13//! - `POST /api/v1/events`: accepts JSON event intake payloads and rejects
14//!   malformed bodies.
15//! - `POST /intake/`: accepts the shared JSON intake endpoint and ignores
16//!   non-event bodies.
17//! - `POST /api/v1/check_run`: accepts service check payloads.
18//! - `GET /api/v1/validate`: accepts Datadog Agent connectivity validation.
19
20mod events;
21mod metrics;
22mod service_checks;
23
24use axum::{
25    extract::DefaultBodyLimit,
26    http::StatusCode,
27    middleware::from_fn,
28    routing::{get, post},
29    Router,
30};
31use tower::ServiceBuilder;
32use tower_http::{decompression::RequestDecompressionLayer, limit::RequestBodyLimitLayer};
33
34use super::{middleware::measure_compressed_size, state::AppState, MAX_COMPRESSED_BODY_BYTES};
35
36pub(super) fn routes() -> Router<AppState> {
37    Router::new()
38        .merge(measured_routes())
39        .merge(decoded_payload_routes())
40        .route("/api/v1/validate", get(|| async { StatusCode::OK }))
41}
42
43/// Routes that fire payload assertions. `measure_compressed_size` runs outermost so Pyld05/Pyld06/Pyld22
44/// and the pre-decompression Content-Encoding are captured before `RequestDecompressionLayer` consumes
45/// them, then the default body limit is lifted so the middleware cap is the backstop. `/api/v2/series`,
46/// the v3 native series endpoint, and `/api/beta/sketches` all assert their envelope and byte properties,
47/// so all three share this measured stack.
48fn measured_routes() -> Router<AppState> {
49    let layers = ServiceBuilder::new()
50        .layer(from_fn(measure_compressed_size))
51        .layer(RequestDecompressionLayer::new().pass_through_unaccepted(true))
52        .layer(DefaultBodyLimit::disable());
53    Router::new()
54        .route("/api/v2/series", post(metrics::handle_series))
55        .route("/api/intake/metrics/v3/series", post(metrics::handle_series_v3))
56        .route("/api/beta/sketches", post(metrics::handle_sketches))
57        .layer(layers)
58}
59
60/// Routes that decompress and parse a body without recording `Measurements`. One shared stack
61/// caps the wire body with `RequestBodyLimitLayer` as it streams. Decompression follows. Each
62/// handler caps the decompressed body with `to_bytes`.
63fn decoded_payload_routes() -> Router<AppState> {
64    Router::new()
65        .route("/api/v1/events_batch", post(events::handle_events_batch))
66        .route("/api/v1/events", post(events::handle_events_v1))
67        .route("/intake/", post(events::handle_intake))
68        .route("/api/v1/check_run", post(service_checks::handle_check_run_v1))
69        .layer(
70            ServiceBuilder::new()
71                .layer(RequestBodyLimitLayer::new(MAX_COMPRESSED_BODY_BYTES))
72                .layer(RequestDecompressionLayer::new().pass_through_unaccepted(true)),
73        )
74}
