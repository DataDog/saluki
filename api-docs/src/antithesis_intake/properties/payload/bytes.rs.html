3use antithesis_sdk::prelude::*;
4use serde_json::json;
5
6use crate::capture::Target;
7
8/// Compressed body cap in bytes.
9const PYLD05_COMPRESSED_CAP_BYTES: u64 = 512_000;
10/// Uncompressed body cap in bytes.
11const PYLD06_UNCOMPRESSED_CAP_BYTES: u64 = 5 * 1024 * 1024;
12
13/// Pyld05 -- compressed body strictly below the cap.
14pub(crate) fn compressed_size(target: Target, compressed_len: u64) -> bool {
15    let ok = compressed_len < PYLD05_COMPRESSED_CAP_BYTES;
16    assert_always!(
17        ok,
18        "Pyld05.compressed_size",
19        &json!({ "lane": target, "compressed_bytes": compressed_len, "cap_bytes": PYLD05_COMPRESSED_CAP_BYTES })
20    );
21    ok
22}
23
24/// Pyld06 -- body at or below the uncompressed cap. The claim is unconditional: `uncompressed_len` is
25/// the body byte count the handler already holds, which for an identity body is its true uncompressed
26/// size, so the cap is asserted on every request regardless of whether the intake decompressed it.
27pub(crate) fn uncompressed_size(target: Target, uncompressed_len: u64) -> bool {
28    let ok = uncompressed_len <= PYLD06_UNCOMPRESSED_CAP_BYTES;
29    assert_always!(
30        ok,
31        "Pyld06.uncompressed_size",
32        &json!({ "lane": target, "uncompressed_bytes": uncompressed_len, "cap_bytes": PYLD06_UNCOMPRESSED_CAP_BYTES })
33    );
34    ok
35}
36
37/// Pyld05 -- a compressed body that overran the intake's buffering cap. The cap sits far above the
38/// spec cap, so reaching here is the violation; the exact size is unknown and the detail is a floor.
39pub(crate) fn compressed_size_over_cap(target: Target, cap_bytes: u64) {
40    assert_always!(
41        false,
42        "Pyld05.compressed_size",
43        &json!({ "lane": target, "compressed_bytes_at_least": cap_bytes, "cap_bytes": PYLD05_COMPRESSED_CAP_BYTES })
44    );
45}
46
47/// Pyld06 -- a decompressed body that overran the intake's buffering cap. As with Pyld05 the cap sits
48/// far above the spec cap, so the detail is a floor rather than the true size.
49pub(crate) fn uncompressed_size_over_cap(target: Target, cap_bytes: u64) {
50    assert_always!(
51        false,
52        "Pyld06.uncompressed_size",
53        &json!({ "lane": target, "uncompressed_bytes_at_least": cap_bytes, "cap_bytes": PYLD06_UNCOMPRESSED_CAP_BYTES })
54    );
55}
56
57/// Pyld22 -- Content-Length absent or equal to the wire body length.
58pub(crate) fn content_length(target: Target, declared: Option<u64>, body_len: u64) {
59    let ok = declared.is_none_or(|d| d == body_len);
60    assert_always!(
61        ok,
62        "Pyld22.content_length",
63        &json!({ "lane": target, "compressed_bytes": body_len, "declared_content_length": declared })
64    );
65}

67#[cfg(test)]
68mod tests {
69    use super::{uncompressed_size, PYLD06_UNCOMPRESSED_CAP_BYTES};
70    use crate::capture::Target;
71
72    // Pyld06 is an unconditional claim: an over-cap body fails whether or not the intake
73    // decompressed it. The identity-encoded body carries its true uncompressed length too.
74    #[test]
75    fn pyld06_rejects_oversized_body_regardless_of_encoding() {
76        assert!(!uncompressed_size(Target::Agent, PYLD06_UNCOMPRESSED_CAP_BYTES + 1));
77        assert!(uncompressed_size(Target::Agent, PYLD06_UNCOMPRESSED_CAP_BYTES));
78    }
