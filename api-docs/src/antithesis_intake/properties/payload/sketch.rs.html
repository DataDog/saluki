1//! Sketch (`/api/beta/sketches`) property checks.
2
3use antithesis_sdk::prelude::*;
4use datadog_protos::metrics::sketch_payload::Sketch;
5use serde_json::json;
6
7use super::constants::{MAX_TAG_LENGTH_BYTES, MAX_TAG_SET_SIZE_BYTES};
8use crate::capture::{Target, MAX_HOST_NAME_LEN, MAX_METRIC_NAME_LEN, MAX_TAG_COUNT};
9
10/// Pyld07-sketch -- the sketch body cleanly decodes. A correct Agent produces a payload the intake
11/// accepts, so malformed wire and a non-UTF-8 non-tag field fail. A feral tag is venerable Agent
12/// behavior, so it is a faithful reject.
13pub(crate) fn decode_faithful(target: Target, cleanly_decoded: bool, outcome: &str, body_len: usize) {
14    assert_always!(
15        cleanly_decoded,
16        "Pyld07-sketch.decode_success",
17        &json!({ "lane": target, "outcome": outcome, "body_len": body_len })
18    );
19}
20
21/// Fire the per-sketch shape assertions on a decoded `Sketch`. Same doctrine as the series shape checks:
22/// a correct Agent stays behind every intake drop or cap threshold. Pyld20/21 do not apply, since the
23/// intake does not NaN-filter sketch summaries and the summary carries no scalar point value.
24pub(crate) fn shape(target: Target, sketch: &Sketch) {
25    let name = sketch.metric();
26    assert_always!(
27        !name.is_empty(),
28        "Pyld09.metric_non_empty",
29        &json!({ "lane": target, "metric": name })
30    );
31    assert_always!(
32        name.len() <= MAX_METRIC_NAME_LEN,
33        "Pyld10.metric_name_length",
34        &json!({ "lane": target, "metric": name, "len": name.len() })
35    );
36    assert_always!(
37        name.bytes().any(|b| b.is_ascii_alphabetic()),
38        "Pyld11.metric_name_alphabetic",
39        &json!({ "lane": target, "metric": name })
40    );
41    assert_always!(
42        sketch.tags.len() <= MAX_TAG_COUNT,
43        "Pyld13.tag_count",
44        &json!({ "lane": target, "metric": name, "tags": sketch.tags.len() })
45    );
46    assert_always!(
47        sketch.host().len() <= MAX_HOST_NAME_LEN,
48        "Pyld19.host_name_length",
49        &json!({ "lane": target, "metric": name, "host_len": sketch.host().len() })
50    );
51    let tag_over = sketch
52        .tags
53        .iter()
54        .map(String::len)
55        .max()
56        .filter(|&len| len > MAX_TAG_LENGTH_BYTES);
57    assert_always!(
58        tag_over.is_none(),
59        "Pyld23.tag_length",
60        &json!({ "lane": target, "metric": name, "observed": tag_over })
61    );
62    let tagset_bytes: usize = sketch.tags.iter().map(String::len).sum();
63    assert_always!(
64        tagset_bytes <= MAX_TAG_SET_SIZE_BYTES,
65        "Pyld24.tag_set_size",
66        &json!({ "lane": target, "metric": name, "bytes": tagset_bytes })
67    );
68    let empty = sketch.dogsketches.is_empty() && sketch.distributions.is_empty();
69    assert_always!(
70        !empty,
71        "Pyld25-sketch.points_non_empty",
72        &json!({ "lane": target, "metric": name })
73    );
74}
