3use antithesis_sdk::prelude::*;
4use datadog_protos::metrics::MetricPayload;
5use serde_json::json;

7use super::constants::MAX_POINTS_PER_PAYLOAD;

10/// Pyld07-v2 -- the body cleanly decodes. A correct Agent produces a payload the intake accepts, so any
11/// reject fails: malformed wire, a non-UTF-8 non-tag field, or a non-UTF-8 tag from a non-`datadog-agent`
12/// source. A feral tag from the `datadog-agent` source is coerced and decodes to `Ok`, so it never
13/// reaches this as a rejection.
14pub(crate) fn decode_production_faithful(
15    target: Target, production_faithful: bool, outcome: &str, body_len: usize, decompression_applied: bool,
16) {
17    assert_always!(
18        production_faithful,
19        "Pyld07.decode_success",
20        &json!({ "lane": target, "outcome": outcome, "body_len": body_len, "decompression_applied": decompression_applied })
21    );
22}
23
24/// Pyld07-v3 -- the v3 body cleanly decodes. Same doctrine as Pyld07-v2: any reject fails, since a
25/// correct Agent produces a payload the intake accepts. v3 tags are universally coerced, so a feral tag
26/// decodes to `Ok` and only a non-UTF-8 non-tag field or malformed wire reaches this as a rejection.
27pub(crate) fn decode_v3(target: Target, cleanly_decoded: bool, outcome: &str, body_len: usize) {
28    assert_always!(
29        cleanly_decoded,
30        "Pyld07-v3.decode_success",
31        &json!({ "lane": target, "outcome": outcome, "body_len": body_len })
32    );
33}

35/// Pyld08 -- total points across the payload at or below the cap.
36pub(crate) fn point_count(target: Target, payload: &MetricPayload) {
37    let total: usize = payload.series.iter().map(|s| s.points.len()).sum();
38    let over = (total > MAX_POINTS_PER_PAYLOAD).then_some(total);
39    assert_always!(
40        over.is_none(),
41        "Pyld08.payload_point_count",
42        &json!({ "lane": target, "max_points": MAX_POINTS_PER_PAYLOAD, "observed": over })
43    );
44}
