111/// Returns whether to keep a trace, based on its ID and a sampling rate.
112///
113/// This assumes trace IDs are nearly uniformly distributed.
114pub fn sample_by_rate(trace_id: u64, rate: f64) -> bool {
115    if rate < 1.0 {
116        trace_id.wrapping_mul(SAMPLER_HASHER) < (rate * MAX_TRACE_ID_FLOAT) as u64
117    } else {
118        true
119    }
120}

122pub fn get_trace_env(trace: &Trace, root_span_idx: usize) -> Option<&MetaString> {
123    // logic taken from here: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/traceutil/trace.go#L19-L20
124    let env = trace
125        .spans()
126        .get(root_span_idx)
127        .and_then(|span| span.attributes.get("env").and_then(AttributeValue::as_string));
128    if let Some(env) = env {
129        return Some(env);
130    }
131    for span in trace.spans().iter() {
132        if let Some(env) = span.attributes.get("env").and_then(AttributeValue::as_string) {
133            return Some(env);
134        }
135    }
136    // Fall back to the payload-level env (set from tracer payload headers or OTLP resource attributes).
137    if !trace.payload.env.is_empty() {
138        return Some(&trace.payload.env);
139    }
140    None
141}

143/// Returns the hostname a trace reports, if any.
144///
145/// The payload hostname wins. The root span's `_dd.hostname` applies only when the payload hostname is empty, which is
146/// how older tracers report it. Every consumer of a trace's hostname uses this function, so they all select the same
147/// root span.
148pub fn get_tracer_hostname(trace: &Trace) -> Option<&MetaString> {
149    if !trace.payload.hostname.is_empty() {
150        return Some(&trace.payload.hostname);
151    }
152    get_root_span_index(trace.spans())
153        .and_then(|idx| trace.spans().get(idx))
154        .and_then(|span| span.attributes.get("_dd.hostname").and_then(AttributeValue::as_string))
155        .filter(|hostname| !hostname.is_empty())
156}

158/// Finds the index of the root span within the given spans.
159///
160/// Scans backwards for the first span with a zero parent ID (clients often report the root last),
161/// falling back to whichever span claims a parent that is absent from the trace. Returns `None`
162/// only when `spans` is empty. The spans are never modified; the parent-claim map is scratch
163/// state local to this function.
164pub fn get_root_span_index(spans: &[Span]) -> Option<usize> {
165    if spans.is_empty() {
166        return None;
167    }
168
169    let length = spans.len();
170    let mut parent_id_to_child: FastHashMap<u64, usize> = FastHashMap::default();
171
172    for i in 0..length {
173        let j = length - 1 - i;
174        if spans[j].parent_id() == 0 {
175            return Some(j);
176        }
177        parent_id_to_child.insert(spans[j].parent_id(), j);
178    }
179
180    for span in spans.iter() {
181        parent_id_to_child.remove(&span.span_id());
182    }
183
184    if parent_id_to_child.len() != 1 {
185        debug!("Didn't reliably find the root span for a trace");
186    }
187
188    if let Some((_, child_idx)) = parent_id_to_child.iter().next() {
189        return Some(*child_idx);
190    }
191
192    Some(length - 1)
193}

195/// Computes top-level marks from the span relationships themselves, marking spans in-place.
196///
197/// A span is top-level when it is a root span, its parent is missing from the trace (the entry
198/// point of a distributed fragment), or its parent is in a different service (the entry point of
199/// this service's subtree). Marks are only ever set, never cleared.
200pub fn compute_top_level(spans: &mut [Span]) {
201    let mut span_id_to_index: FastHashMap<u64, usize> = FastHashMap::default();
202    for (i, span) in spans.iter().enumerate() {
203        span_id_to_index.insert(span.span_id(), i);
204    }
205
206    // Classify while immutably borrowed, then mark in a second pass.
207    let mut top_level_indices = Vec::with_capacity(spans.len());
208    for (i, span) in spans.iter().enumerate() {
209        let parent_id = span.parent_id();
210        if parent_id == 0 {
211            top_level_indices.push(i);
212        } else if let Some(&parent_idx) = span_id_to_index.get(&parent_id) {
213            if spans[parent_idx].service() != span.service() {
214                top_level_indices.push(i);
215            }
216        } else {
217            top_level_indices.push(i);
218        }
219    }
220
221    for i in top_level_indices {
222        spans[i]
223            .attributes
224            .insert(MetaString::from_static(TOP_LEVEL_KEY), AttributeValue::Float(1.0));
225    }
226}

234    fn span(span_id: u64, parent_id: u64, hostname: Option<&str>) -> Span {
235        let mut attributes = FastHashMap::default();
236        if let Some(hostname) = hostname {
237            attributes.insert(
238                MetaString::from("_dd.hostname"),
239                AttributeValue::String(MetaString::from(hostname)),
240            );
241        }
242        Span::new("svc", "op", "res", "web", span_id, parent_id, 0, 1, 0).with_attributes(attributes)
243    }

245    fn tracer_hostname(trace: &Trace) -> Option<&str> {
246        get_tracer_hostname(trace).map(|hostname| hostname.as_ref())
247    }
