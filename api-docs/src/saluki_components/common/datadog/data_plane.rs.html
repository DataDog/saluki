1//! The ADP-specific `data_plane` configuration section.
2//!
3//! These keys live under the Datadog Agent's `data_plane` section, so a consumer reads them through
4//! a nested struct rather than a flattened one. Each consumer composes only the sub-sections it
5//! actually reads: a struct that also accepted unrelated `data_plane` keys would silently change
6//! shape when one of them is set, which the configuration smoke tests treat as a defect.
7
8use facet::Facet;
9use serde::Deserialize;
10
11/// The `data_plane` keys read by a payload encoder.
12#[derive(Clone, Debug, Default, Deserialize, Facet)]
13#[cfg_attr(test, derive(PartialEq, serde::Serialize))]
14pub(crate) struct EncoderDataPlaneConfiguration {
15    /// ADP-specific zstd compression level, taking precedence over the Core Agent's
16    /// `serializer_zstd_compressor_level`.
17    ///
18    /// Defaults to unset, in which case
19    /// [`resolve_zstd_compressor_level`](super::resolve_zstd_compressor_level) picks the level.
20    #[serde(default)]
21    pub(crate) serializer_zstd_compressor_level: Option<i32>,
22}
