22/// Builds a middleware function that will update the request's URI to use the resolved endpoint's URI, and add the API
23/// key as a header.
24///
25/// The request's URI must already contain a relative path component. The resolved endpoint's URI replaces any
26/// existing scheme, host, and port components. If the endpoint has a base path, such as `https://proxy.example.com/prefix`,
27/// the request's path is appended to it, which gives `/prefix/api/v2/series`. A request that goes to a derived logs or
28/// traces intake host doesn't get the base path.
29pub fn for_resolved_endpoint<B>(endpoint: ResolvedEndpoint) -> impl Fn(Request<B>) -> Request<B> + Clone {
30    let new_uri_authority = Authority::try_from(endpoint.endpoint().authority())
31        .expect("should not fail to construct new endpoint authority");
32    let new_uri_scheme =
33        Scheme::try_from(endpoint.endpoint().scheme()).expect("should not fail to construct new endpoint scheme");
34    let base_path = endpoint_base_path(endpoint.endpoint()).to_string();
35    move |mut request| {
36        let path_and_query = request.uri().path_and_query().expect("request path must exist").clone();
37
38        // Select the authority from the request path alone. A derived intake authority is used only for official
39        // Datadog endpoints.
40        let intake_authority = if path_and_query.as_str().starts_with("/api/v2/logs") {
41            endpoint.logs_authority()
42        } else if path_and_query.as_str().starts_with("/api/v0.2/traces")
43            || path_and_query.as_str().starts_with("/api/v0.2/stats")
44        {
45            endpoint.traces_authority()
46        } else {
47            None
48        };
49
50        // Prepend the base path only when the request goes to the endpoint's own authority. The base path belongs to
51        // the configured endpoint, not to a derived intake host.
52        let (authority, path_and_query) = match intake_authority {
53            Some(intake_authority) => (intake_authority.clone(), path_and_query),
54            None if base_path.is_empty() => (new_uri_authority.clone(), path_and_query),
55            None => (
56                new_uri_authority.clone(),
57                PathAndQuery::try_from(format!("{}{}", base_path, path_and_query.as_str()))
58                    .expect("should not fail to construct new path and query"),
59            ),
60        };
61
62        // Build an updated URI from the endpoint's scheme and the selected authority, plus the path and query.
63        let new_uri = Uri::builder()
64            .scheme(new_uri_scheme.clone())
65            .authority(authority)
66            .path_and_query(path_and_query)
67            .build()
68            .expect("should not fail to construct new URI");
69        let api_key = endpoint.api_key();
70        let api_key_value = HeaderValue::from_str(&api_key).expect("should not fail to construct API key header value");
71        *request.uri_mut() = new_uri;
72
73        // Add the API key as a header.
74        request
75            .headers_mut()
76            .insert(DD_API_KEY_HEADER.clone(), api_key_value.clone());
77
78        request
79    }
80}

82/// Builds a middleware function that optionally signals backend support for arbitrary tag values.
83///
84/// When enabled, adds `Allow-Arbitrary-Tag-Value: true` to every outbound request. This mirrors the Datadog Agent's
85/// forwarder behavior for `allow_arbitrary_tags`; it does not perform local tag validation.
86pub fn with_allow_arbitrary_tags<B>(enabled: bool) -> impl Fn(Request<B>) -> Request<B> + Clone {
87    move |mut request| {
88        if enabled {
89            request.headers_mut().insert(
90                ALLOW_ARBITRARY_TAG_VALUE_HEADER.clone(),
91                HeaderValue::from_static("true"),
92            );
93        }
94
95        request
96    }
97}

99/// Returns the full name of the application, formatted for a request header: `Example App` gives `example-app`.
100pub(crate) fn formatted_app_name(app_details: &saluki_metadata::AppDetails) -> String {
101    app_details.full_name().replace([' ', '_'], "-").to_lowercase()
102}

104/// Builds a middleware function that adds request headers indicating the version of the data plane making the request.
105///
106/// Adds the `dd-agent-version` header with the version of the data plane (`x.y.z`), and the `User-Agent` header with
107/// the name of data plane and the version, such as `agent-data-plane/x.y.z`.`
108pub fn with_version_info<B>() -> impl Fn(Request<B>) -> Request<B> + Clone {
109    let app_details = saluki_metadata::get_app_details();
110    let formatted_full_name = formatted_app_name(app_details);
111
112    let agent_version_header_value = HeaderValue::from_static(app_details.version().raw());
113    let raw_user_agent_header_value = format!("{}/{}", formatted_full_name, app_details.version().raw());
114    let user_agent_header_value = HeaderValue::from_maybe_shared(raw_user_agent_header_value)
115        .expect("should not fail to construct User-Agent header value");
116
117    move |mut request| {
118        request
119            .headers_mut()
120            .insert(DD_AGENT_VERSION_HEADER.clone(), agent_version_header_value.clone());
121        request
122            .headers_mut()
123            .insert(http::header::USER_AGENT, user_agent_header_value.clone());
124        request
125    }
126}

128/// A source of the current time, in milliseconds since the Unix epoch.
129pub(crate) type UnixMillisClock = Arc<dyn Fn() -> u64 + Send + Sync>;
130
131/// Returns a clock that reads the system time.
132pub(crate) fn system_clock() -> UnixMillisClock {
133    Arc::new(|| {
134        SystemTime::now()
135            .duration_since(UNIX_EPOCH)
136            .map(|elapsed| elapsed.as_millis() as u64)
137            .unwrap_or_default()
138    })
139}

141/// Builds a middleware function that stamps logs requests with the time they're sent.
142///
143/// Adds the `dd-current-timestamp` header, read from `clock`, to requests for the logs intake paths (`/api/v2/logs`
144/// and `/v1/input`). The path must be the route before endpoint mapping, which can add a base path. Any existing value
145/// is replaced, so a retried request carries the time of its latest attempt rather than its first.
146pub(crate) fn with_logs_current_timestamp<B>(clock: UnixMillisClock) -> impl Fn(Request<B>) -> Request<B> + Clone {
147    move |mut request| {
148        if LOGS_PATHS.contains(&request.uri().path()) {
149            request
150                .headers_mut()
151                .insert(DD_CURRENT_TIMESTAMP_HEADER.clone(), HeaderValue::from(clock()));
152        }
153
154        request
155    }
156}

193        // A resolved Datadog endpoint carries pre-computed logs/traces intake authorities (its host gains

398    fn current_timestamp(request: &Request<()>) -> Option<&str> {
399        request
400            .headers()
401            .get("dd-current-timestamp")
402            .map(|value| value.to_str().expect("header should be ASCII"))
403    }
