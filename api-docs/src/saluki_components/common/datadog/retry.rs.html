18/// Datadog Agent-specific forwarder retry configuration.
19#[derive(Clone)]
20#[cfg_attr(test, derive(Debug, PartialEq))]
21pub struct RetryConfiguration {
22    /// The minimum backoff factor to use when retrying requests.
23    ///
24    /// Controls the interval range that a calculated backoff duration can fall within, such that with a minimum
25    /// backoff factor of 2.0, calculated backoff durations will fall between `d/2` and `d`, where `d` is the calculated
26    /// backoff duration using a purely exponential growth strategy.
27    backoff_factor: f64,
28
29    /// The base growth rate of the backoff duration when retrying requests, in seconds.
30    backoff_base: f64,
31
32    /// The upper bound of the backoff duration when retrying requests, in seconds.
33    backoff_max: f64,
34
35    /// The amount to decrease the error count by when a request is successful.
36    ///
37    /// This essentially controls how quickly we forget about the number of previous errors when calculating the next
38    /// backoff duration for a request that must be retried.
39    recovery_error_decrease_factor: u32,
40
41    /// Whether or not a successful request should completely reset the error count.
42    recovery_reset: bool,
43
44    /// The maximum in-memory size of the retry queue, in bytes.
45    queue_max_size_bytes: u64,
46
47    /// The maximum size of the retry queue on disk, in bytes.
48    ///
49    /// A value of `0` disables disk persistence.
50    storage_max_size_bytes: u64,
51
52    /// The ratio of in-memory retry queue bytes to flush to disk when the queue is full.
53    ///
54    /// When disk persistence is enabled and the in-memory retry queue does not have enough room for a new transaction,
55    /// this controls how much in-memory data ADP moves to disk. For example, `0.5` moves at least half of the configured
56    /// in-memory retry queue size to disk during each overflow. If set to `0`, ADP moves only enough old transactions to
57    /// disk to make room for the new transaction.
58    flush_to_disk_mem_ratio: f64,
59
60    /// The path to the directory where the retry queue will be stored on disk.
61    ///
62    /// This is empty when neither `forwarder_storage_path` nor `run_path` is configured, in which case
63    /// no default path can be calculated.
64    storage_path: PathBuf,
65
66    /// The maximum disk usage ratio for storing transactions on disk.
67    ///
68    /// `0.8` means the Agent can store transactions on disk until `forwarder_storage_max_size_in_bytes`
69    /// is reached or when the disk mount for `forwarder_storage_path` exceeds 80% of the disk capacity,
70    /// whichever is lower.
71    storage_max_disk_ratio: f64,
72
73    /// Maximum age in days for retry files on disk before they are deleted at startup.
74    ///
75    /// When disk persistence is enabled, ADP removes any `retry-*.json` files in the
76    /// per-queue subdirectory of the storage path that are older than this many days
77    /// each time it starts. This prevents unbounded disk growth from stale retry data left
78    /// behind after long outages.
79    outdated_file_in_days: u32,
80
81    /// The time window used to estimate retry queue capacity, in seconds.
82    ///
83    /// ADP records incoming transaction payload bytes over this window and uses that rate to estimate how many seconds
84    /// of data the retry queue can buffer. Values below 10 seconds are clamped to 10 seconds, matching the fixed retry
85    /// queue capacity bucket size.
86    capacity_time_interval_secs: u64,
87}

89impl RetryConfiguration {
90    /// Creates a new `RetryConfiguration` from the resolved forwarder configuration.
91    ///
92    /// When no retry-queue storage path is configured, one is derived from `run_path`, matching the
93    /// Datadog Agent's own layout. Both may be absent, in which case there is no storage path and
94    /// disk persistence cannot be used.
95    pub(super) fn from_configuration(forwarder: &shared::Forwarder, config: &GenericConfiguration) -> Self {
96        Self {
97            backoff_factor: forwarder.backoff_factor,
98            backoff_base: forwarder.backoff_base,
99            backoff_max: forwarder.backoff_max,
100            recovery_error_decrease_factor: forwarder.recovery_interval,
101            recovery_reset: forwarder.recovery_reset,
102            queue_max_size_bytes: forwarder.effective_retry_queue_max_size_bytes(),
103            storage_max_size_bytes: forwarder.storage_max_size_in_bytes,
104            flush_to_disk_mem_ratio: forwarder.flush_to_disk_mem_ratio,
105            storage_path: resolve_storage_path(&forwarder.storage_path, config),
106            storage_max_disk_ratio: forwarder.storage_max_disk_ratio,
107            outdated_file_in_days: forwarder.outdated_file_in_days,
108            capacity_time_interval_secs: forwarder
109                .retry_queue_capacity_time_interval_sec
110                .max(RETRY_QUEUE_CAPACITY_MIN_HISTORY_DURATION_SECS),
111        }
112    }

149    /// Creates a new [`DefaultHttpRetryPolicy`] based on the forwarder configuration.
150    ///
151    /// If a [`GenericConfiguration`] is supplied, the policy captures it and checks whether
152    /// secrets management is active on every 403 Forbidden response. This allows the retry gate to
153    /// pick up runtime changes pushed via the config stream without rebuilding the service. When no
154    /// configuration is supplied, 403 responses retain their default non-retriable behavior.
155    pub fn to_default_http_retry_policy<B: 'static>(
156        &self, live_config: Option<GenericConfiguration>,
157    ) -> DefaultHttpRetryPolicy<B> {
158        let retry_backoff = ExponentialBackoff::with_jitter(
159            Duration::from_secs_f64(self.backoff_base),
160            Duration::from_secs_f64(self.backoff_max),
161            self.backoff_factor,
162        );
163
164        let classifier = if let Some(config) = live_config {
165            let gate: HttpRetryPredicate<B> =
166                Arc::new(move |response| response.status() == StatusCode::FORBIDDEN && secrets_in_use(&config));
167            StandardHttpClassifier::new().with_predicate(gate)
168        } else {
169            StandardHttpClassifier::new()
170        };
171
172        let recovery_error_decrease_factor = (!self.recovery_reset).then_some(self.recovery_error_decrease_factor);
173        DefaultHttpRetryPolicy::with_backoff_and_classifier(retry_backoff, classifier)
174            .with_recovery_error_decrease_factor(recovery_error_decrease_factor)
175    }
176}

178/// Resolves the directory where retry payloads are persisted to disk.
179///
180/// A configured `forwarder_storage_path` is used as-is. Otherwise the path is derived from
181/// `run_path`, which has no typed home: its schema default is an unresolved placeholder, so
182/// promoting it to the typed model would put that placeholder in the model.
183///
184/// TODO: read `run_path` from typed configuration once the placeholder default is resolved.
185fn resolve_storage_path(configured: &Path, config: &GenericConfiguration) -> PathBuf {
186    if configured.parent().is_some() {
187        return configured.to_path_buf();
188    }
189
190    match config.try_get_typed::<PathBuf>("run_path") {
191        Ok(Some(mut run_path)) => {
192            run_path.push(RETRY_TXN_DIR);
193            run_path
194        }
195        Ok(None) => {
196            debug!("`forwarder_storage_path` and `run_path` were empty. Cannot calculate default storage path for forwarder.");
197            PathBuf::new()
198        }
199        Err(e) => {
200            debug!(error = %e, "Failed to read `run_path` from configuration. Cannot calculate default storage path for forwarder.");
201            PathBuf::new()
202        }
203    }
204}

206fn secrets_in_use(config: &GenericConfiguration) -> bool {
207    matches!(config.try_get_typed::<u64>("secret_refresh_on_api_key_failure_interval"), Ok(Some(value)) if value > 0)
208        || matches!(config.try_get_typed::<String>("secret_backend_command"), Ok(Some(value)) if !value.trim().is_empty())
209}

377    async fn policy_403_gate_reflects_dynamic_secrets_config_change() {
378        use std::time::Duration as StdDuration;
379
380        use saluki_config::dynamic::{ConfigSetting, ConfigUpdate};
381
382        let (config, sender) = ConfigurationLoader::for_tests(Some(json!({})), None, true).await;
383        let sender = sender.expect("dynamic configuration sender should be present");
384
385        // Apply an empty initial snapshot and wait for readiness.
386        sender
387            .send(ConfigUpdate::snapshot([]))
388            .await
389            .expect("should send initial snapshot");
390        config.ready().await;
391
392        let retry_config = test_retry_config().await;
393        let mut policy = retry_config.to_default_http_retry_policy(Some(config.clone()));
394
395        // Before secrets are configured, 403 must not be retried.
396        assert!(!would_retry(&mut policy, ok_response(StatusCode::FORBIDDEN)));
397
398        // Push a config update that enables secrets management.
399        let mut watcher = config.watch_for_updates("secret_backend_command");
400        sender
401            .send(ConfigUpdate::Partial(ConfigSetting::explicit(
402                "secret_backend_command",
403                json!("/bin/true"),
404            )))
405            .await
406            .expect("should send partial update");
407
408        tokio::time::timeout(StdDuration::from_secs(2), watcher.changed::<String>())
409            .await
410            .expect("timed out waiting for secret_backend_command update");
411
412        // The same policy instance must now retry 403 because the predicate reads the live cached secrets flag.
413        assert!(would_retry(&mut policy, ok_response(StatusCode::FORBIDDEN)));
414    }
