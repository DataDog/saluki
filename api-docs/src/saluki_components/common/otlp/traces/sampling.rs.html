1//! Head-based sampling rate decoding for OTLP spans.
2
3/// Number of rejected values out of which a `th` threshold is expressed (2^56).
4const THRESHOLD_RANGE: u64 = 1 << 56;
5
6/// Maximum number of hex digits in a `th` value.
7const MAX_THRESHOLD_HEX_DIGITS: usize = 14;
8
9/// Number of hex digits in an `rv` value.
10const RANDOM_VALUE_HEX_DIGITS: usize = 14;
11
12/// Smallest `p` value that means "not sampled" instead of a probability.
13const P_VALUE_NOT_SAMPLED: u64 = 63;
14
15/// Maximum accepted size of a whole tracestate, in bytes.
16const MAX_TRACESTATE_LEN: usize = 1024;
17
18/// Maximum accepted size of the `ot` vendor value, in bytes.
19const MAX_OT_VALUE_LEN: usize = 256;
20
21/// A tracestate with more than this many list members is rejected.
22const MAX_TRACESTATE_MEMBERS: usize = 32;
23
24/// Maximum accepted size of a tracestate key, in bytes.
25const MAX_KEY_LEN: usize = 256;
26
27/// Maximum accepted size of the tenant part of a multi-tenant key, in bytes.
28const MAX_TENANT_LEN: usize = 241;
29
30/// Maximum accepted size of the system part of a multi-tenant key, in bytes.
31const MAX_SYSTEM_LEN: usize = 14;
32
33/// Decodes the head-based sampling probability from a raw W3C tracestate.
34///
35/// Two encodings in the `ot` vendor entry are understood:
36///
37/// - `th:<hex>`: a rejection threshold of up to 14 hex digits with trailing zeros omitted. The probability is
38///   `1 - threshold / 2^56`, so `th:8` is 0.5 and `th:0` is 1.0.
39/// - `p:<n>`: a power-of-two probability of `2^-n`, so `p:0` is 1.0 and `p:4` is 0.0625. Values of 63 or more mean
40///   "not sampled" and carry no probability.
41///
42/// `th` takes precedence over `p`. Returns `None` when the tracestate has no probability, and also when any part of it
43/// is malformed: a tracestate that fails validation is rejected as a whole, even if its `ot` entry looks usable.
44pub(super) fn sample_rate_from_tracestate(tracestate: &str) -> Option<f64> {
45    if tracestate.is_empty() || tracestate.len() > MAX_TRACESTATE_LEN || !is_valid_tracestate(tracestate) {
46        return None;
47    }
48
49    // When the `ot` vendor is listed more than once, the last entry wins.
50    let mut ot_value = None;
51    let mut members = 0;
52    let mut rest = tracestate;
53    while !rest.is_empty() {
54        members += 1;
55        if members > MAX_TRACESTATE_MEMBERS {
56            return None;
57        }
58
59        let (member, tail) = rest.split_once(',').unwrap_or((rest, ""));
60        rest = tail;
61
62        let member = member.trim_matches([' ', '\t']);
63        if member.is_empty() {
64            continue;
65        }
66        let (key, value) = member.split_once('=')?;
67        if key.len() > MAX_KEY_LEN {
68            return None;
69        }
70        if let Some((tenant, system)) = key.split_once('@') {
71            if tenant.len() > MAX_TENANT_LEN || system.len() > MAX_SYSTEM_LEN {
72                return None;
73            }
74        }
75        if key == "ot" {
76            ot_value = Some(parse_ot_value(value)?);
77        }
78    }
79
80    let ot = ot_value?;
81    if let Some(threshold) = ot.threshold {
82        return Some((THRESHOLD_RANGE - threshold) as f64 / THRESHOLD_RANGE as f64);
83    }
84    match ot.p_value?.parse::<u64>() {
85        Ok(0) => Some(1.0),
86        Ok(p) if p < P_VALUE_NOT_SAMPLED => Some(1.0 / (1u64 << p) as f64),
87        _ => None,
88    }
89}

91/// The parts of the `ot` vendor entry that carry a sampling probability.
92struct OtValue<'a> {
93    /// Rejection threshold out of 2^56, from `th`.
94    threshold: Option<u64>,
95    /// Raw value of the first `p` field.
96    p_value: Option<&'a str>,
97}
98
99/// Parses and validates the value of the `ot` vendor entry, a `;`-separated list of `key:value` fields.
100fn parse_ot_value(value: &str) -> Option<OtValue<'_>> {
101    if value.len() > MAX_OT_VALUE_LEN || value.is_empty() || value.ends_with(';') {
102        return None;
103    }
104
105    let mut ot = OtValue {
106        threshold: None,
107        p_value: None,
108    };
109    for field in value.split(';') {
110        let (key, field_value) = field.split_once(':')?;
111        let mut key_bytes = key.bytes();
112        if !key_bytes.next().is_some_and(|b| b.is_ascii_lowercase())
113            || !key_bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit())
114            || !field_value
115                .bytes()
116                .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
117        {
118            return None;
119        }
120
121        match key {
122            "rv" => {
123                if field_value.len() != RANDOM_VALUE_HEX_DIGITS || !field_value.bytes().all(|b| b.is_ascii_hexdigit()) {
124                    return None;
125                }
126            }
127            "th" => {
128                let digits = field_value.len();
129                if digits == 0
130                    || digits > MAX_THRESHOLD_HEX_DIGITS
131                    || !field_value.bytes().all(|b| b.is_ascii_hexdigit())
132                {
133                    return None;
134                }
135                // The omitted trailing zeros are restored by shifting up to 14 digits.
136                let parsed = u64::from_str_radix(field_value, 16).ok()?;
137                ot.threshold = Some(parsed << (4 * (MAX_THRESHOLD_HEX_DIGITS - digits)));
138            }
139            "p" => {
140                ot.p_value.get_or_insert(field_value);
141            }
142            _ => {}
143        }
144    }
145    Some(ot)
146}

148/// Checks the tracestate list-member syntax: `key=value` members separated by commas, with optional surrounding
149/// whitespace and empty members allowed.
150fn is_valid_tracestate(tracestate: &str) -> bool {
151    tracestate.split(',').all(|member| {
152        let member = member.trim_matches([' ', '\t']);
153        if member.is_empty() {
154            return true;
155        }
156        match member.split_once('=') {
157            Some((key, value)) if !key.is_empty() => is_valid_key(key) && is_valid_value(value),
158            _ => false,
159        }
160    })
161}
162
163fn is_valid_key_char(b: u8) -> bool {
164    b.is_ascii_lowercase() || b.is_ascii_digit() || matches!(b, b'_' | b'-' | b'*' | b'/')
165}
166
167/// Either a simple key (`lcalpha` followed by key characters) or a multi-tenant key (`tenant@system`).
168fn is_valid_key(key: &str) -> bool {
169    match key.split_once('@') {
170        Some((tenant, system)) => {
171            let tenant = tenant.as_bytes();
172            let system = system.as_bytes();
173            !tenant.is_empty()
174                && !system.is_empty()
175                && (tenant[0].is_ascii_lowercase() || tenant[0].is_ascii_digit())
176                && system[0].is_ascii_lowercase()
177                && tenant[1..].iter().all(|&b| is_valid_key_char(b))
178                && system[1..].iter().all(|&b| is_valid_key_char(b))
179        }
180        None => {
181            let key = key.as_bytes();
182            key[0].is_ascii_lowercase() && key[1..].iter().all(|&b| is_valid_key_char(b))
183        }
184    }
185}
186
187/// A value is 1 to 256 printable ASCII characters, excluding `,` and `=`, and cannot end in a space.
188fn is_valid_value(value: &str) -> bool {
189    let bytes = value.as_bytes();
190    let is_non_blank = |b: u8| (0x21..=0x7e).contains(&b) && b != b',' && b != b'=';
191    match bytes.split_last() {
192        Some((&last, init)) => {
193            bytes.len() <= 256 && is_non_blank(last) && init.iter().all(|&b| b == b' ' || is_non_blank(b))
194        }
195        None => false,
196    }
197}

205    fn assert_rate(tracestate: &str, expected: Option<f64>) {
206        match (sample_rate_from_tracestate(tracestate), expected) {
207            (Some(got), Some(want)) => assert!(
208                (got - want).abs() < 1e-9,
209                "tracestate {tracestate:?}: expected {want}, got {got}"
210            ),
211            (got, want) => assert_eq!(got, want, "tracestate {tracestate:?}"),
212        }
213    }

215    #[test]
216    fn decodes_threshold_and_power_of_two_encodings() {
217        let cases: &[(&str, &str, Option<f64>)] = &[
218            ("50% sampling (th:8)", "ot=th:8", Some(0.5)),
219            ("100% sampling (th:0)", "ot=th:0", Some(1.0)),
220            ("p-encoding 50% (p:1)", "ot=p:1;r:1", Some(0.5)),
221            ("p-encoding 6.25% (p:4)", "ot=p:4;r:4", Some(0.0625)),
222            ("p-encoding 100% (p:0)", "ot=p:0", Some(1.0)),
223            ("th absent rv present", "ot=rv:abcdefabcdefab", None),
224            ("no tracestate", "", None),
225            ("non-ot vendor field only", "zz=vendorcontent", None),
226        ];
227        for (name, tracestate, expected) in cases {
228            let got = sample_rate_from_tracestate(tracestate);
229            match expected {
230                Some(want) => {
231                    let got = got.unwrap_or_else(|| panic!("{name}: expected {want}, got no rate"));
232                    assert!((got - want).abs() < 1e-9, "{name}: expected {want}, got {got}");
233                }
234                None => assert_eq!(got, None, "{name}"),
235            }
236        }
237    }
