26/// Recognized OTLP resource attributes used for entity tag resolution.
27#[derive(Default)]
28struct OtlpResourceAttributes<'a> {
29    container_id: Option<&'a str>,
30    oci_manifest_digest: Option<&'a str>,
31    ecs_task_arn: Option<&'a str>,
32    k8s_container_name: Option<&'a str>,
33    k8s_deployment_name: Option<&'a str>,
34    k8s_namespace_name: Option<&'a str>,
35    k8s_node_name: Option<&'a str>,
36    k8s_pod_uid: Option<&'a str>,
37    process_pid: Option<i64>,
38    cgroup_inode: Option<i64>,
39    init_container: Option<bool>,
40}
41
42impl<'a> OtlpResourceAttributes<'a> {
43    fn from_attributes(attributes: &'a [otlp_common::KeyValue]) -> Self {
44        let mut extracted = Self::default();
45
46        for attribute in attributes {
47            let Some(value) = attribute.value.as_ref().and_then(|value| value.value.as_ref()) else {
48                continue;
49            };
50
51            match (attribute.key.as_str(), value) {
52                (CONTAINER_ID, Value::StringValue(value)) => extracted.container_id = Some(value),
53                (OCI_MANIFEST_DIGEST, Value::StringValue(value)) => extracted.oci_manifest_digest = Some(value),
54                (ECS_TASK_ARN, Value::StringValue(value)) => extracted.ecs_task_arn = Some(value),
55                (K8S_CONTAINER_NAME, Value::StringValue(value)) => extracted.k8s_container_name = Some(value),
56                (K8S_DEPLOYMENT_NAME, Value::StringValue(value)) => extracted.k8s_deployment_name = Some(value),
57                (K8S_NAMESPACE_NAME, Value::StringValue(value)) => extracted.k8s_namespace_name = Some(value),
58                (K8S_NODE_NAME, Value::StringValue(value)) => extracted.k8s_node_name = Some(value),
59                (K8S_POD_UID, Value::StringValue(value)) => extracted.k8s_pod_uid = Some(value),
60                (PROCESS_PID, Value::IntValue(value)) => extracted.process_pid = Some(*value),
61                (CGROUP_INODE, Value::IntValue(value)) => extracted.cgroup_inode = Some(*value),
62                (INIT_CONTAINER, Value::BoolValue(value)) => extracted.init_container = Some(*value),
63                _ => {}
64            }
65        }
66
67        extracted
68    }
69}

81    /// Resolves entity and global tags for one OTLP metrics resource by entity precedence.
82    pub fn resolve_resource_tags(
83        &self, attributes: &[otlp_common::KeyValue], tag_cardinality: ConfigOriginTagCardinality,
84    ) -> SharedTagSet {
85        let tag_cardinality = match tag_cardinality {
86            ConfigOriginTagCardinality::Low => OriginTagCardinality::Low,
87            ConfigOriginTagCardinality::Orchestrator => OriginTagCardinality::Orchestrator,
88            ConfigOriginTagCardinality::High => OriginTagCardinality::High,
89            ConfigOriginTagCardinality::None => return SharedTagSet::default(),
90        };
91
92        let attributes = OtlpResourceAttributes::from_attributes(attributes);
93        let mut tags = TagSet::default();
94        let mut written_keys = FastHashSet::default();
95
96        if let Some(container_entity) = self.container_entity_from_attributes(&attributes, tag_cardinality) {
97            self.collect_tags_for_entity(container_entity, tag_cardinality, &mut written_keys, &mut tags);
98        }
99
100        for entity_id in entity_ids_from_attributes(&attributes) {
101            self.collect_tags_for_entity(entity_id, tag_cardinality, &mut written_keys, &mut tags);
102        }
103
104        self.collect_tags_for_entity(EntityId::Global, tag_cardinality, &mut written_keys, &mut tags);
105        tags.into_shared()
106    }

108    fn container_entity_from_attributes(
109        &self, attributes: &OtlpResourceAttributes<'_>, tag_cardinality: OriginTagCardinality,
110    ) -> Option<EntityId> {
111        if let Some(container_id) = attributes.container_id.filter(|value| !value.is_empty()) {
112            return Some(EntityId::Container(container_id.into()));
113        }
114
115        let mut fallback_entities = Vec::with_capacity(3);
116        if let Some(process_id) = attributes
117            .process_pid
118            .and_then(|value| u32::try_from(value).ok())
119            .filter(|value| *value != 0)
120        {
121            fallback_entities.push(EntityId::ContainerPid(process_id));
122        }
123        if let Some(cgroup_inode) = attributes
124            .cgroup_inode
125            .and_then(|value| u64::try_from(value).ok())
126            .filter(|value| *value != 0)
127        {
128            fallback_entities.push(EntityId::ContainerInode(cgroup_inode));
129        }
130        if let Some(container_entity) = self.container_entity_from_external_data(attributes) {
131            fallback_entities.push(container_entity);
132        }
133
134        fallback_entities.into_iter().find(|entity_id| {
135            self.workload_provider
136                .get_tags_for_entity(entity_id, tag_cardinality)
137                .is_some_and(|tags| !tags.is_empty())
138        })
139    }
140
141    fn container_entity_from_external_data(&self, attributes: &OtlpResourceAttributes<'_>) -> Option<EntityId> {
142        let pod_uid = attributes.k8s_pod_uid?;
143        let container_name = attributes.k8s_container_name?;
144        let init_container = attributes.init_container.unwrap_or(false);
145        let external_data = format!("pu-{pod_uid},cn-{container_name},it-{init_container}");
146
147        let mut origin = RawOrigin::default();
148        origin.set_external_data(external_data.as_str());
149        self.workload_provider.get_resolved_origin(origin).and_then(|origin| {
150            origin
151                .resolved_external_data()
152                .map(|external_data| external_data.container_entity_id().clone())
153        })
154    }
155
156    fn collect_tags_for_entity(
157        &self, entity_id: EntityId, tag_cardinality: OriginTagCardinality, written_keys: &mut FastHashSet<String>,
158        tags: &mut TagSet,
159    ) {
160        let Some(entity_tags) = self.workload_provider.get_tags_for_entity(&entity_id, tag_cardinality) else {
161            trace!(
162                ?entity_id,
163                cardinality = tag_cardinality.as_str(),
164                "No tags found for entity."
165            );
166            return;
167        };
168
169        for tag in &entity_tags {
170            let Some((key, value)) = tag.as_str().split_once(':') else {
171                continue;
172            };
173            if key.is_empty() || value.is_empty() || !written_keys.insert(key.to_owned()) {
174                continue;
175            }
176            tags.insert_tag(tag.clone());
177        }
178    }
179
180    fn collect_origin_tags(&self, resolved_origin: &saluki_env::workload::origin::ResolvedOrigin) -> SharedTagSet {
181        let mut collected_tags = SharedTagSet::default();
182        let tag_cardinality = resolved_origin.cardinality().unwrap_or(OriginTagCardinality::Low);
183
184        // Logs retain their existing origin-only behavior. Metrics use `resolve_resource_tags` above to include the
185        // full OTLP entity list and global tags.
186        let entity_ids = [
187            resolved_origin.local_data(),
188            resolved_origin.pod_uid(),
189            resolved_origin.process_id(),
190        ];
191
192        for entity_id in entity_ids.iter().flatten() {
193            if let Some(tags) = self.workload_provider.get_tags_for_entity(entity_id, tag_cardinality) {
194                if !tags.is_empty() {
195                    collected_tags.extend_from_shared(&tags);
196                    return collected_tags;
197                }
198            } else {
199                trace!(
200                    ?entity_id,
201                    cardinality = tag_cardinality.as_str(),
202                    "No tags found for entity."
203                );
204            }
205        }
206        collected_tags
207    }
208}

210impl OriginTagsResolver for OtlpOriginTagResolver {
211    fn resolve_origin_tags(&self, origin: RawOrigin<'_>) -> SharedTagSet {
212        match self.workload_provider.get_resolved_origin(origin.clone()) {
213            Some(resolved_origin) => self.collect_origin_tags(&resolved_origin),
214            None => {
215                trace!(?origin, "No resolved origin found for raw origin.");
216                SharedTagSet::default()
217            }
218        }
219    }
220}

222fn entity_ids_from_attributes(attributes: &OtlpResourceAttributes<'_>) -> Vec<EntityId> {
223    let mut entity_ids = Vec::with_capacity(7);
224
225    if let Some(oci_manifest_digest) = attributes.oci_manifest_digest {
226        if let Some((_, digest)) = oci_manifest_digest.split_once("@sha256:") {
227            entity_ids.push(EntityId::ContainerImageMetadata(format!("sha256:{digest}").into()));
228        }
229    }
230    if let Some(ecs_task_arn) = attributes.ecs_task_arn {
231        entity_ids.push(EntityId::EcsTask(ecs_task_arn.into()));
232    }
233    if let (Some(deployment), Some(namespace)) = (attributes.k8s_deployment_name, attributes.k8s_namespace_name) {
234        entity_ids.push(EntityId::KubernetesDeployment(
235            format!("{namespace}/{deployment}").into(),
236        ));
237    }
238    if let Some(namespace) = attributes.k8s_namespace_name {
239        entity_ids.push(EntityId::KubernetesMetadata(format!("/namespaces//{namespace}").into()));
240    }
241    if let Some(node_name) = attributes.k8s_node_name {
242        entity_ids.push(EntityId::KubernetesNode(node_name.into()));
243    }
244    if let Some(pod_uid) = attributes.k8s_pod_uid {
245        if let Some(entity_id) = EntityId::from_pod_uid(pod_uid) {
246            entity_ids.push(entity_id);
247        }
248    }
249    if let Some(process_id) = attributes.process_pid {
250        entity_ids.push(EntityId::Process(MetaString::from(process_id.to_string())));
251    }
252
253    entity_ids
254}

273        let mut tags = tags.into_iter().map(|tag| tag.as_str().to_owned()).collect::<Vec<_>>();

304    fn resolves_all_resource_entities_in_core_order_and_merges_global_tags_last() {
305        let mut provider = TestWorkloadProvider::new();
306        provider
307            .add_entity(
308                EntityId::Container("container-id".into()),
309                &["service:container", "container:present"],
310            )
311            .add_entity(
312                EntityId::ContainerImageMetadata("sha256:digest".into()),
313                &["service:image", "image:present"],
314            )
315            .add_entity(EntityId::EcsTask("task-arn".into()), &["task:present"])
316            .add_entity(
317                EntityId::KubernetesDeployment("default/api".into()),
318                &["deployment:present"],
319            )
320            .add_entity(
321                EntityId::KubernetesMetadata("/namespaces//default".into()),
322                &["namespace:present"],
323            )
324            .add_entity(EntityId::KubernetesNode("node-1".into()), &["node:present"])
325            .add_entity(EntityId::PodUid("pod-1".into()), &["pod:present"])
326            .add_entity(EntityId::Process("42".into()), &["process:present"])
327            .add_entity(EntityId::Global, &["service:global", "global:present"]);
328        let resolver = OtlpOriginTagResolver::new(Arc::new(provider));
329
330        let resource = vec![
331            attribute(CONTAINER_ID, Value::StringValue("container-id".into())),
332            attribute(OCI_MANIFEST_DIGEST, Value::StringValue("image@sha256:digest".into())),

335            attribute(K8S_NAMESPACE_NAME, Value::StringValue("default".into())),
336            attribute(K8S_NODE_NAME, Value::StringValue("node-1".into())),
337            attribute(K8S_POD_UID, Value::StringValue("pod-1".into())),
338            attribute(PROCESS_PID, Value::IntValue(42)),
339        ];
340
341        assert_eq!(
342            tags(resolver.resolve_resource_tags(&resource, ConfigOriginTagCardinality::Low)),
343            vec![
344                "container:present",
345                "deployment:present",
346                "global:present",
347                "image:present",
348                "namespace:present",
349                "node:present",
350                "pod:present",
351                "process:present",
352                "service:container",

356    }
