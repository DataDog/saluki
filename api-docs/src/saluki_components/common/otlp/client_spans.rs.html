1//! Per-client counters for OTLP spans received.
2//!
3//! Tags the counter with the client dimensions an operator needs to attribute traffic: the sending SDK's language, its
4//! version, and the ingest endpoint version.
5
6use std::borrow::Cow;
7use std::collections::HashMap;
8use std::sync::{Arc, OnceLock, RwLock};
9
10use saluki_common::strings::StringBuilder;
11use saluki_metrics::{Counter, MetricsBuilder};
12
13use crate::common::otlp::traces::normalize::{is_normalized_tag_value, normalize_tag_value_into_unchecked};
14
15/// Prefix on the SDK version in `tracer_version`, to tell OTLP clients from Datadog tracers.
16const TRACER_VERSION_PREFIX: &str = "otlp-";
17
18/// Value of the `endpoint_version` tag.
19const ENDPOINT_VERSION: &str = "opentelemetry_grpc_v1";
20
21/// Value of the `lang` and `tracer_version` tags on the shared series that absorbs clients past the cap.
22const OVERFLOW_TAG_VALUE: &str = "overflow";
23
24/// Maximum number of distinct (`lang`, `tracer_version`) series registered per component.
25///
26/// SDK language and version come from the sender, so without a cap a misbehaving client could create series without
27/// limit. Clients past the cap are counted on a single shared overflow series.
28pub(super) const MAX_CLIENT_SERIES: usize = 1000;
29
30/// The registered client series, keyed by normalized language and then by normalized SDK version.
31///
32/// Nesting the maps lets a lookup borrow both keys as `&str`, so finding an existing series never allocates.
33#[derive(Default)]
34struct ClientSeries {
35    by_language: HashMap<Box<str>, HashMap<Box<str>, Counter>>,
36    len: usize,
37}
38
39impl ClientSeries {
40    fn get(&self, language: &str, version: &str) -> Option<&Counter> {
41        self.by_language.get(language)?.get(version)
42    }
43}
44
45/// Counters for received OTLP spans, resolved by the sending client's identity.
46///
47/// Clones share one set of counters.
48#[derive(Clone)]
49pub struct ClientSpanCounters {
50    builder: MetricsBuilder,
51    max_series: usize,
52    series: Arc<RwLock<ClientSeries>>,
53    overflow: Arc<OnceLock<Counter>>,
54}
55
56impl ClientSpanCounters {
57    /// Creates counters that register through `builder`, capped at [`MAX_CLIENT_SERIES`] client series.
58    pub(super) fn new(builder: MetricsBuilder) -> Self {
59        Self::with_max_series(builder, MAX_CLIENT_SERIES)
60    }
61
62    fn with_max_series(builder: MetricsBuilder, max_series: usize) -> Self {
63        Self {
64            builder,
65            max_series,
66            series: Arc::new(RwLock::new(ClientSeries::default())),
67            overflow: Arc::new(OnceLock::new()),
68        }
69    }
70
71    /// Returns the counter for spans sent by the client identified by the raw `telemetry.sdk.language` and
72    /// `telemetry.sdk.version` resource attributes, registering it on first use.
73    ///
74    /// Both values are normalized as tag values. An empty language is left off the series, while an empty version
75    /// still yields `tracer_version:otlp-`. Once [`MAX_CLIENT_SERIES`] series exist, new clients share one overflow
76    /// series.
77    pub fn for_client(&self, sdk_language: &str, sdk_version: &str) -> Counter {
78        let language = normalized(sdk_language);
79        let version = normalized(sdk_version);
80
81        // Hits, and clients past the cap, are served under the read lock. Once the series are full they never grow
82        // again, so the overflow check stays valid and those clients never contend for the write lock.
83        {
84            let series = self.series.read().unwrap();
85            if let Some(counter) = series.get(&language, &version) {
86                return counter.clone();
87            }
88            if series.len >= self.max_series {
89                return self.overflow_counter();
90            }
91        }
92
93        let mut series = self.series.write().unwrap();
94        // Another thread may have registered the series, or filled the last slot, since the read lock was released.
95        if let Some(counter) = series.get(&language, &version) {
96            return counter.clone();
97        }
98        if series.len >= self.max_series {
99            return self.overflow_counter();
100        }
101
102        let counter = register(&self.builder, (!language.is_empty()).then_some(&*language), &version);
103        series
104            .by_language
105            .entry(language.into_owned().into_boxed_str())
106            .or_default()
107            .insert(version.into_owned().into_boxed_str(), counter.clone());
108        series.len += 1;
109        counter
110    }
111
112    fn overflow_counter(&self) -> Counter {
113        self.overflow
114            .get_or_init(|| register(&self.builder, Some(OVERFLOW_TAG_VALUE), OVERFLOW_TAG_VALUE))
115            .clone()
116    }
117}
118
119/// Registers the span counter for a client. `version` is the normalized SDK version, without its `otlp-` prefix.
120fn register(builder: &MetricsBuilder, lang: Option<&str>, version: &str) -> Counter {
121    let mut tags = vec![("message_type", "otlp_spans".to_string())];
122    if let Some(lang) = lang {
123        tags.push(("lang", lang.to_string()));
124    }
125    tags.push(("tracer_version", format!("{TRACER_VERSION_PREFIX}{version}")));
126    tags.push(("endpoint_version", ENDPOINT_VERSION.to_string()));
127
128    builder.register_counter_with_tags("component_events_received_total", tags)
129}
130
131/// Returns `value` normalized as a tag value, borrowing it when it's already normalized.
132fn normalized(value: &str) -> Cow<'_, str> {
133    if value.is_empty() || is_normalized_tag_value(value) {
134        return Cow::Borrowed(value);
135    }
136
137    let mut normalized = StringBuilder::new();
138    normalize_tag_value_into_unchecked(value, &mut normalized);
139    Cow::Owned(normalized.as_str().to_string())
140}

148    fn counters(max_series: usize) -> ClientSpanCounters {
149        let builder = MetricsBuilder::default().add_default_tag("component_id:test");
150        ClientSpanCounters::with_max_series(builder, max_series)
151    }
152
153    fn spans(recorder: &TestRecorder, tags: &'static [(&'static str, &'static str)]) -> Option<u64> {
154        recorder.counter(("component_events_received_total", tags))
155    }
156
157    #[test]
158    fn tags_the_series_with_the_client_identity() {
159        let recorder = TestRecorder::default();
160        let _guard = metrics::set_default_local_recorder(&recorder);
161
162        counters(10).for_client("go", "1.2.3").increment(4);
163
164        assert_eq!(
165            spans(
166                &recorder,
167                &[
168                    ("component_id", "test"),
169                    ("message_type", "otlp_spans"),
170                    ("lang", "go"),
171                    ("tracer_version", "otlp-1.2.3"),
172                    ("endpoint_version", "opentelemetry_grpc_v1"),
173                ]
174            ),
175            Some(4)
176        );
177    }

254    #[test]
255    fn counts_clients_past_the_cap_on_one_overflow_series() {
256        let recorder = TestRecorder::default();
257        let _guard = metrics::set_default_local_recorder(&recorder);
258
259        let counters = counters(2);

261        counters.for_client("go", "2").increment(1);

264        // A client registered before the cap is reached keeps its own series.

266
267        assert_eq!(counters.series.read().unwrap().len, 2);
268        assert_eq!(
269            spans(
270                &recorder,
271                &[
272                    ("component_id", "test"),
273                    ("message_type", "otlp_spans"),
274                    ("lang", "overflow"),
275                    ("tracer_version", "otlp-overflow"),
276                    ("endpoint_version", "opentelemetry_grpc_v1"),
277                ]
278            ),
279            Some(12)
280        );
281        assert_eq!(
282            spans(
283                &recorder,
284                &[
285                    ("component_id", "test"),
286                    ("message_type", "otlp_spans"),
287                    ("lang", "go"),
288                    ("tracer_version", "otlp-1"),
