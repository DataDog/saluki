158    /// Builds and starts the OTLP servers (HTTP and gRPC).
159    ///
160    /// Both servers run on the shared worker pool, since request handling shouldn't contend with the runtime driving
161    /// the topology, and decoding can be compute-heavy for large requests.
162    ///
163    /// # Errors
164    ///
165    /// If the gRPC endpoint isn't a TCP address, the listen addresses can't be bound, or either server can't be
166    /// spawned, an error is returned.
167    pub async fn build<H: OtlpHandler>(
168        self, handler: H, memory_limiter: MemoryLimiter, metrics: Metrics, spawner: &ComponentSpawner,
169    ) -> Result<(), GenericError> {
170        let otlp_handler = Arc::new(handler);
171        let metrics = Arc::new(metrics);
172
173        // Create and spawn the gRPC server.
174        let inner_grpc = GrpcServiceImpl::new(otlp_handler.clone(), memory_limiter.clone(), metrics.clone());
175
176        let grpc_metrics_server =
177            MetricsServiceServer::new(inner_grpc.clone()).max_decoding_message_size(self.grpc_max_recv_msg_size_bytes);
178
179        let grpc_logs_server =
180            LogsServiceServer::new(inner_grpc.clone()).max_decoding_message_size(self.grpc_max_recv_msg_size_bytes);
181
182        let grpc_traces_server =
183            TraceServiceServer::new(inner_grpc).max_decoding_message_size(self.grpc_max_recv_msg_size_bytes);
184
185        let grpc_server = GrpcServer::new(self.grpc_endpoint.clone())
186            .add_service(grpc_metrics_server)
187            .add_service(grpc_logs_server)
188            .add_service(grpc_traces_server);
189
190        spawner
191            .supervisable(grpc_server)
192            .on_worker_pool()
193            .spawn()
194            .await
195            .error_context("Failed to spawn OTLP gRPC server.")?;
196
197        // Create and spawn the HTTP server.
198        let router = Router::new()
199            .route("/v1/metrics", post(http_metrics_handler::<H>))
200            .route("/v1/logs", post(http_logs_handler::<H>))
201            .route("/v1/traces", post(http_traces_handler::<H>))
202            .with_state((otlp_handler, memory_limiter, metrics));
203
204        // Apply CORS middleware when origins are configured.
205        let router = if !self.cors.allowed_origins.is_empty() {
206            router.layer(build_cors_layer(&self.cors))
207        } else {
208            router
209        };
210
211        let service = TowerToHyperService::new(router);
212
213        let http_server = HttpServer::from_listen_address(self.http_endpoint, service);
214
215        spawner
216            .supervisable(http_server)
217            .on_worker_pool()
218            .spawn()
219            .await
220            .error_context("Failed to spawn OTLP HTTP server.")?;
221
222        Ok(())
223    }

226/// Builds a CORS layer from the resolved CORS configuration.
227///
228/// A bare `*` in the list of allowed origins enables allow-all; otherwise the first `*` in an origin is a partial wildcard
229/// (for example, `http://*.domain.com` matches `http://foo.domain.com`).
230fn build_cors_layer(cors: &CorsConfiguration) -> CorsLayer {
231    let mut layer = CorsLayer::new();
232    let allowed_origins = cors
233        .allowed_origins
234        .iter()
235        .map(|origin| origin.to_ascii_lowercase())
236        .collect::<Vec<_>>();
237
238    let allows_any_origin = allowed_origins.iter().any(|origin| origin == "*");
239    if allows_any_origin {
240        layer = layer.allow_origin(Any);
241    } else {
242        layer = layer.allow_origin(AllowOrigin::predicate(move |origin, _request_parts| {
243            let Ok(origin) = origin.to_str() else {
244                return false;
245            };
246            let origin = origin.to_ascii_lowercase();
247
248            allowed_origins.iter().any(|pattern| origin_matches(pattern, &origin))
249        }));
250    }
251
252    // Preflight must permit the methods browser exporters use; without this the browser blocks
253    // the actual request even when the origin is allowed.
254    layer = layer.allow_methods([Method::GET, Method::POST, Method::HEAD]);
255
256    if cors.allowed_headers.iter().any(|h| h == "*") {
257        layer = layer.allow_headers(Any);
258    } else {
259        let mut headers: Vec<HeaderName> = cors
260            .allowed_headers
261            .iter()
262            .filter_map(|h| HeaderName::from_str(h).ok())
263            .collect();
264        headers.extend_from_slice(&IMPLICIT_HEADERS);
265        if cors.allowed_headers.is_empty() {
266            headers.push(HeaderName::from_static("x-requested-with"));
267        }
268        layer = layer.allow_headers(headers);
269    }
270
271    // Exposed headers.
272    if !cors.exposed_headers.is_empty() {
273        let headers: Vec<HeaderName> = cors
274            .exposed_headers
275            .iter()
276            .filter_map(|h| HeaderName::from_str(h).ok())
277            .collect();
278        layer = layer.expose_headers(headers);
279    }
280
281    // Max age.
282    if cors.max_age > 0 {
283        layer = layer.max_age(std::time::Duration::from_secs(cors.max_age));
284    }
285
286    // Wildcard CORS responses cannot permit browser credentials.
287    if !allows_any_origin
288        && !cors.allowed_headers.iter().any(|header| header == "*")
289        && !cors.exposed_headers.iter().any(|header| header == "*")
290    {
291        layer = layer.allow_credentials(true);
292    }
293
294    layer
295}

297/// Matches an Origin header against an allowed-origin pattern with first-`*` wildcard semantics.
298fn origin_matches(pattern: &str, origin: &str) -> bool {
299    let mut parts = pattern.splitn(2, '*');
300    let prefix = parts.next().unwrap_or("");
301    let Some(suffix) = parts.next() else {
302        return pattern == origin;
303    };
304
305    origin.starts_with(prefix) && origin.ends_with(suffix) && origin.len() >= prefix.len() + suffix.len()
306}

308/// HTTP handler for OTLP metrics requests.
309async fn http_metrics_handler<H: OtlpHandler>(
310    State((handler, memory_limiter, metrics)): State<(Arc<H>, MemoryLimiter, Arc<Metrics>)>, body: Bytes,
311) -> (StatusCode, &'static str) {
312    memory_limiter.wait_for_capacity().await;
313
314    metrics.bytes_received().increment(body.len() as u64);
315
316    match handler.handle_metrics(body).await {
317        Ok(()) => (StatusCode::OK, "OK"),
318        Err(e) => {
319            error!(error = %e, "Failed to handle OTLP metrics.");
320            (StatusCode::INTERNAL_SERVER_ERROR, "Internal processing error")
321        }
322    }
323}

386#[async_trait]
387impl<H: OtlpHandler> MetricsService for GrpcServiceImpl<H> {
388    async fn export(
389        &self, request: TonicRequest<ExportMetricsServiceRequest>,
390    ) -> Result<Response<ExportMetricsServiceResponse>, Status> {
391        self.memory_limiter.wait_for_capacity().await;
392
393        let raw_bytes = request.into_inner().encode_to_vec();
394        self.metrics.bytes_received().increment(raw_bytes.len() as u64);
395
396        match self.handler.handle_metrics(Bytes::from(raw_bytes)).await {
397            Ok(()) => Ok(Response::new(ExportMetricsServiceResponse { partial_success: None })),
398            Err(e) => {
399                error!(error = %e, "Failed to handle OTLP metrics.");
400                Err(Status::internal("Internal processing error"))
401            }
402        }
403    }
404}

696    /// Waits for a Unix socket file to appear, retrying briefly to avoid races with async
697    /// server startup.
698    #[cfg(unix)]
699    async fn wait_for_socket(path: &std::path::Path) {
700        for _ in 0..100 {
701            if path.exists() {
702                return;
703            }
704            tokio::time::sleep(std::time::Duration::from_millis(5)).await;
705        }
706        panic!("Unix socket at {} did not appear within 500ms", path.display());
707    }
