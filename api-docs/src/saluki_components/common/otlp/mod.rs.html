268/// TLS configuration for an OTLP receiver, loaded from file paths on disk.
269///
270/// Both `cert_file` and `key_file` are required to enable TLS. When `ca_file` is set, the server requests client
271/// certificates and verifies them if presented, but does not require them (optional verification).
272#[derive(Clone, Debug)]
273pub struct OtlpTlsConfiguration {
274    cert_file: PathBuf,
275    key_file: PathBuf,
276    ca_file: Option<PathBuf>,
277}
278
279impl OtlpTlsConfiguration {
280    /// Creates a new TLS configuration with the given certificate and key file paths.
281    pub fn new(cert_file: PathBuf, key_file: PathBuf) -> Self {
282        Self {
283            cert_file,
284            key_file,
285            ca_file: None,
286        }
287    }
288
289    /// Sets the CA certificate file for optional client certificate verification.
290    pub fn with_ca_file(mut self, ca_file: PathBuf) -> Self {
291        self.ca_file = Some(ca_file);
292        self
293    }
294
295    /// Builds a `rustls::ServerConfig` from the configured file paths.
296    ///
297    /// # Errors
298    ///
299    /// If the certificate or key files cannot be read or parsed, or if the resulting configuration isn't FIPS
300    /// compliant, an error will be returned.
301    fn build(self) -> Result<rustls::ServerConfig, GenericError> {
302        let mut builder = ServerTLSConfigBuilder::new()
303            .with_cert_file(self.cert_file)
304            .with_key_file(self.key_file);
305
306        if let Some(ca_file) = self.ca_file {
307            builder = builder.with_ca_file(ca_file);
308        }
309
310        builder.build()
311    }

363/// How the OTLP servers serve a signal.
364#[derive(Clone, Copy, Debug, Eq, PartialEq)]
365pub enum OtlpRoute {
366    /// The servers don't serve the signal: a gRPC call gets `UNIMPLEMENTED`, and an HTTP request gets 404.
367    Disabled,
368
369    /// The servers decode the signal's requests and pass them to the OTLP handler.
370    Internal,
371
372    /// The servers stream the signal's requests to an upstream OTLP receiver, and stream its responses back.
373    ///
374    /// The proxy doesn't decode or limit the requests: the upstream receiver enforces its own limits.
375    Proxied,
376}

380pub struct OtlpRoutes {

534    /// Builds and starts the OTLP servers (HTTP and gRPC).
535    ///
536    /// Both servers are spawned into the scope of the current process, and they stop as soon as that process is told to
537    /// stop. The handler is dropped once both have exited, so the component can drain a channel that the handler feeds
538    /// until it closes. The servers run on `worker_pool` rather than the component's runtime, since request handling
539    /// shouldn't contend with the runtime driving the topology, and decoding can be compute-heavy for large requests.
540    ///
541    /// # Errors
542    ///
543    /// If the gRPC endpoint isn't a TCP or Unix address, the listen addresses can't be bound, the TLS configuration is
544    /// invalid, or a signal is proxied without a proxy or without a destination for the protocol it arrives on, an
545    /// error is returned.
546    pub async fn build<H: OtlpHandler>(
547        self, handler: H, memory_limiter: MemoryLimiter, metrics: Metrics, worker_pool: &Handle,
548    ) -> Result<(), GenericError> {
549        if self.routes.any_proxied() && self.proxy.is_none() {
550            return Err(generic_error!("An OTLP signal is proxied, but no OTLP proxy is set."));
551        }
552
553        let otlp_handler = Arc::new(handler);
554        let metrics = Arc::new(metrics);
555
556        // Validate TLS configurations before spawning. A server with no endpoint doesn't start, so its TLS
557        // configuration isn't loaded.
558        let http_tls_config = match self.http_tls {
559            Some(tls) if self.http_endpoint.is_some() => Some(tls.build()?),
560            _ => None,
561        };
562        let grpc_tls_config = match self.grpc_tls {
563            Some(tls) if self.grpc_endpoint.is_some() => Some(tls.build()?),
564            _ => None,
565        };
566
567        // Create and spawn the gRPC server.
568        //
569        // The gRPC endpoint is served by an HTTP server: gRPC is HTTP/2 with a distinct route naming convention, so
570        // the generated services are just another route set. It is restricted to HTTP/2 because the OTLP gRPC port
571        // only ever serves gRPC, and an HTTP/1.1 caller belongs on the HTTP endpoint instead.
572        //
573        // A proxied signal takes the whole path prefix of its service, so the upstream answers every method of it. It
574        // replaces the internal service rather than sitting next to it, since both register the same paths.
575        if let Some(grpc_endpoint) = self.grpc_endpoint {
576            let inner_grpc = GrpcServiceImpl::new(otlp_handler.clone(), memory_limiter.clone(), metrics.clone());
577            let max_decoding_message_size = self.grpc_max_recv_msg_size_bytes;
578            let mut grpc_server = HttpServer::from_listen_address(grpc_endpoint);
579            let mut proxied_grpc_routes = Router::new();
580            for signal in Signal::ALL {
581                match (self.routes.get(signal), &self.proxy) {
582                    // The check at the top of `build` rules out a proxied signal without a proxy.
583                    (OtlpRoute::Disabled, _) | (OtlpRoute::Proxied, None) => {}
584                    (OtlpRoute::Internal, _) => {
585                        let inner = inner_grpc.clone();
586                        grpc_server = match signal {
587                            Signal::Metrics => grpc_server.add_grpc_service(
588                                MetricsServiceServer::new(inner).max_decoding_message_size(max_decoding_message_size),
589                            ),
590                            Signal::Logs => grpc_server.add_grpc_service(
591                                LogsServiceServer::new(inner).max_decoding_message_size(max_decoding_message_size),
592                            ),
593                            Signal::Traces => grpc_server.add_grpc_service(
594                                TraceServiceServer::new(inner).max_decoding_message_size(max_decoding_message_size),
595                            ),
596                        };
597                    }
598                    (OtlpRoute::Proxied, Some(proxy)) => {
599                        let path = format!("/{}/{{*method}}", signal.grpc_service_name());
600                        proxied_grpc_routes = proxied_grpc_routes.route(&path, proxy.grpc_service(signal)?);
601                    }
602                }
603            }
604
605            // The server answers misses with `UNIMPLEMENTED` only when it has an internal service, so set that answer
606            // here too, for a server with only proxied or disabled signals.
607            let mut grpc_server = grpc_server
608                .add_routes(proxied_grpc_routes.fallback(unmatched_route))
609                .with_http2_only()
610                .with_http2_config(self.grpc_http2_config)
611                .with_server_id(self.grpc_server_id)
612                .with_worker_pool(worker_pool.clone());
613
614            if let Some(tls_config) = grpc_tls_config {
615                grpc_server = grpc_server.with_tls_config(tls_config);
616            }
617
618            runtime::nested_supervisor(grpc_server.into_supervisor())
619                .needed()
620                .stops_with_owner()
621                .spawn();
622        }
623
624        // Create and spawn the HTTP server.
625        if let Some(http_endpoint) = self.http_endpoint {
626            // Apply an explicit body-size limit. A configured `0` selects the receiver's 20 MiB compatibility default;
627            // a positive value is the limit in bytes. Axum's own default is not the receiver's default and must not be
628            // left implicit.
629            let max_body_size = if self.http_max_request_body_size == 0 {
630                HTTP_DEFAULT_MAX_REQUEST_BODY_SIZE
631            } else {
632                self.http_max_request_body_size as usize
633            };
634
635            // A proxied signal takes every method of its path, so the upstream answers them all, CORS preflights
636            // included. The body limit and CORS layers apply to the internal routes only: the upstream enforces its
637            // own.
638            let mut internal_routes = Router::new();
639            let mut proxied_routes = Router::new();
640            for signal in Signal::ALL {
641                match (self.routes.get(signal), &self.proxy) {
642                    // The check at the top of `build` rules out a proxied signal without a proxy.
643                    (OtlpRoute::Disabled, _) | (OtlpRoute::Proxied, None) => {}
644                    (OtlpRoute::Internal, _) => {
645                        let handler = match signal {
646                            Signal::Metrics => post(http_metrics_handler::<H>),
647                            Signal::Logs => post(http_logs_handler::<H>),
648                            Signal::Traces => post(http_traces_handler::<H>),
649                        };
650                        internal_routes = internal_routes.route(signal.http_path(), handler);
651                    }
652                    (OtlpRoute::Proxied, Some(proxy)) => {
653                        proxied_routes = proxied_routes.route(signal.http_path(), proxy.http_service(signal)?);
654                    }
655                }
656            }
657
658            let router = internal_routes.layer(DefaultBodyLimit::max(max_body_size)).with_state((
659                otlp_handler,
660                memory_limiter,
661                metrics,
662            ));
663
664            // Apply CORS middleware when origins are configured.
665            let router = if !self.cors.allowed_origins.is_empty() {
666                router.layer(build_cors_layer(&self.cors))
667            } else {
668                router
669            };
670            let router = router.merge(proxied_routes);
671
672            let mut http_server = HttpServer::from_listen_address(http_endpoint)
673                .add_routes(router)
674                .with_server_id(self.http_server_id)
675                .with_worker_pool(worker_pool.clone());
676
677            if let Some(tls_config) = http_tls_config {
678                http_server = http_server.with_tls_config(tls_config);
679            }
680
681            runtime::nested_supervisor(http_server.into_supervisor())
682                .needed()
683                .stops_with_owner()
684                .spawn();
685        }
686
687        Ok(())
688    }

691/// Builds a CORS layer from the resolved CORS configuration.
692///
693/// A bare `*` in the list of allowed origins enables allow-all; otherwise the first `*` in an origin is a partial wildcard
694/// (for example, `http://*.domain.com` matches `http://foo.domain.com`).
695fn build_cors_layer(cors: &CorsConfiguration) -> CorsLayer {
696    let mut layer = CorsLayer::new();
697    let allowed_origins = cors
698        .allowed_origins
699        .iter()
700        .map(|origin| origin.to_ascii_lowercase())
701        .collect::<Vec<_>>();
702
703    let allows_any_origin = allowed_origins.iter().any(|origin| origin == "*");
704    if allows_any_origin {
705        layer = layer.allow_origin(Any);
706    } else {
707        layer = layer.allow_origin(AllowOrigin::predicate(move |origin, _request_parts| {
708            let Ok(origin) = origin.to_str() else {
709                return false;
710            };
711            let origin = origin.to_ascii_lowercase();
712
713            allowed_origins.iter().any(|pattern| origin_matches(pattern, &origin))
714        }));
715    }
716
717    // Preflight must permit the methods browser exporters use; without this the browser blocks
718    // the actual request even when the origin is allowed.
719    layer = layer.allow_methods([Method::GET, Method::POST, Method::HEAD]);
720
721    if cors.allowed_headers.iter().any(|h| h == "*") {
722        layer = layer.allow_headers(Any);
723    } else {
724        let mut headers: Vec<HeaderName> = cors
725            .allowed_headers
726            .iter()
727            .filter_map(|h| HeaderName::from_str(h).ok())
728            .collect();
729        headers.extend_from_slice(&IMPLICIT_HEADERS);
730        if cors.allowed_headers.is_empty() {
731            headers.push(HeaderName::from_static("x-requested-with"));
732        }
733        layer = layer.allow_headers(headers);
734    }
735
736    // Exposed headers.
737    if !cors.exposed_headers.is_empty() {
738        let headers: Vec<HeaderName> = cors
739            .exposed_headers
740            .iter()
741            .filter_map(|h| HeaderName::from_str(h).ok())
742            .collect();
743        layer = layer.expose_headers(headers);
744    }
745
746    // Max age.
747    if cors.max_age > 0 {
748        layer = layer.max_age(std::time::Duration::from_secs(cors.max_age));
749    }
750
751    // Wildcard CORS responses cannot permit browser credentials.
752    if !allows_any_origin
753        && !cors.allowed_headers.iter().any(|header| header == "*")
754        && !cors.exposed_headers.iter().any(|header| header == "*")
755    {
756        layer = layer.allow_credentials(true);
757    }
758
759    layer
760}

762/// Matches an Origin header against an allowed-origin pattern with first-`*` wildcard semantics.
763fn origin_matches(pattern: &str, origin: &str) -> bool {
764    let mut parts = pattern.splitn(2, '*');
765    let prefix = parts.next().unwrap_or("");
766    let Some(suffix) = parts.next() else {
767        return pattern == origin;
768    };
769
770    origin.starts_with(prefix) && origin.ends_with(suffix) && origin.len() >= prefix.len() + suffix.len()
771}
