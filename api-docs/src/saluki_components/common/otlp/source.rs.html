1//! Resolves the source of OTLP telemetry from its resource attributes.
2//!
3//! A source is the entity that emitted the telemetry: a host, or a serverless resource such as an AWS Lambda function.
4//! Detectors run in a fixed order, and the first one that matches wins. Resources that match none of them fall back to
5//! the hostname cascade shared with metrics.
6//!
7//! Only traces use this resolver today. Metrics and logs still use the narrower resolvers in [`super::util`].
8

67/// Resolves the source of telemetry from OTLP resource attributes.
68///
69/// Returns `None` when no source can be determined, and also for a GCP Cloud Run or Cloud Functions resource whose
70/// identity is incomplete. Callers decide whether to fall back to a default hostname; see [`is_gcp_serverless`].
71pub fn resolve_source_from_attributes(attributes: &[KeyValue]) -> Option<Source> {
72    if let Some(source) = azure_functions_source(attributes) {
73        return Some(source);
74    }
75    if let Some(source) = aws_lambda_source(attributes) {
76        return Some(source);
77    }
78    if let Some(source) = gcp_cloud_run_jobs_source(attributes) {
79        return Some(source);
80    }
81    // An incomplete GCP serverless identity must not fall through to the host cascade, which would report the
82    // collector's own host.
83    if is_gcp_serverless(attributes) {
84        return gcp_serverless_source(attributes);
85    }
86    if let Some(source) = gke_autopilot_source(attributes) {
87        return Some(source);
88    }
89    if let Some(source) = azure_app_service_source(attributes) {
90        return Some(source);
91    }
92    if let Some(source) = azure_container_apps_source(attributes) {
93        return Some(source);
94    }
95    if let Some(source) = ecs_fargate_source(attributes) {
96        return Some(source);
97    }
98
99    hostname_from_attributes(attributes).map(|identifier| Source {
100        kind: SourceKind::Hostname,
101        identifier,
102    })
103}

105/// Returns whether the resource declares a GCP Cloud Run or Cloud Functions platform.
106///
107/// This holds even when the identity is incomplete, or the resource is a job.
108pub fn is_gcp_serverless(attributes: &[KeyValue]) -> bool {
109    matches!(
110        attr(attributes, CLOUD_PLATFORM),
111        PLATFORM_GCP_CLOUD_FUNCTIONS | PLATFORM_GCP_CLOUD_RUN
112    )
113}

115/// Returns a string attribute, or an empty string if it's missing or not a string.
116fn attr<'a>(attributes: &'a [KeyValue], key: &str) -> &'a str {
117    get_string_attribute(attributes, key).unwrap_or_default()
118}

120fn source(kind: SourceKind, identifier: impl Into<String>) -> Option<Source> {
121    Some(Source {
122        kind,
123        identifier: identifier.into(),
124    })

131/// Fields of an Azure Resource Manager ID, `/subscriptions/{id}/resourceGroups/{name}/providers/{namespace}/{type}/{name}`.
132struct AzureResourceId<'a> {
133    subscription_id: &'a str,
134    resource_group: &'a str,
135    provider_namespace: &'a str,
136    resource_type: &'a str,
137    resource_name: &'a str,
138}
139
140impl<'a> AzureResourceId<'a> {
141    fn parse(resource_id: &'a str) -> Option<Self> {
142        let parts: Vec<&str> = resource_id.split('/').collect();
143        if parts.len() < 9
144            || !parts[0].is_empty()
145            || !parts[1].eq_ignore_ascii_case("subscriptions")
146            || !parts[3].eq_ignore_ascii_case("resourceGroups")
147            || !parts[5].eq_ignore_ascii_case("providers")
148            || [parts[2], parts[4], parts[6], parts[7], parts[8]]
149                .iter()
150                .any(|part| part.is_empty())
151        {
152            return None;
153        }
154
155        Some(Self {
156            subscription_id: parts[2],
157            resource_group: parts[4],
158            provider_namespace: parts[6],
159            resource_type: parts[7],
160            resource_name: parts[8],
161        })
162    }
163
164    fn matches(&self, provider_namespace: &str, resource_type: &str) -> bool {
165        self.provider_namespace.eq_ignore_ascii_case(provider_namespace)
166            && self.resource_type.eq_ignore_ascii_case(resource_type)
167    }
168}

170/// Identity of an Azure app, taken from attributes and filled in from `cloud.resource_id` where absent.
171struct AzureIdentity<'a> {
172    name: &'a str,
173    subscription_id: &'a str,
174    resource_group: &'a str,
175}
176
177impl<'a> AzureIdentity<'a> {
178    fn from_attributes(attributes: &'a [KeyValue], provider_namespace: &str, resource_type: &str) -> Option<Self> {
179        let mut identity = Self {
180            name: attr(attributes, SERVICE_NAME),
181            subscription_id: attr(attributes, CLOUD_ACCOUNT_ID),
182            resource_group: attr(attributes, AZURE_RESOURCE_GROUP_NAME),
183        };
184
185        if let Some(parsed) = AzureResourceId::parse(attr(attributes, CLOUD_RESOURCE_ID)) {
186            if parsed.matches(provider_namespace, resource_type) {
187                if identity.name.is_empty() {
188                    identity.name = parsed.resource_name;
189                }
190                if identity.subscription_id.is_empty() {
191                    identity.subscription_id = parsed.subscription_id;
192                }
193                if identity.resource_group.is_empty() {
194                    identity.resource_group = parsed.resource_group;
195                }
196            }
197        }
198
199        let complete =
200            !identity.name.is_empty() && !identity.subscription_id.is_empty() && !identity.resource_group.is_empty();
201        complete.then_some(identity)
202    }
203}

205/// Returns whether `platform` is an Azure platform ending in `suffix`, such as `azure.functions` or `azure_functions`.
206///
207/// Any run of characters other than lowercase letters and digits separates `azure` from the suffix.
208fn is_azure_cloud_platform(platform: &str, suffix: &str) -> bool {
209    let platform = platform.to_lowercase();
210    if !platform.starts_with("azure") || !platform.ends_with(suffix) {
211        return false;
212    }
213
214    let separator = platform
215        .len()
216        .checked_sub(suffix.len())
217        .and_then(|end| platform.get("azure".len()..end));
218    separator.is_some_and(|separator| {
219        !separator.is_empty()
220            && separator
221                .chars()
222                .all(|c| !(c.is_ascii_lowercase() || c.is_ascii_digit()))
223    })
224}

226fn azure_functions_source(attributes: &[KeyValue]) -> Option<Source> {
227    if !is_azure_cloud_platform(attr(attributes, CLOUD_PLATFORM), "functions") {
228        return None;
229    }
230    let identity = AzureIdentity::from_attributes(attributes, AZURE_WEB_NAMESPACE, AZURE_WEB_SITES)?;
231    source(SourceKind::AzureFunctions, identity.name)
232}
233
234fn azure_app_service_source(attributes: &[KeyValue]) -> Option<Source> {
235    if !is_azure_cloud_platform(attr(attributes, CLOUD_PLATFORM), "app_service") {
236        return None;
237    }
238    let identity = AzureIdentity::from_attributes(attributes, AZURE_WEB_NAMESPACE, AZURE_WEB_SITES)?;
239    source(SourceKind::AzureAppService, identity.name)
240}
241
242fn azure_container_apps_source(attributes: &[KeyValue]) -> Option<Source> {
243    if !is_azure_cloud_platform(attr(attributes, CLOUD_PLATFORM), "container_apps") {
244        return None;
245    }
246    AzureIdentity::from_attributes(attributes, AZURE_APP_NAMESPACE, AZURE_CONTAINER_APPS)?;
247
248    // Some SDKs report the replica in `service.instance.id`; the Azure-specific attribute stays authoritative.
249    let replica = Some(attr(attributes, AZURE_CONTAINER_APP_INSTANCE_ID))
250        .filter(|replica| !replica.is_empty())
251        .unwrap_or_else(|| attr(attributes, SERVICE_INSTANCE_ID));
252    if replica.is_empty() {
253        return None;
254    }
255    source(SourceKind::AzureContainerApps, replica)
256}

262fn aws_lambda_source(attributes: &[KeyValue]) -> Option<Source> {
263    if attr(attributes, CLOUD_PROVIDER) != "aws" {
264        return None;
265    }
266
267    let platform = attr(attributes, CLOUD_PLATFORM);
268    let faas_id = attr(attributes, FAAS_ID);
269    let resource_id = attr(attributes, CLOUD_RESOURCE_ID);
270    let function_arn = if faas_id.is_empty() { resource_id } else { faas_id };
271    let has_faas_attribute = attributes.iter().any(|kv| kv.key.starts_with("faas."));
272    if platform != PLATFORM_AWS_LAMBDA && faas_id.is_empty() && (resource_id.is_empty() || !has_faas_attribute) {
273        return None;
274    }
275
276    let parts: Vec<&str> = function_arn.split(':').collect();
277    if parts.len() < 7 || parts[0] != "arn" || parts[2] != "lambda" || parts[5] != "function" {
278        return None;
279    }
280    source(SourceKind::AwsLambda, parts[..7].join(":"))
281}
282
283fn ecs_fargate_source(attributes: &[KeyValue]) -> Option<Source> {
284    if !attr(attributes, AWS_ECS_LAUNCHTYPE).eq_ignore_ascii_case("fargate") {
285        return None;
286    }
287    let task_arn = get_string_attribute(attributes, AWS_ECS_TASK_ARN)?;
288    source(SourceKind::AwsEcsFargate, task_arn)
289}

295#[derive(Default)]
296struct GcpIdentity<'a> {
297    project_id: &'a str,
298    location: &'a str,
299    workload_name: &'a str,
300}
301
302impl<'a> GcpIdentity<'a> {
303    fn from_pattern(resource_id: &'a str, pattern: &Regex, include_name: bool) -> Self {
304        let Some(captures) = pattern.captures(resource_id) else {
305            return Self::default();
306        };
307        Self {
308            project_id: captures.get(1).map_or("", |m| m.as_str()),
309            location: captures.get(2).map_or("", |m| m.as_str()),
310            workload_name: if include_name {
311                captures.get(3).map_or("", |m| m.as_str())
312            } else {
313                ""
314            },
315        }
316    }
317
318    /// Overrides parsed fields with non-empty attributes.
319    fn with_attributes(mut self, attributes: &'a [KeyValue]) -> Self {
320        for (field, key) in [
321            (&mut self.project_id, CLOUD_ACCOUNT_ID),
322            (&mut self.location, CLOUD_REGION),
323            (&mut self.workload_name, FAAS_NAME),
324        ] {
325            let value = attr(attributes, key);
326            if !value.is_empty() {
327                *field = value;
328            }
329        }
330        self
331    }
332
333    fn is_complete(&self) -> bool {
334        !self.project_id.is_empty() && !self.location.is_empty() && !self.workload_name.is_empty()
335    }
336}
337
338fn has_gcp_cloud_run_job_evidence(attributes: &[KeyValue]) -> bool {
339    attributes.iter().any(|kv| {
340        matches!(
341            kv.key.as_str(),
342            "gcp.cloud_run.job.execution" | "gcp.cloud_run.job.task_index"
343        )
344    }) || GCP_CLOUD_RUN_JOB_RESOURCE.is_match(attr(attributes, CLOUD_RESOURCE_ID))
345}
346
347fn gcp_cloud_run_jobs_source(attributes: &[KeyValue]) -> Option<Source> {
348    if attr(attributes, CLOUD_PLATFORM) != PLATFORM_GCP_CLOUD_RUN || !has_gcp_cloud_run_job_evidence(attributes) {
349        return None;
350    }
351
352    let identity = GcpIdentity::from_pattern(attr(attributes, CLOUD_RESOURCE_ID), &GCP_CLOUD_RUN_JOB_RESOURCE, true)
353        .with_attributes(attributes);
354    let instance_id = attr(attributes, FAAS_INSTANCE);
355    if !identity.is_complete() || instance_id.is_empty() {
356        return None;
357    }
358    source(SourceKind::GcpCloudRunJobs, instance_id)
359}

361fn gcp_serverless_source(attributes: &[KeyValue]) -> Option<Source> {
362    let resource_id = attr(attributes, CLOUD_RESOURCE_ID);
363    let (kind, parsed) = match attr(attributes, CLOUD_PLATFORM) {
364        // Functions take precedence over their underlying Cloud Run service.
365        PLATFORM_GCP_CLOUD_FUNCTIONS => {
366            let parsed = GcpIdentity::from_pattern(resource_id, &GCP_CLOUD_FUNCTIONS_RESOURCE, true);
367            let parsed = if parsed.project_id.is_empty() {
368                GcpIdentity::from_pattern(resource_id, &GCP_CLOUD_RUN_SERVICE_RESOURCE, true)
369            } else {
370                parsed
371            };
372            (SourceKind::GcpCloudFunctions, parsed)
373        }
374        PLATFORM_GCP_CLOUD_RUN => {
375            let parsed = GcpIdentity::from_pattern(resource_id, &GCP_CLOUD_RUN_SERVICE_RESOURCE, true);
376            let parsed = if parsed.project_id.is_empty() {
377                GcpIdentity::from_pattern(resource_id, &GCP_CLOUD_RUN_REVISION_RESOURCE, false)
378            } else {
379                parsed
380            };
381            (SourceKind::GcpCloudRun, parsed)
382        }
383        _ => return None,
384    };
385
386    // A job is out of scope here; `gcp_cloud_run_jobs_source` handles complete ones.
387    if has_gcp_cloud_run_job_evidence(attributes) {
388        return None;
389    }
390
391    let identity = parsed.with_attributes(attributes);
392    let instance_id = attr(attributes, FAAS_INSTANCE);
393    if !identity.is_complete() || instance_id.is_empty() {
394        return None;
395    }
396    source(kind, instance_id)
397}

399fn gke_autopilot_source(attributes: &[KeyValue]) -> Option<Source> {
400    if attr(attributes, CLOUD_PROVIDER) != "gcp" || attr(attributes, CLOUD_PLATFORM) != PLATFORM_GKE {
401        return None;
402    }
403
404    let project_id = attr(attributes, CLOUD_ACCOUNT_ID);
405    let cluster_name = attr(attributes, K8S_CLUSTER_NAME);
406    let hostname = Some(attr(attributes, ATTRIBUTE_HOST))
407        .filter(|host| !host.is_empty())
408        .or_else(|| Some(attr(attributes, ATTRIBUTE_DATADOG_HOSTNAME)).filter(|host| !host.is_empty()))
409        .map(str::to_owned)
410        .or_else(|| gcp_hostname(attributes))
411        .unwrap_or_default();
412    if project_id.is_empty() || cluster_name.is_empty() || !hostname.to_lowercase().starts_with("gk3") {
413        return None;
414    }
415    source(
416        SourceKind::GkeAutopilot,
417        format!("{project_id}:{cluster_name}:{hostname}"),
418    )
419}

427    fn attrs(entries: &[(&str, &str)]) -> Vec<KeyValue> {
428        entries
429            .iter()
430            .map(|(key, value)| KeyValue {
431                key: (*key).to_string(),
432                value: Some(AnyValue {
433                    value: Some(Value::StringValue((*value).to_string())),
434                }),

436            .collect()
437    }
