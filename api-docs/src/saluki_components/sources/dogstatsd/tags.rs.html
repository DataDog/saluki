10/// Regular (non-well-known) tags from a DogStatsD payload.
11///
12/// As an optimization, up to 32 tags are stored inline without allocating. This isn't a limit on the number of tags:
13/// any beyond that are stored on the heap.
14pub type RegularTags<'a> = SmallVec<[&'a str; 32]>;
15
16/// Returns `true` if the given tag is a well-known tag.
17fn is_well_known_tag(tag: &str) -> bool {
18    tag.starts_with(HOST_TAG_KEY_SUFFIXED)
19        || tag.starts_with(ENTITY_ID_TAG_KEY_SUFFIXED)
20        || tag.starts_with(JMX_CHECK_NAME_TAG_KEY_SUFFIXED)
21        || tag.starts_with(CARDINALITY_TAG_KEY_SUFFIXED)
22}
23
24/// Well-known tags shared across DogStatsD payloads.
25///
26/// Well-known tags are tags which are generally set by the DogStatsD client to convey structured information
27/// about the source of the metric, event, or service check.
28#[derive(Default)]
29pub struct WellKnownTags<'a> {
30    pub hostname: Option<&'a str>,
31    pub pod_uid: Option<&'a str>,
32    pub jmx_check_name: Option<&'a str>,
33    pub cardinality: Option<OriginTagCardinality>,
34}
35
36impl<'a> WellKnownTags<'a> {
37    /// Splits the raw tags of a DogStatsD payload into well-known tags and regular tags.
38    ///
39    /// Any tag with a well-known tag key is excluded from the regular tags, even if its value is ignored. All fields
40    /// of the well-known tags default to `None`.
41    ///
42    /// The raw tags are walked exactly once, so callers can iterate over the regular tags as many times as needed
43    /// (such as for both hashing and building a context) without re-parsing the raw tags.
44    pub fn split_raw_tags(tags: &'a RawTags<'_>) -> (Self, RegularTags<'a>) {
45        let mut well_known_tags = Self::default();
46        let mut regular_tags = RegularTags::new();
47
48        for tag in tags.iter() {
49            if !is_well_known_tag(tag) {
50                regular_tags.push(tag);
51                continue;
52            }
53
54            match BorrowedTag::from(tag).name_and_value() {
55                (HOST_TAG_KEY, Some(hostname)) => {
56                    well_known_tags.hostname = Some(hostname);
57                }
58                (ENTITY_ID_TAG_KEY, Some(entity_id)) if entity_id != ENTITY_ID_IGNORE_VALUE => {
59                    well_known_tags.pod_uid = Some(entity_id);
60                }
61                (JMX_CHECK_NAME_TAG_KEY, Some(name)) => {
62                    well_known_tags.jmx_check_name = Some(name);
63                }
64                (CARDINALITY_TAG_KEY, Some(value)) => {
65                    if let Ok(card) = OriginTagCardinality::try_from(value) {
66                        well_known_tags.cardinality = Some(card);
67                    }
68                }
69                _ => {}
70            }
71        }
72
73        (well_known_tags, regular_tags)
74    }
75}

78mod tests {
79    use saluki_core::{
80        constants::datadog::*,
81        data_model::{origin::OriginTagCardinality, tags::RawTags},
82    };

84    use super::WellKnownTags;

86    #[test]
87    fn well_known_tags_are_parsed_from_their_tag() {
88        let tag = |key: &str, value: &str| format!("{key}:{value}");
89        let card = |value| (None, None, None, value);
90        // Expected hostname, pod UID, JMX check name, and cardinality.
91        let cases = [
92            (String::new(), (None, None, None, None)),
93            // Cardinality values are case-insensitive, to match the core Datadog Agent.
94            (tag(CARDINALITY_TAG_KEY, "none"), card(Some(OriginTagCardinality::None))),
95            (tag(CARDINALITY_TAG_KEY, "low"), card(Some(OriginTagCardinality::Low))),
96            (
97                tag(CARDINALITY_TAG_KEY, "orch"),
98                card(Some(OriginTagCardinality::Orchestrator)),
99            ),
100            (
101                tag(CARDINALITY_TAG_KEY, "orchestrator"),
102                card(Some(OriginTagCardinality::Orchestrator)),
103            ),
104            (tag(CARDINALITY_TAG_KEY, "high"), card(Some(OriginTagCardinality::High))),
105            (tag(CARDINALITY_TAG_KEY, "Low"), card(Some(OriginTagCardinality::Low))),
106            (tag(CARDINALITY_TAG_KEY, "HiGH"), card(Some(OriginTagCardinality::High))),
107            (tag(CARDINALITY_TAG_KEY, "MEDIUM"), card(None)),
108            (tag(CARDINALITY_TAG_KEY, "fakelevel"), card(None)),
109            (tag(HOST_TAG_KEY, ""), (Some(""), None, None, None)),
110            (tag(HOST_TAG_KEY, "localhost"), (Some("localhost"), None, None, None)),
111            (tag(JMX_CHECK_NAME_TAG_KEY, ""), (None, None, Some(""), None)),
112            (
113                tag(JMX_CHECK_NAME_TAG_KEY, "check_name"),
114                (None, None, Some("check_name"), None),
115            ),
116            (tag(ENTITY_ID_TAG_KEY, "asdasda"), (None, Some("asdasda"), None, None)),
117            (tag(ENTITY_ID_TAG_KEY, ENTITY_ID_IGNORE_VALUE), (None, None, None, None)),
118        ];
119
120        for (raw_tags, expected) in cases {
121            let tags = RawTags::new(&raw_tags, usize::MAX, usize::MAX);
122            let (wkt, _) = WellKnownTags::split_raw_tags(&tags);
123            let actual = (wkt.hostname, wkt.pod_uid, wkt.jmx_check_name, wkt.cardinality);
124            assert_eq!(actual, expected, "raw tags {raw_tags:?}");
125        }
126    }
127
128    #[test]
129    fn regular_tags_exclude_well_known_tags() {
130        let raw_tags_input = format!(
131            "env:prod,{}:my-host,host,hostname:other,{}:{},{}:bogus,{}:check,env:prod",
132            HOST_TAG_KEY, ENTITY_ID_TAG_KEY, ENTITY_ID_IGNORE_VALUE, CARDINALITY_TAG_KEY, JMX_CHECK_NAME_TAG_KEY
133        );
134        let tags = RawTags::new(&raw_tags_input, usize::MAX, usize::MAX);
135        let (wkt, regular_tags) = WellKnownTags::split_raw_tags(&tags);
136
137        // Well-known tags are excluded even when their value is ignored, while regular tags retain their order and
138        // duplicates.
139        assert_eq!(
140            regular_tags.as_slice(),
141            ["env:prod", "host", "hostname:other", "env:prod"]
142        );
143        assert_eq!(wkt.hostname, Some("my-host"));
144        assert_eq!(wkt.pod_uid, None);
145        assert_eq!(wkt.cardinality, None);
146        assert_eq!(wkt.jmx_check_name, Some("check"));
147    }
