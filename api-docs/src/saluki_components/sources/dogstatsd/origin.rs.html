45/// Origin enrichment configuration.
46///
47/// Origin enrichment controls the when and how of enriching metrics ingested via DogStatsD based on various sources of
48/// "origin" information, such as specific metric tags or UDS socket credentials. Enrichment involves adding additional
49/// metric tags that describe the origin of the metric, such as the Kubernetes pod or container.
50#[derive(Clone)]
51#[cfg_attr(test, derive(Debug, PartialEq))]
52pub struct OriginEnrichmentConfiguration {
53    /// Whether or not to enable origin detection.
54    ///
55    /// If disabled, no origin tags will be added to events even if the origin information is detected.
56    pub enabled: bool,
57
58    /// Whether or not a client-provided entity ID should take precedence over automatically detected origin metadata.
59    ///
60    /// When a client-provided entity ID is specified, and an origin process ID has automatically been detected, setting
61    /// this to `true` will cause the origin process ID to be ignored. This only applies when
62    /// `origin_detection_unified` is `false`.
63    pub entity_id_precedence: bool,
64
65    /// The cardinality of tags to enrich metrics with, when the payload doesn't specify one itself.
66    pub tag_cardinality: OriginTagCardinality,
67
68    /// Whether or not to use the unified origin detection behavior.
69    ///
70    /// When set to `true`, all detected entity IDs -- UDS Origin Detection, `dd.internal.entity_id`, container ID from
71    /// DogStatsD payload -- will be used for querying tags to enrich with. When set to `false`, the original precedence
72    /// behavior will be used, which enriches with the entity ID detected via Origin Detection first, and then
73    /// potentially again with either the client-provided entity ID (`dd.internal.entity_id`) or the container ID from
74    /// the DogStatsD payload, with the client-provided entity ID taking precedence. An entity ID detected via Origin
75    /// Detection is only used when no client-provided entity ID was present, or when `entity_id_precedence` is
76    /// `false`.
77    pub origin_detection_unified: bool,
78
79    /// Whether or not to opt out of origin detection for DogStatsD metrics.
80    ///
81    /// When set to `true`, and the metric explicitly denotes a cardinality of `"none"`, origin enrichment will be
82    /// skipped. This is only applicable to DogStatsD metrics when unified origin detection behavior isn't enabled.
83    pub origin_detection_optout: bool,
84
85    /// Whether or not to parse client-provided origin fields from DogStatsD payloads.
86    ///
87    /// When enabled, the `c:` (Local Data), `e:` (External Data), and `card:` (Cardinality) protocol fields are
88    /// parsed and used for origin enrichment.
89    pub origin_detection_client: bool,
90}

127    fn collect_origin_tags_with_process_entity(
128        &self, origin: &ResolvedOrigin, process_entity_id: Option<&EntityId>,
129    ) -> SharedTagSet {
130        let mut collected_tags = SharedTagSet::default();
131
132        if !self.config.enabled {
133            return collected_tags;
134        }
135
136        // Examine the various possible entity ID values, and based on their state, use one or more of them to grab any
137        // enriched tags attached to the entities. We evalulate a number of possible entity IDs:
138        //
139        // - sender entity (resolved from UDS socket credentials when the packet is received)
140        // - Local Data-based container ID (extracted from special "container ID" extension in DogStatsD protocol; also known as Local Data)
141        // - Local Data-based pod UID (extracted from `dd.internal.entity_id` tag)
142        // - External Data-based container ID (derived from pod UID and container name in External Data)
143        // - External Data-based pod UID (raw pod UID from External Data)
144        let maybe_process_entity_id = process_entity_id;
145        let maybe_local_container_id = origin.local_data();
146        let maybe_local_pod_uid = origin.pod_uid();
147        let maybe_external_container_id = origin.resolved_external_data().map(|red| red.container_entity_id());
148        let maybe_external_pod_uid = origin.resolved_external_data().map(|red| red.pod_entity_id());
149
150        let tag_cardinality = origin.cardinality().unwrap_or(self.config.tag_cardinality);
151
152        if !self.config.origin_detection_unified {
153            if self.config.origin_detection_optout && tag_cardinality == OriginTagCardinality::None {
154                trace!("Skipping origin enrichment for DogStatsD metric with cardinality 'none'.");
155                return collected_tags;
156            }
157
158            // If we discovered an entity ID from socket credentials, and no client-provided entity ID was provided (or
159            // it was, but entity ID precedence is disabled), then try to get tags for the detected entity ID.
160            if let Some(entity_id) = maybe_process_entity_id {
161                if maybe_local_pod_uid.is_none() || !self.config.entity_id_precedence {
162                    if let Some(tags) = self.workload_provider.get_tags_for_entity(entity_id, tag_cardinality) {
163                        collected_tags.extend_from_shared(&tags);
164                    } else {
165                        trace!(
166                            ?entity_id,
167                            cardinality = tag_cardinality.as_str(),
168                            "No tags found for entity."
169                        );
170                    }
171                }
172            }
173
174            // If we have a client-provided entity ID, try to get tags for the entity based on those. A client-provided
175            // entity ID takes precedence over the container ID.
176            let maybe_entity_id = maybe_local_pod_uid.or(maybe_local_container_id);
177            if let Some(entity_id) = maybe_entity_id {
178                if let Some(tags) = self.workload_provider.get_tags_for_entity(entity_id, tag_cardinality) {
179                    collected_tags.extend_from_shared(&tags);
180                } else {
181                    trace!(
182                        ?entity_id,
183                        cardinality = tag_cardinality.as_str(),
184                        "No tags found for entity."
185                    );
186                }
187            }
188        } else {
189            if tag_cardinality == OriginTagCardinality::None {
190                trace!("Skipping origin enrichment for metric with cardinality 'none'.");
191                return collected_tags;
192            }
193
194            // Evaluate all available entity IDs in order of priority: Local Data-based container ID, sender entity,
195            // External Data-based container ID, Local Data-based pod UID, and External Data-based pod UID.
196            //
197            // As soon as the first set of tags for an entity ID is found, we skip the remaining entity IDs.
198            let maybe_entity_ids = &[
199                maybe_local_container_id,
200                maybe_process_entity_id,
201                maybe_external_container_id,
202                maybe_local_pod_uid,
203                maybe_external_pod_uid,
204            ];
205            for entity_id in maybe_entity_ids.iter().flatten() {
206                if let Some(tags) = self.workload_provider.get_tags_for_entity(entity_id, tag_cardinality) {
207                    if !tags.is_empty() {
208                        collected_tags.extend_from_shared(&tags);
209                        break;
210                    }
211                } else {
212                    trace!(
213                        ?entity_id,
214                        cardinality = tag_cardinality.as_str(),
215                        "No tags found for entity."
216                    );
217                }
218            }
219        }
220
221        collected_tags
222    }
223
224    fn collect_origin_tags(&self, origin: &ResolvedOrigin) -> SharedTagSet {
225        self.collect_origin_tags_with_process_entity(origin, origin.process_id())
226    }
227
228    pub(super) fn resolve_origin_tags_with_process_origin(
229        &self, mut origin: RawOrigin<'_>, process_origin: Option<&ProcessOrigin>,
230    ) -> SharedTagSet {
231        match process_origin {
232            Some(ProcessOrigin::Replay(process_id)) => {
233                origin.set_process_id(mark_replay_process_id(*process_id));
234                self.resolve_origin_tags(origin)
235            }
236            Some(ProcessOrigin::Unpinned(process_id)) => {
237                origin.set_process_id(*process_id);
238                self.resolve_origin_tags(origin)
239            }
240            Some(ProcessOrigin::Pinned(process_entity_id)) => {
241                let resolved_origin = self
242                    .workload_provider
243                    .get_resolved_origin(origin.clone())
244                    .unwrap_or_else(|| ResolvedOrigin::from_parts(origin.cardinality(), None, None, None, None));
245                self.collect_origin_tags_with_process_entity(&resolved_origin, process_entity_id.as_ref())
246            }
247            None => self.resolve_origin_tags(origin),
248        }
249    }
250}

252impl OriginTagsResolver for DogStatsDOriginTagResolver {
253    fn resolve_origin_tags(&self, origin: RawOrigin<'_>) -> SharedTagSet {
254        // Replay traffic is tagged by setting the marker bit on the origin process ID. It bypasses the live enrichment
255        // pipeline entirely: the captured `TaggerState` already contains fully-resolved tags per entity, so the
256        // resolver's entity-ID-walking logic has nothing to add.
257        if let Some(captured_process_id) = origin.process_id().and_then(captured_process_id_from_replay) {
258            if let Some(store) = self.captured_tagger.current() {
259                let cardinality = origin.cardinality().unwrap_or(self.config.tag_cardinality);
260                return store
261                    .lookup(captured_process_id as i32, cardinality)
262                    .unwrap_or_default();
263            }
264            trace!(?origin, "Replay-flagged origin but no captured tagger available.");
265            return SharedTagSet::default();
266        }
267
268        match self.workload_provider.get_resolved_origin(origin.clone()) {
269            Some(resolved_origin) => self.collect_origin_tags(&resolved_origin),
270            None => {
271                trace!(?origin, "No resolved origin found for origin.");
272                SharedTagSet::default()
273            }
274        }
275    }

278/// Builds a `RawOrigin` from the origin fields of a DogStatsD packet and its well-known tags.
279///
280/// Metric, event, and service check packets all carry the same origin fields.
281pub fn origin_from_packet<'a>(
282    local_data: Option<&'a str>, external_data: Option<&'a str>, cardinality: Option<OriginTagCardinality>,
283    well_known_tags: &WellKnownTags<'a>,
284) -> RawOrigin<'a> {
285    let cardinality = cardinality.or(well_known_tags.cardinality);
286
287    let mut origin = RawOrigin::default();
288    origin.set_pod_uid(well_known_tags.pod_uid);
289    origin.set_local_data(local_data);
290    origin.set_external_data(external_data);
291    origin.set_cardinality(cardinality);
292    origin
293}

517        // and don't take any other tags.
