46pub(super) struct ConnectedPacketForwarder {
47    pub(super) socket: UdpSocket,
48    pub(super) target: SocketAddr,
49}
50
51impl ConnectedPacketForwarder {
52    pub(super) async fn connect(host: &str, port: u16) -> std::io::Result<Self> {
53        match Self::connect_from_bind_addr(FORWARDER_IPV4_BIND_ADDR, host, port).await {
54            Ok(forwarder) => Ok(forwarder),
55            Err(ipv4_error) => {
56                debug!(
57                    %host,
58                    port,
59                    error = %ipv4_error,
60                    "Could not connect to statsd forward target with IPv4 UDP socket."
61                );
62                Self::connect_from_bind_addr(FORWARDER_IPV6_BIND_ADDR, host, port)
63                    .await
64                    .map_err(|ipv6_error| {
65                        std::io::Error::new(
66                            ipv6_error.kind(),
67                            format!(
68                                "could not connect to statsd forward target with IPv4 or IPv6 UDP socket: \
69                                 IPv4 error: {ipv4_error}; IPv6 error: {ipv6_error}"
70                            ),
71                        )
72                    })
73            }
74        }
75    }
76
77    async fn connect_from_bind_addr(bind_addr: &str, host: &str, port: u16) -> std::io::Result<Self> {
78        let socket = UdpSocket::bind(bind_addr).await?;
79        socket.connect((host, port)).await?;
80        timeout(FORWARDER_SOCKET_READY_TIMEOUT, socket.writable())
81            .await
82            .map_err(|_| {
83                std::io::Error::new(std::io::ErrorKind::TimedOut, "timed out waiting for forwarder socket")
84            })??;
85
86        let target = socket.peer_addr()?;
87        Ok(Self { socket, target })
88    }
89
90    pub(super) async fn run(self, mut packets_rx: mpsc::Receiver<ForwardPacket>, metrics: Metrics) {
91        while let Some(packet) = packets_rx.recv().await {
92            match self.socket.send(&packet.payload).await {
93                Ok(bytes_sent) => {
94                    metrics.packets_forwarded().increment(1);
95                    metrics.bytes_forwarded().increment(bytes_sent as u64);
96                }
97                Err(e) => {
98                    metrics.packet_forwarding_errors().increment(1);
99                    debug!(target = %self.target, error = %e, "Failed to forward DogStatsD packet.");
100                }
101            }
102        }
103    }
104}

116#[derive(Clone)]
117pub(super) struct PacketForwarder {
118    target_host: MetaString,
119    target_port: u16,
120    metrics: Metrics,
121    pub(super) connected: Arc<OnceLock<mpsc::Sender<ForwardPacket>>>,
122}
123
124impl PacketForwarder {
125    /// Starts connecting to the forward target in the background.
126    ///
127    /// Connecting is deliberately off the caller's path -- a listener shouldn't wait on a remote target before it can
128    /// accept traffic -- so this returns as soon as the child is registered, not once the target is reachable.
129    ///
130    /// # Errors
131    ///
132    /// If the component's supervisor is no longer running, an error is returned.
133    pub(super) async fn spawn_connect(&self, spawner: &ComponentSpawner) -> Result<(), SpawnError> {
134        let forwarder = self.clone();
135        let forwarder_spawner = spawner.clone();
136
137        spawner
138            .spawn_interruptible("packet_forwarder_connect", async move {
139                forwarder.connect(&forwarder_spawner).await;
140            })
141            .await?;
142
143        Ok(())
144    }
145
146    async fn connect(self, spawner: &ComponentSpawner) {
147        let host = &self.target_host;
148        let port = self.target_port;
149
150        let forwarder = match timeout(FORWARDER_CONNECT_TIMEOUT, ConnectedPacketForwarder::connect(host, port)).await {
151            Err(e) => {
152                warn!(%host, port, error = %e, "Timed out connecting to statsd forward target. Packet forwarding disabled.");
153                return;
154            }
155            Ok(Err(e)) => {
156                warn!(%host, port, error = %e, "Failed to connect to statsd forward target. Packet forwarding disabled.");
157                return;
158            }
159            Ok(Ok(forwarder)) => forwarder,
160        };
161
162        let target = forwarder.target;
163        let (packets_tx, packets_rx) = mpsc::channel(FORWARDER_QUEUE_CAPACITY);
164
165        // Spawn the packet forwarder in noninterruptible mode so that it attempts to finish forwarding any
166        // in-flight packets prior to shutdown.
167        let metrics = self.metrics.clone();
168        if let Err(e) = spawner
169            .spawn_noninterruptible("packet_forwarder", move |_shutdown| forwarder.run(packets_rx, metrics))
170            .await
171        {
172            warn!(%target, error = %e, "Could not start statsd packet forwarder. Packet forwarding disabled.");
173            return;
174        }
175
176        // Set our packet channel, which signals to the listeners that packets can now be forwarded.
177        //
178        // If we get raced somehow and we lose the `set` call here, the dropping of `packets_tx` will cascade
179        // to our forwarder task causing it to exit, so cleanup is handled automatically.
180        if self.connected.set(packets_tx).is_err() {
181            debug!("DogStatsD packet forwarding was already initialized.");
182        } else {
183            info!(%target, "DogStatsD packet forwarding enabled.");
184        }
185    }
186
187    pub(super) async fn forward(&self, payload: Bytes) {
188        if payload.is_empty() {
189            return;
190        }
191
192        if let Some(packets_tx) = self.connected.get() {
193            let packet = ForwardPacket::from_payload(payload);
194            if packets_tx.send(packet).await.is_err() {
195                self.metrics.packet_forwarding_errors().increment(1);
196                debug!("Failed to enqueue DogStatsD packet for forwarding: receiver dropped.");
197            }
198        }
199    }
200}

225    #[tokio::test]
226    async fn connect_enables_forwarding_and_delivers_payload() {
227        // Connecting to a live UDP target enables forwarding, and `forward` delivers the payload verbatim.
228        let target = UdpSocket::bind("127.0.0.1:0").await.expect("target should bind");
229        let target_addr = target.local_addr().expect("target should have an address");
230
231        let supervisor = TestComponentSupervisor::start("dogstatsd").await;
232        let spawner = supervisor.spawner();
233
234        let forwarder = build_forwarder(&target_addr.ip().to_string(), target_addr.port());
235
236        let forwarder_spawn = forwarder.clone();
237        forwarder_spawn.connect(&spawner).await;
238
239        assert!(
240            forwarder.connected.get().is_some(),
241            "connecting to a live target must enable forwarding"
242        );
243
244        forwarder.forward(Bytes::from_static(b"metric.name:1|c")).await;
245
246        let mut buf = [0u8; 64];
247        let received = timeout(Duration::from_secs(5), target.recv(&mut buf))
248            .await
249            .expect("forwarded payload should arrive before the timeout")
250            .expect("recv should succeed");
251        assert_eq!(&buf[..received], b"metric.name:1|c");
252
253        // The forwarding loop is a supervised child, so it goes away with the component rather than outliving it.
254        assert_eq!(supervisor.active_children(), 1);
255
256        // The loop stops on its senders closing, not on the shutdown signal, so release our clone before shutting
257        // down. Holding it here would leave the loop waiting until the shutdown budget elapsed.
258        drop(forwarder);
259        assert!(supervisor.shutdown().await.is_ok());
260    }

264        // A second connect must not replace the already-established sender (the `OnceLock::set` is a no-op once set).
265        let target = UdpSocket::bind("127.0.0.1:0").await.expect("target should bind");
266        let target_addr = target.local_addr().expect("target should have an address");
267        let forwarder = build_forwarder(&target_addr.ip().to_string(), target_addr.port());
268
269        let supervisor = TestComponentSupervisor::start("dogstatsd").await;
270        let spawner = supervisor.spawner();
271
272        let forwarder_spawn = forwarder.clone();
273        forwarder_spawn.connect(&spawner).await;
274
275        let first = forwarder
276            .connected
277            .get()
278            .expect("first connect should enable forwarding")
279            .clone();
280
281        let forwarder_spawn = forwarder.clone();
282        forwarder_spawn.connect(&spawner).await;
283
284        let second = forwarder
285            .connected
286            .get()
287            .expect("forwarding should stay enabled after a second connect");

326    #[tokio::test]
327    async fn queued_packets_are_forwarded_during_shutdown() {
328        // Packets already queued when shutdown begins must still be sent. An interruptible forwarding loop would be
329        // dropped at its current await point and take the whole queue with it -- measured at 64 of 1024 delivered.
330        let target = UdpSocket::bind("127.0.0.1:0").await.expect("target should bind");
331        let target_addr = target.local_addr().expect("target should have an address");
332
333        let supervisor = TestComponentSupervisor::start("dogstatsd").await;
334        let spawner = supervisor.spawner();
335
336        let forwarder = build_forwarder(&target_addr.ip().to_string(), target_addr.port());
337
338        let forwarder_spawn = forwarder.clone();
339        forwarder_spawn.connect(&spawner).await;
340
341        // Fill the queue via the sender directly. Going through `forward` would await `send`, which yields to the
342        // loop on every call and paces it, so no backlog would ever build up.
343        let packets_tx = forwarder
344            .connected
345            .get()
346            .expect("connecting to a live target must enable forwarding")
347            .clone();
348
349        let mut queued = 0;
350        for i in 0..FORWARDER_QUEUE_CAPACITY {
351            let payload = Bytes::from(format!("queued.metric.{i}:1|c"));
352            if packets_tx.try_send(ForwardPacket::from_payload(payload)).is_err() {
353                break;
354            }
355            queued += 1;
356        }
357        assert_eq!(
358            queued, FORWARDER_QUEUE_CAPACITY,
359            "the queue should have accepted a full batch"
360        );
361
362        // Drain the target concurrently: sending a full batch before reading any of it would overflow the socket's
363        // receive buffer and lose packets in the harness rather than in the code under test.
364        let receiver = tokio::spawn(async move {
365            let mut received = 0;
366            let mut buf = [0u8; 128];
367            while let Ok(Ok(_)) = timeout(RECEIVE_IDLE_TIMEOUT, target.recv(&mut buf)).await {
368                received += 1;
369            }
370            received
371        });
372
373        // Close every sender, which is what lets the loop finish, and then shut down while it is still draining.
374        drop(packets_tx);
375        drop(forwarder);
376        supervisor.shutdown().await.expect("supervisor should stop cleanly");
377
378        let received = receiver.await.expect("receiver should not panic");
379        assert_eq!(
380            received, queued,
381            "every queued packet should have been forwarded before the loop stopped"
382        );
383    }
