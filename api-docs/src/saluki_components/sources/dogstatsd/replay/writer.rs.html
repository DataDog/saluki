63#[derive(Default)]
64struct WriterState {
65    ongoing: bool,
66    accepting: bool,
67    target_path: Option<PathBuf>,
68    compressed: bool,
69    traffic_tx: Option<SyncSender<CaptureRecord>>,

72enum CaptureFileWriter {
73    Plain(BufWriter<File>),
74    Compressed(ZstdEncoder<'static, BufWriter<File>>),
75}

77impl CaptureFileWriter {
78    fn finish(self) -> io::Result<()> {
79        match self {
80            Self::Plain(mut writer) => writer.flush(),
81            Self::Compressed(writer) => {
82                let mut writer = writer.finish()?;
83                writer.flush()
84            }
85        }
86    }
87}

89impl Write for CaptureFileWriter {
90    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
91        match self {
92            Self::Plain(writer) => writer.write(buf),
93            Self::Compressed(writer) => writer.write(buf),
94        }
95    }
96
97    fn flush(&mut self) -> io::Result<()> {
98        match self {
99            Self::Plain(writer) => writer.flush(),
100            Self::Compressed(writer) => writer.flush(),
101        }
102    }
103}

105impl TrafficCaptureWriter {
106    /// Creates a new capture writer with the given queue depth.
107    #[cfg(test)]
108    pub(super) fn new(queue_depth: usize) -> Self {
109        Self::with_workload_provider(queue_depth, None)
110    }
111
112    /// Creates a new capture writer with the given queue depth and optional workload provider.
113    pub(super) fn with_workload_provider(
114        queue_depth: usize, workload_provider: Option<Arc<dyn WorkloadProvider + Send + Sync>>,
115    ) -> Self {
116        Self {
117            queue_depth: queue_depth.max(MIN_CAPTURE_DEPTH),
118            workload_provider,
119            ongoing: Arc::new(AtomicBool::new(false)),
120            state: Arc::new(Mutex::new(WriterState::default())),
121        }
122    }
123
124    /// Returns whether a capture session is currently active.
125    pub(super) fn is_ongoing(&self) -> bool {
126        self.ongoing.load(Ordering::Acquire)
127    }
128
129    /// Starts a new capture session.
130    pub(super) fn start_capture(
131        &self, target_dir: CaptureTargetDir, duration: Duration, compressed: bool,
132    ) -> Result<PathBuf, GenericError> {
133        let mut state = self.state.lock().expect("capture writer mutex poisoned");
134
135        if state.ongoing {
136            return Err(generic_error!("capture already in progress"));
137        }
138
139        let target_path = resolve_target_path(&target_dir)?;
140        let mut writer = open_target_writer(&target_path, compressed)?;
141        write_header(&mut writer).map_err(|e| generic_error!("Failed to write capture file header: {}", e))?;
142        let (traffic_tx, traffic_rx) = mpsc::sync_channel(self.queue_depth);
143
144        state.ongoing = true;
145        state.accepting = true;
146        state.target_path = Some(target_path.clone());
147        state.compressed = compressed;
148        state.traffic_tx = Some(traffic_tx);
149        self.ongoing.store(true, Ordering::Release);
150
151        let shared_state = Arc::clone(&self.state);
152        let ongoing = Arc::clone(&self.ongoing);
153        let workload_provider = self.workload_provider.clone();
154        if let Err(e) = thread::Builder::new()
155            .name("dogstatsd-capture-writer".into())
156            .spawn(move || run_capture_loop(shared_state, ongoing, traffic_rx, writer, duration, workload_provider))
157        {
158            state.ongoing = false;
159            state.accepting = false;
160            state.target_path = None;
161            state.compressed = false;
162            state.traffic_tx = None;
163            self.ongoing.store(false, Ordering::Release);
164            return Err(generic_error!("Failed to spawn capture writer thread: {}", e));
165        }
166
167        Ok(target_path)
168    }
169
170    /// Stops the current capture session, if one is running.
171    pub(super) fn stop_capture(&self) {
172        let traffic_tx = {
173            let mut state = self.state.lock().expect("capture writer mutex poisoned");
174
175            if !state.ongoing {
176                return;
177            }
178
179            state.accepting = false;
180            state.traffic_tx.take()
181        };
182
183        drop(traffic_tx);
184    }
185
186    /// Attempts to enqueue a captured packet for persistence.
187    pub(super) fn enqueue(&self, record: CaptureRecord) -> bool {
188        let traffic_tx = {
189            let state = self.state.lock().expect("capture writer mutex poisoned");
190            if !state.accepting {
191                return false;
192            }
193
194            state.traffic_tx.clone()
195        };
196
197        match traffic_tx {
198            Some(traffic_tx) => traffic_tx.send(record).is_ok(),
199            None => false,
200        }
201    }
202}

204fn resolve_target_path(target_dir: &CaptureTargetDir) -> Result<PathBuf, GenericError> {
205    if target_dir.as_path().as_os_str().is_empty() {
206        return Err(generic_error!(
207            "DogStatsD capture path is not configured and no explicit path was provided."
208        ));
209    }
210
211    ensure_target_dir_exists(target_dir)?;
212
213    let timestamp = get_unix_timestamp();
214
215    Ok(target_dir
216        .as_path()
217        .join(format!("{}-{}", FILE_TEMPLATE_PREFIX, timestamp)))
218}
219
220fn ensure_target_dir_exists(target_dir: &CaptureTargetDir) -> Result<(), GenericError> {
221    match fs::metadata(target_dir.as_path()) {
222        Ok(metadata) => {
223            if !metadata.is_dir() {
224                return Err(generic_error!(
225                    "DogStatsD capture path '{}' is not a directory.",
226                    target_dir.as_path().display()
227                ));
228            }
229        }
230        Err(e) if e.kind() == io::ErrorKind::NotFound => match target_dir {
231            CaptureTargetDir::Implicit(path) => {
232                fs::create_dir_all(path).map_err(|create_err| {
233                    generic_error!(
234                        "Failed to create DogStatsD capture directory '{}': {}",
235                        path.display(),
236                        create_err
237                    )
238                })?;
239            }
240            CaptureTargetDir::Explicit(path) => {
241                return Err(generic_error!(
242                    "DogStatsD capture path '{}' does not exist.",
243                    path.display()
244                ));
245            }
246        },
247        Err(e) => {
248            return Err(generic_error!(
249                "Failed to inspect DogStatsD capture path '{}': {}",
250                target_dir.as_path().display(),
251                e
252            ));
253        }
254    }
255
256    Ok(())
257}

259fn open_target_writer(target_path: &Path, compressed: bool) -> Result<CaptureFileWriter, GenericError> {
260    let file = File::options()
261        .write(true)
262        .create_new(true)
263        .open(target_path)
264        .map_err(|e| {
265            generic_error!(
266                "Failed to create DogStatsD capture file '{}': {}",
267                target_path.display(),
268                e
269            )
270        })?;
271    let writer = BufWriter::new(file);
272
273    if compressed {
274        let writer = ZstdEncoder::new(writer, 0).map_err(|e| {
275            generic_error!(
276                "Failed to create compressed DogStatsD capture writer for '{}': {}",
277                target_path.display(),
278                e
279            )
280        })?;
281        Ok(CaptureFileWriter::Compressed(writer))
282    } else {
283        Ok(CaptureFileWriter::Plain(writer))
284    }
285}

287fn run_capture_loop(
288    state: Arc<Mutex<WriterState>>, ongoing: Arc<AtomicBool>, traffic_rx: Receiver<CaptureRecord>,
289    mut writer: CaptureFileWriter, duration: Duration,
290    workload_provider: Option<Arc<dyn WorkloadProvider + Send + Sync>>,
291) {
292    let mut pid_map = FastHashMap::<i32, String>::default();
293    let start = std::time::Instant::now();
294
295    loop {
296        let remaining = duration.saturating_sub(start.elapsed());
297        if remaining.is_zero() {
298            debug!("DogStatsD capture reached requested duration.");
299            break;
300        }
301
302        match traffic_rx.recv_timeout(remaining) {
303            Ok(record) => {
304                if let Err(e) = write_record(&mut writer, record, &mut pid_map) {
305                    error!("Failed to write captured DogStatsD record: {}", e);
306                    break;
307                }
308            }
309            Err(RecvTimeoutError::Timeout) => break,
310            Err(RecvTimeoutError::Disconnected) => break,
311        }
312    }
313
314    if let Err(e) = write_state(&mut writer, pid_map, duration, workload_provider.as_deref()) {
315        warn!("Failed to write DogStatsD capture state trailer: {}", e);
316    }
317
318    if let Err(e) = writer.finish() {
319        error!("Failed to finish DogStatsD capture file: {}", e);
320    }
321
322    let mut state = state.lock().expect("capture writer mutex poisoned");
323    state.ongoing = false;
324    state.accepting = false;
325    state.target_path = None;
326    state.compressed = false;
327    state.traffic_tx = None;
328    ongoing.store(false, Ordering::Release);
329}

331fn write_record(
332    writer: &mut CaptureFileWriter, record: CaptureRecord, pid_map: &mut FastHashMap<i32, String>,
333) -> io::Result<()> {
334    let CaptureRecord {
335        timestamp_ns,
336        payload,
337        pid,
338        ancillary,
339        container_id,
340    } = record;
341
342    if let (Some(pid), Some(container_id)) = (pid, container_id) {
343        pid_map.insert(pid, container_id);
344    }
345
346    let payload_size = payload.len() as i32;
347    let ancillary_size = ancillary.len() as i32;
348    let message = UnixDogstatsdMsg {
349        timestamp: timestamp_ns,
350        payload_size,
351        payload,
352        pid: pid.unwrap_or_default(),
353        ancillary_size,
354        ancillary,
355    };
356    let serialized = message.encode_to_vec();
357
358    writer.write_all(&(serialized.len() as u32).to_le_bytes())?;
359    writer.write_all(&serialized)?;
360    Ok(())
361}

363fn write_state(
364    writer: &mut CaptureFileWriter, pid_map: FastHashMap<i32, String>, duration: Duration,
365    workload_provider: Option<&(dyn WorkloadProvider + Send + Sync)>,
366) -> io::Result<()> {
367    let state = TaggerState {
368        state: build_saved_entity_state(&pid_map, workload_provider)
369            .into_iter()
370            .collect(),
371        pid_map: pid_map.into_iter().collect(),
372        duration: duration.as_millis().min(i64::MAX as u128) as i64,
373    };
374    let serialized = state.encode_to_vec();
375
376    writer.write_all(&[0, 0, 0, 0])?;
377    writer.write_all(&serialized)?;
378    writer.write_all(&(serialized.len() as u32).to_le_bytes())?;
379    Ok(())
380}

382fn build_saved_entity_state(
383    pid_map: &FastHashMap<i32, String>, workload_provider: Option<&(dyn WorkloadProvider + Send + Sync)>,
384) -> FastHashMap<String, Entity> {
385    let Some(workload_provider) = workload_provider else {
386        return FastHashMap::default();
387    };
388
389    let mut saved_entities = FastHashMap::default();
390    let mut seen_entities = HashSet::new();
391
392    for raw_entity_id in pid_map.values() {
393        let Some(entity_id) = parse_entity_id_str(raw_entity_id) else {
394            warn!(
395                raw_entity_id,
396                "Skipping captured entity with an unsupported ID while building capture state trailer."
397            );
398            continue;
399        };
400
401        if !seen_entities.insert(entity_id.clone()) {
402            continue;
403        }
404
405        let Some(entity) = build_saved_entity(workload_provider, &entity_id) else {
406            continue;
407        };
408
409        saved_entities.insert(entity_id.to_string(), entity);
410    }
411
412    saved_entities
413}

415fn build_saved_entity(
416    workload_provider: &(dyn WorkloadProvider + Send + Sync), entity_id: &EntityId,
417) -> Option<Entity> {
418    let low_tags =
419        sorted_unique_tag_strings(workload_provider.get_tags_for_entity(entity_id, OriginTagCardinality::Low));
420    let orchestrator_cumulative =
421        sorted_unique_tag_strings(workload_provider.get_tags_for_entity(entity_id, OriginTagCardinality::Orchestrator));
422    let high_cumulative =
423        sorted_unique_tag_strings(workload_provider.get_tags_for_entity(entity_id, OriginTagCardinality::High));
424
425    let orchestrator_tags = subtract_tag_strings(orchestrator_cumulative.clone(), &low_tags);
426    let high_tags = subtract_tag_strings(high_cumulative, &orchestrator_cumulative);
427
428    if low_tags.is_empty() && orchestrator_tags.is_empty() && high_tags.is_empty() {
429        return None;
430    }
431
432    Some(Entity {
433        id: Some(entity_id_to_remote_entity_id(entity_id)),
434        low_cardinality_tags: low_tags,
435        orchestrator_cardinality_tags: orchestrator_tags,
436        high_cardinality_tags: high_tags,
437        ..Default::default()
438    })
439}

441fn sorted_unique_tag_strings(tags: Option<SharedTagSet>) -> Vec<String> {
442    let Some(tags) = tags else {
443        return Vec::new();
444    };
445
446    let unique = (&tags)
447        .into_iter()
448        .map(|tag| tag.as_str().to_string())
449        .collect::<BTreeSet<_>>();
450    unique.into_iter().collect()
451}
452
453fn subtract_tag_strings(current: Vec<String>, lower: &[String]) -> Vec<String> {
454    if lower.is_empty() {
455        return current;
456    }
457
458    let lower = lower.iter().map(String::as_str).collect::<HashSet<_>>();
459    current
460        .into_iter()
461        .filter(|tag| !lower.contains(tag.as_str()))
462        .collect()
463}

465fn entity_id_to_remote_entity_id(entity_id: &EntityId) -> RemoteEntityId {
466    match entity_id {
467        EntityId::Container(container_id) => RemoteEntityId {
468            prefix: "container_id".to_string(),
469            uid: container_id.to_string(),
470        },
471        EntityId::PodUid(pod_uid) => RemoteEntityId {
472            prefix: "kubernetes_pod_uid".to_string(),
473            uid: pod_uid.to_string(),
474        },
475        EntityId::Global => RemoteEntityId {
476            prefix: "internal".to_string(),
477            uid: "global-entity-id".to_string(),
478        },
479        EntityId::ContainerPid(pid) => RemoteEntityId {
480            prefix: "container_pid".to_string(),
481            uid: pid.to_string(),
482        },
483        EntityId::ContainerInode(inode) => RemoteEntityId {
484            prefix: "container_inode".to_string(),
485            uid: inode.to_string(),
486        },
487    }
488}

490pub(super) fn parse_entity_id_str(value: &str) -> Option<EntityId> {
491    const CONTAINER_ID_PREFIX: &str = "container_id://";
492    const POD_UID_PREFIX: &str = "kubernetes_pod_uid://";
493    const CONTAINER_PID_PREFIX: &str = "container_pid://";
494    const CONTAINER_INODE_PREFIX: &str = "container_inode://";
495
496    if let Some(container_id) = value.strip_prefix(CONTAINER_ID_PREFIX) {
497        Some(EntityId::Container(container_id.into()))
498    } else if let Some(pod_uid) = value.strip_prefix(POD_UID_PREFIX) {
499        Some(EntityId::PodUid(pod_uid.into()))
500    } else if let Some(pid) = value.strip_prefix(CONTAINER_PID_PREFIX) {
501        pid.parse().ok().map(EntityId::ContainerPid)
502    } else if let Some(inode) = value.strip_prefix(CONTAINER_INODE_PREFIX) {
503        inode.parse().ok().map(EntityId::ContainerInode)
504    } else if value == "system://global" {
505        Some(EntityId::Global)
506    } else {
507        None
508    }
509}

693    fn assert_capture_contents(bytes: &[u8]) {
694        assert!(valid_header(bytes));
695
696        let header_len = 8;
697        let record_size =
698            u32::from_le_bytes(bytes[header_len..header_len + 4].try_into().expect("record size")) as usize;
699        let record_start = header_len + 4;
700        let record_end = record_start + record_size;
701        let record = UnixDogstatsdMsg::decode(&bytes[record_start..record_end]).expect("record should decode");
702
703        assert_eq!(record.timestamp, 123);
704        assert_eq!(record.payload, b"test");
705        assert_eq!(record.payload_size, 4);
706        assert_eq!(record.pid, 42);
707        assert!(record.ancillary.is_empty());
708        assert_eq!(record.ancillary_size, 0);
709
710        assert_eq!(&bytes[record_end..record_end + 4], &[0, 0, 0, 0]);
711
712        let state = decode_capture_state(bytes);
713
714        assert!(
715            state.state.is_empty(),
716            "captures without a workload provider should not include entity state"
717        );
718        assert_eq!(
719            state.pid_map.get(&42).map(String::as_str),
720            Some("container_id://container-123")
721        );
722        assert!(!state.duration.is_negative());
723    }
