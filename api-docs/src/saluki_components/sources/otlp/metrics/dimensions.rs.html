13/// A helper struct for building the identity of a metric.
14#[derive(Clone, Debug, Default)]
15pub struct Dimensions {
16    pub name: String,
17    pub tags: SharedTagSet,
18    pub host: Option<MetaString>,
19    pub origin_id: Option<String>,
20}
21
22impl Dimensions {
23    /// Creates a new `Dimensions` with a suffix added to the name.
24    pub fn with_suffix(&self, suffix: &str) -> Self {
25        Self {
26            name: format!("{}.{}", self.name, suffix),
27            tags: self.tags.clone(),
28            host: self.host.clone(),
29            origin_id: self.origin_id.clone(),
30        }
31    }
32
33    /// Creates a new `Dimensions` with additional tags, structurally sharing the existing tags.
34    pub fn add_tags<I>(&self, tags_to_add: I) -> Self
35    where
36        I: IntoIterator<Item = String>,
37    {
38        let additions: SharedTagSet = tags_to_add
39            .into_iter()
40            .filter(|tag| !self.tags.has_tag(tag))
41            .map(Tag::from)
42            .collect();
43
44        let mut tags = self.tags.clone();
45        tags.extend_from_shared(&additions);
46
47        Self {
48            name: self.name.clone(),
49            tags,
50            host: self.host.clone(),
51            origin_id: self.origin_id.clone(),
52        }
53    }
54
55    /// Creates a new `Dimensions` with tags from an OTLP attribute map.
56    ///
57    /// When `shadowing_resource_attributes` is `Some`, any data-point attribute whose key also
58    /// appears in the resource attributes is skipped. This matches the Agent, where resource
59    /// attributes overwrite colliding data-point attributes when they are added as tags.
60    pub fn with_attribute_map(
61        &self, attributes: &[otlp_common::KeyValue], shadowing_resource_attributes: Option<&[otlp_common::KeyValue]>,
62    ) -> Self {
63        if attributes.is_empty() {
64            return self.clone();
65        }
66
67        let mut tag_set = TagSet::default();
68        for kv in attributes {
69            if let Some(resource_attributes) = shadowing_resource_attributes {
70                if resource_attributes.iter().any(|resource_kv| resource_kv.key == kv.key) {
71                    continue;
72                }
73            }
74
75            if let Some(value) = kv.value.as_ref().and_then(|v| v.value.as_ref()) {
76                let v_str = match value {
77                    otlp_common::any_value::Value::StringValue(s) => s.clone(),
78                    otlp_common::any_value::Value::BoolValue(b) => b.to_string(),
79                    otlp_common::any_value::Value::IntValue(i) => i.to_string(),
80                    otlp_common::any_value::Value::DoubleValue(d) => d.to_string(),
81                    // Bytes, arrays, and key-value lists are not converted to tags.
82                    _ => continue,
83                };
84                // Empty values render as `n/a`, matching the Agent's `FormatKeyValueTag`.
85                tag_set.insert_tag(format_key_value_tag(&kv.key, &v_str));
86            }
87        }
88
89        let mut tags = tag_set.into_shared();
90        tags.extend_from_shared(&self.tags);
91
92        Self {
93            name: self.name.clone(),
94            tags,
95            host: self.host.clone(),
96            origin_id: self.origin_id.clone(),
97        }
98    }
99
100    /// Creates a canonical cache key for the dimension set.
101    #[cfg(test)]
102    pub fn get_cache_key(&self) -> String {
103        let mut dimensions: Vec<String> = self.tags.into_iter().map(|t| t.to_string()).collect();
104
105        dimensions.push(format!("name:{}", self.name));
106
107        if let Some(host) = &self.host {
108            dimensions.push(format!("host:{}", host));
109        }
110        if let Some(origin_id) = &self.origin_id {
111            dimensions.push(format!("originID:{}", origin_id));
112        }
113
114        // Sort the dimensions alphabetically to ensure a canonical key.
115        dimensions.sort();
116
117        // Join with a null character separator.
118        dimensions.join("\0")
119    }
120
121    /// Creates a compact context key for the dimension set.
122    ///
123    /// The host remains a distinct scalar dimension, while the origin ID is hashed as an origin tag.
124    pub fn context_key(&self) -> ContextKey {
125        let (context_key, _) =
126            hash_context_with_host(&self.name, self.host.as_deref(), &self.tags, self.origin_id.as_deref());
127
128        context_key
129    }
130}

132#[cfg(test)]
133mod tests {
134    use saluki_context::tags::{Tag, TagSet};
135
136    use super::*;
137
138    fn dims_with_tags(name: &str, tags: &[&str]) -> Dimensions {
139        let tag_set: TagSet = tags.iter().map(|t| Tag::from(*t)).collect();
140        Dimensions {
141            name: name.to_string(),
142            tags: tag_set.into_shared(),
143            host: None,
144            origin_id: None,
145        }
146    }
147
148    #[test]
149    fn with_suffix_and_add_tags_preserve_origin_id() {
150        let dimensions = Dimensions {
151            name: "http.request.duration".to_string(),
152            tags: TagSet::default().into_shared(),
153            host: None,
154            origin_id: Some("container_id://container-123".to_string()),
155        };
156
157        assert_eq!(
158            dimensions.with_suffix("count").origin_id.as_deref(),
159            Some("container_id://container-123")
160        );
161        assert_eq!(
162            dimensions.add_tags(["env:prod".to_string()]).origin_id.as_deref(),
163            Some("container_id://container-123")
164        );
165    }
166
167    #[test]
168    fn add_tags_preserves_order_deduplication_and_base() {
169        let base = dims_with_tags("http.request.duration", &["env:prod", "service:web", "lower_bound:1.0"]);
170        let extended = base.add_tags(["lower_bound:1.0".to_string(), "upper_bound:2.0".to_string()]);
171
172        let extended_tags: Vec<&str> = extended.tags.into_iter().map(|t| t.as_str()).collect();
173        assert_eq!(
174            extended_tags,
175            vec!["env:prod", "service:web", "lower_bound:1.0", "upper_bound:2.0"]
176        );
177
178        let base_tags: Vec<&str> = base.tags.into_iter().map(|t| t.as_str()).collect();
179        assert_eq!(base_tags, vec!["env:prod", "service:web", "lower_bound:1.0"]);
180    }
181
182    #[test]
183    fn cache_key_is_independent_of_tag_order() {
184        let first = dims_with_tags("http.request.duration", &["service:web", "env:prod"]);
185        let second = dims_with_tags("http.request.duration", &["env:prod", "service:web"]);
186
187        assert_eq!(first.get_cache_key(), second.get_cache_key());
188    }
189
190    #[test]
191    fn context_key_is_independent_of_tag_order() {
192        let first = dims_with_tags("http.request.duration", &["service:web", "env:prod"]);
193        let second = dims_with_tags("http.request.duration", &["env:prod", "service:web"]);

195        assert_eq!(first.context_key(), second.context_key());
196    }

198    #[test]
199    fn add_tags_preserves_canonical_cache_key() {
200        let base = dims_with_tags("http.request.duration", &["service:web", "env:prod"]);
201        let extended = base.add_tags(["lower_bound:1.0".to_string(), "upper_bound:2.0".to_string()]);
202
203        let expected = [
204            "env:prod",
205            "lower_bound:1.0",
206            "name:http.request.duration",
207            "service:web",
208            "upper_bound:2.0",
209        ]
210        .join("\0");
211        assert_eq!(extended.get_cache_key(), expected);
212    }
