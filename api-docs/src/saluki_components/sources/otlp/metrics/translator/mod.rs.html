84fn infer_delta_interval(start_ts: u64, ts: u64) -> i64 {
85    if start_ts == 0 || start_ts > ts {
86        return 0;
87    }
88    // Convert OTLP nanosecond timestamps to seconds because Datadog interval inference operates on whole-second intervals.
89    let delta = (ts - start_ts) as f64 / 1e9;
90    let rounded_delta = f64::round(delta);
91
92    if f64::abs(rounded_delta - delta) < 0.05 {
93        return rounded_delta as i64;
94    }
95    0
96}

132    /// Translates a batch of OTLP `ResourceMetrics` into Saluki `Event`s.
133    /// This is the Rust equivalent of the Go `MapMetrics` function.
134    ///
135    /// Returns the translated events and the set of runtime languages detected from metric names
136    /// in this `ResourceMetrics`.
137    ///
138    /// The processing duration is recorded on the translator's latency histogram regardless of whether the
139    /// translation succeeds or fails, so the histogram captures the full distribution of processing times.
140    pub fn translate_metrics(
141        &mut self, resource_metrics: OtlpResourceMetrics, metrics: &Metrics,
142    ) -> Result<(IntoIter<Event>, FastHashSet<&'static str>), GenericError> {
143        let start = Instant::now();
144        let result = self.translate_metrics_inner(resource_metrics, metrics);
145        self.translator_metrics
146            .processing_duration()
147            .record(start.elapsed().as_secs_f64());
148        if result.is_err() {
149            self.translator_metrics.errors_translate().increment(1);
150        }
151        result
152    }

154    fn translate_metrics_inner(
155        &mut self, resource_metrics: OtlpResourceMetrics, metrics: &Metrics,
156    ) -> Result<(IntoIter<Event>, FastHashSet<&'static str>), GenericError> {
157        let mut events = Vec::new();
158        let mut detected_languages = FastHashSet::default();
159        let resource = resource_metrics.resource.unwrap_or_default();
160        let source = self.attribute_translator.resource_to_metric_source(&resource);
161
162        let resource_tag_mode = if self.config.resource_attributes_as_tags {
163            ResourceAttributeTagMode::All
164        } else {
165            ResourceAttributeTagMode::Mapped
166        };
167        let resource_attribute_tags = self
168            .attribute_translator
169            .tags_from_attributes(&resource.attributes, resource_tag_mode)
170            .into_shared();
171        self.resolved_origin_tags = self
172            .origin_tag_resolver
173            .resolve_resource_tags(&resource.attributes, self.config.tag_cardinality);
174
175        // Combine configured and resource-derived tags once per resource, then reuse them for every
176        // instrumentation scope.
177        let mut resource_tags = self.metric_tags.clone();
178        resource_tags.extend_from_shared(&resource_attribute_tags);
179
180        let host = match source {
181            Some(Source {
182                kind: SourceKind::HostnameKind,
183                identifier,
184            }) => Some(MetaString::from(identifier)),
185            Some(Source {
186                kind: SourceKind::AwsEcsFargateKind,
187                ..
188            }) => None,
189            None => Some(self.default_hostname.clone()),
190        };
191
192        for scope_metrics in resource_metrics.scope_metrics {
193            let scope_tags = {
194                let mut tags = TagSet::default();
195
196                if self.config.instrumentation_scope_metadata_as_tags {
197                    // Always add instrumentation scope tags, even if scope is `None`
198                    // to match the datadog agent's behavior which adds "n/a" values
199                    let scope_tags = match &scope_metrics.scope {
200                        Some(scope) => instrumentationscope::tags_from_instrumentation_scope_metadata(scope),
201                        None => instrumentationscope::tags_from_empty_instrumentation_scope(),
202                    };
203                    for tag in scope_tags {
204                        tags.insert_tag(tag);
205                    }
206                } else if self.config.instrumentation_library_metadata_as_tags {
207                    if let Some(scope) = &scope_metrics.scope {
208                        for tag in instrumentationlibrary::tags_from_instrumentation_library_metadata(scope) {
209                            tags.insert_tag(tag);
210                        }
211                    }
212                }
213
214                tags.into_shared()
215            };
216
217            let mut tags = resource_tags.clone();
218            tags.extend_from_shared(&scope_tags);
219
220            let mut new_metrics: Vec<OtlpMetric> = Vec::new();
221            for mut metric in scope_metrics.metrics {
222                if let Some(mappings) = RUNTIME_METRICS_MAPPINGS.get(metric.name.as_str()) {
223                    for (prefix, language) in RUNTIME_METRIC_PREFIX_LANGUAGE_MAP.iter() {
224                        if metric.name.starts_with(prefix) {
225                            detected_languages.insert(*language);
226                        }
227                    }
228
229                    if RUNTIME_REMAPPING_ENABLED {
230                        for mapping in mappings {
231                            if mapping.attributes.is_empty() {
232                                // If there are no attributes to match, just duplicate the metric with the new name.
233                                let mut new_metric = metric.clone();
234                                new_metric.name = mapping.mapped_name.to_string();
235                                new_metrics.push(new_metric);
236                                break;
237                            }
238                            if let Some(ref data) = metric.data {
239                                match data {
240                                    OtlpMetricData::Sum(_) => {
241                                        map_sum_runtime_metric_with_attributes(&metric, &mut new_metrics, mapping);
242                                    }
243                                    OtlpMetricData::Gauge(_) => {
244                                        map_gauge_runtime_metric_with_attributes(&metric, &mut new_metrics, mapping);
245                                    }
246                                    OtlpMetricData::Histogram(_) => {
247                                        map_histogram_runtime_metric_with_attributes(
248                                            &metric,
249                                            &mut new_metrics,
250                                            mapping,
251                                        );
252                                    }
253                                    _ => {}
254                                }
255                            }
256                        }
257                    }
258                }
259
260                if self.config.with_remapping {
261                    remap::remap_metrics(&mut new_metrics, &metric);
262                }
263
264                if self.config.with_otel_prefix {
265                    remap::rename_metric(&mut metric);
266                }
267
268                let mut translated_events =
269                    self.map_to_dd_format(metric, &tags, host.clone(), &resource.attributes, metrics);
270                events.append(&mut translated_events);
271            }
272
273            for metric in new_metrics {
274                let mut translated_events =
275                    self.map_to_dd_format(metric, &tags, host.clone(), &resource.attributes, metrics);
276                events.append(&mut translated_events);
277            }
278        }
279
280        metrics.metrics_received().increment(events.len() as u64);
281
282        Ok((events.into_iter(), detected_languages))
283    }

316    /// Translates a single OTLP `Metric` into a collection of Saluki `Event`s.
317    fn map_to_dd_format(
318        &mut self, metric: OtlpMetric, attribute_tags: &SharedTagSet, host: Option<MetaString>,
319        resource_attributes: &[OtlpKeyValue], metrics: &Metrics,
320    ) -> Vec<Event> {
321        let origin_id = self.attribute_translator.origin_id_from_attributes(resource_attributes);
322        let base_dims = Dimensions {
323            name: metric.name,
324            tags: attribute_tags.clone(),
325            host,
326            origin_id,
327        };
328
329        let context = TranslationContext {
330            resource_attributes,
331            metrics,
332        };
333
334        if let Some(data) = metric.data {
335            match data {
336                OtlpMetricData::Gauge(gauge) => {
337                    self.map_number_metrics(base_dims, gauge.data_points, DataType::Gauge, &context)
338                }
339                OtlpMetricData::Sum(sum) => match AggregationTemporality::try_from(sum.aggregation_temporality) {
340                    Ok(AggregationTemporality::Cumulative) => {
341                        if sum.is_monotonic {
342                            match self.config.cumulative_monotonic_mode {
343                                CumulativeMonotonicMode::ToDelta => {
344                                    self.map_number_monotonic_metrics(base_dims, sum.data_points, &context)
345                                }
346                                CumulativeMonotonicMode::RawValue => {
347                                    self.map_number_metrics(base_dims, sum.data_points, DataType::Gauge, &context)
348                                }
349                            }
350                        } else {
351                            // Cumulative non-monotonic sums are handled as gauges.
352                            self.map_number_metrics(base_dims, sum.data_points, DataType::Gauge, &context)
353                        }
354                    }
355                    Ok(AggregationTemporality::Delta) => {
356                        self.map_number_metrics(base_dims, sum.data_points, DataType::Count, &context)
357                    }
358                    _ => {
359                        warn!(
360                            metric_name = base_dims.name,
361                            temporality = sum.aggregation_temporality,
362                            "Unsupported or unknown aggregation temporality for Sum metric."
363                        );
364                        self.translator_metrics
365                            .dropped_unsupported_temporality()
366                            .increment(sum.data_points.len() as u64);
367                        Vec::new()
368                    }
369                },
370                OtlpMetricData::Histogram(histogram) => {
371                    match AggregationTemporality::try_from(histogram.aggregation_temporality) {
372                        Ok(AggregationTemporality::Cumulative) => {
373                            self.map_histogram_metrics(base_dims, histogram.data_points, false, &context)
374                        }
375                        Ok(AggregationTemporality::Delta) => {
376                            self.map_histogram_metrics(base_dims, histogram.data_points, true, &context)
377                        }
378                        _ => {
379                            warn!(
380                                metric_name = base_dims.name,
381                                temporality = histogram.aggregation_temporality,
382                                "Unsupported or unknown aggregation temporality for Histogram metric."
383                            );
384                            self.translator_metrics
385                                .dropped_unsupported_temporality()
386                                .increment(histogram.data_points.len() as u64);
387                            Vec::new()
388                        }
389                    }
390                }
391                OtlpMetricData::Summary(summary) => self.map_summary_metrics(base_dims, summary.data_points, &context),
392                OtlpMetricData::ExponentialHistogram(exponential_histogram) => {
393                    match AggregationTemporality::try_from(exponential_histogram.aggregation_temporality) {
394                        Ok(AggregationTemporality::Delta) => self.map_exponential_histogram_metrics(
395                            base_dims,
396                            exponential_histogram.data_points,
397                            true,
398                            &context,
399                        ),
400                        _ => {
401                            debug!(
402                                metric_name = base_dims.name,
403                                temporality = exponential_histogram.aggregation_temporality,
404                                "Unknown or unsupported aggregation temporality"
405                            );
406                            self.translator_metrics
407                                .dropped_unsupported_temporality()
408                                .increment(exponential_histogram.data_points.len() as u64);
409                            Vec::new()
410                        }
411                    }
412                }
413            }
414        } else {
415            Vec::new()
416        }
417    }

419    /// Centralized helper to create a metric event and push it to the events vector.
420    /// TODO: how do we handle timestamp, hostname, origin?
421    fn record_metric_event(
422        &mut self, dims: &Dimensions, value: f64, timestamp_ns: u64, data_type: DataType, events: &mut Vec<Event>,
423        context: &TranslationContext,
424    ) {
425        context.metrics.metrics_received().increment(1);
426
427        match self.context_resolver.resolve_with_optional_host_and_origin_tags(
428            &dims.name,
429            dims.host.as_deref(),
430            &dims.tags,
431            self.resolved_origin_tags.clone(),
432        ) {
433            Some(resolved_context) => {
434                let timestamp_s = timestamp_ns / 1_000_000_000;
435                let values = match data_type {
436                    DataType::Gauge => MetricValues::gauge((timestamp_s, value)),
437                    DataType::Count => MetricValues::counter((timestamp_s, value)),
438                    DataType::Rate => MetricValues::rate((timestamp_s, value), Duration::ZERO),
439                };
440
441                let metric = Metric::from_parts(resolved_context, values, MetricMetadata::default());
442                events.push(Event::Metric(metric));
443            }
444            None => {
445                warn!("Failed to resolve context for metric: {}", dims.name);
446            }
447        }
448    }

450    fn record_sketch_event(
451        &mut self, dims: &Dimensions, sketch: DDSketch, timestamp_ns: u64, events: &mut Vec<Event>,
452        context: &TranslationContext, interval: i64,
453    ) {
454        context.metrics.metrics_received().increment(1);
455        match self.context_resolver.resolve_with_optional_host_and_origin_tags(
456            &dims.name,
457            dims.host.as_deref(),
458            &dims.tags,
459            self.resolved_origin_tags.clone(),
460        ) {
461            Some(resolved_context) => {
462                if interval != 0 {
463                    trace!(
464                        metric = %dims.name,
465                        interval,
466                        "Inferred delta interval for OTLP metric."
467                    );
468                }
469                let timestamp_s = timestamp_ns / 1_000_000_000;
470                let values = MetricValues::distribution((timestamp_s, sketch));
471                let metric = Metric::from_parts(resolved_context, values, MetricMetadata::default());
472                events.push(Event::Metric(metric));
473            }
474            None => {
475                warn!("Failed to resolve context for metric: {}", dims.name);
476            }
477        }
478    }

480    /// Determines if the initial value of a cumulative monotonic metric should be consumed.
481    fn should_consume_initial_value(&self, start_ts: u64, ts: u64) -> bool {
482        match self.config.initial_cumulative_monotonic_value {
483            InitialCumulativeMonotonicValue::Auto => {
484                // We report the first value if the timeseries started after the translator process started.
485                self.process_start_time_ns < start_ts && start_ts != ts
486            }
487            InitialCumulativeMonotonicValue::Keep => true,
488            InitialCumulativeMonotonicValue::Drop => false,
489        }
490    }
