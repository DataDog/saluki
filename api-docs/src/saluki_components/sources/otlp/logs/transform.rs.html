47/// Transforms an OTLP log record, with its resource and scope, into a log.
48pub(super) fn transform_log_record(
49    lr: LogRecord, resource: &Resource, scope: Option<&otlp_common::InstrumentationScope>,
50    record_host: Option<MetaString>, record_service: Option<MetaString>, mut tags: TagSet,
51) -> Log {
52    let mut properties = Properties::default();
53
54    // Resource attributes overwrite record attributes, and scope attributes overwrite both.
55    let reserved = apply_record_attributes(&lr.attributes, &mut properties);
56    if let Some(ddtags) = reserved.ddtags {
57        add_ddtags(&mut tags, ddtags);
58    }
59    apply_resource_attributes(&resource.attributes, &mut properties);
60    if let Some(scope) = scope {
61        apply_scope(scope, &mut properties);
62    }
63    apply_trace_context(&lr, &mut properties);
64
65    let status = derive_status(reserved.status, &lr.severity_text, lr.severity_number);
66    // The Agent sends the derived status in place of any resource or scope attribute named `status`. The encoder
67    // writes additional properties over the status, so the attribute must not be among them.
68    properties.remove("status");
69
70    apply_severity(&lr, &mut properties);
71    apply_timestamp(&lr, &mut properties);
72    // The event name overwrites an `otel.event_name` attribute from any source.
73    if !lr.event_name.is_empty() {
74        properties.insert_static(OTEL_EVENT_NAME, lr.event_name);
75    }
76
77    Log::new(message(reserved.message, lr.body.as_ref()))
78        .with_status(status)
79        .with_source(DEFAULT_SOURCE.clone())
80        .with_hostname(record_host)
81        .with_service(record_service)
82        .with_tags(tags)
83        .with_additional_properties(properties.into_map())
84}

86/// The additional properties of a log, which the content renderer writes over the log's fields.
87#[derive(Default)]
88struct Properties(HashMap<MetaString, JsonValue>);
89
90impl Properties {
91    /// Sets a property, overwriting an earlier one with the same key.
92    fn insert(&mut self, key: impl Into<MetaString>, value: JsonValue) {
93        self.0.insert(key.into(), value);
94    }
95
96    /// Sets a string property under a fixed key, overwriting an earlier one.
97    fn insert_static(&mut self, key: &'static str, value: String) {
98        self.0.insert(MetaString::from_static(key), JsonValue::String(value));
99    }
100
101    /// Sets a string property under a fixed key, unless the key is already set.
102    fn insert_static_if_missing(&mut self, key: &'static str, value: &str) {
103        self.0
104            .entry(MetaString::from_static(key))
105            .or_insert_with(|| JsonValue::String(value.to_string()));
106    }
107
108    fn contains(&self, key: &str) -> bool {
109        self.0.contains_key(key)
110    }
111
112    fn remove(&mut self, key: &str) {
113        self.0.remove(key);
114    }
115
116    fn into_map(self) -> HashMap<MetaString, JsonValue> {
117        self.0
118    }
119}

121/// The reserved record attributes that become fields of the log instead of properties.
122///
123/// When a reserved key repeats, the last attribute wins.
124#[derive(Default)]
125struct ReservedAttributes<'a> {
126    message: Option<&'a str>,
127    status: Option<&'a str>,
128    ddtags: Option<&'a str>,
129}
130
131/// Sets the properties of the record attributes, and returns the reserved ones.
132///
133/// Reserved keys match in any case, and every other key keeps its original spelling. The first trace and span IDs
134/// that decode win.
135fn apply_record_attributes<'a>(
136    attributes: &'a [otlp_common::KeyValue], properties: &mut Properties,
137) -> ReservedAttributes<'a> {
138    let mut reserved = ReservedAttributes::default();
139    for kv in attributes {
140        match kv.key.as_str() {
141            k if is_reserved(k, MESSAGE_KEYS) => reserved.message = Some(string_value_or_empty(kv)),
142            k if is_reserved(k, STATUS_KEYS) => reserved.status = Some(string_value_or_empty(kv)),
143            k if is_reserved(k, TRACE_ID_ATTR_KEYS) => {
144                if let Some(OtlpStringValue(trace_hex)) = kv.value.as_ref().and_then(|av| av.value.as_ref()) {
145                    if !properties.contains(DD_TRACE_ID) {
146                        if let Some(bytes) = decode_hex_exact_to_bytes(trace_hex, 16) {
147                            insert_trace_id(properties, trace_hex.clone(), &bytes);
148                        }
149                    }
150                }
151            }
152            k if is_reserved(k, SPAN_ID_ATTR_KEYS) => {
153                if let Some(OtlpStringValue(span_hex)) = kv.value.as_ref().and_then(|av| av.value.as_ref()) {
154                    if !properties.contains(DD_SPAN_ID) {
155                        if let Some(bytes) = decode_hex_exact_to_bytes(span_hex, 8) {
156                            insert_span_id(properties, span_hex.clone(), &bytes);
157                        }
158                    }
159                }
160            }
161            k if k.eq_ignore_ascii_case(DDTAGS_ATTR) => reserved.ddtags = Some(string_value_or_empty(kv)),
162            k if k.eq_ignore_ascii_case(OTEL_LIBRARY_NAME) => {
163                insert_scope_alias(properties, attributes, OTEL_SCOPE_NAME, kv)
164            }
165            k if k.eq_ignore_ascii_case(OTEL_LIBRARY_VERSION) => {
166                insert_scope_alias(properties, attributes, OTEL_SCOPE_VERSION, kv)
167            }
168            _ => {
169                if let Some(av) = kv.value.as_ref() {
170                    for (key, val) in flatten_attribute(&kv.key, av, 1) {
171                        if !val.is_null() {
172                            properties.insert(key, val);
173                        }
174                    }
175                }
176            }
177        }
178    }
179    reserved
180}

182/// Adds the tags of a `ddtags` attribute. The resource tags are already in `tags`.
183fn add_ddtags(tags: &mut TagSet, ddtags: &str) {
184    for tag in ddtags.split(',').map(str::trim) {
185        if !tag.is_empty() {
186            tags.insert_tag(tag);
187        }
188    }
189}

191/// Sets the properties of the resource attributes.
192///
193/// Unlike record attributes, their aliases match case-sensitively. `hostname` and `service` are renamed to
194/// `otel.hostname` and `otel.service`, so they don't overwrite the log's own fields.
195fn apply_resource_attributes(attributes: &[otlp_common::KeyValue], properties: &mut Properties) {
196    for kv in attributes {
197        match kv.key.as_str() {
198            OTEL_LIBRARY_NAME => insert_scope_alias(properties, attributes, OTEL_SCOPE_NAME, kv),
199            OTEL_LIBRARY_VERSION => insert_scope_alias(properties, attributes, OTEL_SCOPE_VERSION, kv),
200            key => {
201                if let Some(av) = kv.value.as_ref() {
202                    let val = match av.value.as_ref() {
203                        Some(OtlpStringValue(s)) => JsonValue::String(s.clone()),
204                        _ => any_value_to_json(av),
205                    };
206                    if !val.is_null() {
207                        match key {
208                            "hostname" => properties.insert(MetaString::from_static(OTEL_HOSTNAME), val),
209                            "service" => properties.insert(MetaString::from_static(OTEL_SERVICE), val),
210                            _ => properties.insert(key, val),
211                        }
212                    }
213                }
214            }
215        }
216    }
217}

219/// Sets the properties of the scope's attributes, then its name and version.
220///
221/// The attributes keep their own keys: their aliases aren't mapped, and `hostname` and `service` aren't renamed. The
222/// name and version fill in only the keys that are still missing.
223fn apply_scope(scope: &otlp_common::InstrumentationScope, properties: &mut Properties) {
224    for kv in &scope.attributes {
225        if let Some(av) = kv.value.as_ref() {
226            let val = match av.value.as_ref() {
227                Some(OtlpStringValue(s)) => JsonValue::String(s.clone()),
228                _ => any_value_to_json(av),
229            };
230            if !val.is_null() {
231                properties.insert(kv.key.as_str(), val);
232            }
233        }
234    }
235    for (key, value) in [(OTEL_SCOPE_NAME, &scope.name), (OTEL_SCOPE_VERSION, &scope.version)] {
236        if !value.is_empty() {
237            properties.insert_static_if_missing(key, value);
238        }
239    }
240}

242/// Sets the trace and span IDs of the record, overwriting the IDs from its attributes.
243fn apply_trace_context(lr: &LogRecord, properties: &mut Properties) {
244    if lr.trace_id.iter().any(|&b| b != 0) {
245        insert_trace_id(properties, bytes_to_hex_lowercase(&lr.trace_id), &lr.trace_id);
246    }
247    if lr.span_id.iter().any(|&b| b != 0) {
248        insert_span_id(properties, bytes_to_hex_lowercase(&lr.span_id), &lr.span_id);
249    }
250}

252/// Sets `otel.trace_id` to the hex form of a trace ID, and `dd.trace_id` to the decimal form of its last 8 bytes.
253fn insert_trace_id(properties: &mut Properties, hex: String, bytes: &[u8]) {
254    properties.insert_static(OTEL_TRACE_ID, hex);
255    properties.insert_static(DD_TRACE_ID, u64_from_last_8(bytes).to_string());
256}

258/// Sets `otel.span_id` to the hex form of a span ID, and `dd.span_id` to the decimal form of its first 8 bytes.
259fn insert_span_id(properties: &mut Properties, hex: String, bytes: &[u8]) {
260    properties.insert_static(OTEL_SPAN_ID, hex);
261    properties.insert_static(DD_SPAN_ID, u64_from_first_8(bytes).to_string());
262}

264/// Sets the severity text and number of the record, when they're set.
265fn apply_severity(lr: &LogRecord, properties: &mut Properties) {
266    if !lr.severity_text.is_empty() {
267        properties.insert_static(OTEL_SEVERITY_TEXT, lr.severity_text.clone());
268    }
269    if lr.severity_number != 0 {
270        properties.insert_static(OTEL_SEVERITY_NUMBER, lr.severity_number.to_string());
271    }
272}

274/// Sets the timestamp of the record, in nanoseconds and in the Go layout `2006-01-02T15:04:05.000Z07:00`, in UTC.
275///
276/// The observed timestamp is the fallback when the timestamp isn't set. Nanoseconds are read as signed, so a
277/// timestamp past `i64::MAX` wraps to before 1970.
278fn apply_timestamp(lr: &LogRecord, properties: &mut Properties) {
279    let timestamp = if lr.time_unix_nano != 0 {
280        lr.time_unix_nano
281    } else {
282        lr.observed_time_unix_nano
283    };
284    if timestamp == 0 {
285        return;
286    }
287    let nanos = timestamp as i64;
288    properties.insert_static(OTEL_TIMESTAMP, nanos.to_string());
289    let instant = Timestamp::from_unix_nanos(i128::from(nanos))
290        .expect("any signed 64-bit count of nanoseconds should be a valid timestamp");
291    properties.insert_static(
292        DD_TIMESTAMP,
293        instant.format(TimestampFormat::Rfc3339UtcMillis).to_string(),
294    );
295}

297/// Returns the message: the last message attribute, or the body when that attribute is missing or empty.
298fn message(attribute: Option<&str>, body: Option<&otlp_common::AnyValue>) -> MetaString {
299    match attribute {
300        Some(message) if !message.is_empty() => MetaString::from(message),
301        _ => body
302            .map(any_value_to_message_string)
303            .map(MetaString::from)
304            .unwrap_or_default(),
305    }
306}

308/// Derives a log's status from its status attributes, severity text, and severity number.
309///
310/// The status is the last status attribute (`status`, `severity`, `level`, or `syslog.severity`), then the severity
311/// text, then the name of the severity number's range, taking the first that isn't empty. Attribute and severity
312/// text are used as they are, without normalization. A severity number of 0 has no name, so a log with nothing else
313/// gets an empty status.
314pub(crate) fn derive_status(status: Option<&str>, severity_text: &str, severity_number: i32) -> MetaString {
315    match status {
316        Some(status) if !status.is_empty() => MetaString::from(status),
317        _ if !severity_text.is_empty() => MetaString::from(severity_text),
318        _ if severity_number != 0 => MetaString::from_static(status_from_severity_number(severity_number)),
319        _ => MetaString::empty(),
320    }
321}

323/// Returns the name of the range that a severity number falls in.
324///
325/// The names follow the ranges of the OpenTelemetry log data model, and numbers above the last range are errors:
326/// <https://github.com/open-telemetry/opentelemetry-specification/blob/main/specification/logs/data-model.md#field-severitynumber>
327fn status_from_severity_number(severity_number: i32) -> &'static str {
328    match severity_number {
329        ..=4 => "trace",
330        5..=8 => "debug",
331        9..=12 => "info",
332        13..=16 => "warn",
333        17..=20 => "error",
334        21..=24 => "fatal",
335        _ => "error",
336    }
337}

339/// Returns the last 8 bytes of a trace ID as a big-endian integer. A shorter ID is padded with leading zeros.
340fn u64_from_last_8(bytes: &[u8]) -> u64 {
341    let n = bytes.len();
342    let slice = if n >= 8 { &bytes[n - 8..n] } else { bytes };
343    let mut buf = [0u8; 8];
344    let start = 8 - slice.len();
345    buf[start..].copy_from_slice(slice);
346    u64::from_be_bytes(buf)
347}
348
349/// Returns the first 8 bytes of a span ID as a big-endian integer. A shorter ID is padded with leading zeros.
350fn u64_from_first_8(bytes: &[u8]) -> u64 {
351    let slice = if bytes.len() >= 8 { &bytes[..8] } else { bytes };
352    let mut buf = [0u8; 8];
353    let start = 8 - slice.len();
354    buf[start..].copy_from_slice(slice);
355    u64::from_be_bytes(buf)
356}

358/// Decodes a hex string of exactly `expected_len_bytes` bytes.
359fn decode_hex_exact_to_bytes(s: &str, expected_len_bytes: usize) -> Option<Vec<u8>> {
360    if s.len() != expected_len_bytes * 2 {
361        return None;
362    }
363    let mut out = vec![0u8; expected_len_bytes];
364    faster_hex::hex_decode(s.as_bytes(), &mut out).ok()?;
365    Some(out)
366}

368/// Returns whether `key` is one of the `reserved` keys, ignoring ASCII case.
369///
370/// Only ASCII letters are folded. Go's `strings.ToLower` also turns the dotted `İ` (U+0130) into `i`, so a key such
371/// as `TRACEİD` is reserved under Go's matching. We choose to ignore that edge case.
372fn is_reserved(key: &str, reserved: &[&str]) -> bool {
373    reserved.iter().any(|reserved| reserved.eq_ignore_ascii_case(key))
374}

376/// Returns an attribute's string value, or an empty string for any other value.
377///
378/// Converting other values to strings, as `pcommon.Value.AsString` does, is still missing. Until then, a non-string
379/// message, status, or `ddtags` attribute counts as empty, and so does a non-string duplicate that comes after a string
380/// one.
381fn string_value_or_empty(kv: &otlp_common::KeyValue) -> &str {
382    match kv.value.as_ref().and_then(|av| av.value.as_ref()) {
383        Some(OtlpStringValue(s)) => s,
384        _ => "",
385    }
386}

388/// Maps a deprecated `otel.library.*` attribute to its canonical `otel.scope.*` key, unless the attribute's own
389/// collection also has the canonical key, spelled exactly.
390fn insert_scope_alias(
391    properties: &mut Properties, collection: &[otlp_common::KeyValue], canonical: &'static str,
392    kv: &otlp_common::KeyValue,
393) {
394    if collection.iter().any(|other| other.key == canonical) {
395        return;
396    }
397    let value = kv.value.as_ref().map(any_value_to_message_string).unwrap_or_default();
398    properties.insert_static(canonical, value);
399}

401/// Renders a value as the message: a string as it is, and any other value as JSON.
402fn any_value_to_message_string(av: &otlp_common::AnyValue) -> String {
403    match av.value.as_ref() {
404        Some(OtlpStringValue(s)) => s.clone(),
405        _ => serde_json::to_string(&any_value_to_json(av)).unwrap_or_else(|_| String::new()),
406    }
407}

409/// Converts a value into JSON.
410fn any_value_to_json(av: &otlp_common::AnyValue) -> JsonValue {
411    match av.value.as_ref() {
412        Some(BoolValue(b)) => JsonValue::Bool(*b),
413        Some(IntValue(i)) => JsonValue::from(*i),
414        Some(DoubleValue(d)) => JsonValue::from(*d),
415        Some(OtlpStringValue(s)) => JsonValue::String(s.clone()),
416        Some(BytesValue(bytes)) => match String::from_utf8(bytes.clone()) {
417            Ok(s) => JsonValue::String(s),
418            Err(_) => JsonValue::String(format!("<{} bytes>", bytes.len())),
419        },
420        Some(ArrayValue(arr)) => JsonValue::Array(arr.values.iter().map(any_value_to_json).collect::<Vec<_>>()),
421        Some(KvlistValue(kvl)) => {
422            let mut obj = serde_json::Map::new();
423            for kv in &kvl.values {
424                if let Some(val) = kv.value.as_ref() {
425                    obj.insert(kv.key.clone(), any_value_to_json(val));
426                }
427            }
428            JsonValue::Object(obj)
429        }
430        None => JsonValue::Null,
431    }
432}

434/// Flattens a map into dotted keys, up to a maximum depth.
435///
436/// Leaf strings, integers, Booleans, and doubles keep their types, and other leaf values become strings.
437fn flatten_attribute(base_key: &str, av: &otlp_common::AnyValue, depth: usize) -> Vec<(String, JsonValue)> {
438    const MAX_DEPTH: usize = 10;
439
440    fn to_leaf_json(av: &otlp_common::AnyValue) -> JsonValue {
441        match av.value.as_ref() {
442            Some(OtlpStringValue(s)) => JsonValue::String(s.clone()),
443            Some(IntValue(i)) => JsonValue::from(*i),
444            Some(BoolValue(b)) => JsonValue::Bool(*b),
445            Some(DoubleValue(d)) => JsonValue::from(*d),
446            Some(BytesValue(bytes)) => match String::from_utf8(bytes.clone()) {
447                Ok(s) => JsonValue::String(s),
448                Err(_) => JsonValue::String(format!("<{} bytes>", bytes.len())),
449            },
450            _ => JsonValue::String(serde_json::to_string(&any_value_to_json(av)).unwrap_or_else(|_| String::new())),
451        }
452    }
453
454    if depth < MAX_DEPTH {
455        if let Some(KvlistValue(kvl)) = av.value.as_ref() {
456            let mut out = Vec::new();
457            for kv in &kvl.values {
458                if let Some(v) = kv.value.as_ref() {
459                    let new_key = format!("{}.{}", base_key, kv.key);
460                    let nested = flatten_attribute(&new_key, v, depth + 1);
461                    if nested.is_empty() {
462                        out.push((new_key, to_leaf_json(v)));
463                    } else {
464                        out.extend(nested);
465                    }
466                }
467            }
468            return out;
469        }
470    }
471
472    vec![(base_key.to_string(), to_leaf_json(av))]
473}
