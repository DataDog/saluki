52/// Transforms an OTLP log record, with its resource and scope, into a log.
53///
54/// The message, status, `ddtags`, and trace and span ID attributes, the `otel.library.*` aliases, the resource and scope
55/// attributes, and the body become strings. Other record attributes keep their types, as [`Flattener`] describes.
56///
57/// The source is the final `datadog.log.source` property when it's a non-empty string, and `otlp_log_ingestion`
58/// otherwise, as the Agent's logs agent exporter chooses it. The property stays in the content.
59///
60/// # Errors
61///
62/// Returns an error when a record attribute holds a NaN or an infinite double that no later property overwrites. The
63/// caller drops the log, as the Agent drops a log that it fails to marshal.
64pub(super) fn transform_log_record(
65    lr: LogRecord, resource: &Resource, scope: Option<&otlp_common::InstrumentationScope>,
66    record_host: Option<MetaString>, record_service: Option<MetaString>, mut tags: TagSet,
67) -> Result<Log, GenericError> {
68    let mut properties = Properties::default();
69    let mut flattener = Flattener::default();
70
71    // Resource attributes overwrite record attributes, and scope attributes overwrite both.
72    let reserved = apply_record_attributes(&lr.attributes, &mut flattener, &mut properties);
73    if let Some(ddtags) = reserved.ddtags.as_deref() {
74        add_ddtags(&mut tags, ddtags);
75    }
76    apply_resource_attributes(&resource.attributes, &mut properties);
77    if let Some(scope) = scope {
78        apply_scope(scope, &mut properties);
79    }
80    apply_trace_context(&lr, &mut properties);
81
82    let status = derive_status(reserved.status.as_deref(), &lr.severity_text, lr.severity_number);
83    // The Agent sends the derived status in place of any resource or scope attribute named `status`. The encoder
84    // writes additional properties over the status, so the attribute must not be among them.
85    properties.remove("status");
86
87    apply_severity(&lr, &mut properties);
88    apply_timestamp(&lr, &mut properties);
89    // The event name overwrites an `otel.event_name` attribute from any source.
90    if !lr.event_name.is_empty() {
91        properties.insert_static(OTEL_EVENT_NAME, lr.event_name);
92    }
93
94    let message = message(reserved.message, lr.body.as_ref());
95
96    // A non-finite double is still a placeholder only if no later property overwrote it.
97    if flattener.wrote_non_finite {
98        if let Some(key) = properties.non_finite_placeholder() {
99            return Err(generic_error!(
100                "Log attribute '{}' holds a NaN or infinite double, which JSON can't represent.",
101                key
102            ));
103        }
104    }
105
106    let source = match properties.get(LOG_SOURCE_ATTR) {
107        Some(JsonValue::String(source)) if !source.is_empty() => MetaString::from(source.as_str()),
108        _ => DEFAULT_SOURCE.clone(),
109    };
110
111    Ok(Log::new(message)
112        .with_status(status)
113        .with_source(source)
114        .with_hostname(record_host)
115        .with_service(record_service)
116        .with_tags(tags)
117        .with_additional_properties(properties.into_map()))
118}

120/// The additional properties of a log, which the content renderer writes over the log's fields.
121#[derive(Default)]
122struct Properties(HashMap<MetaString, JsonValue>);
123
124impl Properties {
125    /// Sets a property, overwriting an earlier one with the same key.
126    fn insert(&mut self, key: impl Into<MetaString>, value: JsonValue) {
127        self.0.insert(key.into(), value);
128    }
129
130    /// Sets a string property under a fixed key, overwriting an earlier one.
131    fn insert_static(&mut self, key: &'static str, value: String) {
132        self.0.insert(MetaString::from_static(key), JsonValue::String(value));
133    }
134
135    /// Sets a string property under a fixed key, unless the key is already set.
136    fn insert_static_if_missing(&mut self, key: &'static str, value: &str) {
137        self.0
138            .entry(MetaString::from_static(key))
139            .or_insert_with(|| JsonValue::String(value.to_string()));
140    }
141
142    fn get(&self, key: &str) -> Option<&JsonValue> {
143        self.0.get(key)
144    }
145
146    fn contains(&self, key: &str) -> bool {
147        self.0.contains_key(key)
148    }
149
150    fn remove(&mut self, key: &str) {
151        self.0.remove(key);
152    }
153
154    fn non_finite_placeholder(&self) -> Option<&MetaString> {
155        self.0
156            .iter()
157            .find(|(_, value)| holds_non_finite_placeholder(value))
158            .map(|(key, _)| key)
159    }
160
161    fn into_map(self) -> HashMap<MetaString, JsonValue> {
162        self.0
163    }
164}

166/// The reserved record attributes that become fields of the log instead of properties.
167///
168/// When a reserved key repeats, the last attribute wins.
169#[derive(Default)]
170struct ReservedAttributes<'a> {
171    message: Option<Cow<'a, str>>,
172    status: Option<Cow<'a, str>>,
173    ddtags: Option<Cow<'a, str>>,
174}
175
176/// Sets the properties of the record attributes, and returns the reserved ones.
177///
178/// Reserved keys match in any case, and every other key keeps its original spelling. The first trace and span IDs
179/// that decode win.
180fn apply_record_attributes<'a>(
181    attributes: &'a [otlp_common::KeyValue], flattener: &mut Flattener, properties: &mut Properties,
182) -> ReservedAttributes<'a> {
183    let mut reserved = ReservedAttributes::default();
184    for kv in attributes {
185        match kv.key.as_str() {
186            k if is_reserved(k, MESSAGE_KEYS) => reserved.message = Some(as_string(kv)),
187            k if is_reserved(k, STATUS_KEYS) => reserved.status = Some(as_string(kv)),
188            k if is_reserved(k, TRACE_ID_ATTR_KEYS) => {
189                if !properties.contains(DD_TRACE_ID) {
190                    let trace_hex = as_string(kv);
191                    if let Some(bytes) = decode_hex_exact_to_bytes(&trace_hex, 16) {
192                        insert_trace_id(properties, trace_hex.into_owned(), &bytes);
193                    }
194                }
195            }
196            k if is_reserved(k, SPAN_ID_ATTR_KEYS) => {
197                if !properties.contains(DD_SPAN_ID) {
198                    let span_hex = as_string(kv);
199                    if let Some(bytes) = decode_hex_exact_to_bytes(&span_hex, 8) {
200                        insert_span_id(properties, span_hex.into_owned(), &bytes);
201                    }
202                }
203            }
204            k if k.eq_ignore_ascii_case(DDTAGS_ATTR) => reserved.ddtags = Some(as_string(kv)),
205            k if k.eq_ignore_ascii_case(OTEL_LIBRARY_NAME) => {
206                insert_scope_alias(properties, attributes, OTEL_SCOPE_NAME, kv)
207            }
208            k if k.eq_ignore_ascii_case(OTEL_LIBRARY_VERSION) => {
209                insert_scope_alias(properties, attributes, OTEL_SCOPE_VERSION, kv)
210            }
211            _ => flattener.flatten(Cow::Borrowed(kv.key.as_str()), attribute_value(kv), 1, properties),
212        }
213    }
214    reserved
215}

217/// Adds the tags of a `ddtags` attribute. The resource tags are already in `tags`.
218fn add_ddtags(tags: &mut TagSet, ddtags: &str) {
219    for tag in ddtags.split(',').map(str::trim) {
220        if !tag.is_empty() {
221            tags.insert_tag(tag);
222        }
223    }
224}

226/// Sets the properties of the resource attributes.
227///
228/// Unlike record attributes, their aliases match case-sensitively. `hostname` and `service` are renamed to
229/// `otel.hostname` and `otel.service`, so they don't overwrite the log's own fields.
230fn apply_resource_attributes(attributes: &[otlp_common::KeyValue], properties: &mut Properties) {
231    for kv in attributes {
232        match kv.key.as_str() {
233            OTEL_LIBRARY_NAME => insert_scope_alias(properties, attributes, OTEL_SCOPE_NAME, kv),
234            OTEL_LIBRARY_VERSION => insert_scope_alias(properties, attributes, OTEL_SCOPE_VERSION, kv),
235            key => {
236                let value = JsonValue::String(as_string(kv).into_owned());
237                match key {
238                    "hostname" => properties.insert(MetaString::from_static(OTEL_HOSTNAME), value),
239                    "service" => properties.insert(MetaString::from_static(OTEL_SERVICE), value),
240                    _ => properties.insert(key, value),
241                }
242            }
243        }
244    }
245}

247/// Sets the properties of the scope's attributes, then its name and version.
248///
249/// The attributes keep their own keys: their aliases aren't mapped, and `hostname` and `service` aren't renamed. The
250/// name and version fill in only the keys that are still missing.
251fn apply_scope(scope: &otlp_common::InstrumentationScope, properties: &mut Properties) {
252    for kv in &scope.attributes {
253        properties.insert(kv.key.as_str(), JsonValue::String(as_string(kv).into_owned()));
254    }
255    for (key, value) in [(OTEL_SCOPE_NAME, &scope.name), (OTEL_SCOPE_VERSION, &scope.version)] {
256        if !value.is_empty() {
257            properties.insert_static_if_missing(key, value);
258        }
259    }
260}

262/// Sets the trace and span IDs of the record, overwriting the IDs from its attributes.
263fn apply_trace_context(lr: &LogRecord, properties: &mut Properties) {
264    if lr.trace_id.iter().any(|&b| b != 0) {
265        insert_trace_id(properties, bytes_to_hex_lowercase(&lr.trace_id), &lr.trace_id);
266    }
267    if lr.span_id.iter().any(|&b| b != 0) {
268        insert_span_id(properties, bytes_to_hex_lowercase(&lr.span_id), &lr.span_id);
269    }
270}

272/// Sets `otel.trace_id` to the hex form of a trace ID, and `dd.trace_id` to the decimal form of its last 8 bytes.
273fn insert_trace_id(properties: &mut Properties, hex: String, bytes: &[u8]) {
274    properties.insert_static(OTEL_TRACE_ID, hex);
275    properties.insert_static(DD_TRACE_ID, u64_from_last_8(bytes).to_string());
276}

278/// Sets `otel.span_id` to the hex form of a span ID, and `dd.span_id` to the decimal form of its first 8 bytes.
279fn insert_span_id(properties: &mut Properties, hex: String, bytes: &[u8]) {
280    properties.insert_static(OTEL_SPAN_ID, hex);
281    properties.insert_static(DD_SPAN_ID, u64_from_first_8(bytes).to_string());
282}

284/// Sets the severity text and number of the record, when they're set.
285fn apply_severity(lr: &LogRecord, properties: &mut Properties) {
286    if !lr.severity_text.is_empty() {
287        properties.insert_static(OTEL_SEVERITY_TEXT, lr.severity_text.clone());
288    }
289    if lr.severity_number != 0 {
290        properties.insert_static(OTEL_SEVERITY_NUMBER, lr.severity_number.to_string());
291    }
292}

294/// Sets the timestamp of the record, in nanoseconds and in the Go layout `2006-01-02T15:04:05.000Z07:00`, in UTC.
295///
296/// The observed timestamp is the fallback when the timestamp isn't set. Nanoseconds are read as signed, so a
297/// timestamp past `i64::MAX` wraps to before 1970.
298fn apply_timestamp(lr: &LogRecord, properties: &mut Properties) {
299    let timestamp = if lr.time_unix_nano != 0 {
300        lr.time_unix_nano
301    } else {
302        lr.observed_time_unix_nano
303    };
304    if timestamp == 0 {
305        return;
306    }
307    let nanos = timestamp as i64;
308    properties.insert_static(OTEL_TIMESTAMP, nanos.to_string());
309    let instant = Timestamp::from_unix_nanos(i128::from(nanos))
310        .expect("any signed 64-bit count of nanoseconds should be a valid timestamp");
311    properties.insert_static(
312        DD_TIMESTAMP,
313        instant.format(TimestampFormat::Rfc3339UtcMillis).to_string(),
314    );
315}

317/// Returns the message: the last message attribute, or the body when that attribute is missing or empty.
318fn message(attribute: Option<Cow<'_, str>>, body: Option<&otlp_common::AnyValue>) -> MetaString {
319    match attribute {
320        Some(message) if !message.is_empty() => MetaString::from(message),
321        _ => MetaString::from(otlp_value_as_string(body.and_then(|body| body.value.as_ref()))),
322    }

325/// Derives a log's status from its status attributes, severity text, and severity number.
326///
327/// The status is the last status attribute (`status`, `severity`, `level`, or `syslog.severity`), then the severity
328/// text, then the name of the severity number's range, taking the first that isn't empty. Attribute and severity
329/// text are used as they are, without normalization. A severity number of 0 has no name, so a log with nothing else
330/// gets an empty status.
331pub(crate) fn derive_status(status: Option<&str>, severity_text: &str, severity_number: i32) -> MetaString {
332    match status {
333        Some(status) if !status.is_empty() => MetaString::from(status),
334        _ if !severity_text.is_empty() => MetaString::from(severity_text),
335        _ if severity_number != 0 => MetaString::from_static(status_from_severity_number(severity_number)),
336        _ => MetaString::empty(),
337    }
338}
339
340/// Returns the name of the range that a severity number falls in.
341///
342/// The names follow the ranges of the OpenTelemetry log data model, and numbers above the last range are errors:
343/// <https://github.com/open-telemetry/opentelemetry-specification/blob/main/specification/logs/data-model.md#field-severitynumber>
344fn status_from_severity_number(severity_number: i32) -> &'static str {
345    match severity_number {
346        ..=4 => "trace",
347        5..=8 => "debug",
348        9..=12 => "info",
349        13..=16 => "warn",
350        17..=20 => "error",
351        21..=24 => "fatal",
352        _ => "error",
353    }
354}

356/// Returns the last 8 bytes of a trace ID as a big-endian integer. A shorter ID is padded with leading zeros.
357fn u64_from_last_8(bytes: &[u8]) -> u64 {
358    let n = bytes.len();
359    let slice = if n >= 8 { &bytes[n - 8..n] } else { bytes };
360    let mut buf = [0u8; 8];
361    let start = 8 - slice.len();
362    buf[start..].copy_from_slice(slice);
363    u64::from_be_bytes(buf)
364}
365
366/// Returns the first 8 bytes of a span ID as a big-endian integer. A shorter ID is padded with leading zeros.
367fn u64_from_first_8(bytes: &[u8]) -> u64 {
368    let slice = if bytes.len() >= 8 { &bytes[..8] } else { bytes };
369    let mut buf = [0u8; 8];
370    let start = 8 - slice.len();
371    buf[start..].copy_from_slice(slice);
372    u64::from_be_bytes(buf)
373}

375/// Decodes a hex string of exactly `expected_len_bytes` bytes.
376fn decode_hex_exact_to_bytes(s: &str, expected_len_bytes: usize) -> Option<Vec<u8>> {
377    if s.len() != expected_len_bytes * 2 {
378        return None;
379    }
380    let mut out = vec![0u8; expected_len_bytes];
381    faster_hex::hex_decode(s.as_bytes(), &mut out).ok()?;
382    Some(out)
383}

385/// Returns whether `key` is one of the `reserved` keys, ignoring ASCII case.
386///
387/// Only ASCII letters are folded. Go's `strings.ToLower` also turns the dotted `İ` (U+0130) into `i`, so a key such
388/// as `TRACEİD` is reserved under Go's matching. We choose to ignore that edge case.
389fn is_reserved(key: &str, reserved: &[&str]) -> bool {
390    reserved.iter().any(|reserved| reserved.eq_ignore_ascii_case(key))
391}

393/// Maps a deprecated `otel.library.*` attribute to its canonical `otel.scope.*` key, unless the attribute's own
394/// collection also has the canonical key, spelled exactly.
395fn insert_scope_alias(
396    properties: &mut Properties, collection: &[otlp_common::KeyValue], canonical: &'static str,
397    kv: &otlp_common::KeyValue,
398) {
399    if collection.iter().any(|other| other.key == canonical) {
400        return;
401    }
402    properties.insert_static(canonical, as_string(kv).into_owned());
403}

413/// The depth at which [`Flattener`] renders a map as a string instead of flattening it.
414const FLATTEN_STRING_DEPTH: usize = 10;

416/// Where [`Flattener`] writes the properties of a record attribute.
417trait PropertySink {
418    fn set(&mut self, key: String, value: JsonValue);
419}
420
421impl PropertySink for Properties {
422    fn set(&mut self, key: String, value: JsonValue) {
423        self.insert(key, value);
424    }
425}

427impl PropertySink for serde_json::Map<String, JsonValue> {
428    fn set(&mut self, key: String, value: JsonValue) {
429        self.insert(key, value);
430    }

433/// Flattens record attributes as `flattenAttribute` in the Agent does.
434///
435/// - A map spreads its entries into dotted keys. Under an empty key, the entries keep their own keys.
436/// - An array stays an array at any depth, as [`Flattener::array_element`] describes.
437/// - Strings, integers, Booleans, and doubles keep their types. Bytes, empty values, and maps at depth 10 become
438///   strings.
439///
440/// A map becomes a string only at exactly depth 10. An array at depth 10 moves its elements to depth 11, so the maps
441/// below it flatten without a limit. This is a quirk of the Agent since it added array support
442/// (DataDog/datadog-agent#45708): the array branch comes before the depth check, so depth can skip past 10. We copy it.
443///
444/// JSON can't hold a NaN or an infinite double, and the Agent fails to marshal a log that holds one. A non-finite
445/// double becomes a `null` placeholder, and the flattener remembers that it wrote one, so the caller can drop the log
446/// if the placeholder is still there once every property is set. No other value becomes `null`.
447#[derive(Default)]
448struct Flattener {
449    wrote_non_finite: bool,
450}
451
452impl Flattener {
453    /// Flattens a value under `key` into `out`. A record attribute starts at depth 1.
454    fn flatten<S: PropertySink>(&mut self, key: Cow<'_, str>, value: Option<&OtlpValue>, depth: usize, out: &mut S) {
455        match value {
456            Some(KvlistValue(map)) if depth != FLATTEN_STRING_DEPTH => {
457                for kv in &map.values {
458                    let nested_key = if key.is_empty() {
459                        Cow::Borrowed(kv.key.as_str())
460                    } else {
461                        Cow::Owned(format!("{}.{}", key, kv.key))
462                    };
463                    self.flatten(nested_key, attribute_value(kv), depth + 1, out);
464                }
465            }
466            _ => {
467                let value = self.value(value, depth);
468                out.set(key.into_owned(), value);
469            }
470        }
471    }
472
473    /// Converts a value that doesn't spread into dotted keys.
474    fn value(&mut self, value: Option<&OtlpValue>, depth: usize) -> JsonValue {
475        match value {
476            Some(ArrayValue(array)) => JsonValue::Array(
477                array
478                    .values
479                    .iter()
480                    .map(|element| self.array_element(element.value.as_ref(), depth + 1))
481                    .collect(),
482            ),
483            Some(OtlpStringValue(s)) => JsonValue::String(s.clone()),
484            Some(IntValue(i)) => JsonValue::from(*i),
485            Some(BoolValue(b)) => JsonValue::Bool(*b),
486            Some(DoubleValue(d)) => match serde_json::Number::from_f64(*d) {
487                Some(number) => JsonValue::Number(number),
488                None => {
489                    self.wrote_non_finite = true;
490                    JsonValue::Null
491                }
492            },
493            other => JsonValue::String(otlp_value_as_string(other).into_owned()),
494        }
495    }
496
497    /// Converts an element of an array. The element is one level deeper than its array.
498    ///
499    /// The Agent flattens each element under an empty key. An element whose result has an empty key becomes that
500    /// value, and any other element becomes an object of its results. So a map that spreads becomes an object of its
501    /// dotted keys, unless it has an empty key itself, and every other element becomes a single value.
502    fn array_element(&mut self, element: Option<&OtlpValue>, depth: usize) -> JsonValue {
503        match element {
504            Some(map @ KvlistValue(_)) if depth != FLATTEN_STRING_DEPTH => {
505                let mut flattened = serde_json::Map::new();
506                self.flatten(Cow::Borrowed(""), Some(map), depth, &mut flattened);
507                match flattened.remove("") {
508                    Some(value) => value,
509                    None => JsonValue::Object(flattened),
510                }
511            }
512            other => self.value(other, depth),
513        }
514    }
515}
516
517fn holds_non_finite_placeholder(value: &JsonValue) -> bool {
518    match value {
519        JsonValue::Null => true,
520        JsonValue::Array(values) => values.iter().any(holds_non_finite_placeholder),
521        JsonValue::Object(map) => map.values().any(holds_non_finite_placeholder),
522        _ => false,
523    }
524}
