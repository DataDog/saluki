1//! OTLP logs rendering.
2//!
3//! Renders each translated OTLP log into the content of its log message, as the last step of OTLP logs translation.
4//! See [`ContentRenderer`] for what the content holds.
5//!
6//! Enrichment of OTLP logs belongs before rendering, since it changes the properties that go into the content.
7//! Processing rules belong after it, since they apply to the content.
8

20/// The most serialization buffer capacity that a [`ContentRenderer`] keeps between logs.
21///
22/// A renderer keeps the capacity of the largest content it has written, so one enormous log would otherwise stay held
23/// for the renderer's lifetime. The limit sits well above the content of an ordinary log: such a log keeps its reuse,
24/// and a log past it has the excess released before the renderer is used again.
25pub const CONTENT_BUFFER_RETENTION_LIMIT_BYTES: usize = 64 * 1024;
26
27/// The most field list capacity, in fields, that a [`ContentRenderer`] keeps between logs.
28///
29/// Like the serialization buffer, the field list keeps the capacity of the log with the most fields, so it's bounded
30/// too. The limit sits well above the fields of an ordinary log.
31pub const CONTENT_FIELDS_RETENTION_LIMIT: usize = 1024;
32
33/// The memory that the most field list capacity a [`ContentRenderer`] keeps between logs takes.
34pub const CONTENT_FIELDS_RETENTION_LIMIT_BYTES: usize =
35    CONTENT_FIELDS_RETENTION_LIMIT * mem::size_of::<(&str, ContentField<'_>)>();
36
37/// A field of a log's content.
38enum ContentField<'a> {
39    String(&'a str),
40    Value(&'a JsonValue),
41}
42
43/// Renders translated OTLP logs into the content of their log messages.
44///
45/// The content is the log as a `datadogV2.HTTPLogItem` without its `ddtags`, serialized by `HTTPLogItem.MarshalJSON`.
46/// It holds:
47///
48/// - `hostname` and `service`, when the log has them, and `message`.
49/// - Every additional property, overwriting a top-level field of the same name. A `datadog.log.source` property, and a
50///   `ddtags` property from a resource attribute, stay in the content.
51/// - `status`, the log's status, even when it's empty. It's set after every other property, so it overwrites a
52///   `status` property.
53///
54/// The content is serialized with `serde_json`, with its keys sorted by their bytes, as Go sorts map keys. Its escaping
55/// and number formatting differ from Go's `encoding/json` (`gap/json-encoding`): `<`, `>`, `&`, U+2028, and U+2029
56/// aren't escaped, and floats use `serde_json`'s layout, such as `1.0` and `1e21` where Go writes `1` and `1e+21`.
57///
58/// A renderer keeps its serialization buffer between logs, up to [`CONTENT_BUFFER_RETENTION_LIMIT_BYTES`], and its
59/// field list, up to [`CONTENT_FIELDS_RETENTION_LIMIT`], so rendering many logs reuses their allocations.
60#[derive(Default)]
61pub struct ContentRenderer {
62    buffer: Vec<u8>,
63    // Always empty between logs. It borrows nothing, and only holds the allocation of the field list.
64    fields: Vec<(&'static str, ContentField<'static>)>,
65}
66
67impl std::fmt::Debug for ContentRenderer {
68    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
69        f.debug_struct("ContentRenderer")
70            .field("buffer_capacity", &self.buffer.capacity())
71            .field("fields_capacity", &self.fields.capacity())
72            .finish()
73    }
74}
75
76impl ContentRenderer {
77    /// Renders a log's content.
78    ///
79    /// # Errors
80    ///
81    /// If a property can't be serialized, an error is returned, and the log should be dropped.
82    pub fn render(&mut self, log: &Log) -> Result<MetaString, GenericError> {
83        let content = self.write_content(log).and_then(|()| {
84            simdutf8::basic::from_utf8(&self.buffer)
85                .map(MetaString::from)
86                .map_err(|e| generic_error!("Rendered log content isn't valid UTF-8: {}", e))
87        });
88
89        if self.buffer.capacity() > CONTENT_BUFFER_RETENTION_LIMIT_BYTES {
90            self.buffer.clear();
91            self.buffer.shrink_to(CONTENT_BUFFER_RETENTION_LIMIT_BYTES);
92        }
93
94        content
95    }
96
97    /// Writes a log's content into the buffer.
98    fn write_content(&mut self, log: &Log) -> Result<(), GenericError> {
99        // The fields with each key once. `HTTPLogItem.MarshalJSON` writes the top-level fields, then every property,
100        // overwriting a top-level field of the same name, and the status is set after every property. So a property
101        // wins over `hostname`, `message`, and `service`, and `status` wins over a property. Properties already have
102        // unique keys.
103        let properties = log.additional_properties();
104        let mut fields = reuse_empty(mem::take(&mut self.fields));
105        fields.reserve(properties.len() + 4);
106        let (mut hostname_overwritten, mut message_overwritten, mut service_overwritten) = (false, false, false);
107        for (key, value) in properties {
108            match key.as_ref() {
109                "status" => continue,
110                "hostname" => hostname_overwritten = true,
111                "message" => message_overwritten = true,
112                "service" => service_overwritten = true,
113                _ => {}
114            }
115            fields.push((key.as_ref(), ContentField::Value(value)));
116        }
117        if !hostname_overwritten && !log.hostname().is_empty() {
118            fields.push(("hostname", ContentField::String(log.hostname())));
119        }
120        if !message_overwritten {
121            fields.push(("message", ContentField::String(log.message())));
122        }
123        if !service_overwritten && !log.service().is_empty() {
124            fields.push(("service", ContentField::String(log.service())));
125        }
126        fields.push(("status", ContentField::String(log.status())));
127
128        // Go sorts map keys by their bytes, as `str` orders them. Every key is unique, so an unstable sort, which
129        // doesn't allocate, gives the same order as a stable one.
130        fields.sort_unstable_by_key(|(key, _)| *key);
131
132        let written = self.write_fields(&fields);
133
134        fields.clear();
135        fields.shrink_to(CONTENT_FIELDS_RETENTION_LIMIT);
136        self.fields = reuse_empty(fields);
137        written
138    }
139
140    /// Writes a log's fields into the buffer, as a JSON object.
141    fn write_fields(&mut self, fields: &[(&str, ContentField<'_>)]) -> Result<(), GenericError> {
142        self.buffer.clear();
143        self.buffer.push(b'{');
144        let mut first = true;
145        for (key, field) in fields {
146            if !first {
147                self.buffer.push(b',');
148            }
149            first = false;
150
151            let buffer = &mut self.buffer;
152            let written = serde_json::to_writer(&mut *buffer, key).and_then(|()| {
153                buffer.push(b':');
154                match field {
155                    ContentField::String(value) => serde_json::to_writer(&mut *buffer, value),
156                    ContentField::Value(value) => serde_json::to_writer(&mut *buffer, value),
157                }
158            });
159            written.map_err(|e| generic_error!("Failed to serialize log property '{}': {}", key, e))?;
160        }
161        self.buffer.push(b'}');
162        Ok(())
163    }
164}

166/// Returns an empty vector with the allocation of `fields`, for fields that borrow for a different lifetime.
167///
168/// The element types differ only in their lifetimes, so collecting the empty vector's iterator reuses its allocation in
169/// place.
170fn reuse_empty<'a, 'b>(mut fields: Vec<(&'a str, ContentField<'a>)>) -> Vec<(&'b str, ContentField<'b>)> {
171    fields.clear();
172    fields
173        .into_iter()
174        .map(|_| unreachable!("the fields are empty"))
175        .collect()
176}

178/// The shortest time between two logged render failures.
179const RENDER_FAILURE_LOG_INTERVAL: Duration = Duration::from_secs(10);
180
181/// OTLP logs renderer.
182///
183/// Replaces the message of each log with its rendered content, and clears its additional properties, which the
184/// content now holds. The status, hostname, service, source, and tags stay on the log, for the envelope.
185#[derive(Default)]
186pub struct OtlpLogsRenderer {
187    renderer: ContentRenderer,
188    failures_since_logged: u64,
189    last_failure_logged: Option<Instant>,
190}
191
192impl OtlpLogsRenderer {
193    /// Renders a log's content into its message.
194    ///
195    /// If the log can't be rendered, it's counted in `metrics` as dropped and `None` is returned. That only happens for
196    /// a property that can't be serialized.
197    pub fn render(&mut self, log: Log, metrics: &Metrics) -> Option<Log> {
198        match self.renderer.render(&log) {
199            Ok(content) => Some(log.with_message(content).with_additional_properties(None)),
200            Err(error) => {
201                metrics.logs_dropped_render_failure().increment(1);
202                self.failures_since_logged += 1;
203
204                let now = Instant::now();
205                let log_due = self
206                    .last_failure_logged
207                    .is_none_or(|last| now.duration_since(last) >= RENDER_FAILURE_LOG_INTERVAL);
208                if log_due {
209                    self.last_failure_logged = Some(now);
210                    let failures = mem::take(&mut self.failures_since_logged);
211                    error!(%error, failures, "Failed to render OTLP log content. Dropping logs.");
212                }
213                None
214            }
215        }
216    }
217}

256    fn log_with_properties(message: &str, properties: JsonValue) -> Log {
257        let properties = match properties {
258            JsonValue::Object(properties) => properties

261                .collect::<HashMap<_, _>>(),

458    #[test]
459    fn matches_the_reference_byte_for_byte_except_known_gaps() {
460        // One renderer for every case, as the transform uses one for every log.
461        let mut renderer = ContentRenderer::default();
462        let mut failures = Vec::new();
463        for case in RENDER_CASES {
464            let content = renderer
465                .render(&(case.log)())
466                .unwrap_or_else(|e| panic!("{}: render failed: {}", case.name, e));
467            let content = &*content;
468            match &case.gap {
469                None if content != case.target => failures.push(format!(
470                    "{}: content differs from the target with no known gap\n  target:  {}\n  current: {}",
471                    case.name, case.target, content
472                )),
473                Some(gap) if content == case.target => failures.push(format!(
474                    "{}: content now matches the target; remove its entry for {}",
475                    case.name, gap.gap
476                )),
477                Some(gap) if content != gap.current => failures.push(format!(
478                    "{}: current content changed\n  expected: {}\n  current:  {}",
479                    case.name, gap.current, content
480                )),
481                _ => {}
482            }
483        }
484        assert!(failures.is_empty(), "render case failures:\n{}", failures.join("\n"));
485    }

546    fn releases_the_excess_of_an_oversized_buffer() {
547        let mut renderer = ContentRenderer::default();
548        let ordinary = log_with_properties("m", json!({"padding": "x".repeat(1024)}));
549        renderer.render(&ordinary).expect("log should render");
550        let ordinary_capacity = renderer.buffer.capacity();
551        assert!(ordinary_capacity >= 1024);
552        assert!(ordinary_capacity <= CONTENT_BUFFER_RETENTION_LIMIT_BYTES);
553
554        // An ordinary log keeps the buffer it grew.
555        renderer.render(&Log::new("m")).expect("log should render");
556        assert_eq!(renderer.buffer.capacity(), ordinary_capacity);
557
558        // An oversized log renders in full, then has the excess released.
559        let oversized = "x".repeat(2 * CONTENT_BUFFER_RETENTION_LIMIT_BYTES);
560        let content = renderer
561            .render(&Log::new(oversized.as_str()))
562            .expect("log should render");
563        assert!(content.contains(&oversized));
564        assert!(renderer.buffer.capacity() <= CONTENT_BUFFER_RETENTION_LIMIT_BYTES);
565    }
