1//! The trace-agent's `/info` discovery endpoint.
2//!
3//! Tracers read `/info` to learn which endpoints and features the agent supports, and how it is configured. The relay
4//! builds the payload from ADP's configuration, as the trace-agent's `makeInfoHandler` builds it from its own.
5//!
6//! Every response the trace-agent sends carries `Datadog-Agent-State`, the SHA-256 of the `/info` body, so that a
7//! tracer re-fetches `/info` when the hash changes. The relay's body can't match the trace-agent's byte for byte: Go
8//! orders `feature_flags` and `span_kinds_stats_computed` by map iteration, which is random. So the relay owns the
9//! header instead, and replaces the trace-agent's on every response it proxies.
10//!
11//! The payload follows the trace-agent's latest shape, whatever the version of the Agent ADP runs alongside.
12

53/// Serializes a float as Go's `encoding/json` does where the two differ: an integral value has no fractional part,
54/// so `10.0` is written `10`.
55fn serialize_go_float<S: Serializer>(value: &f64, serializer: S) -> Result<S::Ok, S::Error> {
56    // Go writes integral values below 1e21 without an exponent; past `i64`'s range, the float form is close enough.
57    if value.fract() == 0.0 && value.abs() < 9e18 {
58        serializer.serialize_i64(*value as i64)
59    } else {
60        serializer.serialize_f64(*value)
61    }
62}

64/// A float serialized as Go's `encoding/json` does; see [`serialize_go_float`].
65struct GoFloat(f64);
66
67impl Serialize for GoFloat {
68    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
69        serialize_go_float(&self.0, serializer)
70    }
71}

309/// A rendered `/info` body, with the headers derived from it.
310pub(super) struct RenderedInfo {
311    body: Bytes,
312    state: HeaderValue,
313    version: HeaderValue,
314}
315
316impl RenderedInfo {
317    fn render(
318        config: &InfoConfiguration, registry: &Registry, agent: &RunningAgentVersion, org_prop_marker: &str,
319    ) -> Self {
320        let payload = InfoPayload::new(config, registry, agent, org_prop_marker);
321
322        // The trace-agent indents with tabs; only the hash, not the layout, matters to tracers.
323        let mut body = Vec::new();
324        let mut serializer =
325            serde_json::Serializer::with_formatter(&mut body, serde_json::ser::PrettyFormatter::with_indent(b"\t"));
326        payload
327            .serialize(&mut serializer)
328            .expect("the info payload only holds serializable values");
329
330        let state = faster_hex::hex_string(&Sha256::digest(&body));
331        Self {
332            body: Bytes::from(body),
333            state: HeaderValue::from_str(&state).expect("a hex digest is a valid header value"),
334            version: HeaderValue::from_str(&agent.version).unwrap_or_else(|_| HeaderValue::from_static("")),
335        }
336    }
337}

339/// The `/info` endpoint: its payload, and the headers every other response carries.
340///
341/// Clones share the payload. The setters and [`Self::refresh_semantic_registry`] re-render it from every input, so they
342/// must be called from a single task, as the relay does, for the last render to see every update.
343#[derive(Clone)]
344pub(super) struct Info {
345    config: Arc<InfoConfiguration>,
346    semantic_registry: SemanticRegistryProvider,
347    agent: Arc<ArcSwap<RunningAgentVersion>>,
348    org_prop_marker: Arc<ArcSwap<String>>,
349    rendered: Arc<ArcSwap<RenderedInfo>>,
350}
351
352impl Info {
353    /// Renders the payload of `config` with the current semantic registry, before the Agent version and the org
354    /// propagation marker are known.
355    pub(super) fn new(config: InfoConfiguration, semantic_registry: SemanticRegistryProvider) -> Self {
356        let agent = RunningAgentVersion::default();
357        let rendered = RenderedInfo::render(&config, &semantic_registry.snapshot(), &agent, "");
358        Self {
359            config: Arc::new(config),
360            semantic_registry,
361            agent: Arc::new(ArcSwap::from_pointee(agent)),
362            org_prop_marker: Arc::new(ArcSwap::from_pointee(String::new())),
363            rendered: Arc::new(ArcSwap::from_pointee(rendered)),
364        }
365    }
366
367    /// Re-renders the payload with the org propagation marker of the API key.
368    pub(super) fn set_org_prop_marker(&self, marker: String) {
369        self.org_prop_marker.store(Arc::new(marker));
370        self.rerender();
371    }
372
373    /// Re-renders the payload with the version of the Agent.
374    pub(super) fn set_agent_version(&self, agent: RunningAgentVersion) {
375        self.agent.store(Arc::new(agent));
376        self.rerender();
377    }
378
379    /// Re-renders the payload with the current semantic registry, whose mappings select the peer tags.
380    ///
381    /// An unchanged registry renders the same body, so `Datadog-Agent-State` only changes when the peer tags do.
382    pub(super) fn refresh_semantic_registry(&self) {
383        self.rerender();
384    }
385
386    fn rerender(&self) {
387        let agent = self.agent.load();
388        let marker = self.org_prop_marker.load();
389        let rendered = RenderedInfo::render(&self.config, &self.semantic_registry.snapshot(), &agent, &marker);
390        self.rendered.store(Arc::new(rendered));
391    }
392
393    /// Answers `GET /info` with the current payload.
394    ///
395    /// As in the trace-agent, the response doesn't carry `Datadog-Agent-State` itself.
396    pub(super) fn response(&self) -> Response {
397        let rendered = self.rendered.load();
398        // The trace-agent sets no content type, so Go's sniffing labels the body as plain text.
399        (
400            [(header::CONTENT_TYPE, "text/plain; charset=utf-8")],
401            rendered.body.clone(),
402        )
403            .into_response()
404    }
405
406    /// Sets `Datadog-Agent-State` and `Datadog-Agent-Version` on a response the relay served itself.
407    pub(super) fn set_headers(&self, headers: &mut HeaderMap) {
408        let rendered = self.rendered.load();
409        headers.insert(AGENT_STATE_HEADER, rendered.state.clone());
410        headers.insert(AGENT_VERSION_HEADER, rendered.version.clone());
411    }
412
413    /// Replaces `Datadog-Agent-State` and `Datadog-Agent-Version` on a response proxied from the trace-agent.
414    ///
415    /// Only a header the trace-agent set is replaced, so a response the trace-agent sent without them, such as for an
416    /// endpoint it doesn't serve, stays without them.
417    pub(super) fn replace_headers(&self, headers: &mut HeaderMap) {
418        let rendered = self.rendered.load();
419        if headers.contains_key(AGENT_STATE_HEADER) {
420            headers.insert(AGENT_STATE_HEADER, rendered.state.clone());
421        }
422        if headers.contains_key(AGENT_VERSION_HEADER) {
423            headers.insert(AGENT_VERSION_HEADER, rendered.version.clone());
424        }
425    }
426}

428/// The `/info` payload, with the trace-agent's field names and order (`infoPayload`).
429#[derive(Serialize)]
430struct InfoPayload<'a> {
431    version: &'a str,
432    git_commit: &'a str,
433    endpoints: Vec<&'static str>,
434    #[serde(skip_serializing_if = "Vec::is_empty")]
435    feature_flags: Vec<&'a str>,
436    client_drop_p0s: bool,
437    span_meta_structs: bool,
438    long_running_spans: bool,
439    span_events: bool,
440    evp_proxy_allowed_headers: &'static [&'static str],
441    config: ReducedConfig<'a>,
442    peer_tags: Option<Vec<String>>,
443    span_kinds_stats_computed: Option<&'static [&'static str]>,
444    obfuscation_version: u32,
445    filter_tags: RenderedTagFilters,
446    filter_tags_regex: RenderedTagFilters,
447    #[serde(skip_serializing_if = "<[String]>::is_empty")]
448    ignore_resources: &'a [String],
449    #[serde(skip_serializing_if = "str::is_empty")]
450    org_prop_marker: &'a str,
451}
452
453impl<'a> InfoPayload<'a> {
454    fn new(
455        config: &'a InfoConfiguration, registry: &Registry, agent: &'a RunningAgentVersion, org_prop_marker: &'a str,
456    ) -> Self {
457        let endpoints: Vec<&'static str> = ENDPOINTS
458            .iter()
459            .filter(|(_, requires)| match requires {
460                Requires::Nothing => true,
461                Requires::Telemetry => config.telemetry_enabled,
462                Requires::RemoteConfiguration => config.remote_configuration_enabled,
463            })
464            .map(|(pattern, _)| *pattern)
465            .collect();
466
467        // The trace-agent reads the features into a set.
468        let mut feature_flags: Vec<&str> = Vec::with_capacity(config.features.len());
469        for feature in &config.features {
470            if !feature_flags.contains(&feature.as_str()) {
471                feature_flags.push(feature);
472            }
473        }
474
475        let sql = EffectiveSql::new(config);
476        let obfuscation_version = if sql.diverges_from_tracers() {
477            DIVERGENT_SQL_OBFUSCATION_VERSION
478        } else {
479            OBFUSCATION_VERSION
480        };
481
482        let peer_tags = config.peer_tags_aggregation.then(|| {
483            let custom: Vec<MetaString> = config
484                .custom_peer_tags
485                .iter()
486                .map(|tag| MetaString::from(tag.as_str()))
487                .collect();
488            let keys = PeerTagKeys::build(registry, &custom);
489            keys.keys().iter().map(|key| key.to_string()).collect::<Vec<_>>()
490        });
491
492        Self {
493            version: &agent.version,
494            git_commit: &agent.commit,
495            // The trace-agent lets tracers drop unsampled traces when it computes stats, unless the probabilistic
496            // sampler needs to see every trace.
497            client_drop_p0s: !config.probabilistic_sampler_enabled && endpoints.contains(&"/v0.6/stats"),
498            endpoints,
499            feature_flags,
500            span_meta_structs: true,
501            long_running_spans: true,
502            span_events: true,
503            evp_proxy_allowed_headers: EVP_PROXY_ALLOWED_HEADERS,
504            config: ReducedConfig::new(config, sql),
505            // The trace-agent reports no peer tags as `null`, not as an empty list.
506            peer_tags: peer_tags.filter(|tags| !tags.is_empty()),
507            span_kinds_stats_computed: config.compute_stats_by_span_kind.then_some(SPAN_KINDS_STATS_COMPUTED),
508            obfuscation_version,
509            filter_tags: RenderedTagFilters::new(&config.filter_tags),
510            filter_tags_regex: RenderedTagFilters::new(&config.filter_tags_regex),
511            ignore_resources: &config.ignore_resources,
512            org_prop_marker,
513        }
514    }
515}

517/// The SQL obfuscation the trace-agent applies (`EffectiveSQLConfig`), which beta feature flags select rather than
518/// the `apm_config.obfuscation.sql` settings.
519struct EffectiveSql {
520    replace_digits: bool,
521    keep_sql_alias: bool,
522    dollar_quoted_func: bool,
523    obfuscation_mode: String,
524}
525
526impl EffectiveSql {
527    fn new(config: &InfoConfiguration) -> Self {
528        let has = |feature: &str| config.features.iter().any(|enabled| enabled == feature);
529        // An explicit mode wins; otherwise the `sqllexer` feature selects the `sqllexer` obfuscator.
530        let obfuscation_mode = if !config.obfuscation.sql_obfuscation_mode.is_empty() {
531            config.obfuscation.sql_obfuscation_mode.clone()
532        } else if has("sqllexer") {
533            "obfuscate_only".to_string()
534        } else {
535            String::new()
536        };
537        Self {
538            replace_digits: has("quantize_sql_tables") || has("replace_sql_digits"),
539            keep_sql_alias: has("keep_sql_alias"),
540            dollar_quoted_func: has("dollar_quoted_func"),
541            obfuscation_mode,
542        }
543    }
544
545    /// Returns whether this SQL obfuscation changes the obfuscated queries from those tracers produce, which the
546    /// trace-agent signals with `obfuscation_version`.
547    ///
548    /// Table name collection is left out: it only collects metadata, which stats don't keep.
549    fn diverges_from_tracers(&self) -> bool {
550        self.replace_digits || self.keep_sql_alias || self.dollar_quoted_func || !self.obfuscation_mode.is_empty()
551    }
552}

554/// The trace-agent's `reducedConfig`.
555#[derive(Serialize)]
556struct ReducedConfig<'a> {
557    default_env: &'a str,
558    #[serde(serialize_with = "serialize_go_float")]
559    target_tps: f64,
560    #[serde(serialize_with = "serialize_go_float")]
561    max_eps: f64,
562    receiver_port: u16,
563    receiver_socket: &'a str,
564    connection_limit: u32,
565    receiver_timeout: u32,
566    max_request_bytes: u64,
567    statsd_port: u16,
568    #[serde(serialize_with = "serialize_go_float")]
569    max_memory: f64,
570    #[serde(serialize_with = "serialize_go_float")]
571    max_cpu: f64,
572    // Go sorts map keys, and a stable order keeps the hash stable.
573    analyzed_spans_by_service: BTreeMap<&'a str, BTreeMap<&'a str, GoFloat>>,
574    obfuscation: ReducedObfuscation<'a>,
575}
576
577impl<'a> ReducedConfig<'a> {
578    fn new(config: &'a InfoConfiguration, sql: EffectiveSql) -> Self {
579        Self {
580            default_env: &config.default_env,
581            target_tps: config.target_traces_per_second,
582            max_eps: config.max_events_per_second,
583            receiver_port: config.receiver_port,
584            receiver_socket: &config.receiver_socket,
585            connection_limit: config.connection_limit,
586            receiver_timeout: config.receiver_timeout,
587            max_request_bytes: config.max_request_bytes,
588            statsd_port: config.statsd_port,
589            max_memory: config.max_memory,
590            max_cpu: config.max_cpu,
591            analyzed_spans_by_service: config
592                .analyzed_spans_by_service
593                .iter()
594                .map(|(service, rates)| {
595                    let rates = rates
596                        .iter()

607/// The trace-agent's `reducedObfuscationConfig`.
608#[derive(Serialize)]
609struct ReducedObfuscation<'a> {
610    elastic_search: bool,
611    mongo: bool,
612    sql_exec_plan: bool,
613    sql_exec_plan_normalize: bool,
614    sql_obfuscation_mode: String,
615    // No rules are reported as `null`.
616    tag_replace_rules: Option<Vec<ReducedReplaceRule<'a>>>,
617    http: ReducedHttp,
618    remove_stack_traces: bool,
619    redis: ReducedCache,
620    valkey: ReducedCache,
621    memcached: ReducedMemcached,
622    credit_cards: ReducedCreditCards<'a>,
623    sql: ReducedSql,
624    elasticsearch: ReducedJson<'a>,
625    opensearch: ReducedJson<'a>,
626    mongodb: ReducedJson<'a>,
627}
628
629impl<'a> ReducedObfuscation<'a> {
630    fn new(config: &'a InfoConfiguration, sql: EffectiveSql) -> Self {
631        let obfuscation = &config.obfuscation;
632        let tag_replace_rules = (!config.replace_tags.is_empty()).then(|| {
633            config
634                .replace_tags
635                .iter()
636                .map(|rule| ReducedReplaceRule {
637                    name: &rule.name,
638                    pattern: &rule.pattern,
639                    repl: &rule.repl,
640                })
641                .collect()
642        });
643        Self {
644            elastic_search: obfuscation.elasticsearch.enabled,
645            mongo: obfuscation.mongodb.enabled,
646            sql_exec_plan: obfuscation.sql_exec_plan,
647            sql_exec_plan_normalize: obfuscation.sql_exec_plan_normalize,
648            sql_obfuscation_mode: sql.obfuscation_mode.clone(),
649            tag_replace_rules,
650            http: ReducedHttp {
651                remove_query_string: obfuscation.http_remove_query_string,
652                remove_path_digits: obfuscation.http_remove_paths_with_digits,
653            },
654            remove_stack_traces: obfuscation.remove_stack_traces,
655            redis: ReducedCache::new(&obfuscation.redis),
656            valkey: ReducedCache::new(&obfuscation.valkey),
657            memcached: ReducedMemcached {
658                enabled: obfuscation.memcached_enabled,
659                keep_command: obfuscation.memcached_keep_command,
660            },
661            credit_cards: ReducedCreditCards {
662                enabled: obfuscation.credit_cards_enabled,
663                luhn: obfuscation.credit_cards_luhn,
664                keep_values: &obfuscation.credit_cards_keep_values,
665            },
666            sql: ReducedSql::new(sql),
667            elasticsearch: ReducedJson::new(&obfuscation.elasticsearch),
668            opensearch: ReducedJson::new(&obfuscation.opensearch),
669            mongodb: ReducedJson::new(&obfuscation.mongodb),
670        }
671    }
672}
