1//! Tracer flare proxy.
2//!
3//! Tracers upload flares, a multipart form holding a zip archive and the support case it belongs to, to
4//! `/tracer_flare/v1`. Like the trace-agent, the relay forwards them as they are to the Datadog serverless flare
5//! endpoint, adding the API key.

25/// Path of the Datadog endpoint flares are forwarded to.
26const FLARE_ENDPOINT_PATH: &str = "/api/ui/support/serverless/flare";

28/// Site flares are sent to when none is configured, as in the trace-agent.

31/// Name the API key source is logged under.

34const DD_API_KEY_HEADER: HeaderName = HeaderName::from_static("dd-api-key");
35const FORWARDED_HEADER: HeaderName = HeaderName::from_static("forwarded");
36const X_FORWARDED_FOR_HEADER: HeaderName = HeaderName::from_static("x-forwarded-for");
37const X_FORWARDED_HOST_HEADER: HeaderName = HeaderName::from_static("x-forwarded-host");
38const X_FORWARDED_PROTO_HEADER: HeaderName = HeaderName::from_static("x-forwarded-proto");

40/// Settings of the tracer flare proxy.
41#[derive(Clone)]
42pub struct TracerFlareConfiguration {
43    /// Datadog site flares are sent to, such as `datadoghq.com`.
44    ///
45    /// Empty means `datadoghq.com`.
46    pub site: String,
47
48    /// API key flares are sent with.
49    pub api_key: Live<String>,
50
51    /// Settings of the Datadog intake forwarder.
52    ///
53    /// Only the settings of its HTTP client are used: the request timeout, the TLS settings (handshake timeout, key log
54    /// file, and certificate validation, including `skip_ssl_validation`), the proxy, the HTTP protocol, and the
55    /// connection reset interval. The destination comes from `site`, and the API key from `api_key`.
56    pub forwarder: ForwarderConfiguration,
57}

72#[derive(Debug)]

81impl TracerFlareProxy {
82    /// Creates a proxy forwarding flares to the flare endpoint of the configured site.
83    ///
84    /// # Errors
85    ///
86    /// If the site isn't a valid host, the API key isn't a valid header value, or the HTTP client can't be built, an
87    /// error is returned.
88    pub(super) fn from_config(config: &TracerFlareConfiguration) -> Result<Self, GenericError> {
89        let site = match config.site.trim() {
90            "" => DEFAULT_SITE,
91            site => site,
92        };
93        let endpoint = flare_endpoint(site, &flare_version_label(agent_version::version()))?;
94        let client_builder =
95            intake_http_client_builder(&config.forwarder, config.forwarder.effective_min_tls_version())?;
96        Self::new(client_builder, endpoint, config.api_key.as_str())
97    }
98
99    fn new(client_builder: HttpClientBuilder, endpoint: Uri, api_key: &str) -> Result<Self, GenericError> {
100        let api_key = ApiKeyCell::new(api_key)
101            .map_err(|e| generic_error!("The configured API key is not a valid HTTP header value: {e}"))?;
102        debug!(%endpoint, "Tracer flares are forwarded to Datadog.");
103
104        Ok(Self {
105            client: UpstreamClient::new(client_builder.build()?),
106            endpoint,
107            api_key,
108        })
109    }
110
111    /// Replaces the API key flares are sent with.
112    pub(super) fn refresh_api_key(&self, api_key: &str) {
113        self.api_key.refresh(API_KEY_SOURCE, api_key);
114    }
115
116    /// Forwards the flare in `request` to Datadog and returns its response.
117    ///
118    /// Any response from Datadog is returned as is, including an error status. The request keeps its method, headers,
119    /// and body, but not its path or query, as the trace-agent does.
120    ///
121    /// # Errors
122    ///
123    /// If Datadog can't be reached, an error is returned, answered with `502 Bad Gateway`.
124    pub(super) async fn forward(&self, request: AxumRequest) -> Result<Response, ForwardError> {
125        let (mut parts, body) = request.into_parts();
126
127        let forwarded_host = request_host(&parts.uri, &parts.headers);
128        let peer_ip =
129            parts
130                .extensions
131                .get::<ConnectInfo<ConnectionAddress>>()
132                .and_then(|ConnectInfo(addr)| match addr {
133                    ConnectionAddress::SocketLike(addr) => Some(addr.ip().to_string()),
134                    ConnectionAddress::ProcessLike(_) => None,
135                });
136
137        parts.uri = self.endpoint.clone();
138        // The relay's server also accepts HTTP/2: the client picks the protocol of its own connection.
139        parts.version = Version::default();
140        set_upstream_headers(
141            &mut parts.headers,
142            peer_ip.as_deref(),
143            forwarded_host.as_ref(),
144            &self.api_key.load(),
145        );
146
147        self.client
148            .send(http::Request::from_parts(parts, body))
149            .await
150            .map_err(|e| {
151                warn!(endpoint = %self.endpoint, error = %e, "Failed to forward tracer flare to Datadog.");
152                ForwardError
153            })
154    }
155}

157/// Returns the host a request was sent to, from its URI for HTTP/2 or its `Host` header for HTTP/1.1.
158fn request_host(uri: &Uri, headers: &HeaderMap) -> Option<HeaderValue> {
159    match uri.authority() {
160        Some(authority) => HeaderValue::from_str(authority.as_str()).ok(),
161        None => headers.get(header::HOST).cloned(),
162    }
163}

165/// Rewrites the headers of a flare request for Datadog, as Go's `httputil.ReverseProxy` does for the trace-agent.
166///
167/// Headers set by an earlier proxy are replaced rather than added to, so the client can't pass itself off as another.
168fn set_upstream_headers(
169    headers: &mut HeaderMap, peer_ip: Option<&str>, forwarded_host: Option<&HeaderValue>, api_key: &str,
170) {
171    strip_hop_by_hop_headers(headers);
172    // The client sets `Host` from the upstream URI.
173    headers.remove(header::HOST);
174    headers.remove(FORWARDED_HEADER);
175    headers.remove(X_FORWARDED_FOR_HEADER);
176    headers.remove(X_FORWARDED_HOST_HEADER);
177
178    if let Some(peer_ip) = peer_ip.and_then(|ip| HeaderValue::from_str(ip).ok()) {
179        headers.insert(X_FORWARDED_FOR_HEADER, peer_ip);
180    }
181    if let Some(host) = forwarded_host {
182        headers.insert(X_FORWARDED_HOST_HEADER, host.clone());
183    }
184    // The relay only serves plain HTTP.
185    headers.insert(X_FORWARDED_PROTO_HEADER, HeaderValue::from_static("http"));
186
187    match HeaderValue::from_str(api_key) {
188        Ok(mut value) => {
189            value.set_sensitive(true);
190            headers.insert(DD_API_KEY_HEADER, value);
191        }
192        // `ApiKeyCell` only holds valid header values.
193        Err(_) => {
194            headers.remove(DD_API_KEY_HEADER);
195        }
196    }
197}

199/// Returns the version label of the flare host, such as `7-50-0-flare`.
200///
201/// The trace-agent uses its own version. When the Agent version wasn't known at build time, or carries no version
202/// number, ADP's version is used instead.
203fn flare_version_label(agent_version: Option<AgentVersion>) -> String {
204    // A development build with no version number, such as `nightly`, has a major version of `0`.
205    let (major, minor, patch) = match agent_version.filter(|version| version.major() != 0) {
206        Some(version) => (version.major(), version.minor(), version.patch()),
207        None => {
208            let version = saluki_metadata::get_app_details().version();
209            (version.major().into(), version.minor().into(), version.patch().into())
210        }
211    };
212    format!("{major}-{minor}-{patch}-flare")
213}

215/// Returns the flare endpoint of `site`, a Datadog site such as `datadoghq.com` or a URL.
216///
217/// This follows the trace-agent's `getServerlessFlareEndpoint`: an official Datadog application host has its leftmost
218/// label replaced with `version_label` (a bare official site is treated as its `app.` host), and any other host is
219/// kept. The scheme is always `https`, and the path is always that of the flare endpoint.
220///
221/// It differs from the trace-agent in two ways:
222///
223/// - A site with a subdomain but no scheme, such as `us3.datadoghq.com`, is taken as the host, where the trace-agent
224///   leaves the host empty.
225/// - Official hosts are recognized as the forwarder recognizes them, so a Multi-Region Failover host such as
226///   `app.mrf.datadoghq.com` is versioned too, where the trace-agent keeps it.
227///
228/// # Errors
229///
230/// If `site` isn't a valid URL with a host, an error is returned.
231fn flare_endpoint(site: &str, version_label: &str) -> Result<Uri, GenericError> {
232    let site = site.trim();
233    let invalid = |e: &dyn std::fmt::Display| {
234        generic_error!("Datadog site `{site}` is not a valid host to send tracer flares to: {e}")
235    };
236
237    let mut url = Url::parse(&endpoint_with_default_scheme(site)).map_err(|e| invalid(&e))?;
238    if let Some(app_host) = url
239        .host_str()
240        .filter(|host| is_datadog_site(host))
241        .map(|host| format!("app.{host}"))
242    {
243        url.set_host(Some(&app_host)).map_err(|e| invalid(&e))?;
244    }
245    let mut url = add_version_prefix(url, version_label).map_err(|e| invalid(&e))?;
246
247    // `http` and `https` are both special schemes, so switching between them can't fail.
248    let _ = url.set_scheme("https");
249    // The URL has a host, so it can carry neither credentials nor a fragment that would fail to clear.
250    let _ = url.set_username("");
251    let _ = url.set_password(None);
252    url.set_path(FLARE_ENDPOINT_PATH);
253    url.set_query(None);
254    url.set_fragment(None);
255
256    url.as_str().parse().map_err(|e| invalid(&e))
257}

259/// A proxy forwarding to `base` over plain HTTP, standing in for the flare endpoint, as the trace-agent's test does.

263        .with_tls_config(|builder| builder.with_root_cert_store(rustls::RootCertStore::empty()))
264        .with_http_protocol(saluki_io::net::client::http::HttpProtocol::Http1);

266        .parse()
267        .expect("endpoint should parse");
268    TracerFlareProxy::new(builder, endpoint, api_key).expect("proxy should build")
269}

388    /// What the fake intake saw of the last request it received.
389    #[derive(Clone, Debug, Default)]
390    struct Seen {
391        method: String,
392        uri: String,
393        headers: HeaderMap,
394        body: Vec<u8>,
395    }
396
397    /// A stand-in flare intake: records each request and answers with `status`.
398    async fn spawn_intake(status: StatusCode) -> (String, Arc<AsyncMutex<Seen>>) {
399        let seen = Arc::new(AsyncMutex::new(Seen::default()));
400        let router = Router::new().fallback(any({
401            let seen = Arc::clone(&seen);
402            move |request: AxumRequest| {
403                let seen = Arc::clone(&seen);
404                async move {
405                    let (parts, body) = request.into_parts();
406                    let body = to_bytes(body, usize::MAX)
407                        .await
408                        .expect("request body should be readable");
409                    *seen.lock().await = Seen {
410                        method: parts.method.to_string(),
411                        uri: parts.uri.to_string(),
412                        headers: parts.headers,
413                        body: body.to_vec(),
414                    };
415                    (status, [("x-intake", "yes"), ("connection", "close")], "from intake")
416                }
417            }
418        }));
419        let listener = TcpListener::bind("127.0.0.1:0")
420            .await
421            .expect("should bind an ephemeral port");
422        let addr = listener.local_addr().expect("listener should have a local address");
423        tokio::spawn(async move { axum::serve(listener, router).await });
424        (format!("http://{addr}"), seen)
425    }

427    const MULTIPART_BODY: &str = "--boundary\r\n\
428        Content-Disposition: form-data; name=\"case_id\"\r\n\r\ncase_id\r\n\
429        --boundary\r\n\
430        Content-Disposition: form-data; name=\"flare_file\"; filename=\"flare.zip\"\r\n\
431        Content-Type: application/octet-stream\r\n\r\nzip bytes\r\n\
432        --boundary--\r\n";
433
434    fn flare_request(uri: &str, headers: &[(&str, &str)], peer: Option<ConnectionAddress>) -> AxumRequest {
435        let mut builder = http::Request::builder()
436            .method("POST")
437            .uri(uri)
438            .header("content-type", "multipart/form-data; boundary=boundary")
439            .header("host", "localhost:8126");
440        for (name, value) in headers {
441            builder = builder.header(*name, *value);
442        }
443        let mut request = builder.body(Body::from(MULTIPART_BODY)).expect("request should build");
444        if let Some(peer) = peer {
445            request.extensions_mut().insert(ConnectInfo(peer));
446        }
447        request
448    }

450    #[tokio::test]
451    async fn forwards_the_flare_to_the_flare_endpoint() {
452        let (base, seen) = spawn_intake(StatusCode::OK).await;
453        let proxy = plain_http_proxy(&base, "test-api-key");
454
455        let response = proxy
456            .forward(flare_request(
457                "/tracer_flare/v1?foo=bar",
458                &[],
459                Some(SocketAddr::from(([192, 0, 2, 7], 4242)).into()),
460            ))
461            .await
462            .expect("flare should be forwarded");
463
464        assert_eq!(response.status(), StatusCode::OK);
465        assert_eq!(response.headers()["x-intake"], "yes");
466        assert!(!response.headers().contains_key("connection"));
467        let body = to_bytes(response.into_body(), usize::MAX).await.unwrap();
468        assert_eq!(&body[..], b"from intake");
469
470        let seen = seen.lock().await.clone();
471        assert_eq!(seen.method, "POST");
472        assert_eq!(seen.uri, FLARE_ENDPOINT_PATH, "the path and query should be replaced");
473        assert_eq!(seen.headers["dd-api-key"], "test-api-key");
474        assert_eq!(seen.headers["x-forwarded-for"], "192.0.2.7");
475        assert_eq!(seen.headers["x-forwarded-host"], "localhost:8126");
476        assert_eq!(seen.headers["x-forwarded-proto"], "http");
477        assert_eq!(seen.headers["content-type"], "multipart/form-data; boundary=boundary");
478        assert_ne!(seen.headers["host"], "localhost:8126");
479        assert_eq!(seen.body, MULTIPART_BODY.as_bytes());
480    }
