3//! The marker identifies the Datadog organization of the API key without revealing it: the trace-agent fetches the

23/// Header the API key is sent in.
24const DD_API_KEY_HEADER: &str = "DD-API-KEY";
25
26/// Number of retries after the first attempt, as in the trace-agent.
27const MAX_RETRIES: u32 = 3;
28
29/// Fetches the org propagation marker of the API key.
30pub(super) struct OrgPropagationMarkerFetcher {
31    client: HttpClient,
32    endpoint: String,
33    api_key: ApiKeyCell,
34    request_timeout: Duration,
35}
36
37impl OrgPropagationMarkerFetcher {
38    /// Creates a fetcher that queries `endpoint`, the URL of `/api/v2/validate`, with the current key of `api_key`.
39    ///
40    /// The HTTP client uses the forwarder's settings, as the trace-agent's client does, and the forwarder's request
41    /// timeout bounds each attempt, response body included.
42    ///
43    /// # Errors
44    ///
45    /// If the HTTP client can't be built, an error is returned.
46    pub(super) fn new(
47        forwarder: &ForwarderConfiguration, endpoint: String, api_key: ApiKeyCell,
48    ) -> Result<Self, GenericError> {
49        let client = intake_http_client_builder(forwarder, forwarder.effective_min_tls_version())?.build()?;
50        Ok(Self::with_client(
51            client,
52            endpoint,
53            api_key,
54            forwarder.request_timeout(),
55        ))
56    }
57
58    fn with_client(client: HttpClient, endpoint: String, api_key: ApiKeyCell, request_timeout: Duration) -> Self {
59        Self {
60            client,
61            endpoint,
62            api_key,
63            request_timeout,
64        }
65    }
66
67    /// Fetches the marker with the trace-agent's retries: up to four attempts, `base_delay`, twice, and four times
68    /// `base_delay` apart.
69    ///
70    /// Returns `None` once every attempt failed, as the trace-agent then gives up.
71    pub(super) async fn fetch_with_retries(self, base_delay: Duration) -> Option<String> {
72        let Self {
73            client,
74            endpoint,
75            api_key,
76            request_timeout,
77        } = self;
78        let (endpoint, api_key) = (endpoint.as_str(), &api_key);
79        // The closure owns the client, as a borrowed one would keep the future from being `Send`.
80        let fetch = move || fetch_marker(client.clone(), endpoint, api_key, request_timeout);
81        let result = fetch
82            .retry(
83                ExponentialBuilder::default()
84                    .with_min_delay(base_delay)
85                    .with_factor(2.0)
86                    .with_max_times(MAX_RETRIES as usize),
87            )
88            .notify(|error, delay| {
89                warn!(%error, "Failed to fetch the org propagation marker. Retrying in {delay:?}...");
90            })
91            .await;
92        match result {
93            Ok(marker) => {
94                debug!(marker, "Fetched the org propagation marker.");
95                Some(marker)
96            }
97            Err(error) => {
98                warn!(
99                    %error,
100                    "Org propagation marker fetch failed after {} attempts; `/info` reports none.",
101                    MAX_RETRIES + 1
102                );
103                None
104            }
105        }
106    }
107}

109/// Fetches the marker of the key in `api_key` from `endpoint`, once, within `request_timeout`.
110async fn fetch_marker(
111    mut client: HttpClient, endpoint: &str, api_key: &ApiKeyCell, request_timeout: Duration,
112) -> Result<String, GenericError> {
113    let uri: Uri = endpoint
114        .parse()
115        .with_error_context(|| format!("The validate endpoint `{endpoint}` is not a valid URL."))?;
116    let request = Request::get(uri)
117        .header(DD_API_KEY_HEADER, api_key.load().as_ref())
118        .body(Empty::<Bytes>::new())
119        .error_context("Failed to build the validate request.")?;
120
121    // The client's timeout ends once the headers arrive, whereas the trace-agent's also bounds the body.
122    let (status, body) = tokio::time::timeout(request_timeout, async {
123        let response = client.send(request).await?;
124        let status = response.status();
125        let body = response
126            .into_body()
127            .collect()
128            .await
129            .error_context("Failed to read the validate response.")?
130            .to_bytes();
131        Ok::<_, GenericError>((status, body))
132    })
133    .await
134    .map_err(|_| generic_error!("timed out after {:?}", request_timeout))??;
135    let org_uuid = parse_validate_response(status, &body)?;
136    Ok(compute_marker(&org_uuid))
137}

139/// The part of the `/api/v2/validate` response the marker comes from.
140#[derive(Deserialize)]
141struct ValidateResponse {
142    data: ValidateData,
143}
144
145#[derive(Deserialize)]
146struct ValidateData {
147    #[serde(default)]
148    id: String,
149}
150
151/// Returns the organization UUID from a `/api/v2/validate` response.
152fn parse_validate_response(status: StatusCode, body: &[u8]) -> Result<String, GenericError> {
153    if status != StatusCode::OK {
154        return Err(generic_error!("unexpected status {}", status.as_u16()));
155    }
156    let response: ValidateResponse = serde_json::from_slice(body).error_context("decode response")?;
157    if response.data.id.is_empty() {
158        return Err(generic_error!("empty org UUID in validate response"));
159    }
160    Ok(response.data.id)
161}
162
163/// Computes the marker of an organization UUID, as the trace-agent's `computeOPM` does: the first eight bytes of its
164/// SHA-256, with the low four bits of the last one cleared, in URL-safe base64 without padding.
165fn compute_marker(org_uuid: &str) -> String {
166    let digest = Sha256::digest(org_uuid.as_bytes());
167    let mut truncated = [0u8; 8];
168    truncated.copy_from_slice(&digest[..8]);
169    truncated[7] &= 0xF0;
170    URL_SAFE_NO_PAD.encode(truncated)
171}

185    #[test]
186    fn the_marker_matches_the_trace_agent() {
187        // Expected values are the output of the trace-agent's `computeOPM`.
188        assert_eq!(compute_marker("8dee7c38-00cb-11ea-a77b-8b5a08d3b091"), "PuAey90vQlA");
189        assert_eq!(compute_marker("org"), "6Hy0XAWtOJA");
190        assert_eq!(compute_marker(""), "47DEQpj8HBA");
191    }

193    #[test]
194    fn the_org_uuid_comes_from_a_successful_validate_response() {
195        let body = br#"{"valid":true,"data":{"id":"org-uuid","type":"api_key"}}"#;
196        assert_eq!(parse_validate_response(StatusCode::OK, body).unwrap(), "org-uuid");
197
198        assert!(parse_validate_response(StatusCode::FORBIDDEN, body).is_err());
199        assert!(parse_validate_response(StatusCode::OK, br#"{"data":{}}"#).is_err());
200        assert!(parse_validate_response(StatusCode::OK, b"not json").is_err());
201    }

203    /// Starts a validate endpoint that fails the first `failures` requests, and records the API key of each request.
204    async fn spawn_validate_endpoint(failures: usize) -> (String, Arc<Mutex<Vec<String>>>) {
205        let seen = Arc::new(Mutex::new(Vec::new()));
206        let router = Router::new()
207            .route(
208                "/api/v2/validate",
209                get(
210                    move |State(seen): State<Arc<Mutex<Vec<String>>>>, headers: HeaderMap| async move {
211                        let mut seen = seen.lock().unwrap();
212                        seen.push(headers[DD_API_KEY_HEADER].to_str().unwrap().to_string());
213                        if seen.len() <= failures {
214                            (StatusCode::INTERNAL_SERVER_ERROR, String::new())
215                        } else {
216                            (StatusCode::OK, r#"{"valid":true,"data":{"id":"org"}}"#.to_string())
217                        }
218                    },
219                ),
220            )
221            .with_state(Arc::clone(&seen));
222        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
223        let addr = listener.local_addr().unwrap();
224        tokio::spawn(async move { axum::serve(listener, router).await });
225        (format!("http://{addr}/api/v2/validate"), seen)
226    }

228    fn fetcher(endpoint: String, api_key: &ApiKeyCell, request_timeout: Duration) -> OrgPropagationMarkerFetcher {
229        let client = HttpClient::builder().build().unwrap();
230        OrgPropagationMarkerFetcher::with_client(client, endpoint, api_key.clone(), request_timeout)
231    }

233    #[tokio::test]
234    async fn a_stalled_response_body_fails_the_attempt() {
235        crate::common::datadog::test_util::init_tls();
236        let attempts = Arc::new(Mutex::new(0));
237        let router = Router::new()
238            .route(
239                "/api/v2/validate",
240                get(move |State(attempts): State<Arc<Mutex<usize>>>| async move {
241                    *attempts.lock().unwrap() += 1;
242                    // Headers go out, but the body never completes.
243                    axum::body::Body::from_stream(futures::stream::pending::<Result<Bytes, std::io::Error>>())
244                }),
245            )
246            .with_state(Arc::clone(&attempts));
247        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
248        let addr = listener.local_addr().unwrap();
249        tokio::spawn(async move { axum::serve(listener, router).await });
250        let api_key = ApiKeyCell::new("key").unwrap();
251
252        let marker = tokio::time::timeout(
253            Duration::from_secs(10),
254            fetcher(
255                format!("http://{addr}/api/v2/validate"),
256                &api_key,
257                Duration::from_millis(50),
258            )
259            .fetch_with_retries(Duration::from_millis(1)),
260        )
261        .await
262        .expect("each attempt should time out");
263
264        assert_eq!(marker, None);
265        assert_eq!(*attempts.lock().unwrap(), 4);
266    }
