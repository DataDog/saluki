1//! Builds endpoint routing groups from exact-name metric allow lists.
2
3use std::collections::{BTreeMap, HashMap};
4
5use agent_data_plane_config::shared;
6use saluki_error::{generic_error, GenericError};
7
8/// A group of configured endpoints that share one exact-name metric allowlist.
9#[derive(Clone, Debug, Eq, PartialEq)]
10pub struct EndpointAllowlistGroup {
11    /// Configured endpoint identities targeted by this policy.
12    pub endpoints: Vec<String>,
13
14    /// Exact metric names permitted to reach these endpoints.
15    pub metric_allowlist: Vec<String>,
16}
17
18/// Endpoint-aware selective metric-routing configuration.
19#[derive(Debug, Eq, PartialEq)]
20pub struct MetricsEndpointRoutingConfiguration {
21    selected_endpoints: Vec<String>,
22    policy_groups: Vec<EndpointAllowlistGroup>,
23}
24
25impl MetricsEndpointRoutingConfiguration {
26    /// Resolves endpoint policies against the configured primary and additional endpoints.
27    ///
28    /// An empty policy map preserves ordinary endpoint routing. Every policy key must exactly match a key in
29    /// `additional_endpoints` or the effective primary endpoint; this prevents a typo from silently leaving an intended
30    /// capacity-saving destination on the unfiltered path. Equivalent allow lists are grouped so they can share one
31    /// filtered encoder branch.
32    ///
33    /// # Errors
34    ///
35    /// Returns an error if a policy names neither the primary endpoint nor an endpoint configured in
36    /// `additional_endpoints`.
37    pub fn from_configuration(
38        metric_allowlists: &HashMap<String, Vec<String>>, endpoints: &shared::Endpoints,
39    ) -> Result<Self, GenericError> {
40        let mut selected_endpoints = Vec::with_capacity(metric_allowlists.len());
41        let mut grouped_endpoints = BTreeMap::<Vec<String>, Vec<String>>::new();
42        let primary_endpoint = endpoints.primary_endpoint();
43
44        for (endpoint, metric_allowlist) in metric_allowlists {
45            if endpoint != &primary_endpoint && !endpoints.additional_endpoints.contains_key(endpoint) {
46                return Err(generic_error!(
47                    "Experimental metrics endpoint-routing policy endpoint '{}' does not match the configured primary \
48                     endpoint and is not present in `additional_endpoints`; correct the endpoint, add it and its API \
49                     key to `additional_endpoints`, or remove it from \
50                     `experimental.metrics_endpoint_routing.metric_allowlist`.",
51                    endpoint
52                ));
53            }
54
55            selected_endpoints.push(endpoint.clone());
56
57            let mut canonical_allowlist = metric_allowlist.clone();
58            canonical_allowlist.sort_unstable();
59            canonical_allowlist.dedup();
60            if !canonical_allowlist.is_empty() {
61                grouped_endpoints
62                    .entry(canonical_allowlist)
63                    .or_default()
64                    .push(endpoint.clone());
65            }
66        }
67
68        selected_endpoints.sort_unstable();
69        let policy_groups = grouped_endpoints
70            .into_iter()
71            .map(|(metric_allowlist, mut endpoints)| {
72                endpoints.sort_unstable();
73                EndpointAllowlistGroup {
74                    endpoints,
75                    metric_allowlist,
76                }
77            })
78            .collect();
79
80        Ok(Self {
81            selected_endpoints,
82            policy_groups,
83        })
84    }
85
86    /// Returns every configured endpoint removed from the ordinary unfiltered metric path.
87    pub fn selected_endpoints(&self) -> &[String] {
88        &self.selected_endpoints
89    }
90
91    /// Returns the filtered routing groups that require an encoder branch.
92    ///
93    /// Selected endpoints with an empty allowlist are absent because they receive no metric payloads.
94    pub fn policy_groups(&self) -> &[EndpointAllowlistGroup] {
95        &self.policy_groups
96    }
97}

99#[cfg(test)]
100mod tests {
101    use std::collections::HashMap;
102
103    use agent_data_plane_config::ConfigValue;
104
105    use super::*;
106
107    const PRIMARY: &str = "https://primary.example.com";
108
109    fn endpoints() -> shared::Endpoints {
110        shared::Endpoints {
111            dd_url: ConfigValue::explicit(PRIMARY.to_string()),
112            additional_endpoints: HashMap::from([
113                ("https://secondary-a.example.com".to_string(), vec!["key-a".to_string()]),
114                ("https://secondary-b.example.com".to_string(), vec!["key-b".to_string()]),
115                ("https://secondary-c.example.com".to_string(), vec!["key-c".to_string()]),
116            ]),
117            ..Default::default()
118        }
119    }
120
121    #[test]
122    fn empty_policy_map_preserves_the_ordinary_endpoint_path() {
123        let allowlists = HashMap::new();
124
125        let config = MetricsEndpointRoutingConfiguration::from_configuration(&allowlists, &endpoints())
126            .expect("empty routing should be valid");
127        assert!(config.selected_endpoints().is_empty());
128        assert!(config.policy_groups().is_empty());
129    }
130
131    #[test]
132    fn groups_equivalent_primary_and_additional_allowlists_and_keeps_empty_policies_selected() {
133        let allowlists = HashMap::from([
134            (
135                PRIMARY.to_string(),
136                vec!["metric.b".to_string(), "metric.a".to_string(), "metric.a".to_string()],
137            ),
138            (
139                "https://secondary-b.example.com".to_string(),
140                vec!["metric.a".to_string(), "metric.b".to_string()],
141            ),
142            ("https://secondary-c.example.com".to_string(), Vec::new()),
143        ]);
144
145        let config = MetricsEndpointRoutingConfiguration::from_configuration(&allowlists, &endpoints())
146            .expect("configured endpoints should resolve");
147        assert_eq!(
148            config.selected_endpoints(),
149            [
150                "https://primary.example.com",
151                "https://secondary-b.example.com",
152                "https://secondary-c.example.com"
153            ]
154        );
155        assert_eq!(
156            config.policy_groups(),
157            [EndpointAllowlistGroup {
158                endpoints: vec![
159                    "https://primary.example.com".to_string(),
160                    "https://secondary-b.example.com".to_string()
161                ],
162                metric_allowlist: vec!["metric.a".to_string(), "metric.b".to_string()]
163            }]
164        );
165    }
166
167    #[test]
168    fn accepts_a_site_derived_primary_endpoint() {
169        let endpoints = shared::Endpoints {
170            site: ConfigValue::explicit("us5.datadoghq.com".to_string()),
171            dd_url: ConfigValue::defaulted("https://app.datadoghq.com".to_string()),
172            ..Default::default()
173        };
174        let allowlists = HashMap::from([(
175            "https://app.us5.datadoghq.com".to_string(),
176            vec!["metric.a".to_string()],
177        )]);
178
179        let config = MetricsEndpointRoutingConfiguration::from_configuration(&allowlists, &endpoints)
180            .expect("site-derived primary endpoint should resolve");
181        assert_eq!(config.selected_endpoints(), ["https://app.us5.datadoghq.com"]);
182    }
183
184    #[test]
185    fn rejects_a_policy_that_does_not_name_a_configured_endpoint() {
186        let allowlists = HashMap::from([("https://typo.example.com".to_string(), vec!["metric.a".to_string()])]);
187
188        let error = MetricsEndpointRoutingConfiguration::from_configuration(&allowlists, &endpoints())
189            .expect_err("unknown endpoint should be rejected");
190        let message = error.to_string();
191        assert!(message.contains("https://typo.example.com"));
192        assert!(message.contains("primary endpoint"));
193        assert!(message.contains("additional_endpoints"));
194    }
