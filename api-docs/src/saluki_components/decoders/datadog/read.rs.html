1//! Low-level MessagePack read helpers for the v1.0 APM trace wire format.
2//!
3//! Each helper operates on a `&mut &[u8]` cursor: reads advance the slice in place, mirroring the
4//! byte-slice threading (`o []byte`) used by the reference Go decoder. Failures are mapped to
5//! [`DecodeError`] with a static context string naming the field being read.
6
7use std::fmt::Display;
8
9use super::error::DecodeError;
10
11/// Maximum element count permitted in any array or map header.
12///
13/// Protects the decoder from payloads that declare an enormous collection size to force large
14/// allocations. Matches the reference decoder's `maxSize` (25,000,000). This alone doesn't stop a
15/// tiny payload from claiming a count near this ceiling; [`read_array_len`] and [`read_map_len`]
16/// additionally reject counts that couldn't possibly be backed by the remaining input, via
17/// [`check_slab_count`].
18pub const MAX_SIZE: u32 = 25_000_000;
19
20/// Conservative lower bound on the wire size of one array element: every element is at least a
21/// one-byte `nil` or fixint.
22const MIN_BYTES_PER_ARRAY_ELEMENT: u32 = 1;
23
24/// Conservative lower bound on the wire size of one map entry: a key and a value, each at least
25/// one byte.
26const MIN_BYTES_PER_MAP_ENTRY: u32 = 2;
27
28/// Maximum nesting depth when skipping an unknown value.
29///
30/// Bounds recursion so a deeply nested unknown field cannot overflow the stack. Matches default OTel limit.
31const MAX_SKIP_DEPTH: usize = 64;
32
33/// Builds a closure that maps a MessagePack decoder error into a [`DecodeError::Msgpack`] carrying
34/// the given static context.
35fn mp<E: Display>(context: &'static str) -> impl FnOnce(E) -> DecodeError {
36    move |e| DecodeError::Msgpack {
37        context,
38        detail: e.to_string(),
39    }
40}
41
42/// Guards a collection pre-allocation (`Vec::with_capacity(len)`) against a claimed element count
43/// that couldn't possibly be backed by the remaining bytes.
44///
45/// Without this, a small malicious payload could set an array or map header to millions of
46/// entries and force a large allocation before decoding ever reaches the missing bytes and fails
47/// naturally. This is a much tighter bound than [`MAX_SIZE`] for small inputs, since it scales
48/// with the actual data available rather than a fixed ceiling.
49///
50/// # Errors
51///
52/// Returns an error if `len * min_bytes_per_entry` exceeds the number of bytes remaining on the
53/// cursor.
54fn check_slab_count(
55    len: u32, min_bytes_per_entry: u32, remaining: &[u8], context: &'static str,
56) -> Result<(), DecodeError> {
57    if u64::from(len) * u64::from(min_bytes_per_entry) > remaining.len() as u64 {
58        return Err(DecodeError::ImplausibleHeaderCount {
59            context,
60            len,
61            remaining: remaining.len(),
62        });
63    }
64    Ok(())
65}

67/// Reads an array header, returning its element count.
68///
69/// # Errors
70///
71/// Returns an error if the next value is not an array header, if the declared length exceeds
72/// [`MAX_SIZE`], or if the declared length couldn't possibly be backed by the remaining input (see
73/// [`check_slab_count`]).
74pub fn read_array_len(r: &mut &[u8], context: &'static str) -> Result<u32, DecodeError> {
75    read_array_len_with_minimum(r, MIN_BYTES_PER_ARRAY_ELEMENT, context)
76}
77
78/// Reads an array header and checks that the remaining input can plausibly contain its elements.
79///
80/// `min_bytes_per_element` is a format-specific plausibility estimate for the average encoded size
81/// of each element. Use [`read_array_len`] when the element type does not have a tighter bound than
82/// one byte.
83///
84/// # Errors
85///
86/// Returns an error if the next value is not an array header, if the declared length exceeds
87/// [`MAX_SIZE`], or if the declared elements would require more bytes than remain on the cursor.
88pub fn read_array_len_with_minimum(
89    r: &mut &[u8], min_bytes_per_element: u32, context: &'static str,
90) -> Result<u32, DecodeError> {
91    let len = rmp::decode::read_array_len(r).map_err(mp(context))?;
92    if len > MAX_SIZE {
93        return Err(DecodeError::OversizeHeader {
94            context,
95            len,
96            max: MAX_SIZE,
97        });
98    }
99    check_slab_count(len, min_bytes_per_element, r, context)?;
100    Ok(len)
101}

103/// Reads a map header, returning its entry count.
104///
105/// # Errors
106///
107/// Returns an error if the next value is not a map header, if the declared length exceeds
108/// [`MAX_SIZE`], or if the declared length couldn't possibly be backed by the remaining input (see
109/// [`check_slab_count`]).
110pub fn read_map_len(r: &mut &[u8], context: &'static str) -> Result<u32, DecodeError> {
111    let len = rmp::decode::read_map_len(r).map_err(mp(context))?;
112    if len > MAX_SIZE {
113        return Err(DecodeError::OversizeHeader {
114            context,
115            len,
116            max: MAX_SIZE,
117        });
118    }
119    check_slab_count(len, MIN_BYTES_PER_MAP_ENTRY, r, context)?;
120    Ok(len)
121}

153/// Reads a UTF-8 string, borrowing from the input for the lifetime of the underlying buffer.
154pub fn read_str<'a>(r: &mut &'a [u8], context: &'static str) -> Result<&'a str, DecodeError> {
155    let (s, tail) = rmp::decode::read_str_from_slice(*r).map_err(mp(context))?;
156    *r = tail;
157    Ok(s)
158}

160/// Reads a binary blob, copying it into an owned `Vec<u8>`.
161pub fn read_bytes(r: &mut &[u8], context: &'static str) -> Result<Vec<u8>, DecodeError> {
162    let len = rmp::decode::read_bin_len(r).map_err(mp(context))? as usize;
163    if r.len() < len {
164        return Err(DecodeError::Msgpack {
165            context,
166            detail: format!("expected {len} bytes but only {} remain", r.len()),
167        });
168    }
169    let (head, tail) = (*r).split_at(len);
170    *r = tail;
171    Ok(head.to_vec())
172}

174/// Returns `true` if the next value on the cursor is encoded as a MessagePack string.
175///
176/// Used to distinguish the two encodings of a streaming string: a literal string (to add to the
177/// table) versus a `uint32` index (a reference into the table). Does not advance the cursor.
178pub fn peek_is_str(r: &&[u8]) -> bool {
179    match r.first() {
180        // fixstr (0xa0..=0xbf), str8 (0xd9), str16 (0xda), str32 (0xdb).
181        Some(&b) => is_fixstr(b) || matches!(b, 0xd9..=0xdb),
182        None => false,
183    }
184}
185
186/// Returns `true` if the next value on the cursor is encoded as a MessagePack array.
187///
188/// Used when walking an unknown field's value, where the shape is not known ahead of time. Does not
189/// advance the cursor.
190pub fn peek_is_array(r: &&[u8]) -> bool {
191    match r.first() {
192        // fixarray (0x90..=0x9f), array16 (0xdc), array32 (0xdd).
193        Some(&b) => b & 0xf0 == 0x90 || matches!(b, 0xdc | 0xdd),
194        None => false,
195    }
196}

198/// Returns `true` if the next value on the cursor is encoded as a MessagePack map.
199///
200/// Used when walking an unknown field's value, where the shape is not known ahead of time. Does not
201/// advance the cursor.
202pub fn peek_is_map(r: &&[u8]) -> bool {
203    match r.first() {
204        // fixmap (0x80..=0x8f), map16 (0xde), map32 (0xdf).
205        Some(&b) => b & 0xf0 == 0x80 || matches!(b, 0xde | 0xdf),
206        None => false,
207    }
208}

210/// Returns `true` if the marker byte is a MessagePack fixstr.
211fn is_fixstr(b: u8) -> bool {
212    b & 0xe0 == 0xa0
213}

215/// Advances the cursor by `n` bytes, returning an error if fewer remain.
216fn advance(r: &mut &[u8], n: usize, context: &'static str) -> Result<(), DecodeError> {
217    if r.len() < n {
218        return Err(DecodeError::Msgpack {
219            context,
220            detail: format!("expected {n} bytes but only {} remain", r.len()),
221        });
222    }
223    *r = &r[n..];
224    Ok(())
225}
226
227/// Reads `n` big-endian length bytes into a `usize`.
228fn read_len(r: &mut &[u8], n: usize, context: &'static str) -> Result<usize, DecodeError> {
229    if r.len() < n {
230        return Err(DecodeError::Msgpack {
231            context,
232            detail: format!("expected {n} length bytes but only {} remain", r.len()),
233        });
234    }
235    let mut len = 0usize;
236    for &b in &r[..n] {
237        len = (len << 8) | b as usize;
238    }
239    *r = &r[n..];
240    Ok(len)
241}

243/// Skips over a single MessagePack value of any type without interpreting it.
244///
245/// This keeps the cursor aligned but does nothing else. Unknown *fields* must not be skipped with
246/// this directly: any inline string they carry still occupies a slot in the payload's streaming
247/// string table, so it has to be harvested. Use
248/// [`value::harvest_unknown_field`](super::value::harvest_unknown_field) for that; it falls back to
249/// this function for scalars, which cannot contain strings.
250///
251/// # Errors
252///
253/// Returns an error on truncated input, a reserved marker, or nesting deeper than
254/// [`MAX_SKIP_DEPTH`].
255pub fn skip_value(r: &mut &[u8], context: &'static str) -> Result<(), DecodeError> {
256    skip_value_depth(r, context, 0)
257}
258
259fn skip_value_depth(r: &mut &[u8], context: &'static str, depth: usize) -> Result<(), DecodeError> {
260    use rmp::Marker::*;
261
262    if depth > MAX_SKIP_DEPTH {
263        return Err(DecodeError::DepthExceeded { max: MAX_SKIP_DEPTH });
264    }
265
266    let marker = rmp::decode::read_marker(r).map_err(|e| DecodeError::Msgpack {
267        context,
268        detail: e.0.to_string(),
269    })?;
270
271    match marker {
272        Null | True | False | FixPos(_) | FixNeg(_) => Ok(()),
273        U8 | I8 => advance(r, 1, context),
274        U16 | I16 => advance(r, 2, context),
275        U32 | I32 | F32 => advance(r, 4, context),
276        U64 | I64 | F64 => advance(r, 8, context),
277        FixStr(n) => advance(r, n as usize, context),
278        Str8 | Bin8 => {
279            let n = read_len(r, 1, context)?;
280            advance(r, n, context)
281        }
282        Str16 | Bin16 => {
283            let n = read_len(r, 2, context)?;
284            advance(r, n, context)
285        }
286        Str32 | Bin32 => {
287            let n = read_len(r, 4, context)?;
288            advance(r, n, context)
289        }
290        FixArray(n) => skip_seq(r, n as usize, context, depth),
291        Array16 => {
292            let n = read_len(r, 2, context)?;
293            skip_seq(r, n, context, depth)
294        }
295        Array32 => {
296            let n = read_len(r, 4, context)?;
297            skip_seq(r, n, context, depth)
298        }
299        FixMap(n) => skip_seq(r, n as usize * 2, context, depth),
300        Map16 => {
301            let n = read_len(r, 2, context)?;
302            skip_seq(r, n * 2, context, depth)
303        }
304        Map32 => {
305            let n = read_len(r, 4, context)?;
306            skip_seq(r, n * 2, context, depth)
307        }
308        // Extension types: a type byte plus a fixed or length-prefixed data payload.
309        FixExt1 => advance(r, 1 + 1, context),
310        FixExt2 => advance(r, 1 + 2, context),
311        FixExt4 => advance(r, 1 + 4, context),
312        FixExt8 => advance(r, 1 + 8, context),
313        FixExt16 => advance(r, 1 + 16, context),
314        Ext8 => {
315            let n = read_len(r, 1, context)?;
316            advance(r, 1 + n, context)
317        }
318        Ext16 => {
319            let n = read_len(r, 2, context)?;
320            advance(r, 1 + n, context)
321        }
322        Ext32 => {
323            let n = read_len(r, 4, context)?;
324            advance(r, 1 + n, context)
325        }
326        Reserved => Err(DecodeError::Msgpack {
327            context,
328            detail: "encountered reserved marker 0xc1".to_string(),
329        }),
330    }
331}

333/// Skips `count` consecutive values (array elements or flattened map key/value slots).
334fn skip_seq(r: &mut &[u8], count: usize, context: &'static str, depth: usize) -> Result<(), DecodeError> {
335    for _ in 0..count {
336        skip_value_depth(r, context, depth + 1)?;
337    }
338    Ok(())
