49/// Configuration for the Datadog APM Stats encoder.
50pub struct DatadogApmStatsEncoderConfiguration {
51    flush_timeout: Duration,
52    agent_hostname: Option<String>,
53    agent_version: String,
54    agent_env: AgentEnv,
55}
56
57impl DatadogApmStatsEncoderConfiguration {
58    /// Creates a new `DatadogApmStatsEncoderConfiguration`.
59    ///
60    /// The encoder waits up to `flush_timeout` before flushing a partially filled payload; a zero duration flushes
61    /// almost immediately. Each payload carries `agent_env` as its agent environment.
62    pub fn new(flush_timeout: Duration, agent_env: AgentEnv) -> Self {
63        let app_details = saluki_metadata::get_app_details();
64        Self {
65            flush_timeout,
66            agent_hostname: None,
67            agent_version: format!("agent-data-plane/{}", app_details.version().raw()),
68            agent_env,
69        }
70    }
71
72    /// Sets the agent hostname using the environment provider.
73    pub async fn with_environment_provider<E>(mut self, environment_provider: E) -> Result<Self, GenericError>
74    where
75        E: EnvironmentProvider<Host = BoxedHostProvider>,
76    {
77        let host_provider = environment_provider.host();
78        let hostname = host_provider.get_hostname().await?;
79        self.agent_hostname = Some(hostname);
80        Ok(self)
81    }
82
83    fn endpoint_encoder(&self) -> StatsEndpointEncoder {
84        StatsEndpointEncoder::new(
85            MetaString::from(self.agent_hostname.clone().unwrap_or_default()),
86            MetaString::from(self.agent_version.clone()),
87            self.agent_env.clone().into(),
88        )
89    }
90}

92#[async_trait]
93impl EncoderBuilder for DatadogApmStatsEncoderConfiguration {
94    fn input_event_type(&self) -> EventType {
95        EventType::TraceStats
96    }
97
98    fn output_payload_type(&self) -> PayloadType {
99        PayloadType::Http
100    }
101
102    async fn build(&self, context: BuildContext) -> Result<Box<dyn Encoder + Send>, GenericError> {
103        let metrics_builder = MetricsBuilder::from_component_context(context.component_context());
104        let telemetry = ComponentTelemetry::from_builder(&metrics_builder);
105        let payload_telemetry = StatsPayloadTelemetry::from_builder(&metrics_builder);
106        let compression_scheme = CompressionScheme::gzip_default();
107
108        let mut stats_rb =
109            RequestBuilder::new(self.endpoint_encoder(), compression_scheme, RB_BUFFER_CHUNK_SIZE).await?;
110        stats_rb.with_max_inputs_per_payload(MAX_STATS_PER_PAYLOAD);
111
112        let flush_timeout = if self.flush_timeout.is_zero() {
113            // We always give ourselves a minimum flush timeout of 10ms to allow for some very minimal amount of
114            // batching, while still practically flushing things almost immediately.
115            Duration::from_millis(10)
116        } else {
117            self.flush_timeout
118        };
119
120        Ok(Box::new(DatadogStats {
121            stats_rb,
122            telemetry,
123            payload_telemetry,
124            flush_timeout,
125        }))
126    }
127}

150/// Telemetry for what the stats encoder writes into payloads.
151struct StatsPayloadTelemetry {
152    client_payloads: Counter,
153    buckets: Counter,
154    entries: Counter,
155    bytes_uncompressed: Counter,
156}
157
158impl StatsPayloadTelemetry {
159    fn from_builder(builder: &MetricsBuilder) -> Self {
160        Self {
161            client_payloads: builder.register_counter("component_stats_client_payloads_encoded_total"),
162            buckets: builder.register_counter("component_stats_buckets_encoded_total"),
163            entries: builder.register_counter("component_stats_entries_encoded_total"),
164            bytes_uncompressed: builder.register_counter("component_bytes_uncompressed_total"),
165        }
166    }
167
168    /// Tracks stats written into a payload.
169    fn track_stats(&self, counts: StatsCounts) {
170        self.client_payloads.increment(counts.client_payloads);
171        self.buckets.increment(counts.buckets);
172        self.entries.increment(counts.entries);
173    }
174
175    /// Tracks a flushed payload, given its size before compression.
176    fn track_payload(&self, uncompressed_len: usize) {
177        self.bytes_uncompressed.increment(uncompressed_len as u64);
178    }
179}

181/// Client payloads, buckets, and grouped entries that stats add to a payload.
182#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
183struct StatsCounts {
184    client_payloads: u64,
185    buckets: u64,
186    entries: u64,
187}
188
189impl StatsCounts {
190    /// Counts the client payloads, buckets, and grouped entries of stats.
191    fn from_trace_stats(stats: &TraceStats) -> Self {
192        let client_payloads = stats.stats();
193        let buckets = client_payloads.iter().flat_map(|client_payload| client_payload.stats());
194        Self {
195            client_payloads: client_payloads.len() as u64,
196            buckets: buckets.clone().count() as u64,
197            entries: buckets.map(|bucket| bucket.stats().len() as u64).sum(),
198        }
199    }
200}

202#[async_trait]
203impl Encoder for DatadogStats {
204    async fn run(mut self: Box<Self>, mut context: EncoderContext) -> Result<(), GenericError> {
205        let Self {
206            stats_rb,
207            telemetry,
208            payload_telemetry,
209            flush_timeout,
210        } = *self;
211
212        let mut health = context.take_health_handle();
213
214        let (events_tx, events_rx) = mpsc::channel(8);
215        let (payloads_tx, mut payloads_rx) = mpsc::channel(8);
216
217        // Run our request builder task on the worker pool.
218        //
219        // The request builder task ignores the shutdown signal on purpose: it drains its incoming event buffer channel
220        // until the channel closes, which is what guarantees every buffered metric is encoded and dispatched.
221        let request_builder_fut = run_request_builder(
222            stats_rb,
223            telemetry,
224            payload_telemetry,
225            events_rx,
226            payloads_tx,
227            flush_timeout,
228        );
229        runtime::worker("request_builder", request_builder_fut)
230            .on_runtime(context.topology_context().global_thread_pool().clone())
231            .needed()
232            .spawn();
233
234        health.mark_ready();
235        debug!("Datadog APM Stats encoder started.");
236
237        loop {
238            select! {
239                biased;
240
241                _ = health.live() => continue,
242                maybe_payload = payloads_rx.recv() => match maybe_payload {
243                    Some(payload) => {
244                        if let Err(e) = context.dispatcher().dispatch(payload).await {
245                            error!("Failed to dispatch payload: {}", e);
246                        }
247                    }
248                    None => break,
249                },
250                maybe_event_buffer = context.events().next() => match maybe_event_buffer {
251                    Some(event_buffer) => events_tx.send(event_buffer).await
252                        .error_context("Failed to send event buffer to request builder task.")?,
253                    None => break,
254                },
255            }
256        }
257
258        drop(events_tx);
259
260        // Continue draining the payloads receiver until it is closed.
261        while let Some(payload) = payloads_rx.recv().await {
262            if let Err(e) = context.dispatcher().dispatch(payload).await {
263                error!("Failed to dispatch payload: {}", e);
264            }
265        }
266
267        // Draining `payloads_rx` to completion already implies the request builder finished: it owns the only sender,
268        // so the channel only closes once that child's future has run to completion (or been dropped).
269        debug!("Datadog APM Stats encoder stopped.");
270
271        Ok(())
272    }
273}

275async fn run_request_builder(
276    mut stats_request_builder: RequestBuilder<StatsEndpointEncoder>, telemetry: ComponentTelemetry,
277    payload_telemetry: StatsPayloadTelemetry, mut events_rx: Receiver<EventsBuffer>,
278    payloads_tx: Sender<PayloadsBuffer>, flush_timeout: std::time::Duration,
279) -> Result<(), GenericError> {
280    let mut pending_flush = false;
281    let pending_flush_timeout = sleep(flush_timeout);
282    pin!(pending_flush_timeout);
283
284    loop {
285        select! {
286            Some(event_buffer) = events_rx.recv() => {
287                for trace_stats in event_buffer.into_events::<TraceStats>() {
288
289                    // Encode the stats. If we get it back, that means the current request is full, and we need to
290                    // flush it before we can try to encode the stats again.
291                    let stats_counts = StatsCounts::from_trace_stats(&trace_stats);
292                    let stats_to_retry = match stats_request_builder.encode(trace_stats).await {
293                        Ok(None) => {
294                            payload_telemetry.track_stats(stats_counts);
295                            continue;
296                        }
297                        Ok(Some(stats)) => stats,
298                        Err(e) => {
299                            error!(error = %e, "Failed to encode stats.");
300                            telemetry.events_dropped_encoder().increment(1);
301                            continue;
302                        }
303                    };
304
305                    let maybe_requests = stats_request_builder.flush().await;
306                    if maybe_requests.is_empty() {
307                        panic!("builder told us to flush, but gave us nothing");
308                    }
309
310                    for maybe_request in maybe_requests {
311                        match maybe_request {
312                            Ok((events, _data_points, uncompressed_len, request)) => {
313                                payload_telemetry.track_payload(uncompressed_len);
314                                let payload_meta = PayloadMetadata::from_event_count(events);
315                                let http_payload = HttpPayload::new(payload_meta, request);
316                                let payload = Payload::Http(http_payload);
317
318                                payloads_tx.send(payload).await
319                                    .map_err(|_| generic_error!("Failed to send payload to encoder."))?;
320                            },
321                            Err(e) => if e.is_recoverable() {
322                                continue;
323                            } else {
324                                return Err(GenericError::from(e).context("Failed to flush request."));
325                            }
326                        }
327                    }
328
329                    match stats_request_builder.encode(stats_to_retry).await {
330                        Ok(None) => payload_telemetry.track_stats(stats_counts),
331                        Ok(Some(_)) => {
332                            error!("Stats did not fit in a request after flushing; dropping them.");
333                            telemetry.events_dropped_encoder().increment(1);
334                        }
335                        Err(e) => {
336                            error!(error = %e, "Failed to encode stats.");
337                            telemetry.events_dropped_encoder().increment(1);
338                        }
339                    }
340                }
341
342                debug!("Processed event buffer.");
343
344                // If we're not already pending a flush, we'll start the countdown.
345                if !pending_flush {
346                    pending_flush_timeout.as_mut().reset(tokio::time::Instant::now() + flush_timeout);
347                    pending_flush = true;
348                }
349            },
350            _ = &mut pending_flush_timeout, if pending_flush => {
351                debug!("Flushing pending request(s).");
352
353                pending_flush = false;
354
355                let maybe_stats_requests = stats_request_builder.flush().await;
356                for maybe_request in maybe_stats_requests {
357                    match maybe_request {
358                        Ok((events, _data_points, uncompressed_len, request)) => {
359                            payload_telemetry.track_payload(uncompressed_len);
360                            let payload_meta = PayloadMetadata::from_event_count(events);
361                            let http_payload = HttpPayload::new(payload_meta, request);
362                            let payload = Payload::Http(http_payload);
363
364                            payloads_tx.send(payload).await
365                                .map_err(|_| generic_error!("Failed to send payload to encoder."))?;
366                        },
367                        Err(e) => if e.is_recoverable() {
368                            continue;
369                        } else {
370                            return Err(GenericError::from(e).context("Failed to flush request."));
371                        }
372                    }
373                }
374
375                debug!("All flushed requests sent to I/O task. Waiting for next event buffer...");
376            },
377
378            else => break,
379        }
380    }
381
382    Ok(())
383}

386struct StatsEndpointEncoder {

388    agent_version: MetaString,
389    agent_env: MetaString,
390}

392impl StatsEndpointEncoder {
393    fn new(agent_hostname: MetaString, agent_version: MetaString, agent_env: MetaString) -> Self {
394        Self {
395            agent_hostname,
396            agent_version,
397            agent_env,
398        }
399    }
400
401    fn to_proto_stats_payload(&self, stats: &TraceStats) -> ProtoStatsPayload {
402        let mut payload = ProtoStatsPayload::new();
403        payload.set_agentHostname(self.agent_hostname.to_string());
404        payload.set_agentEnv(self.agent_env.to_string());
405        payload.set_agentVersion(self.agent_version.to_string());
406        payload.set_clientComputed(false);
407        payload.set_splitPayload(stats.is_split_payload());
408        payload.set_stats(stats.stats().iter().map(convert_client_stats_payload).collect());
409
410        payload
411    }

472/// Error type for stats encoding.

476impl std::fmt::Display for StatsEncodeError {
477    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
478        write!(f, "failed to encode stats as MessagePack: {}", self.0)
479    }
480}

482impl std::error::Error for StatsEncodeError {
483    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
484        Some(&self.0)
485    }

634        let (events_tx, events_rx) = mpsc::channel(1);
635        let (payloads_tx, mut payloads_rx) = mpsc::channel(8);
636
637        let mut events = EventsBuffer::default();
638        for stats in all_stats {
639            assert!(events.try_push(Event::TraceStats(stats)).is_none());
640        }
641        events_tx.send(events).await.expect("events should be sent");
642        drop(events_tx);
643
644        run_request_builder(
645            request_builder,
646            ComponentTelemetry::from_builder(&metrics_builder),
647            StatsPayloadTelemetry::from_builder(&metrics_builder),
648            events_rx,
649            payloads_tx,
650            Duration::from_millis(1),
651        )
652        .await
653        .expect("request builder should run to completion");
654
655        let mut payloads = 0;
656        while payloads_rx.recv().await.is_some() {
657            payloads += 1;
658        }
659        assert_eq!(payloads, 1);
660
661        assert_eq!(
662            recorder.counter("component_stats_client_payloads_encoded_total"),
663            Some(3)
664        );
665        assert_eq!(recorder.counter("component_stats_buckets_encoded_total"), Some(4));
666        assert_eq!(recorder.counter("component_stats_entries_encoded_total"), Some(10));
667        assert_eq!(
668            recorder.counter("component_bytes_uncompressed_total"),
669            Some(expected_uncompressed_len)
670        );
