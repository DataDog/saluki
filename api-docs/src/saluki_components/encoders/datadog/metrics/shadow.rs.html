19pub(super) struct SketchShadowState<'a> {
20    config: &'a SketchShadowConfig,
21    sampling_decided: bool,
22    sampled_destinations: Option<Arc<[SketchShadowDestination]>>,
23    metrics: Vec<Metric>,
24    ratio: V3CompressionRatio,
25}
26
27impl<'a> SketchShadowState<'a> {
28    pub fn new(config: &'a SketchShadowConfig) -> Self {
29        Self {
30            config,
31            sampling_decided: false,
32            sampled_destinations: None,
33            metrics: Vec::new(),
34            ratio: V3CompressionRatio::default(),
35        }
36    }
37
38    pub fn is_enabled(&self) -> bool {
39        !self.config.eligible_destinations.is_empty()
40    }
41
42    fn sample_batch(&mut self) {
43        self.sample_batch_with_rng(|| rand::rng().random::<f64>());
44    }
45
46    // Each V2 builder batch is a complete validation group with its own destination IDs.
47    // The validator waits for V2 and V3 independently, so their request counts can differ.
48    fn sample_batch_with_rng(&mut self, mut random: impl FnMut() -> f64) {
49        if self.sampling_decided {
50            return;
51        }
52        self.sampling_decided = true;
53        let mut sampled_destinations = Vec::new();
54        for endpoint in &self.config.eligible_destinations {
55            if random() < self.config.sample_rate {
56                let id = Uuid::now_v7();
57                let mut buffer = Uuid::encode_buffer();
58                sampled_destinations.push(SketchShadowDestination {
59                    configured_endpoint: Arc::clone(endpoint),
60                    batch_id: HeaderValue::from_str(id.hyphenated().encode_lower(&mut buffer)).unwrap(),
61                });
62            }
63        }
64        if !sampled_destinations.is_empty() {
65            self.sampled_destinations = Some(sampled_destinations.into());
66        }
67    }
68
69    pub async fn encode(
70        &mut self, builder: &mut RequestBuilder<MetricsEndpointEncoder>, metric: Metric, context: V3FlushContext<'_>,
71        tx: &mut Sender<Payload>,
72    ) -> Result<EncodeResult, GenericError> {
73        self.sample_batch();
74        let mut copy = self.sampled_destinations.as_ref().map(|_| metric.clone());
75        let mut flushed = false;
76        let result = match builder.encode(metric).await {
77            Ok(Some(metric)) => {
78                // The rejected metric starts the next V2 batch; it must not enter the previous shadow.
79                self.flush(builder, context, tx).await?;
80                flushed = true;
81                self.sample_batch();
82                copy = self.sampled_destinations.as_ref().map(|_| metric.clone());
83                builder.encode(metric).await
84            }
85            result => result,
86        };
87        match result {
88            Ok(None) => {
89                if let Some(metric) = copy {
90                    self.metrics.push(metric);
91                }
92                Ok(EncodeResult::new(true, flushed))
93            }
94            result => {
95                error!(?result, "Failed to encode sketch metric.");
96                context.telemetry.events_dropped_encoder().increment(1);
97                Ok(EncodeResult::new(false, flushed))
98            }
99        }
100    }
101
102    pub async fn flush(
103        &mut self, builder: &mut RequestBuilder<MetricsEndpointEncoder>, context: V3FlushContext<'_>,
104        tx: &mut Sender<Payload>,
105    ) -> Result<(), GenericError> {
106        let mut metrics = std::mem::take(&mut self.metrics);
107        let result = self.flush_batch(builder, context, tx, &metrics).await;
108        metrics.clear();
109        self.metrics = metrics;
110        result
111    }
112
113    async fn flush_batch(
114        &mut self, builder: &mut RequestBuilder<MetricsEndpointEncoder>, context: V3FlushContext<'_>,
115        tx: &mut Sender<Payload>, metrics: &[Metric],
116    ) -> Result<(), GenericError> {
117        let Some(sampled_destinations) = self.sampled_destinations.take() else {
118            // Allow a fresh sampling decision for the next batch.
119            self.sampling_decided = false;
120            return flush_v2_metrics(builder, tx, Some(MetricsPayloadInfo::v2_sketches()))
121                .await
122                .map(|_| ());
123        };
124        // Allow a fresh sampling decision for the next batch.
125        self.sampling_decided = false;
126        let mut requests = Vec::new();
127        let mut baseline_failed = false;
128        let mut fatal_error = None;
129        for request in builder.flush().await {
130            match request {
131                Ok(request) => requests.push(request),
132                Err(error) => {
133                    baseline_failed = true;
134                    warn!(%error, "Failed to flush sketch baseline; skipping its shadow.");
135                    if !error.is_recoverable() {
136                        fatal_error = Some(error);
137                    }
138                }
139            }
140        }
141        // Splitting can discard a V2 subrequest without returning an error.
142        baseline_failed |= requests.iter().map(|(events, _, _, _)| *events).sum::<usize>() != metrics.len();
143        let request_count = requests.len();
144        for (request_index, (events, points, _uncompressed_len, request)) in requests.into_iter().enumerate() {
145            enqueue_sketch_request(
146                request,
147                events,
148                points,
149                tx,
150                SketchShadowMetadata {
151                    sampled_destinations: Arc::clone(&sampled_destinations),
152                    request_index,
153                    request_count,
154                    is_shadow: false,
155                },
156            )
157            .await?;
158        }
159        if let Some(error) = fatal_error {
160            return Err(GenericError::from(error).context("Failed to flush request."));
161        }
162        if !baseline_failed && request_count != 0 {
163            // The baseline is already dispatched. An invalid shadow route or failed shadow encode cannot suppress it.
164            encode_v3_payload_requests(
165                &self.config.beta_route,
166                metrics,
167                V3FlushContext {
168                    is_shadow: true,
169                    ..context
170                },
171                "sketches",
172                &mut self.ratio,
173                tx,
174                Some(MetricsPayloadInfo::v3_sketches()),
175                Some(sampled_destinations),
176            )
177            .await?;
178        }
179        Ok(())
180    }
181}

183pub(super) async fn enqueue_sketch_request(
184    request: Request<FrozenChunkedBytesBuffer>, events: usize, points: usize, tx: &mut Sender<Payload>,
185    shadow: SketchShadowMetadata,
186) -> Result<(), GenericError> {
187    let info = if shadow.is_shadow {
188        MetricsPayloadInfo::v3_sketches()
189    } else {
190        MetricsPayloadInfo::v2_sketches()
191    };
192    let metadata = PayloadMetadata::from_event_and_data_point_count(events, points)
193        .with(info)
194        .with(shadow);
195    tx.send(Payload::Http(HttpPayload::new(metadata, request)))
196        .await
197        .error_context("Failed to send sketch payload.")
198}

225    #[test]
226    fn shadow_eligibility_matches_resolved_destinations_and_routing() {
227        // Core #54581: site-gated V2 destinations, with one decision across a resolver's API keys.
228        let mut config = configuration();
229        config
230            .additional_endpoints
231            .insert("https://other.datadoghq.com".into(), vec!["a".into(), "b".into()]);
232        config
233            .additional_endpoints
234            .insert("https://app.us3.datadoghq.com".into(), vec!["c".into()]);
235        let eligible = |config: &DatadogMetricsConfiguration, disabled_by_compressor| {
236            config
237                .build_sketch_shadow_config(disabled_by_compressor)
238                .unwrap()
239                .eligible_destinations
240        };
241        assert_eq!(eligible(&config, false).len(), 2);
242        assert!(eligible(&config, true).is_empty());
243        config.v3_api.sketches.endpoints.push(config.primary_endpoint.clone());
244        assert_eq!(
245            eligible(&config, false).as_slice(),
246            [Arc::<str>::from("https://other.datadoghq.com")]
247        );
248        let catalog =
249            Arc::new(RoutingTargetCatalog::new(&config.primary_endpoint, &config.additional_endpoints).unwrap());
250        let primary = [catalog.primary()].into_iter().collect();
251        assert!(eligible(&config.with_routing(catalog, primary), false).is_empty());
252        assert!(eligible(
253            &configuration().with_metrics_endpoint_override("https://app.datadoghq.com".into()),
254            false
255        )
256        .is_empty());
257        assert!(eligible(&configuration().with_v2_series_only(), false).is_empty());
258    }

261    fn sampling_is_per_destination_and_stable_until_flush() {
262        let config = SketchShadowConfig {
263            eligible_destinations: vec![Arc::from("first"), Arc::from("second")],
264            sample_rate: 0.5,
265            ..Default::default()
266        };
267        let mut state = SketchShadowState::new(&config);
268        let mut samples = [0.4, 0.5].into_iter();
269        state.sample_batch_with_rng(|| samples.next().unwrap());
270        assert_eq!(samples.next(), None);
271        let selected = state.sampled_destinations.as_ref().unwrap();
272        assert_eq!(selected.len(), 1);
273        assert_eq!(selected[0].configured_endpoint.as_ref(), "first");
274        let id = selected[0].batch_id.clone();
275        state.sample_batch_with_rng(|| panic!("must not resample an active batch"));
276        assert_eq!(state.sampled_destinations.as_ref().unwrap()[0].batch_id, id);
277    }

279    #[tokio::test]
280    async fn sketch_shadow_preserves_batches_across_v2_overflow_and_v3_splits() {
281        let mut config = configuration();
282        config.max_metrics_per_payload = 2;
283        config.v3_api.sketches.max_points_per_payload = 1;
284        let (control, mut output) = TestComponentDriver::encoder(config).await.unwrap();
285        let mut events = ["shadow.one", "shadow.two", "shadow.three"]
286            .into_iter()
287            .map(|name| Event::Metric(Metric::distribution(name, [(123, 1.0)])))
288            .collect::<Vec<_>>();
289        events.push(Event::Metric(Metric::counter("series.stays.v3", 1.0)));
290        control.send_events(events).await;
291        let (result, payloads) = tokio::join!(control.shutdown(), output.collect());
292        result.unwrap();
293        let mut batches = BTreeMap::<String, (usize, usize, Vec<usize>, usize)>::new();
294        let mut series = 0;
295        for payload in payloads {
296            let (metadata, request) = payload.try_into_http_payload().unwrap().into_parts();
297            if request.uri() == METRICS_SERIES_V3_PATH {
298                series += metadata.event_count();
299                assert!(metadata.get::<SketchShadowMetadata>().is_none());
300                continue;
301            }
302            let shadow = metadata
303                .get::<SketchShadowMetadata>()
304                .expect("both sketch versions are correlated");
305            assert_eq!(shadow.sampled_destinations.len(), 1);
306            let id = shadow.sampled_destinations[0].batch_id.to_str().unwrap().to_owned();
307            let batch = batches.entry(id).or_default();
308            if shadow.is_shadow {
309                assert_eq!(request.uri(), "/custom/sketch-shadow");
310                batch.1 += metadata.event_count();
311                batch.2.push(shadow.request_index);
312                batch.3 = shadow.request_count;
313            } else {
314                assert_eq!(request.uri(), METRICS_SKETCHES_PATH);
315                assert_eq!((shadow.request_index, shadow.request_count), (0, 1));
316                batch.0 += metadata.event_count();
317            }
318        }
319        assert_eq!(series, 1);
320        assert_eq!(batches.len(), 2, "overflow starts a fresh correlated batch");
321        assert_eq!(batches.values().map(|batch| batch.0).sum::<usize>(), 3);
322        for (baseline, shadow, sequences, request_count) in batches.values() {
323            assert_eq!(baseline, shadow, "the retried metric belongs only to the next batch");
324            assert_eq!(sequences, &(0..*request_count).collect::<Vec<_>>());
325            assert_eq!(
326                shadow, request_count,
327                "the V3 point limit splits each sketch separately"
328            );
329        }
330    }

332    #[tokio::test]
333    async fn incomplete_v2_baseline_skips_shadow() {
334        let config = configuration();
335        let shadow_config = config.build_sketch_shadow_config(false).unwrap();
336        let mut state = SketchShadowState::new(&shadow_config);
337        let endpoint_config = EndpointConfiguration::new(CompressionScheme::noop(), 10, usize::MAX, None);
338        let mut builder = v2::create_v2_request_builder(MetricsEndpoint::Sketches, &endpoint_config)
339            .await
340            .unwrap();
341        let telemetry = ComponentTelemetry::from_builder(&MetricsBuilder::default());
342        let serializer_telemetry = V3SerializerTelemetry::from_builder(&MetricsBuilder::default());
343        let context = V3FlushContext {
344            endpoint_config: &endpoint_config,
345            payload_limits: config.v3_sketch_payload_limits(),
346            series_endpoint_uri: METRICS_SERIES_V3_PATH,
347            serializer_telemetry: &serializer_telemetry,
348            telemetry: &telemetry,
349            is_shadow: false,
350        };
351        let (mut tx, mut rx) = mpsc::channel(8);
352        for name in ["a".repeat(256), "b".repeat(256), "small.one".into(), "small.two".into()] {
353            let metric = Metric::distribution(Context::from_parts(name, TagSet::default()), [(123, 1.0)]);
354            state.encode(&mut builder, metric, context, &mut tx).await.unwrap();
355        }
356        // Force a split: each large metric fits alone, but their half is silently discarded during re-encoding.
357        builder.with_len_limits(512, 512).unwrap();
358        state.flush(&mut builder, context, &mut tx).await.unwrap();
359        drop(tx);
360
361        let (metadata, request) = rx.recv().await.unwrap().try_into_http_payload().unwrap().into_parts();
362        assert_eq!(request.uri(), METRICS_SKETCHES_PATH);
363        assert_eq!((metadata.event_count(), metadata.data_point_count()), (2, 2));
364        assert!(
365            rx.recv().await.is_none(),
366            "an incomplete baseline must not emit a shadow"
367        );
368    }

370    #[tokio::test]
371    async fn disabled_or_failed_shadow_preserves_baseline() {
372        for (rate, route, max_points) in [
373            (0.0, "/custom/sketch-shadow", usize::MAX),
374            (1.0, "invalid route", usize::MAX),
375            // URI parsing accepts these, but endpoint rewriting requires an absolute path.
376            (1.0, "localhost:8080", usize::MAX),
377            (1.0, "*", usize::MAX),
378            // Dropping the oversized metric used to advertise the surviving V3 metric as a complete group.
379            (1.0, "/custom/sketch-shadow", 1),
380        ] {
381            let mut config = configuration();
382            config.v3_api.sketches.shadow_sample_rate = rate;
383            config.v3_api.sketches.beta_route = route.into();
384            config.v3_api.sketches.max_points_per_payload = max_points;
385            let (control, mut output) = TestComponentDriver::encoder(config).await.unwrap();
386            control
387                .send_events([
388                    Event::Metric(Metric::distribution("baseline.large", [(123, 1.0), (124, 2.0)])),
389                    Event::Metric(Metric::distribution("baseline.small", [(123, 1.0)])),
390                ])
391                .await;
392            let (result, mut payloads) = tokio::join!(control.shutdown(), output.collect());
393            result.unwrap();
394            assert_eq!(payloads.len(), 1);
395            let (metadata, request) = payloads.pop().unwrap().try_into_http_payload().unwrap().into_parts();
396            assert_eq!(request.uri(), METRICS_SKETCHES_PATH);
397            assert_eq!((metadata.event_count(), metadata.data_point_count()), (2, 3));
398            if rate == 0.0 {
399                assert!(metadata.get::<SketchShadowMetadata>().is_none());
400            }
401        }
402    }
