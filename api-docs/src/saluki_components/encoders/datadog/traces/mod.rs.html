60/// String interning table for the ETP (Efficient Trace Payload) tracer payload format.
61///
62/// Index 0 is always the empty string. All other strings are assigned indices in insertion order.
63#[derive(Debug)]
64struct StringTable {
65    indices: FastIndexSet<MetaString>,
66}
67
68impl Default for StringTable {
69    fn default() -> Self {
70        Self::new()
71    }
72}
73
74impl StringTable {
75    fn new() -> Self {
76        let mut t = Self {
77            indices: FastIndexSet::default(),
78        };
79        t.intern("");
80        t
81    }
82
83    fn clear(&mut self) {
84        self.indices.clear();
85        self.intern("");
86    }
87
88    /// Interns `s`, returning its index. If `s` was already interned, returns the existing index.
89    fn intern(&mut self, s: &str) -> u32 {
90        // We use get_index_of to check if the string is already interned to avoid reconstructing a new meta string if it's already interned.
91        if let Some(idx) = self.indices.get_index_of(s) {
92            idx as u32
93        } else {
94            let (idx, _) = self.indices.insert_full(MetaString::from(s));
95            idx as u32
96        }
97    }
98}

100/// Configuration for the Datadog Traces encoder.
101///
102/// This encoder converts trace events into Datadog's TracerPayload protobuf format and sends them
103/// to the Datadog traces intake endpoint (`/api/v0.2/traces`). It handles batching, compression,
104/// and enrichment with metadata such as hostname, environment, and container tags.
105pub struct DatadogTraceConfiguration {
106    /// Compression algorithm applied to outgoing payloads.
107    compressor_kind: String,
108
109    /// Effective zstd compression level, resolved by the configuration layer.
110    zstd_compressor_level: i32,
111
112    /// How long the encoder waits before flushing a partially filled payload.
113    ///
114    /// A zero duration is treated as "flush almost immediately" during `build`.
115    flush_timeout: Duration,
116
117    /// Global environment tag applied to emitted payloads.
118    env: String,
119
120    /// Target sampled traces per second, forwarded to the intake as `target_tps`.
121    target_traces_per_second: f64,
122
123    /// Target sampled error traces per second, forwarded to the intake as `error_tps`.
124    errors_per_second: f64,
125
126    /// Whether Error Tracking standalone mode is enabled.
127    error_tracking_standalone: bool,
128
129    /// Whether spans missing intake-required fields are ingested rather than rejected.
130    ignore_missing_datadog_fields: bool,
131
132    /// Percentage of OTLP traces the probabilistic sampler keeps.
133    sampling_percentage: f64,
134
135    /// Default hostname, resolved from the environment provider.
136    default_hostname: Option<String>,
137
138    /// ADP version string embedded in emitted payloads.
139    version: String,
140}

142impl DatadogTraceConfiguration {
143    /// Creates a new `DatadogTraceConfiguration` from the resolved traces and shared configuration.
144    ///
145    /// The OTLP trace settings live in their own domain, so they arrive as a separate slice rather
146    /// than through the traces domain.
147    pub fn from_configuration(
148        traces: &domains::traces::Domain, otlp_traces: &domains::otlp::Traces, shared: &SharedConfiguration,
149    ) -> Self {
150        let app_details = saluki_metadata::get_app_details();
151        let version = format!("agent-data-plane/{}", app_details.version().raw());
152
153        let compression = &shared.endpoints.compression;
154
155        // ADP defaults the global environment tag to `none` rather than the Core Agent's empty
156        // string, so normalize an empty resolved value back to `none`.
157        let env = if traces.env.is_empty() {
158            DEFAULT_TRACE_ENV.to_owned()
159        } else {
160            traces.env.clone()
161        };
162
163        Self {
164            compressor_kind: compression.compressor_kind.clone(),
165            zstd_compressor_level: compression.effective_zstd_level(),
166            flush_timeout: shared.metrics_encoding.flush_timeout,
167            env,
168            target_traces_per_second: traces.target_traces_per_second,
169            errors_per_second: traces.errors_per_second,
170            error_tracking_standalone: traces.error_tracking_standalone_enabled,
171            ignore_missing_datadog_fields: otlp_traces.ignore_missing_datadog_fields,
172            sampling_percentage: otlp_traces.probabilistic_sampler_sampling_percentage,
173            default_hostname: None,
174            version,
175        }
176    }

182    {

187    }

200    async fn build(&self, context: BuildContext) -> Result<Box<dyn Encoder + Send>, GenericError> {
201        let metrics_builder = MetricsBuilder::from_component_context(context.component_context());
202        let telemetry = ComponentTelemetry::from_builder(&metrics_builder);
203        let compression_scheme = CompressionScheme::new(&self.compressor_kind, self.zstd_compressor_level);
204
205        let default_hostname = self.default_hostname.clone().unwrap_or_default();
206        let default_hostname = MetaString::from(default_hostname);
207
208        // Create request builder for traces which is used to generate HTTP requests.
209
210        let mut trace_rb = RequestBuilder::new(
211            TraceEndpointEncoder::new(
212                default_hostname,
213                self.version.clone(),
214                self.env.clone(),
215                self.target_traces_per_second,
216                self.errors_per_second,
217                self.error_tracking_standalone,
218                self.ignore_missing_datadog_fields,
219                self.sampling_percentage,
220            ),
221            compression_scheme,
222            RB_BUFFER_CHUNK_SIZE,
223        )
224        .await?;
225        trace_rb.with_max_inputs_per_payload(MAX_TRACES_PER_PAYLOAD);
226
227        let flush_timeout = if self.flush_timeout.is_zero() {
228            // We always give ourselves a minimum flush timeout of 10ms to allow for some very minimal amount of
229            // batching, while still practically flushing things almost immediately.
230            Duration::from_millis(10)
231        } else {
232            self.flush_timeout
233        };
234
235        Ok(Box::new(DatadogTrace {
236            trace_rb,
237            telemetry,
238            flush_timeout,
239        }))
240    }

264// Encodes Trace events to TracerPayloads.
265#[async_trait]
266impl Encoder for DatadogTrace {
267    async fn run(mut self: Box<Self>, mut context: EncoderContext) -> Result<(), GenericError> {
268        let Self {
269            trace_rb,
270            telemetry,
271            flush_timeout,
272        } = *self;
273
274        let mut health = context.take_health_handle();
275
276        let (events_tx, events_rx) = mpsc::channel(8);
277        let (payloads_tx, mut payloads_rx) = mpsc::channel(8);
278
279        // Run our request builder task on the worker pool.
280        //
281        // The request builder task ignores the shutdown signal on purpose: it drains its incoming event buffer channel
282        // until the channel closes, which is what guarantees every buffered metric is encoded and dispatched.
283        let request_builder_fut = run_request_builder(trace_rb, telemetry, events_rx, payloads_tx, flush_timeout);
284        context
285            .spawner()
286            .noninterruptible("request_builder", |_shutdown| request_builder_fut)
287            .on_worker_pool()
288            .spawn()
289            .await
290            .error_context("Failed to spawn request builder task.")?;
291
292        health.mark_ready();
293        debug!("Datadog Trace encoder started.");
294
295        loop {
296            select! {
297                biased; // makes the branches of the select statement be evaluated in order.
298
299                _ = health.live() => continue,
300                maybe_payload = payloads_rx.recv() => match maybe_payload {
301                    Some(payload) => {
302                        // Dispatch an HTTP payload to the dispatcher.
303                        if let Err(e) = context.dispatcher().dispatch(payload).await {
304                            error!("Failed to dispatch payload: {}", e);
305                        }
306                    }
307                    None => break,
308                },
309                maybe_event_buffer = context.events().next() => match maybe_event_buffer {
310                    Some(event_buffer) => events_tx.send(event_buffer).await
311                        .error_context("Failed to send event buffer to request builder task.")?,
312                    None => break,
313                },
314            }
315        }
316
317        // Drop the events sender, which signals the request builder task to stop.
318        drop(events_tx);
319
320        // Continue draining the payloads receiver until it is closed.
321        while let Some(payload) = payloads_rx.recv().await {
322            if let Err(e) = context.dispatcher().dispatch(payload).await {
323                error!("Failed to dispatch payload: {}", e);
324            }
325        }
326
327        // Draining `payloads_rx` to completion already implies the request builder finished: it owns the only sender,
328        // so the channel only closes once that child's future has run to completion (or been dropped).
329        debug!("Datadog Trace encoder stopped.");
330
331        Ok(())
332    }
333}

335async fn run_request_builder(
336    mut trace_request_builder: RequestBuilder<TraceEndpointEncoder>, telemetry: ComponentTelemetry,
337    mut events_rx: Receiver<EventsBuffer>, payloads_tx: Sender<PayloadsBuffer>, flush_timeout: std::time::Duration,
338) -> Result<(), GenericError> {
339    let mut pending_flush = false;
340    let pending_flush_timeout = sleep(flush_timeout);
341    pin!(pending_flush_timeout);
342
343    loop {
344        select! {
345            Some(event_buffer) = events_rx.recv() => {
346                for event in event_buffer {
347                    let trace = match event.try_into_trace() {
348                        Some(trace) => trace,
349                        None => continue,
350                    };
351                    // Encode the trace. If we get it back, that means the current request is full, and we need to
352                    // flush it before we can try to encode the trace again.
353                    let trace_to_retry = match trace_request_builder.encode(trace).await {
354                        Ok(None) => continue,
355                        Ok(Some(trace)) => trace,
356                        Err(e) => {
357                            error!(error = %e, "Failed to encode trace.");
358                            telemetry.events_dropped_encoder().increment(1);
359                            continue;
360                        }
361                    };
362
363                    let maybe_requests = trace_request_builder.flush().await;
364                    if maybe_requests.is_empty() {
365                        panic!("builder told us to flush, but gave us nothing");
366                    }
367
368                    for maybe_request in maybe_requests {
369                        match maybe_request {
370                            Ok((events, _data_points, request)) => {
371                                let payload_meta = PayloadMetadata::from_event_count(events);
372                                let http_payload = HttpPayload::new(payload_meta, request);
373                                let payload = Payload::Http(http_payload);
374
375                                payloads_tx.send(payload).await
376                                    .map_err(|_| generic_error!("Failed to send payload to encoder."))?;
377                            },
378                            Err(e) => if e.is_recoverable() {
379                                // If the error is recoverable, we'll hold on to the trace to retry it later.
380                                continue;
381                            } else {
382                                return Err(GenericError::from(e).context("Failed to flush request."));
383                            }
384                        }
385                    }
386
387                    // Now try to encode the trace again.
388                    if let Err(e) = trace_request_builder.encode(trace_to_retry).await {
389                        error!(error = %e, "Failed to encode trace.");
390                        telemetry.events_dropped_encoder().increment(1);
391                    }
392                }
393
394                debug!("Processed event buffer.");
395
396                // If we're not already pending a flush, we'll start the countdown.
397                if !pending_flush {
398                    pending_flush_timeout.as_mut().reset(tokio::time::Instant::now() + flush_timeout);
399                    pending_flush = true;
400                }
401            },
402            _ = &mut pending_flush_timeout, if pending_flush => {
403                debug!("Flushing pending request(s).");
404
405                pending_flush = false;
406
407                // Once we've encoded and written all traces, we flush the request builders to generate a request with
408                // anything left over. Again, we'll enqueue those requests to be sent immediately.
409                let maybe_trace_requests = trace_request_builder.flush().await;
410                for maybe_request in maybe_trace_requests {
411                    match maybe_request {
412                        Ok((events, _data_points, request)) => {
413                            let payload_meta = PayloadMetadata::from_event_count(events);
414                            let http_payload = HttpPayload::new(payload_meta, request);
415                            let payload = Payload::Http(http_payload);
416
417                            payloads_tx.send(payload).await
418                                .map_err(|_| generic_error!("Failed to send payload to encoder."))?;
419                        },
420                        Err(e) => if e.is_recoverable() {
421                            continue;
422                        } else {
423                            return Err(GenericError::from(e).context("Failed to flush request."));
424                        }
425                    }
426                }
427
428                debug!("All flushed requests sent to I/O task. Waiting for next event buffer...");
429            },
430
431            // Event buffers channel has been closed, and we have no pending flushing, so we're all done.
432            else => break,
433        }
434    }
435
436    Ok(())
437}

440struct TraceEndpointEncoder {
441    scratch: ScratchWriter<Vec<u8>>,
442    default_hostname: MetaString,
443    agent_hostname: String,
444    version: String,
445    env: String,
446    target_traces_per_second: f64,
447    errors_per_second: f64,
448    ignore_missing_datadog_fields: bool,
449    sampling_percentage: f64,
450    string_builder: StringBuilder,
451    string_table: StringTable,
452    error_tracking_standalone: bool,
453    extra_headers: Vec<(HeaderName, HeaderValue)>,
454}
455
456impl TraceEndpointEncoder {
457    fn new(
458        default_hostname: MetaString, version: String, env: String, target_traces_per_second: f64,
459        errors_per_second: f64, error_tracking_standalone: bool, ignore_missing_datadog_fields: bool,
460        sampling_percentage: f64,
461    ) -> Self {
462        let extra_headers = if error_tracking_standalone {
463            vec![(
464                HeaderName::from_static("x-datadog-error-tracking-standalone"),
465                HeaderValue::from_static("true"),
466            )]
467        } else {
468            Vec::new()
469        };
470        Self {
471            scratch: ScratchWriter::new(Vec::with_capacity(8192)),
472            agent_hostname: default_hostname.as_ref().to_string(),
473            default_hostname,
474            version,
475            env,
476            target_traces_per_second,
477            errors_per_second,
478            ignore_missing_datadog_fields,
479            sampling_percentage,
480            string_builder: StringBuilder::new(),
481            string_table: StringTable::new(),
482            error_tracking_standalone,
483            extra_headers,
484        }
485    }
486
487    fn encode_tracer_payload(&mut self, trace: &Trace, output_buffer: &mut Vec<u8>) -> std::io::Result<()> {
488        let sampling_rate = self.sampling_rate();
489        let source = attributes_to_source(&trace.attributes);
490
491        // Resolve computed metadata strings (may produce strings not directly present on trace fields).
492        let tracer_version = format!("otlp-{}", &trace.payload.tracer_version);
493        let container_tags =
494            resolve_container_tags_from_attrs(&trace.attributes, source.as_ref(), self.ignore_missing_datadog_fields);
495        let env_str: Option<&str> = if !trace.payload.env.is_empty() {
496            Some(&trace.payload.env)
497        } else if self.ignore_missing_datadog_fields {
498            Some("")
499        } else {
500            None
501        };
502        let hostname_str: Option<&str> = resolve_hostname_from_payload(
503            &trace.payload.hostname,
504            source.as_ref(),
505            Some(self.default_hostname.as_ref()),
506            self.ignore_missing_datadog_fields,
507        );
508        let decision_maker = trace.decision_maker.as_deref();
509        let priority = trace.priority.unwrap_or(DEFAULT_CHUNK_PRIORITY);
510        let dropped_trace = trace.dropped_trace;
511        let otlp_sr = trace.otlp_sampling_rate.unwrap_or(sampling_rate);
512        self.string_builder.clear();
513        write!(&mut self.string_builder, "{:.2}", otlp_sr).expect("should never fail to format sampling rate");
514
515        // Build 128-bit big-endian trace ID bytes for the chunk.
516        let mut trace_id_bytes = [0u8; 16];
517        trace_id_bytes[..8].copy_from_slice(&trace.trace_id_high.to_be_bytes());
518        trace_id_bytes[8..].copy_from_slice(&trace.trace_id_low.to_be_bytes());
519
520        // Reset the string table; strings are interned on the fly during encoding below.
521        self.string_table.clear();
522
523        let mut ap_builder = AgentPayloadBuilder::new(&mut self.scratch);
524
525        ap_builder
526            .host_name(&self.agent_hostname)?
527            .env(&self.env)?
528            .agent_version(&self.version)?
529            .target_tps(self.target_traces_per_second)?
530            .error_tps(self.errors_per_second)?;
531
532        ap_builder.add_idx_tracer_payloads(|tp| {
533            // Tracer payload metadata refs (skip default/empty values).
534            // Strings are interned on the fly; the string table is written at the end of this
535            // closure so it is complete by the time tp.strings() is called.
536            if !trace.payload.container_id.is_empty() {
537                tp.container_id_ref(self.string_table.intern(&trace.payload.container_id))?;
538            }
539            if !trace.payload.language_name.is_empty() {
540                tp.language_name_ref(self.string_table.intern(&trace.payload.language_name))?;
541            }
542            if !trace.payload.language_version.is_empty() {
543                tp.language_version_ref(self.string_table.intern(&trace.payload.language_version))?;
544            }
545            tp.tracer_version_ref(self.string_table.intern(tracer_version.as_str()))?;
546            if !trace.payload.runtime_id.is_empty() {
547                tp.runtime_id_ref(self.string_table.intern(&trace.payload.runtime_id))?;
548            }
549            if let Some(e) = env_str {
550                tp.env_ref(self.string_table.intern(e))?;
551            }
552            if let Some(h) = hostname_str {
553                tp.hostname_ref(self.string_table.intern(h))?;
554            }
555            if !trace.payload.app_version.is_empty() {
556                tp.app_version_ref(self.string_table.intern(&trace.payload.app_version))?;
557            }
558
559            // Container tags go in the payload-level attributes map.
560            if let Some(ct) = &container_tags {
561                let k_ref = self.string_table.intern(CONTAINER_TAGS_META_KEY);
562                let v_ref = self.string_table.intern(ct);
563                tp.attributes().write_entry(k_ref, |av: &mut _| {
564                    av.value(|vo| vo.string_value_ref(v_ref)).map(|_| ())
565                })?;
566            }
567
568            // Single TraceChunk containing all spans.
569            tp.add_chunks(|chunk| {
570                chunk.priority(priority)?;
571
572                if !trace.origin.is_empty() {
573                    chunk.origin_ref(self.string_table.intern(&trace.origin))?;
574                }
575
576                // Write 128-bit trace ID.
577                chunk.trace_id(&trace_id_bytes)?;
578
579                // Sampling mechanism (only write when non-zero).
580                if trace.sampling_mechanism != 0 {
581                    chunk.sampling_mechanism(trace.sampling_mechanism)?;
582                }
583
584                // Spans.
585                for span in trace.spans() {
586                    let service_ref = self.string_table.intern(span.service());
587                    let name_ref = self.string_table.intern(span.name());
588                    let resource_ref = self.string_table.intern(span.resource());
589                    let type_ref = self.string_table.intern(span.span_type());
590                    let env_ref = (!span.env.is_empty()).then(|| self.string_table.intern(&span.env));
591                    let version_ref = (!span.version.is_empty()).then(|| self.string_table.intern(&span.version));
592                    let component_ref = (!span.component.is_empty()).then(|| self.string_table.intern(&span.component));
593
594                    chunk.add_spans(|s| {
595                        s.service_ref(service_ref)?
596                            .name_ref(name_ref)?
597                            .resource_ref(resource_ref)?
598                            .span_id(span.span_id())?
599                            .parent_id(span.parent_id())?
600                            .start(span.start())?
601                            .duration(span.duration())?
602                            .error(span.error() != 0)?;
603
604                        // Unified attribute map (replaces separate meta/metrics/meta_struct).
605                        {
606                            let mut attrs = s.attributes();
607                            for (k, v) in &span.attributes {
608                                let k_ref = self.string_table.intern(k);
609                                attrs.write_entry(k_ref, |av: &mut _| {
610                                    encode_etp_attribute_value(av, v, &mut self.string_table)
611                                })?;
612                            }
613                        }
614
615                        s.type_ref(type_ref)?;
616
617                        if let Some(er) = env_ref {
618                            s.env_ref(er)?;
619                        }
620                        if let Some(vr) = version_ref {
621                            s.version_ref(vr)?;
622                        }
623                        if let Some(cr) = component_ref {
624                            s.component_ref(cr)?;
625                        }
626                        if span.kind != 0 {
627                            s.kind(SpanKind::from(span.kind as i32))?;
628                        }
629
630                        // Span links.
631                        for link in span.span_links() {
632                            let mut link_trace_id_bytes = [0u8; 16];
633                            link_trace_id_bytes[..8].copy_from_slice(&link.trace_id_high().to_be_bytes());
634                            link_trace_id_bytes[8..].copy_from_slice(&link.trace_id().to_be_bytes());
635                            let tracestate_ref = self.string_table.intern(link.tracestate());
636
637                            s.add_links(|sl| {
638                                sl.trace_id(&link_trace_id_bytes)?.span_id(link.span_id())?;
639                                {
640                                    let mut lattrs = sl.attributes();
641                                    for (k, v) in link.attributes() {
642                                        let k_ref = self.string_table.intern(k);
643                                        lattrs.write_entry(k_ref, |av: &mut _| {
644                                            encode_etp_attribute_value(av, v, &mut self.string_table)
645                                        })?;
646                                    }
647                                }
648                                sl.tracestate_ref(tracestate_ref)?.flags(link.flags())?;
649                                Ok(())
650                            })?;
651                        }
652
653                        // Span events.
654                        for event in span.span_events() {
655                            let name_ref = self.string_table.intern(event.name());
656                            s.add_events(|se| {
657                                se.time(event.time_unix_nano())?.name_ref(name_ref)?;
658                                {
659                                    let mut eattrs = se.attributes();
660                                    for (k, v) in event.attributes() {
661                                        let k_ref = self.string_table.intern(k);
662                                        eattrs.write_entry(k_ref, |av: &mut _| {
663                                            encode_etp_attribute_value(av, v, &mut self.string_table)
664                                        })?;
665                                    }
666                                }
667                                Ok(())
668                            })?;
669                        }
670
671                        Ok(())
672                    })?;
673                }
674
675                // Chunk attributes: decision maker, ETS tag, OTLP sampling rate.
676                {
677                    let mut cattrs = chunk.attributes();
678                    if let Some(dm) = decision_maker {
679                        let k_ref = self.string_table.intern(TAG_DECISION_MAKER);
680                        let v_ref = self.string_table.intern(dm);
681                        cattrs.write_entry(k_ref, |av: &mut _| {
682                            av.value(|vo| vo.string_value_ref(v_ref)).map(|_| ())
683                        })?;
684                    }
685                    if self.error_tracking_standalone {
686                        let trace_has_error = trace.spans().iter().any(|span| {
687                            span.error() != 0
688                                || span
689                                    .attributes
690                                    .get("_dd.span_events.has_exception")
691                                    .and_then(AttributeValue::as_string)
692                                    .is_some_and(|v| v == "true")
693                        });
694                        if trace_has_error {
695                            let k_ref = self.string_table.intern(TAG_ETS_STANDALONE_ERROR_KEY);
696                            let v_ref = self.string_table.intern(TAG_ETS_STANDALONE_ERROR_VALUE);
697                            cattrs.write_entry(k_ref, |av: &mut _| {
698                                av.value(|vo| vo.string_value_ref(v_ref)).map(|_| ())
699                            })?;
700                        }
701                    }
702                    {
703                        let k_ref = self.string_table.intern(TAG_OTLP_SAMPLING_RATE);
704                        let v_ref = self.string_table.intern(self.string_builder.as_str());
705                        cattrs.write_entry(k_ref, |av: &mut _| {
706                            av.value(|vo| vo.string_value_ref(v_ref)).map(|_| ())
707                        })?;
708                    }
709                }
710
711                if dropped_trace {
712                    chunk.dropped_trace(true)?;
713                }
714
715                Ok(())
716            })?;
717
718            // Write the string table after all refs so the table is complete.
719            // Protobuf allows fields in any order; decoders that do a full parse before
720            // resolving refs handle strings-after-refs correctly.
721            tp.strings(|sb| sb.add_many_mapped(&self.string_table.indices, |s| &**s))?;
722
723            Ok(())
724        })?;
725
726        ap_builder.finish(output_buffer)?;
727
728        Ok(())
729    }
730
731    fn sampling_rate(&self) -> f64 {
732        let rate = self.sampling_percentage / 100.0;
733        if rate <= 0.0 || rate >= 1.0 {
734            return 1.0;
735        }
736        rate
737    }

776/// Encodes an [`AttributeValue`] into an ETP-format `AnyValue` builder, interning any
777/// string values into `st` on the fly.
778fn encode_etp_attribute_value<S: ScratchBuffer>(
779    builder: &mut datadog_protos::traces::builders::idx::AnyValueBuilder<'_, S>, value: &AttributeValue,
780    st: &mut StringTable,
781) -> std::io::Result<()> {
782    builder
783        .value(|vo| match value {
784            AttributeValue::String(s) => vo.string_value_ref(st.intern(s)),
785            AttributeValue::Bool(b) => vo.bool_value(*b),
786            AttributeValue::Int(i) => vo.int_value(*i),
787            AttributeValue::Float(f) => vo.double_value(*f),
788            AttributeValue::Bytes(b) => vo.bytes_value(b),
789            AttributeValue::Array(values) => vo.array_value(|arr| {
790                for v in values {
791                    arr.add_values(|av| encode_etp_attribute_value(av, v, st))?;
792                }
793                Ok(())
794            }),
795            AttributeValue::KeyValueList(kvs) => vo.key_value_list(|kvl| {
796                for (k, v) in kvs {
797                    kvl.add_key_values(|kv| {
798                        kv.key(st.intern(k))?
799                            .value(|av| encode_etp_attribute_value(av, v, st))?;
800                        Ok(())
801                    })?;
802                }
803                Ok(())
804            }),
805        })
806        .map(|_| ())
807}

809fn resolve_hostname_from_payload<'a>(
810    payload_hostname: &'a str, source: Option<&'a OtlpSource>, default_hostname: Option<&'a str>,
811    ignore_missing_fields: bool,
812) -> Option<&'a str> {
813    if !payload_hostname.is_empty() {
814        return Some(payload_hostname);
815    }
816    if ignore_missing_fields {
817        return Some("");
818    }
819    match source {
820        Some(src) => match src.kind {
821            OtlpSourceKind::HostnameKind => Some(src.identifier.as_str()),
822            _ => Some(""),
823        },
824        None => default_hostname,
825    }
826}

828fn resolve_container_tags_from_attrs(
829    attributes: &FastHashMap<MetaString, AttributeValue>, source: Option<&OtlpSource>, ignore_missing_fields: bool,
830) -> Option<MetaString> {
831    if let Some(AttributeValue::String(tags)) = attributes.get(KEY_DATADOG_CONTAINER_TAGS) {
832        if !tags.is_empty() {
833            return Some(tags.clone());
834        }
835    }
836
837    if ignore_missing_fields {
838        return None;
839    }
840    let mut container_tags = TagSet::default();
841    extract_container_tags_from_attributes_map(attributes, &mut container_tags);
842    let is_fargate_source = source.is_some_and(|src| src.kind == OtlpSourceKind::AwsEcsFargateKind);
843    if container_tags.is_empty() && !is_fargate_source {
844        return None;
845    }
846
847    let mut flattened = flatten_container_tag(container_tags);
848    if is_fargate_source {
849        if let Some(src) = source {
850            append_tags(&mut flattened, &src.tag());
851        }
852    }
853
854    if flattened.is_empty() {
855        None
856    } else {
857        Some(MetaString::from(flattened))
858    }
859}
860
861fn flatten_container_tag(tags: TagSet) -> String {
862    let mut flattened = String::new();
863    for tag in tags {
864        if !flattened.is_empty() {
865            flattened.push(',');
866        }
867        flattened.push_str(tag.as_str());
868    }
869    flattened
870}
871
872fn append_tags(target: &mut String, tags: &str) {
873    if tags.is_empty() {
874        return;
875    }
876    if !target.is_empty() {
877        target.push(',');
878    }
879    target.push_str(tags);
880}

895    // Decode helpers for assertions on the encoded ETP `AgentPayload`.
896    //
897    // The encoder emits the wire format via `piecemeal` builders; here we decode it back
898    // with the independently code-generated `rust-protobuf` types (`idx::TracerPayload`
899    // and friends). A full-message parse resolves protobuf field ordering for us, so the
900    // fact that the encoder writes string refs before the string table is a non-issue.
901    //
902    // The only ETP-specific work left is resolving `u32` string-table refs and unwrapping
903    // the `AnyValue` oneof, which the two helpers below handle.
904    // ---------------------------------------------------------------------------
905
906    /// Resolves a string-table ref against a tracer payload's string table.
907    fn resolve_ref(strings: &[String], string_ref: u32) -> &str {
908        strings.get(string_ref as usize).map(String::as_str).unwrap_or_default()
909    }
910
911    /// Resolves the string-valued entries of an ETP attribute map into a readable
912    /// `key -> value` map, keeping only entries whose `AnyValue` is a string ref and
913    /// whose key resolves to a non-empty string. Keys and values are resolved through
914    /// the payload string table.
915    fn string_attrs(attrs: &HashMap<u32, idx::AnyValue>, strings: &[String]) -> HashMap<String, String> {
916        let mut out = HashMap::new();
917        for (k_ref, value) in attrs {
918            if let Some(idx::any_value::Value::StringValueRef(v_ref)) = &value.value {
919                let key = resolve_ref(strings, *k_ref);
920                if !key.is_empty() {
921                    out.insert(key.to_string(), resolve_ref(strings, *v_ref).to_string());
922                }
923            }
924        }
925        out
926    }

928    /// Collects the resolved string-valued attributes of every chunk across all ETP
929    /// tracer payloads in an encoded `AgentPayload`, one map per chunk.
930    fn decode_etp_chunk_attributes(buf: &[u8]) -> Vec<HashMap<String, String>> {
931        let payload = AgentPayload::parse_from_bytes(buf).expect("AgentPayload should decode");
932        payload
933            .idxTracerPayloads
934            .iter()
935            .flat_map(|tp| {
936                let strings = tp.strings();
937                tp.chunks()
938                    .iter()
939                    .map(move |chunk| string_attrs(chunk.attributes(), strings))
940            })
941            .collect()
942    }

946        let payload = AgentPayload::parse_from_bytes(buf).expect("AgentPayload should decode");
947        payload
948            .idxTracerPayloads
949            .iter()
950            .map(|tp| resolve_ref(tp.strings(), tp.tracerVersionRef()).to_string())
951            .collect()
952    }
