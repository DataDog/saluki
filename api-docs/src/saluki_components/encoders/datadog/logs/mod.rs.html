28const MAX_LOGS_PER_PAYLOAD: usize = 1000;
29
30/// The most tags buffer capacity that the encoder keeps between logs.
31///
32/// The encoder keeps the capacity of the longest tags it has written, so one log with enormous tags would otherwise
33/// stay held for the encoder's lifetime. The limit sits well above the tags of an ordinary log: such a log keeps its
34/// reuse, and a log past it has the excess released before the next log is encoded.
35const TAGS_BUFFER_RETENTION_LIMIT_BYTES: usize = 64 * 1024;
36
37/// The envelope status of a log that has none.
38const DEFAULT_STATUS: &str = "info";
39
40static CONTENT_TYPE_JSON: HeaderValue = HeaderValue::from_static("application/json");

75#[async_trait]
76impl IncrementalEncoderBuilder for DatadogLogsConfiguration {
77    type Output = DatadogLogs;
78
79    fn input_event_type(&self) -> EventType {
80        EventType::Log
81    }
82
83    fn output_payload_type(&self) -> PayloadType {
84        PayloadType::Http
85    }
86
87    async fn build(&self, context: BuildContext) -> Result<Self::Output, GenericError> {
88        let metrics_builder = MetricsBuilder::from_component_context(context.component_context());
89        let telemetry = ComponentTelemetry::from_builder(&metrics_builder);
90        let compression_scheme = CompressionScheme::new(&self.compressor_kind, self.zstd_level);
91
92        let mut request_builder = RequestBuilder::new(
93            LogsEndpointEncoder::new(self.clock),
94            compression_scheme,
95            RB_BUFFER_CHUNK_SIZE,
96        )
97        .await?;
98        request_builder.with_max_inputs_per_payload(MAX_LOGS_PER_PAYLOAD);
99
100        Ok(DatadogLogs {
101            request_builder,
102            telemetry,
103        })
104    }
105}

124#[async_trait]
125impl IncrementalEncoder for DatadogLogs {
126    async fn process_event(&mut self, event: Event) -> Result<ProcessResult, GenericError> {
127        let log: Log = match event {
128            Event::Log(log) => log,
129            _ => return Ok(ProcessResult::Continue),
130        };
131        match self.request_builder.encode(log).await {
132            Ok(None) => Ok(ProcessResult::Continue),
133            Ok(Some(log)) => Ok(ProcessResult::FlushRequired(Event::Log(log))),
134            Err(e) => {
135                if e.is_recoverable() {
136                    warn!(error = %e, "Failed to encode Datadog log due to recoverable error. Continuing...");
137
138                    // TODO: Get the actual number of events dropped from the error itself.
139                    self.telemetry.events_dropped_encoder().increment(1);
140                    Ok(ProcessResult::Continue)
141                } else {
142                    Err(e).error_context("Failed to encode Datadog log due to unrecoverable error.")
143                }
144            }
145        }
146    }
147
148    async fn flush(&mut self, dispatcher: &PayloadsDispatcher) -> Result<usize, FlushError> {
149        let mut dispatched_events = 0;
150        let pending_events = self.request_builder.pending_inputs();
151
152        let maybe_requests = self.request_builder.flush().await;
153        for maybe_request in maybe_requests {
154            match maybe_request {
155                Ok((events, _data_points, request)) => {
156                    let payload_meta = PayloadMetadata::from_event_count(events);
157                    let http_payload = HttpPayload::new(payload_meta, request);
158                    let payload = Payload::Http(http_payload);
159                    dispatcher
160                        .dispatch(payload)
161                        .await
162                        .map_err(|source| FlushError::new(dispatched_events, source))?;
163                    dispatched_events += events;
164                }
165                Err(e) => error!(error = %e, "Failed to build Datadog logs payload. Continuing..."),
166            }
167        }
168
169        // Events that were queued but not dispatched were in a payload that couldn't be built, such as one that exceeded
170        // the size limit and couldn't be split. They're gone, so report them as dropped.
171        let dropped_events = pending_events.saturating_sub(dispatched_events);
172        if dropped_events > 0 {
173            self.telemetry.events_dropped_encoder().increment(dropped_events as u64);
174        }
175
176        Ok(dispatched_events)
177    }
178}

180#[derive(Debug)]
181struct LogsEndpointEncoder {
182    tags_deduplicator: ReusableDeduplicator<Tag>,
183    tags_buffer: String,
184    clock: fn() -> DateTime<Utc>,
185}
186
187impl LogsEndpointEncoder {
188    fn new(clock: fn() -> DateTime<Utc>) -> Self {
189        Self {
190            tags_deduplicator: ReusableDeduplicator::new(),
191            tags_buffer: String::new(),
192            clock,
193        }
194    }
195
196    /// Writes the envelope of a log.
197    ///
198    /// The envelope has the same fields, in the same order, as the logs intake's JSON envelope: the message, the
199    /// status, the time the log was encoded in milliseconds, the hostname, the service, the source, and the tags. Every
200    /// field is present, empty when the log has no value for it. The message is written as it is: a log's producer
201    /// renders whatever content the message should carry, such as a JSON object holding the log's properties, so the
202    /// log's additional properties aren't written.
203    fn write_envelope(&mut self, log: &Log, buffer: &mut Vec<u8>) -> Result<(), serde_json::Error> {
204        // A log without a status is sent as `info`.
205        let status = if log.status().is_empty() {
206            DEFAULT_STATUS
207        } else {
208            log.status()
209        };
210        let source = log.source().as_deref().unwrap_or_default();
211
212        buffer.extend_from_slice(b"{\"message\":");
213        serde_json::to_writer(&mut *buffer, log.message())?;
214        buffer.extend_from_slice(b",\"status\":");
215        serde_json::to_writer(&mut *buffer, status)?;
216        buffer.extend_from_slice(b",\"timestamp\":");
217        buffer.extend_from_slice(itoa::Buffer::new().format((self.clock)().timestamp_millis()).as_bytes());
218        buffer.extend_from_slice(b",\"hostname\":");
219        serde_json::to_writer(&mut *buffer, log.hostname())?;
220        buffer.extend_from_slice(b",\"service\":");
221        serde_json::to_writer(&mut *buffer, log.service())?;
222        buffer.extend_from_slice(b",\"ddsource\":");
223        serde_json::to_writer(&mut *buffer, source)?;
224
225        // ddtags: comma-separated, deduplicated
226        buffer.extend_from_slice(b",\"ddtags\":");
227        self.tags_buffer.clear();
228        for (index, tag) in self.tags_deduplicator.deduplicated(log.tags().into_iter()).enumerate() {
229            if index > 0 {
230                self.tags_buffer.push(',');
231            }
232            self.tags_buffer.push_str(tag.as_str());
233        }
234        let written = serde_json::to_writer(&mut *buffer, &self.tags_buffer);
235        if self.tags_buffer.capacity() > TAGS_BUFFER_RETENTION_LIMIT_BYTES {
236            self.tags_buffer.clear();
237            self.tags_buffer.shrink_to(TAGS_BUFFER_RETENTION_LIMIT_BYTES);
238        }
239        written?;
240        buffer.push(b'}');
241        Ok(())
242    }
243}

245impl EndpointEncoder for LogsEndpointEncoder {
246    type Input = Log;
247    type EncodeError = serde_json::Error;
248
249    fn encoder_name() -> &'static str {
250        "logs"
251    }
252
253    fn compressed_size_limit(&self) -> usize {
254        DEFAULT_INTAKE_COMPRESSED_SIZE_LIMIT
255    }
256
257    fn uncompressed_size_limit(&self) -> usize {
258        DEFAULT_INTAKE_UNCOMPRESSED_SIZE_LIMIT
259    }
260
261    fn get_payload_prefix(&self) -> Option<&'static [u8]> {
262        Some(b"[")
263    }
264
265    fn get_payload_suffix(&self) -> Option<&'static [u8]> {
266        Some(b"]")
267    }
268
269    fn get_input_separator(&self) -> Option<&'static [u8]> {
270        Some(b",")
271    }
272
273    fn encode(&mut self, input: &Self::Input, buffer: &mut Vec<u8>) -> Result<(), Self::EncodeError> {
274        self.write_envelope(input, buffer)
275    }

278        PathAndQuery::from_static("/api/v2/logs").into()
279    }
280

282        Method::POST
283    }
284
285    fn content_type(&self) -> HeaderValue {
286        CONTENT_TYPE_JSON.clone()
287    }

311    fn fixed_clock() -> DateTime<Utc> {

319        let mut buffer = Vec::new();

322            .expect("envelope should serialize");

327        tags.into_iter().map(Tag::from_static).collect()

330    #[test]
331    fn releases_the_excess_of_an_oversized_tags_buffer() {
332        let mut encoder = LogsEndpointEncoder::new(fixed_clock);
333        let mut buffer = Vec::new();
334
335        let ordinary = Log::new("m").with_tags(tag_set(["env:prod"]));
336        encoder
337            .write_envelope(&ordinary, &mut buffer)
338            .expect("envelope should serialize");
339        let ordinary_capacity = encoder.tags_buffer.capacity();
340        assert!(ordinary_capacity >= "env:prod".len());
341        assert!(ordinary_capacity <= TAGS_BUFFER_RETENTION_LIMIT_BYTES);
342
343        // An ordinary log keeps the buffer it grew.
344        encoder
345            .write_envelope(&ordinary, &mut buffer)
346            .expect("envelope should serialize");
347        assert_eq!(encoder.tags_buffer.capacity(), ordinary_capacity);
348
349        // An oversized log is written in full, then has the excess released.
350        let tag = format!("big:{}", "x".repeat(2 * TAGS_BUFFER_RETENTION_LIMIT_BYTES));
351        let oversized = Log::new("m").with_tags([Tag::from(tag.clone())].into_iter().collect::<TagSet>());
352        encoder
353            .write_envelope(&oversized, &mut buffer)
354            .expect("envelope should serialize");
355        assert!(String::from_utf8_lossy(&buffer).contains(&tag));
356        assert!(encoder.tags_buffer.capacity() <= TAGS_BUFFER_RETENTION_LIMIT_BYTES);
357    }
