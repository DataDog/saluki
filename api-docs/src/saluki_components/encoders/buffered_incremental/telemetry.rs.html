1use std::time::Duration;
2
3use metrics::{Counter, Gauge};
4use saluki_metrics::MetricsBuilder;
5
6/// Incremental encoder-specific telemetry.
7#[derive(Clone)]
8pub struct ComponentTelemetry {
9    events_dropped_encoder: Counter,
10    events_flushed: Counter,
11    last_flush_events: Gauge,
12    last_flush_duration: Gauge,
13}
14
15impl ComponentTelemetry {
16    /// Creates a new `ComponentTelemetry` instance with default tags derived from the given component context.
17    pub fn from_builder(builder: &MetricsBuilder) -> Self {
18        Self {
19            events_dropped_encoder: builder.register_counter_with_tags(
20                "component_events_dropped_total",
21                ["intentional:false", "drop_reason:encoder_failure"],
22            ),
23            events_flushed: builder.register_counter("encoder_flushed_events_total"),
24            last_flush_events: builder.register_gauge("encoder_last_flush_events"),
25            last_flush_duration: builder.register_gauge("encoder_last_flush_duration_nanoseconds"),
26        }
27    }
28
29    /// Returns a reference to the "events dropped (encoder)" counter.
30    pub fn events_dropped_encoder(&self) -> &Counter {
31        &self.events_dropped_encoder
32    }
33
34    /// Records a completed flush of the given number of events, which took the given duration.
35    pub fn record_flush(&self, events: u64, duration: Duration) {
36        self.events_flushed.increment(events);
37        self.last_flush_events.set(events as f64);
38        self.last_flush_duration.set(duration.as_nanos() as f64);
39    }
40}

42#[cfg(test)]
43mod tests {
44    use saluki_metrics::test::TestRecorder;
45
46    use super::*;
47
48    #[test]
49    fn record_flush_tracks_totals_and_last_flush() {
50        let recorder = TestRecorder::default();
51        let _recorder_guard = metrics::set_default_local_recorder(&recorder);
52        let telemetry = ComponentTelemetry::from_builder(&MetricsBuilder::default());
53
54        telemetry.record_flush(3, Duration::from_micros(5));
55        telemetry.record_flush(2, Duration::from_micros(7));
56
57        assert_eq!(recorder.counter("encoder_flushed_events_total"), Some(5));
58        assert_eq!(recorder.gauge("encoder_last_flush_events"), Some(2.0));
59        assert_eq!(recorder.gauge("encoder_last_flush_duration_nanoseconds"), Some(7_000.0));
60    }
