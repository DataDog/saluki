1//! Trace tag replacer transform.
2//!
3//! Applies regex-based find-and-replace rules to trace tags, span resources, and span-event
4//! attributes. Rules run in order, after obfuscation and truncation and before stats and
5//! sampling, so durable telemetry only ever sees rewritten values.
6

23/// The compiled form of one [`ReplaceRule`].
24#[derive(Clone, Debug)]
25struct CompiledRule {
26    target: RuleTarget,
27    re: Regex,
28    repl: String,
29}
30
31/// The set of fields a rule may rewrite, derived from the rule's `name`.
32#[derive(Clone, Debug, PartialEq)]
33enum RuleTarget {
34    All,
35    Resource,
36    Tag(String),
37}
38
39/// A regex-based trace tag replacement rule.
40#[derive(Clone, Debug)]
41pub struct ReplaceRule {
42    /// Tag key the rule targets: `"*"`, `"resource.name"`, or a literal tag key.
43    pub name: String,
44
45    /// Regular expression matched against each targeted value.
46    pub pattern: String,
47
48    /// Text spliced in place of each match; `$1`-style group references are supported.
49    pub repl: String,
50}
51
52/// Trace tag replacer configuration.
53pub struct TraceTagReplacerConfiguration {
54    rules: Vec<ReplaceRule>,
55}
56
57impl TraceTagReplacerConfiguration {
58    /// Creates a new `TraceTagReplacerConfiguration` from the given rules.
59    ///
60    /// Patterns compile at build time, so an invalid one fails startup naming the rule.
61    pub fn new(rules: Vec<ReplaceRule>) -> Self {
62        Self { rules }
63    }
64}
65
66#[async_trait]
67impl SynchronousTransformBuilder for TraceTagReplacerConfiguration {
68    async fn build(&self, _context: BuildContext) -> Result<Box<dyn SynchronousTransform + Send>, GenericError> {
69        let rules = compile_rules(&self.rules)?;
70        Ok(Box::new(TraceTagReplacer { rules }))
71    }
72}
73
74impl MemoryBounds for TraceTagReplacerConfiguration {
75    fn specify_bounds(&self, builder: &mut MemoryBoundsBuilder) {
76        builder
77            .minimum()
78            .with_single_value::<TraceTagReplacer>("component struct")
79            .with_single_value::<Vec<ReplaceRule>>("replacement rules");
80    }
81}

83/// The tag replacer transform that processes traces.
84pub struct TraceTagReplacer {
85    rules: Vec<CompiledRule>,
86}
87
88impl TraceTagReplacer {
89    fn replace_span(&self, span: &mut Span) {
90        for rule in &self.rules {
91            match &rule.target {
92                RuleTarget::All => self.replace_everywhere(span, &rule.re, &rule.repl),
93                RuleTarget::Resource => self.replace_resource(span, &rule.re, &rule.repl),
94                RuleTarget::Tag(key) => self.replace_named(span, key, &rule.re, &rule.repl),
95            }
96        }
97    }
98
99    fn replace_everywhere(&self, span: &mut Span, re: &Regex, repl: &str) {
100        rewrite_non_hidden(span.attributes.iter_mut(), re, repl);
101
102        self.replace_resource(span, re, repl);
103
104        for event in span.span_events_mut() {
105            rewrite_non_hidden(event.attributes_mut().iter_mut(), re, repl);
106        }
107    }
108
109    fn replace_resource(&self, span: &mut Span, re: &Regex, repl: &str) {
110        if let Some(resource) = replace_str(span.resource(), re, repl) {
111            span.set_resource(resource);
112        }
113    }
114
115    fn replace_named(&self, span: &mut Span, key: &str, re: &Regex, repl: &str) {
116        if let Some(value) = span.attributes.get_mut(key) {
117            if let Some(replaced) = replace_value(value, re, repl) {
118                *value = replaced;
119            }
120        }
121
122        for event in span.span_events_mut() {
123            if let Some(value) = event.attributes_mut().get_mut(key) {
124                if let Some(replaced) = replace_value(value, re, repl) {
125                    *value = replaced;
126                }
127            }
128        }
129    }
130}
131
132impl SynchronousTransform for TraceTagReplacer {
133    fn transform_buffer(&mut self, buffer: &mut EventsBuffer) {
134        if self.rules.is_empty() {
135            return;
136        }
137
138        for event in buffer {
139            if let Event::Trace(ref mut trace) = event {
140                for span in trace.spans_mut() {
141                    self.replace_span(span);
142                }
143            }
144        }
145    }
146}
147
148fn rewrite_non_hidden<'a, I>(entries: I, re: &Regex, repl: &str)
149where
150    I: Iterator<Item = (&'a MetaString, &'a mut AttributeValue)>,
151{
152    for (key, value) in entries {
153        if key.as_ref().starts_with('_') {
154            continue;
155        }
156        if let Some(replaced) = replace_value(value, re, repl) {
157            *value = replaced;
158        }
159    }
160}

162/// Formats a scalar value for regex matching; string values borrow, and composite values have no
163/// single string form and are skipped.
164fn value_as_cow(value: &AttributeValue) -> Option<Cow<'_, str>> {
165    match value {
166        AttributeValue::String(s) => Some(Cow::Borrowed(s.as_ref())),
167        AttributeValue::Bool(b) => Some(Cow::Owned(b.to_string())),
168        AttributeValue::Int(i) => Some(Cow::Owned(i.to_string())),
169        AttributeValue::Float(f) => Some(Cow::Owned(format!("{}", f))),
170        _ => None,
171    }
172}
173
174/// Returns the replacement only when the text changed; replacements are stored as strings. An
175/// unmatched value never allocates: the regex runs over the borrowed form, and only a changed
176/// value is materialized.
177fn replace_value(value: &AttributeValue, re: &Regex, repl: &str) -> Option<AttributeValue> {
178    let as_string = value_as_cow(value)?;
179    match re.replace_all(as_string.as_ref(), repl) {
180        // A borrowed result means the regex matched nothing.
181        Cow::Borrowed(_) => None,
182        Cow::Owned(replaced) => {
183            if replaced != as_string.as_ref() {
184                Some(AttributeValue::String(MetaString::from(replaced)))
185            } else {
186                None
187            }
188        }
189    }
190}
191
192fn replace_str(value: &str, re: &Regex, repl: &str) -> Option<String> {
193    match re.replace_all(value, repl) {
194        Cow::Borrowed(_) => None,
195        Cow::Owned(replaced) => (replaced != value).then_some(replaced),
196    }
197}

199/// Compiles the raw rules into their working form.
200///
201/// # Errors
202///
203/// Returns an error if a rule's `pattern` fails to compile, naming the rule. Startup fails rather
204/// than dropping the rule.
205fn compile_rules(rules: &[ReplaceRule]) -> Result<Vec<CompiledRule>, GenericError> {
206    let mut compiled = Vec::with_capacity(rules.len());
207    for rule in rules {
208        let re = Regex::new(&rule.pattern).map_err(|e| {
209            generic_error!(
210                "apm_config.replace_tags: rule with name '{}' has an invalid pattern '{}': {}. Each rule must \
211                 provide 'name', 'pattern', and 'repl', where 'pattern' is a valid regular expression.",
212                rule.name,
213                rule.pattern,
214                e
215            )
216        })?;
217        let target = match rule.name.as_str() {
218            "*" => RuleTarget::All,
219            "resource.name" => RuleTarget::Resource,
220            _ => RuleTarget::Tag(rule.name.clone()),
221        };
222        compiled.push(CompiledRule {
223            target,
224            re,
225            repl: rule.repl.clone(),
226        });
227    }
228    Ok(compiled)
229}

238    fn attrs(pairs: &[(&str, &str)]) -> FastHashMap<MetaString, AttributeValue> {
239        pairs
240            .iter()
241            .map(|(k, v)| (MetaString::from(*k), AttributeValue::String(MetaString::from(*v))))
242            .collect()
243    }

245    fn event(time: u64, name: &str, pairs: &[(&str, &str)]) -> SpanEvent {
246        SpanEvent::new(time, name).with_attributes(attrs(pairs))
247    }

249    fn replacer(rules: &[(&str, &str, &str)]) -> TraceTagReplacer {
250        let raw = rules
251            .iter()
252            .map(|(name, pattern, repl)| ReplaceRule {
253                name: (*name).to_owned(),
254                pattern: (*pattern).to_owned(),
255                repl: (*repl).to_owned(),
256            })
257            .collect::<Vec<_>>();
258        TraceTagReplacer {
259            rules: compile_rules(&raw).expect("test rules compile"),
260        }
261    }

388    #[test]
389    fn matched_numeric_tag_becomes_a_string() {
390        let mut attrs: FastHashMap<MetaString, AttributeValue> = FastHashMap::default();
391        attrs.insert(MetaString::from("http.status_code"), AttributeValue::Int(200));
392        attrs.insert(MetaString::from("http.port"), AttributeValue::Int(8080));
393        attrs.insert(MetaString::from("http.retry"), AttributeValue::Int(3));
394
395        let mut span = span_with("POST /pay/checkout", attrs, vec![]);
396        replacer(&[("*", "20", "2x"), ("*", "3", "3")]).replace_span(&mut span);
397
398        let value = attr(&span, "http.status_code").expect("tag survives");
399        assert_eq!(
400            value.as_string().map(|s| s.as_ref()),
401            Some("2x0"),
402            "written back as a string"
403        );
404        assert_eq!(value.as_int(), None, "the numeric form does not survive a rewrite");
405        assert_eq!(
406            attr(&span, "http.port").and_then(|v| v.as_int()),
407            Some(8080),
408            "an unmatched numeric stays numeric"
409        );
410        assert_eq!(
411            attr(&span, "http.retry").and_then(|v| v.as_int()),
412            Some(3),
413            "a match whose replacement equals the matched text is left as-is"
414        );
415    }
