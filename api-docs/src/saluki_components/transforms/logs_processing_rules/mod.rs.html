1//! Logs processing rules transform.
2//!
3//! Applies an ordered list of processing rules to the message of each log, which can drop the log or mask parts of
4//! its message. Only the message changes; the status, hostname, service, source, tags, and additional properties are
5//! left as they are.
6//!
7//! The rules run in order, and each one sees the message as the rules before it left it:
8//!
9//! - `exclude_at_match` drops the log if its pattern matches.
10//! - `include_at_match` drops the log if its pattern doesn't match.
11//! - `mask_sequences` replaces every match of its pattern with its placeholder.
12//!
13//! The first rule that drops a log stops the processing of that log.
14//!
15//! Patterns and placeholders use the syntax of the `regex` crate, where `$1` and `${name}` refer to capture groups.
16//! The rules are expected to have been validated already: building the transform only fails when a pattern doesn't
17//! compile.
18

34/// A log processing rule.
35#[derive(Clone, Debug, PartialEq)]
36pub enum LogsProcessingRule {
37    /// Drops logs whose content matches `pattern`.
38    ExcludeAtMatch {
39        /// Name of the rule.
40        name: String,
41        /// Regular expression matched against the content.
42        pattern: String,
43    },
44
45    /// Drops logs whose content doesn't match `pattern`.
46    IncludeAtMatch {
47        /// Name of the rule.
48        name: String,
49        /// Regular expression matched against the content.
50        pattern: String,
51    },
52
53    /// Replaces every match of `pattern` in the content with `replace_placeholder`.
54    MaskSequences {
55        /// Name of the rule.
56        name: String,
57        /// Regular expression matched against the content.
58        pattern: String,
59        /// Replacement for each match, where `$1` and `${name}` refer to capture groups.
60        replace_placeholder: String,
61    },
62}

64/// Logs processing rules transform configuration.
65pub struct LogsProcessingRulesConfiguration {
66    rules: Vec<LogsProcessingRule>,
67}

69impl LogsProcessingRulesConfiguration {
70    /// Creates a new `LogsProcessingRulesConfiguration` that applies the given rules, in order.
71    pub fn new(rules: Vec<LogsProcessingRule>) -> Self {
72        Self { rules }
73    }
74}

76#[async_trait]
77impl SynchronousTransformBuilder for LogsProcessingRulesConfiguration {
78    async fn build(&self, context: BuildContext) -> Result<Box<dyn SynchronousTransform + Send>, GenericError> {
79        let builder = MetricsBuilder::from_component_context(context.component_context());
80        Ok(Box::new(LogsProcessingRules {
81            rules: compile_rules(&self.rules)?,
82            events_dropped: builder.register_counter_with_tags(
83                "component_events_dropped_total",
84                ["intentional:true", "drop_reason:processing_rule"],
85            ),
86        }))
87    }
88}

90impl MemoryBounds for LogsProcessingRulesConfiguration {
91    fn specify_bounds(&self, builder: &mut MemoryBoundsBuilder) {
92        // The memory that the compiled regular expressions hold internally isn't known, so it isn't counted.
93        let placeholder_bytes = self
94            .rules
95            .iter()
96            .map(|rule| match rule {
97                LogsProcessingRule::MaskSequences {
98                    replace_placeholder, ..
99                } => replace_placeholder.len(),
100                _ => 0,
101            })
102            .sum();
103        builder
104            .minimum()
105            .with_single_value::<LogsProcessingRules>("component struct")
106            .with_array::<CompiledRule>("compiled rules", self.rules.len())
107            .with_fixed_amount("placeholders", placeholder_bytes);
108    }
109}

111/// A rule with its pattern compiled.
112#[derive(Debug)]
113enum CompiledRule {
114    Exclude(Regex),
115    Include(Regex),
116    Mask { regex: Regex, placeholder: String },
117}
118
119/// Compiles the rules, in order.
120///
121/// # Errors
122///
123/// Returns an error naming the rule if a pattern doesn't compile.
124fn compile_rules(rules: &[LogsProcessingRule]) -> Result<Vec<CompiledRule>, GenericError> {
125    let compile = |name: &str, pattern: &str| {
126        Regex::new(pattern)
127            .with_error_context(|| format!("Invalid pattern '{pattern}' for logs processing rule '{name}'."))
128    };
129
130    rules
131        .iter()
132        .map(|rule| match rule {
133            LogsProcessingRule::ExcludeAtMatch { name, pattern } => Ok(CompiledRule::Exclude(compile(name, pattern)?)),
134            LogsProcessingRule::IncludeAtMatch { name, pattern } => Ok(CompiledRule::Include(compile(name, pattern)?)),
135            LogsProcessingRule::MaskSequences {
136                name,
137                pattern,
138                replace_placeholder,
139            } => Ok(CompiledRule::Mask {
140                regex: compile(name, pattern)?,
141                placeholder: replace_placeholder.clone(),
142            }),
143        })
144        .collect()
145}

147/// Logs processing rules transform.
148struct LogsProcessingRules {
149    rules: Vec<CompiledRule>,
150    events_dropped: Counter,
151}
152
153impl LogsProcessingRules {
154    /// Applies the rules to `log`, returning `false` if a rule drops it.
155    fn process(&self, log: &mut Log) -> bool {
156        let mut content = Cow::Borrowed(log.message());
157        for rule in &self.rules {
158            match rule {
159                CompiledRule::Exclude(regex) => {
160                    if regex.is_match(&content) {
161                        return false;
162                    }
163                }
164                CompiledRule::Include(regex) => {
165                    if !regex.is_match(&content) {
166                        return false;
167                    }
168                }
169                CompiledRule::Mask { regex, placeholder } => {
170                    let masked = match regex.replace_all(&content, placeholder.as_str()) {
171                        Cow::Borrowed(_) => None,
172                        Cow::Owned(masked) => Some(masked),
173                    };
174                    if let Some(masked) = masked {
175                        content = Cow::Owned(masked);
176                    }
177                }
178            }
179        }
180
181        // A mask can replace a match with the same text, so an owned result isn't necessarily a changed one.
182        let changed = match content {
183            Cow::Owned(content) if content != log.message() => Some(content),
184            _ => None,
185        };
186        if let Some(content) = changed {
187            log.set_message(content);
188        }
189        true
190    }
191}
192
193impl SynchronousTransform for LogsProcessingRules {
194    fn transform_buffer(&mut self, buffer: &mut EventsBuffer) {
195        if self.rules.is_empty() {
196            return;
197        }
198
199        let mut dropped = 0;
200        buffer.retain_events::<Log, _>(|log| {
201            let keep = self.process(log);
202            if !keep {
203                dropped += 1;
204            }
205            keep
206        });
207        if dropped > 0 {
208            self.events_dropped.increment(dropped);
209        }
210    }
211}
