1//! Peer tag key set derivation from the semantic registry.
2//!
3//! Peer tags identify the remote endpoint a span talked to, and their values are hashed into the
4//! stats aggregation key in the order of the configured key list. The snapshot also records the
5//! registry's content hash so it can be rebuilt when the registry changes.
6
7use std::sync::Arc;
8
9use saluki_common::collections::FastHashSet;
10use stringtheory::MetaString;
11
12use crate::common::otlp::semantics::{Concept, Registry};
13
14/// The peer concepts whose precedence lists make up the base key set.
15const PEER_CONCEPTS: &[Concept] = &[
16    Concept::PeerService,
17    Concept::PeerHostname,
18    Concept::PeerDbName,
19    Concept::PeerDbSystem,
20    Concept::PeerCassandraContactPoints,
21    Concept::PeerCouchbaseSeedNodes,
22    Concept::PeerMessagingDestination,
23    Concept::PeerMessagingSystem,
24    Concept::PeerKafkaBootstrapServers,
25    Concept::PeerRpcService,
26    Concept::PeerRpcSystem,
27    Concept::PeerAwsS3Bucket,
28    Concept::PeerAwsSqsQueue,
29    Concept::PeerAwsDynamoDbTable,
30    Concept::PeerAwsKinesisStream,
31    Concept::DdBaseService,
32];
33
34/// A snapshot of the peer tag key set, pinned to a registry content hash.
35///
36/// The key list is `Arc`-backed, so snapshots clone cheaply and can be shared. Whether the snapshot
37/// is stale is answered with a single `u64` comparison against the live registry's content hash.
38#[derive(Clone)]
39pub(crate) struct PeerTagKeys {
40    content_hash: u64,
41    keys: Arc<[MetaString]>,
42}
43
44impl PeerTagKeys {
45    /// Builds the key set from the registry's peer concept precedence lists, plus operator-configured tags.
46    ///
47    /// The result is sorted and deduplicated, which makes the key order canonical regardless of
48    /// registry iteration order or the order of the configured tags.
49    pub(crate) fn build(registry: &Registry, custom_peer_tags: &[MetaString]) -> Self {
50        let mut keys = base_peer_tag_keys(registry);
51        keys.extend(custom_peer_tags.iter().cloned());
52        keys.sort_unstable();
53        keys.dedup();
54
55        Self {
56            content_hash: registry.content_hash(),
57            keys: keys.into(),
58        }
59    }
60
61    /// Rebuilds the key set if the registry content hash changed since this snapshot was taken.
62    ///
63    /// Returns `true` when a rebuild happened, `false` when the snapshot was already current. The
64    /// comparison is a single integer check, so calling this on every flush cycle is effectively
65    /// free; only an actual registry change pays the cost of re-deriving the key set.
66    pub(crate) fn refresh(&mut self, registry: &Registry, custom_peer_tags: &[MetaString]) -> bool {
67        if self.content_hash == registry.content_hash() {
68            return false;
69        }
70
71        *self = Self::build(registry, custom_peer_tags);
72        true
73    }
74
75    /// Returns the snapshot's peer tag keys, in canonical (sorted, deduplicated) order.
76    pub(crate) fn keys(&self) -> &[MetaString] {
77        &self.keys
78    }
79
80    /// Returns `true` if the key set is empty.
81    pub(crate) fn is_empty(&self) -> bool {
82        self.keys.is_empty()
83    }
84}
85
86/// Collects every attribute name from every peer concept's precedence list.
87///
88/// A concept maps to several attributes ordered by precedence; for key-set purposes, all of the
89/// names are collected since the span's raw attributes are keyed by name, not by concept.
90fn base_peer_tag_keys(registry: &Registry) -> Vec<MetaString> {
91    let mut names = FastHashSet::default();
92    for concept in PEER_CONCEPTS {
93        if let Some(fallbacks) = registry.get_attribute_precedence(*concept) {
94            names.extend(fallbacks.iter().map(|tag| tag.name.as_str()));
95        }
96    }
97    names.into_iter().map(MetaString::from).collect()
98}

160    fn base_keys_match_upstream_peer_tags() {
161        let keys = PeerTagKeys::build(&REGISTRY, &[]);
162        assert_eq!(
163            keys.keys().iter().map(|k| k.as_ref()).collect::<Vec<&str>>(),
164            EXPECTED_BASE_KEYS,

169    #[test]
170    fn property_test_keys_are_sorted_and_deduplicated() {
171        proptest!(|(custom_tags in proptest::collection::vec("[a-z_]{1,12}", 0..16))| {
172            let custom: Vec<MetaString> =
173                custom_tags.iter().map(|t: &String| MetaString::from(t.as_str())).collect();
174            let keys = PeerTagKeys::build(&REGISTRY, &custom);
175
176            for window in keys.keys().windows(2) {
177                prop_assert!(window[0] < window[1], "keys must be strictly sorted");
178            }
179        });
180    }

182    #[test]
183    fn custom_tags_are_appended_and_deduplicated() {
184        let custom = vec![
185            MetaString::from("my.custom.peer.tag"),
186            // A duplicate of a base key: must not appear twice.
187            MetaString::from("db.name"),
188            // A duplicate within the custom list itself.
189            MetaString::from("my.custom.peer.tag"),
190        ];
191        let keys = PeerTagKeys::build(&REGISTRY, &custom);
192        let names: Vec<&str> = keys.keys().iter().map(|k| k.as_ref()).collect();
193
194        assert!(names.contains(&"my.custom.peer.tag"));
195        assert_eq!(names.iter().filter(|n| **n == "my.custom.peer.tag").count(), 1);
196        assert_eq!(names.iter().filter(|n| **n == "db.name").count(), 1);
197    }

205    #[test]
206    fn refresh_rebuilds_when_registry_content_changes() {
207        let mut keys = PeerTagKeys::build(&REGISTRY, &[]);
208
209        // A modified registry: one new attribute in the `peer.service` precedence list.
210        let modified = r#"{
211            "version": "modified",
212            "concepts": {
213                "peer.service": {
214                    "fallbacks": [{"name": "peer.service", "provider": "otel", "type": "string"},
215                                  {"name": "custom.remote.service", "provider": "otel", "type": "string"}]
216                }
217            }
218        }"#;
219        let registry = Registry::from_json(modified).expect("modified registry should parse");
220
221        assert!(keys.refresh(&registry, &[]), "content change must trigger a rebuild");
222
223        let names: Vec<&str> = keys.keys().iter().map(|k| k.as_ref()).collect();
224        assert_eq!(names, ["custom.remote.service", "peer.service"]);
225        assert!(
226            !keys.refresh(&registry, &[]),
227            "a second refresh with no change must be a no-op"
228        );
229    }
