1//! Quantization of IP addresses in peer tag values before stats aggregation.
2//!
3//! Peer tag values become APM stats dimensions, and a raw peer IP address turns every host into
4//! its own aggregation key, so bucket cardinality grows with the fleet size and per-service views
5//! fragment. The quantizer collapses the host portion of peer values that are IP addresses into a
6//! single `blocked-ip-address` placeholder, while preserving prefixes (`ip-`, `scheme://`), ports,
7//! and suffixes, and passing through anything that is not an IP address untouched.
8//!
9//! The behavior here is a direct port of the server-side `QuantizePeerIPAddresses` helper, and the
10//! placeholder string is a wire-visible contract: mixed clusters only aggregate cleanly when both
11//! produce byte-identical values, so the semantics—including quirks like lenient IPv4 field
12//! separators—are pinned deliberately rather than replaced with a "nicer" parser.
13
14use std::borrow::Cow;
15use std::net::Ipv6Addr;
16
17/// The placeholder that replaces non-allowlisted IP address hosts.
18const BLOCKED_IP_ADDRESS: &str = "blocked-ip-address";
19
20/// The schemes whose prefixes survive quantization, for example `http://` in `http://10.0.0.1:8080`.
21const SCHEMES: [&str; 5] = ["dnspoll", "ftp", "file", "http", "https"];
22
23/// IP addresses that survive quantization unchanged.
24///
25/// These are localhost (both loopback families) and link-local cloud provider metadata server
26/// addresses. Unlike regular peer IP addresses, these values carry actual meaning: they identify
27/// talking to the metadata endpoint or to localhost, both of which are low-cardinality and useful
28/// as dimensions on their own.
29const ALLOWED_IP_ADDRESSES: [&str; 5] = [
30    // localhost
31    "127.0.0.1",
32    "::1",
33    // link-local cloud provider metadata server addresses
34    "169.254.169.254",
35    "fd00:ec2::254",
36    // ECS task metadata
37    "169.254.170.2",
38];
39
40/// Quantizes the IP addresses in a comma-separated list of hosts.
41///
42/// Each entry that is an IP address (with an optional port, prefix, and suffix) is replaced using
43/// [`quantize_ip`], and duplicate entries post-quantization are collapsed into a single unique
44/// value. Entries that are not IP addresses are left unchanged. Comma-separated host lists are
45/// common for peer tags like `db.cassandra.contact.points`, `db.couchbase.seed.nodes`, and
46/// `messaging.kafka.bootstrap.servers`.
47///
48/// When no entry is rewritten and no duplicates are collapsed, the input is returned borrowed.
49/// The common case—a single peer value that is not an IP address—takes the single-entry fast path
50/// and does not allocate.
51pub(crate) fn quantize_peer_ip_addresses(raw: &str) -> Cow<'_, str> {
52    // Fast path: a single-entry value cannot contain duplicates, so the entry quantizer alone
53    // decides whether anything changes. This is the dominant case, and it allocates nothing at all
54    // when the value passes through unchanged.
55    if !raw.contains(',') {
56        return quantize_ip(raw);
57    }
58
59    let mut out: Vec<Cow<'_, str>> = Vec::new();
60    let mut changed = false;
61
62    for value in raw.split(',') {
63        let quantized = quantize_ip(value);
64        // An owned entry means the quantizer rewrote this value. (`Cow::is_owned` would say this
65        // directly, but it is unstable, so check the variant instead.)
66        if matches!(quantized, Cow::Owned(_)) {
67            changed = true;
68        }
69
70        // Peer value lists are short (a handful of seed nodes or bootstrap servers), so a linear
71        // scan is cheaper than a set and keeps the deduped entries borrowed from the input.
72        let is_new = out.iter().all(|existing| existing.as_ref() != quantized.as_ref());
73        if is_new {
74            out.push(quantized);
75        } else {
76            changed = true;
77        }
78    }
79
80    if !changed {
81        Cow::Borrowed(raw)
82    } else {
83        let mut joined = String::with_capacity(raw.len());
84        for (index, value) in out.iter().enumerate() {
85            if index > 0 {
86                joined.push(',');
87            }
88            joined.push_str(value.as_ref());
89        }
90        Cow::Owned(joined)
91    }
92}

94/// Quantizes the IP address in the provided value, if the value is exactly an IP address with an
95/// optional port, prefix, and suffix.
96///
97/// If the value is not an IP address, it is returned unchanged. If the host is allowlisted, the
98/// value is returned unchanged. Otherwise the host is replaced by `blocked-ip-address` while the
99/// prefix, port, and suffix are preserved: ports are much lower cardinality than IP addresses and
100/// tend to correspond to a protocol (for example, `443` is HTTPS), so keeping them in is both safe
101/// and useful for aggregation.
102fn quantize_ip(raw: &str) -> Cow<'_, str> {
103    let (prefix, rest) = split_prefix(raw);
104    let Some((host, port, suffix)) = parse_ip_and_port(rest) else {
105        return Cow::Borrowed(raw);
106    };
107
108    if ALLOWED_IP_ADDRESSES.contains(&host) {
109        return Cow::Borrowed(raw);
110    }
111
112    let port_len = if port.is_empty() { 0 } else { port.len() + 1 };
113    let mut quantized = String::with_capacity(prefix.len() + BLOCKED_IP_ADDRESS.len() + port_len + suffix.len());
114    quantized.push_str(prefix);
115    quantized.push_str(BLOCKED_IP_ADDRESS);
116    if !port.is_empty() {
117        quantized.push(':');
118        quantized.push_str(port);
119    }
120    quantized.push_str(suffix);
121
122    Cow::Owned(quantized)
123}

125/// Splits a leading `ip-` or scheme prefix off of a value.
126///
127/// The `ip-` prefix covers AWS EC2 hostnames like `ip-10-123-4-567.ec2.internal`. The scheme
128/// prefix covers values like `http://10.0.0.1:8080/health`, including the three-slash `file:///`
129/// form; the scheme may appear anywhere in the value, matching the server-side behavior.
130fn split_prefix(raw: &str) -> (&str, &str) {
131    if let Some(rest) = raw.strip_prefix("ip-") {
132        return ("ip-", rest);
133    }
134
135    for scheme in SCHEMES {
136        let Some(scheme_index) = raw.find(scheme) else {
137            continue;
138        };
139
140        // Slicing here is safe because `find` returns a character boundary and the schemes are
141        // ASCII. Matching the separator with `strip_prefix` then avoids computing any further
142        // index, which is what could otherwise land inside a multi-byte character and panic.
143        let after_scheme = &raw[scheme_index + scheme.len()..];
144
145        // `:///` is tried first because `://` is a prefix of it. The separator must be followed
146        // by at least one character.
147        for separator in [":///", "://"] {
148            if let Some(rest) = after_scheme.strip_prefix(separator) {
149                if !rest.is_empty() {
150                    let scheme_end = scheme_index + scheme.len() + separator.len();
151                    return (&raw[scheme_index..scheme_end], rest);
152                }
153            }
154        }
155    }
156
157    ("", raw)
158}

160/// Parses a value into an IP address host with an optional port and trailing suffix.
161///
162/// Returns `Some((host, port, suffix))` when the host portion is a valid IP address, where `suffix`
163/// covers any trailing characters after the address (for example, `/health` in a URL) and is
164/// preserved after quantization to keep paths distinct. Returns `None` when the value is not an IP
165/// address.
166fn parse_ip_and_port(value: &str) -> Option<(&str, &str, &str)> {
167    let (mut host, mut port) = (value, "");
168    if let Some((split_host, split_port)) = split_host_port(value) {
169        host = split_host;
170        port = split_port;
171    }
172
173    let end = is_parseable_ip(host)?;
174    Some((&host[..end], port, &host[end..]))
175}
176
177/// Validates a string as an IP address and returns the index of the first character after the
178/// address, which is the start of the suffix.
179///
180/// Returns `None` when the string is not an IP address.
181fn is_parseable_ip(s: &str) -> Option<usize> {
182    if s.is_empty() {
183        return None;
184    }
185
186    // Must start with a hex digit, or IPv6 can have a preceding ':'.
187    match s.as_bytes()[0] {
188        b'0'..=b'9' | b'a'..=b'f' | b'A'..=b'F' | b':' => {}
189        _ => return None,
190    }
191
192    let bytes = s.as_bytes();
193    for &b in bytes.iter() {
194        match b {
195            b'.' | b'_' | b'-' => return parse_ipv4(s, b),
196            b':' => {
197                // IPv6: the whole remaining string must parse.
198                if parse_ipv6_with_optional_zone(s) {
199                    return Some(s.len());
200                }
201                return None;
202            }
203            b'%' => {
204                // Assume that this was trying to be an IPv6 address with a zone specifier, but
205                // the address is missing.
206                return None;
207            }
208            _ => continue,
209        }
210    }
211
212    None
213}

215/// Parses an IPv6 address with an optional trailing `%zone`, mirroring Go's `netip.ParseAddr`.
216///
217/// `Ipv6Addr` rejects scoped addresses like `fe80::1%eth0`, so the zone is split off before
218/// parsing the address portion. The zone must be non-empty, matching `netip`'s requirement.
219fn parse_ipv6_with_optional_zone(s: &str) -> bool {
220    match s.split_once('%') {
221        Some((addr, zone)) => !zone.is_empty() && addr.parse::<Ipv6Addr>().is_ok(),
222        None => s.parse::<Ipv6Addr>().is_ok(),
223    }
224}

226/// Parses `s` as an IPv4 address and returns the index of the first character after the address.
227///
228/// This is a modified port of the standard library's IPv4 parsing that accepts alternate field
229/// separators besides `.` (for example, `10-1-2-3` or `10_1_2_3`), because instrumentation libraries
230/// sometimes mangle the separator, and also returns the index of trailing characters after the
231/// address so that suffixes (for example, `.ec2.internal`) can be preserved.
232///
233/// Returns `None` when the string is not an IPv4 address.
234fn parse_ipv4(s: &str, sep: u8) -> Option<usize> {
235    let bytes = s.as_bytes();
236    let mut field_value = 0u32;
237    let mut field_count = 0;
238    let mut digit_len = 0;
239
240    for (i, &b) in bytes.iter().enumerate() {
241        if b.is_ascii_digit() {
242            // No leading zeros.
243            if digit_len == 1 && field_value == 0 {
244                return None;
245            }
246            field_value = field_value * 10 + (b - b'0') as u32;
247            digit_len += 1;
248            if field_value > 255 {
249                return None;
250            }
251        } else if b == sep {
252            // Reject `.1.2.3`, `1.2.3.`, and `1..2.3`-style forms.
253            if i == 0 || i == bytes.len() - 1 || bytes[i - 1] == sep {
254                return None;
255            }
256            // `1.2.3.4.5`—the fifth field is the start of the suffix.
257            if field_count == 3 {
258                return Some(i);
259            }
260            field_count += 1;
261            field_value = 0;
262            digit_len = 0;
263        } else if field_count == 3 && digit_len > 0 {
264            // First non-digit character after a complete four-field address: the suffix begins
265            // here (for example, the `.` in `10.0.0.1.ec2.internal`).
266            return Some(i);
267        } else {
268            return None;
269        }
270    }
271
272    if field_count < 3 {
273        return None;
274    }
275
276    Some(s.len())
277}

279/// Splits a network address of the form `host:port` or `[host]:port` into its host and port.
280///
281/// A literal IPv6 address in a host:port form must be enclosed in square brackets, as in
282/// `[::1]:80`. Returns `None` when the value is not a valid host:port form; callers fall back to
283/// treating the whole value as a bare host in that case.
284fn split_host_port(hostport: &str) -> Option<(&str, &str)> {
285    // The port starts after the last colon.
286    let i = hostport.rfind(':')?;
287
288    let (host, open_close) = if hostport.starts_with('[') {
289        // Expect the first ']' just before the last ':'.
290        let end = hostport.find(']')?;
291        if end + 1 != i {
292            // Either ']' isn't followed by a colon, or it is followed by a colon that is not the
293            // last one.
294            return None;
295        }
296        (&hostport[1..end], (1, end + 1))
297    } else {
298        let host = &hostport[..i];
299        if host.contains(':') {
300            return None;
301        }
302        (host, (0, 0))
303    };
304
305    // There can't be a '[' or ']' before/after these positions.
306    if hostport[open_close.0..].contains('[') {
307        return None;
308    }
309    if hostport[open_close.1..].contains(']') {
310        return None;
311    }
312
313    Some((host, &hostport[i + 1..]))
314}

516    fn property_test_quantize_is_idempotent() {
517        proptest!(|(value in "[A-Za-z0-9:\\.\\-_%/]{1,64}")| {
518            let once = quantize_peer_ip_addresses(&value);
519            let once_owned = once.clone().into_owned();
520            let twice = quantize_peer_ip_addresses(&once_owned);
521            prop_assert_eq!(once, twice, "quantizing a quantized value must be a no-op");
522        });

526    fn property_test_quantize_never_grows_cardinality() {
527        proptest!(|(hosts in proptest::collection::vec("[A-Za-z0-9:\\.\\-_%/]{1,32}", 1..8))| {
