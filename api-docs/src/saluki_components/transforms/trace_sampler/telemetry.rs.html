1//! Sampler telemetry, emitted as metric events.
2//!
3//! The telemetry is split by who can write it: [`SamplerCounters`] holds the fixed counters the
4//! individual samplers record through shared references, with the recording sampler's identity
5//! baked into each counter name, and [`DecisionWindow`] holds the per-decision counts, keyed at
6//! runtime by the sampler that decided. The window is owned and flushed exclusively by the
7//! transform's loop, so it needs no lock; only the counters shared into the inner samplers stay
8//! atomic.
9//!
10//! Decisions accumulate in a map for one window; [`DecisionWindow::take_window_events`]
11//! reports every entry as seen/kept delta counters, the signature-count gauges, and the
12//! rare-sampler counters, then resets the window. The events flow through the sampler transform's
13//! metrics output into the metrics pipeline, so the metrics reach the backend under their
14//! published names without internal-registry involvement. The window map is capped at
15//! [`MAX_DECISION_KEYS`] entries; combinations beyond the cap count under one overflow bucket per
16//! sampler. Hits and misses are window deltas; shrinks accumulate for the process lifetime and are
17//! published as a gauge.
18

28/// How long a telemetry window accumulates decisions before reporting.
29pub(super) const WINDOW: Duration = Duration::from_secs(10);
30
31/// Cap on distinct decision keys held in one window.
32///
33/// Ten times the priority catalog's default capacity. Once a window holds this many keys, unseen
34/// combinations count under one overflow bucket per sampler while existing keys keep their own
35/// buckets, so burst cardinality cannot grow the map without bound. Past this many combinations
36/// in one window, per-combination counts carry no actionable signal. Deliberately not a tuning
37/// knob.
38const MAX_DECISION_KEYS: usize = 50_000;
39
40/// Service tag value marking decisions rolled into a sampler's overflow bucket.
41const OVERFLOW_SERVICE: &str = "other";
42
43const METRIC_SAMPLER_SEEN: &str = "datadog.trace_agent.sampler.seen";
44const METRIC_SAMPLER_KEPT: &str = "datadog.trace_agent.sampler.kept";
45const METRIC_SAMPLER_SIZE: &str = "datadog.trace_agent.sampler.size";
46const METRIC_RARE_HITS: &str = "datadog.trace_agent.sampler.rare.hits";
47const METRIC_RARE_MISSES: &str = "datadog.trace_agent.sampler.rare.misses";
48const METRIC_RARE_SHRINKS: &str = "datadog.trace_agent.sampler.rare.shrinks";

50/// The sampler that decided a trace's fate.
51#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
52pub(super) enum SamplerName {
53    /// No sampler ran, or none claimed the decision.
54    Unknown,
55    Priority,
56    NoPriority,
57    Error,
58    Rare,
59    Probabilistic,
60}
61
62impl SamplerName {
63    /// Returns the `sampler` tag value.
64    fn as_str(self) -> &'static str {
65        match self {
66            Self::Unknown => "unknown",
67            Self::Priority => "priority",
68            Self::NoPriority => "no_priority",
69            Self::Error => "error",
70            Self::Rare => "rare",
71            Self::Probabilistic => "probabilistic",
72        }
73    }
74
75    /// Returns whether decisions under this name carry a `target_env` tag.
76    fn carries_env_tag(self) -> bool {
77        matches!(self, Self::Priority | Self::NoPriority | Self::Error | Self::Rare)
78    }
79}

81/// A sampling priority.
82#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
83pub(super) enum Priority {
84    ManualDrop,
85    AutoDrop,
86    AutoKeep,
87    ManualKeep,
88    /// Any value outside -1..=2, kept so the conversion round-trips.
89    Unknown(i32),
90}
91
92impl Priority {
93    /// Returns the `sampling_priority` tag value; unknown priorities read as `none`.
94    pub(super) const fn as_str(self) -> &'static str {
95        match self {
96            Self::ManualDrop => "manual_drop",
97            Self::AutoDrop => "auto_drop",
98            Self::AutoKeep => "auto_keep",
99            Self::ManualKeep => "manual_keep",
100            Self::Unknown(_) => "none",
101        }
102    }
103}
104
105impl From<i32> for Priority {
106    fn from(value: i32) -> Self {
107        match value {
108            -1 => Self::ManualDrop,
109            0 => Self::AutoDrop,
110            1 => Self::AutoKeep,
111            2 => Self::ManualKeep,
112            other => Self::Unknown(other),
113        }
114    }
115}
116
117impl From<Priority> for i32 {
118    fn from(priority: Priority) -> Self {
119        match priority {
120            Priority::ManualDrop => -1,
121            Priority::AutoDrop => 0,
122            Priority::AutoKeep => 1,
123            Priority::ManualKeep => 2,
124            Priority::Unknown(value) => value,
125        }
126    }
127}

129/// The trace dimensions a decision is counted under.
130///
131/// Service and env are `MetaString` so typical names clone without heap allocation on the per-trace
132/// recording path.
133#[derive(Clone, Debug, Eq, Hash, PartialEq)]
134pub(super) struct DecisionKey {
135    pub(super) sampler: SamplerName,
136    pub(super) service: MetaString,
137    pub(super) env: MetaString,
138    pub(super) priority: Option<Priority>,
139}
140
141impl DecisionKey {
142    /// Assembles the metric tags for this key.
143    ///
144    /// The conditional tags are the contract: the priority tag belongs to the priority sampler
145    /// only, the env tag to the adaptive samplers, and empty dimensions are omitted.
146    fn tags(&self) -> Vec<(&'static str, String)> {
147        let mut tags = Vec::with_capacity(4);
148        tags.push(("sampler", self.sampler.as_str().to_string()));
149        if self.sampler == SamplerName::Priority {
150            if let Some(priority) = self.priority {
151                tags.push(("sampling_priority", priority.as_str().to_string()));
152            }
153        }
154        if !self.service.is_empty() {
155            tags.push(("target_service", self.service.to_string()));
156        }
157        if !self.env.is_empty() && self.sampler.carries_env_tag() {
158            tags.push(("target_env", self.env.to_string()));
159        }
160        tags
161    }
162
163    /// Assembles the metric tag set for this key, in `name:value` form.
164    fn tag_set(&self) -> TagSet {
165        let tags = self.tags();
166        let mut tag_set = TagSet::with_capacity(tags.len());
167        for (name, value) in tags {
168            tag_set.insert_tag(MetaString::from(format!("{}:{}", name, value)));
169        }
170        tag_set
171    }
172}

174/// Windowed decision counts for one key.
175#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
176pub(super) struct DecisionCounts {
177    /// Traces the sampler evaluated.
178    pub(super) seen: u64,
179    /// Traces the sampler kept.
180    pub(super) kept: u64,
181}
182
183/// Counters shared with the individual samplers.
184///
185/// The inner samplers record through shared references, so the counters are atomic; cloning shares
186/// them. Attribution lives in the counter names themselves: each counter is specific to the
187/// sampler that records it, so the counters carry no runtime sampler identity.
188#[derive(Clone)]
189pub(super) struct SamplerCounters {
190    inner: Arc<CountersInner>,
191}
192
193struct CountersInner {
194    rare_hits: AtomicU64,
195    rare_misses: AtomicU64,
196    rare_shrinks: AtomicU64,
197}
198
199impl SamplerCounters {
200    /// Creates a telemetry with all counters at zero.
201    pub(super) fn new() -> Self {
202        Self {
203            inner: Arc::new(CountersInner {
204                rare_hits: AtomicU64::new(0),
205                rare_misses: AtomicU64::new(0),
206                rare_shrinks: AtomicU64::new(0),
207            }),
208        }
209    }
210
211    /// Records a rare-sampler keep.
212    pub(super) fn record_rare_hit(&self) {
213        self.inner.rare_hits.fetch_add(1, Ordering::Relaxed);
214    }
215
216    /// Records a rare-sampler drop.
217    pub(super) fn record_rare_miss(&self) {
218        self.inner.rare_misses.fetch_add(1, Ordering::Relaxed);
219    }
220
221    /// Records a rare-sampler signature-table shrink.
222    pub(super) fn record_rare_shrink(&self) {
223        self.inner.rare_shrinks.fetch_add(1, Ordering::Relaxed);
224    }
225
226    #[cfg(test)]
227    pub(super) fn rare_totals(&self) -> (u64, u64, u64) {
228        (
229            self.inner.rare_hits.load(Ordering::Relaxed),
230            self.inner.rare_misses.load(Ordering::Relaxed),
231            self.inner.rare_shrinks.load(Ordering::Relaxed),
232        )
233    }
234}

236/// The decision window, owned by the transform's loop.
237///
238/// Decisions accumulate in a map for one window and are reported as metric events by
239/// [`DecisionWindow::take_window_events`], which also resets the window. Each decision is keyed at
240/// runtime by the sampler that made it, unlike [`SamplerCounters`] where the recording sampler is
241/// fixed per counter. The loop holds this state exclusively, so the decision map is a plain map;
242/// the counters shared into the inner samplers stay atomic behind [`SamplerCounters`].
243pub(super) struct DecisionWindow {
244    decisions: FastHashMap<DecisionKey, DecisionCounts>,
245    max_decision_keys: usize,
246    counters: SamplerCounters,
247}
248
249impl DecisionWindow {
250    /// Creates a telemetry with the default decision-key cap.
251    pub(super) fn new() -> Self {
252        Self::with_decision_key_cap(MAX_DECISION_KEYS)
253    }
254
255    /// Creates a telemetry with a reduced decision-key cap, for exercising the overflow roll-over
256    /// without recording tens of thousands of decisions.
257    fn with_decision_key_cap(max_decision_keys: usize) -> Self {
258        Self {
259            decisions: FastHashMap::default(),
260            max_decision_keys,
261            counters: SamplerCounters::new(),
262        }
263    }
264
265    /// Returns the counters shared with the individual samplers.
266    pub(super) fn counters(&self) -> &SamplerCounters {
267        &self.counters
268    }
269
270    /// Records one sampler decision.
271    ///
272    /// The priority is carried only by the priority sampler; every other sampler counts under a
273    /// key without it.
274    pub(super) fn record_decision(
275        &mut self, kept: bool, sampler: SamplerName, priority: i32, service: MetaString, env: MetaString,
276    ) {
277        let key = DecisionKey {
278            sampler,
279            service,
280            env,
281            priority: (sampler == SamplerName::Priority).then(|| Priority::from(priority)),
282        };
283        // A full window rolls unseen combinations into one overflow bucket per sampler, so burst
284        // cardinality cannot grow the map without bound; keys already present keep counting.
285        let counts = if self.decisions.len() >= self.max_decision_keys && !self.decisions.contains_key(&key) {
286            self.decisions.entry(self.overflow_key(sampler)).or_default()
287        } else {
288            self.decisions.entry(key).or_default()
289        };
290        counts.seen += 1;
291        counts.kept += u64::from(kept);
292    }
293
294    /// Returns the key counting decisions rolled past the window's cap.
295    ///
296    /// One bucket per sampler: the service collapses to [`OVERFLOW_SERVICE`] and the env and
297    /// priority dimensions are dropped, so the roll-up itself stays bounded.
298    fn overflow_key(&self, sampler: SamplerName) -> DecisionKey {
299        DecisionKey {
300            sampler,
301            service: MetaString::from_static(OVERFLOW_SERVICE),
302            env: MetaString::empty(),
303            priority: None,
304        }
305    }
306
307    /// Reports the window as metric events and starts a new one.
308    ///
309    /// Every decision reports its seen count, and keys that kept nothing never report a kept
310    /// series. The signature-count gauges and the rare-sampler counters are reported every window,
311    /// including quiet ones, so readers see gauges republished unchanged between active windows.
312    pub(super) fn take_window_events(
313        &mut self, priority_tracked: u64, no_priority_tracked: u64, error_tracked: u64,
314    ) -> Vec<Event> {
315        let mut events = Vec::new();
316
317        // Drain the window map in place so its capacity carries into the next window.
318        for (key, counts) in self.decisions.drain() {
319            let tags = key.tag_set();
320            events.push(Event::Metric(Metric::counter(
321                Context::from_parts(METRIC_SAMPLER_SEEN, tags.clone()),
322                counts.seen as f64,
323            )));
324            if counts.kept > 0 {
325                events.push(Event::Metric(Metric::counter(
326                    Context::from_parts(METRIC_SAMPLER_KEPT, tags),
327                    counts.kept as f64,
328                )));
329            }
330        }
331
332        events.push(gauge_event(
333            METRIC_SAMPLER_SIZE,
334            &[("sampler", SamplerName::Priority.as_str())],
335            priority_tracked as f64,
336        ));
337        events.push(gauge_event(
338            METRIC_SAMPLER_SIZE,
339            &[("sampler", SamplerName::NoPriority.as_str())],
340            no_priority_tracked as f64,
341        ));
342        events.push(gauge_event(
343            METRIC_SAMPLER_SIZE,
344            &[("sampler", SamplerName::Error.as_str())],
345            error_tracked as f64,
346        ));
347
348        let hits = self.counters.inner.rare_hits.swap(0, Ordering::Relaxed);
349        if hits > 0 {
350            events.push(Event::Metric(Metric::counter(
351                Context::from_parts(METRIC_RARE_HITS, TagSet::default()),
352                hits as f64,
353            )));
354        }
355        let misses = self.counters.inner.rare_misses.swap(0, Ordering::Relaxed);
356        if misses > 0 {
357            events.push(Event::Metric(Metric::counter(
358                Context::from_parts(METRIC_RARE_MISSES, TagSet::default()),
359                misses as f64,
360            )));
361        }
362        events.push(gauge_event(
363            METRIC_RARE_SHRINKS,
364            &[],
365            self.counters.inner.rare_shrinks.load(Ordering::Relaxed) as f64,
366        ));
367
368        events
369    }

377fn gauge_event(name: &str, tags: &[(&str, &str)], value: f64) -> Event {
378    let mut tag_set = TagSet::with_capacity(tags.len());
379    for (name, value) in tags {
380        tag_set.insert_tag(MetaString::from(format!("{}:{}", name, value)));
381    }
382    Event::Metric(Metric::gauge(Context::from_parts(name, tag_set), value))
383}

395    fn metric_with_tags(events: &[Event], name: &str, tags: &[&str]) -> bool {
396        events.iter().any(|event| {
397            let Some(metric) = event.try_as_metric() else {
398                return false;
399            };
400            metric.context().name() == name && tags.iter().all(|tag| metric.context().tags().has_tag(tag))
401        })
402    }

404    fn has_counter(events: &[Event], name: &str, tags: &[&str], value: f64) -> bool {
405        events.iter().any(|event| {
406            let Some(metric) = event.try_as_metric() else {
407                return false;
408            };
409            metric.context().name() == name
410                && tags.iter().all(|tag| metric.context().tags().has_tag(tag))
411                && *metric.values() == MetricValues::counter(value)
412        })
413    }

416        events.iter().any(|event| {
417            let Some(metric) = event.try_as_metric() else {
418                return false;
419            };
420            metric.context().name() == name
421                && tags.iter().all(|tag| metric.context().tags().has_tag(tag))
422                && *metric.values() == MetricValues::gauge(value)
423        })
424    }
