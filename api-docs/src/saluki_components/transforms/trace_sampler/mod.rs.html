54fn normalize_sampling_rate(rate: f64) -> f64 {
55    if rate <= 0.0 || rate >= 1.0 {
56        1.0
57    } else {
58        rate
59    }

189impl TraceSampler {
190    /// Find the root span index of a trace.
191    fn get_root_span_index(&self, trace: &Trace) -> Option<usize> {
192        // The shared root finder, so every consumer anchors metadata to identical root spans.
193        get_root_span_index(trace.spans())
194    }
195
196    /// Check for user-set sampling priority in trace
197    fn get_user_priority(&self, trace: &Trace, root_span_idx: usize) -> Option<i32> {
198        // First check trace-level sampling priority (last-seen priority from OTLP ingest)
199        if let Some(priority) = trace.priority {
200            return Some(priority);
201        }
202
203        if trace.spans().is_empty() {
204            return None;
205        }
206
207        // Fall back to checking spans (for compatibility with non-OTLP traces)
208        // Prefer the root span (common case), but fall back to scanning all spans to be robust to ordering.
209        if let Some(root) = trace.spans().get(root_span_idx) {
210            if let Some(p) = root
211                .attributes
212                .get(SAMPLING_PRIORITY_METRIC_KEY)
213                .and_then(AttributeValue::as_num)
214            {
215                return Some(p as i32);
216            }
217        }
218        let spans = trace.spans();
219        spans.iter().find_map(|span| {
220            span.attributes
221                .get(SAMPLING_PRIORITY_METRIC_KEY)
222                .and_then(AttributeValue::as_num)
223                .map(|p| p as i32)
224        })
225    }
226
227    /// Returns `true` if the given trace ID should be probabilistically sampled.
228    fn sample_probabilistic(&self, trace_id_high: u64, trace_id_low: u64) -> bool {
229        self.probabilistic
230            .sample(trace_id_high, trace_id_low, self.sampling_rate)
231    }

233    fn is_otlp_trace(&self, trace: &Trace, root_span_idx: usize) -> bool {
234        trace
235            .spans()
236            .get(root_span_idx)
237            .map(|span| {
238                span.attributes
239                    .contains_key(&MetaString::from_static(OTEL_TRACE_ID_META_KEY))
240            })
241            .unwrap_or(false)
242    }

244    /// Returns `true` if the trace contains a span with an error.
245    fn trace_contains_error(&self, trace: &Trace, consider_exception_span_events: bool) -> bool {
246        trace.spans().iter().any(|span| {
247            span.error() != 0 || (consider_exception_span_events && self.span_contains_exception_span_event(span))
248        })
249    }

251    /// Returns `true` if the span has exception span events.
252    ///
253    /// This checks for the `_dd.span_events.has_exception` meta field set to `"true"`.
254    fn span_contains_exception_span_event(&self, span: &Span) -> bool {
255        if let Some(has_exception) = span
256            .attributes
257            .get("_dd.span_events.has_exception")
258            .and_then(AttributeValue::as_string)
259        {
260            return has_exception == "true";
261        }
262        false
263    }

265    /// Computes the OTLP pre-sampling priority and decision maker for a trace, mirroring
266    /// `OTLPReceiver.createChunks` in DDA which runs before `runSamplersV1`.
267    ///
268    /// Returns `Some((priority, dm))` for OTLP traces when the probabilistic sampler is disabled,
269    /// or `None` if pre-sampling doesn't apply.
270    ///
271    /// See: https://github.com/DataDog/datadog-agent/blob/be33ac1490c4a34602cbc65a211406b73ad6d00b/pkg/trace/api/otlp.go#L561-L585
272    fn otlp_pre_sample(&mut self, trace: &mut Trace, root_span_idx: usize) -> Option<(i32, &'static str)> {
273        if self.probabilistic_sampler_enabled || !self.is_otlp_trace(trace, root_span_idx) {
274            return None;
275        }
276        let (priority, dm) = if let Some(user_priority) = self.get_user_priority(trace, root_span_idx) {
277            (user_priority, DECISION_MAKER_MANUAL)
278        } else {
279            let root_trace_id = trace.trace_id_low;
280            if sample_by_rate(root_trace_id, self.otlp_sampling_rate) {
281                (PRIORITY_AUTO_KEEP, DECISION_MAKER_PROBABILISTIC)
282            } else {
283                (PRIORITY_AUTO_DROP, DECISION_MAKER_PROBABILISTIC)
284            }
285        };
286        if priority == PRIORITY_AUTO_KEEP {
287            if let Some(root_span) = trace.spans_mut().get_mut(root_span_idx) {
288                root_span.attributes.remove(PROB_RATE_KEY);
289            }
290        }
291        Some((priority, dm))
292    }

294    /// Apply analyzed span sampling to the trace.
295    ///
296    /// Returns `true` if the trace was modified.
297    fn analyzed_span_sampling(&self, trace: &mut Trace) -> bool {
298        let retained = trace.retain_spans(|_, span| span.attributes.contains_key(KEY_ANALYZED_SPANS));
299        if retained > 0 {
300            trace.dropped_trace = false;
301            trace.priority = Some(PRIORITY_USER_KEEP);
302            trace.otlp_sampling_rate = Some(self.sampling_rate);
303            true
304        } else {
305            false
306        }
307    }

311        trace
312            .spans()
313            .iter()
314            .any(|span| span.attributes.contains_key(KEY_ANALYZED_SPANS))

317    /// Apply Single Span Sampling to the trace
318    /// Returns true if the trace was modified
319    fn single_span_sampling(&self, trace: &mut Trace) -> bool {
320        let retained = trace.retain_spans(|_, span| span.attributes.contains_key(KEY_SPAN_SAMPLING_MECHANISM));
321        if retained > 0 {
322            trace.dropped_trace = false;
323            trace.priority = Some(PRIORITY_USER_KEEP);
324            trace.otlp_sampling_rate = Some(self.sampling_rate);
325            true
326        } else {
327            false
328        }
329    }

331    /// Evaluates the given trace against all configured samplers.
332    ///
333    /// Return a tuple containing whether or not the trace should be kept, the decision maker tag (which sampler is responsible),
334    /// and the index of the root span used for evaluation.
335    fn run_samplers(&mut self, trace: &mut Trace) -> (bool, i32, &'static str, Option<usize>) {
336        // logic taken from: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/agent/agent.go#L1066
337        // Empty trace check
338        if trace.spans().is_empty() {
339            return (false, PRIORITY_AUTO_DROP, "", None);
340        }
341
342        let Some(root_span_idx) = self.get_root_span_index(trace) else {
343            return (false, PRIORITY_AUTO_DROP, "", None);
344        };
345
346        // ETS: only sample traces containing errors (including exception span events); skip all other samplers.
347        // logic taken from: https://github.com/DataDog/datadog-agent/blob/be33ac1490c4a34602cbc65a211406b73ad6d00b/pkg/trace/agent/agent.go#L1068
348        if self.error_tracking_standalone {
349            let otlp_pre_sample = self.otlp_pre_sample(trace, root_span_idx);
350            if self.trace_contains_error(trace, true) {
351                let now = std::time::SystemTime::now();
352                let keep = self.error_sampler.sample_error(now, trace, root_span_idx);
353                let default_priority = if keep { PRIORITY_AUTO_KEEP } else { PRIORITY_AUTO_DROP };
354                let (priority, dm) = otlp_pre_sample.unwrap_or((default_priority, ""));
355                return (keep, priority, dm, Some(root_span_idx));
356            }
357            let (pre_priority, pre_dm) = otlp_pre_sample.unwrap_or((PRIORITY_AUTO_DROP, ""));
358            return (false, pre_priority, pre_dm, Some(root_span_idx));
359        }
360
361        // Run the rare sampler early, before all other samplers. This mirrors the Go agent behavior
362        // where the rare sampler runs first to catch traces that would otherwise be dropped entirely.
363        // logic taken from: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/agent/agent.go#L1078
364        let rare = self.rare_sampler.sample(trace, root_span_idx);
365
366        // Modern path: ProbabilisticSamplerEnabled = true
367        if self.probabilistic_sampler_enabled {
368            let mut prob_keep = false;
369            let mut decision_maker = "";
370
371            if rare {
372                // Rare sampler wins over probabilistic sampling.
373                prob_keep = true;
374            } else {
375                if self.sample_probabilistic(trace.trace_id_high, trace.trace_id_low) {
376                    decision_maker = DECISION_MAKER_PROBABILISTIC;
377                    prob_keep = true;
378
379                    if let Some(root_span) = trace.spans_mut().get_mut(root_span_idx) {
380                        root_span.attributes.insert(
381                            MetaString::from_static(PROB_RATE_KEY),
382                            AttributeValue::Float(self.sampling_rate),
383                        );
384                    }
385                } else if self.error_sampling_enabled && self.trace_contains_error(trace, false) {
386                    let now = std::time::SystemTime::now();
387                    prob_keep = self.error_sampler.sample_error(now, trace, root_span_idx);
388                }
389            }
390
391            let priority = if prob_keep {
392                PRIORITY_AUTO_KEEP
393            } else {
394                PRIORITY_AUTO_DROP
395            };
396
397            return (prob_keep, priority, decision_maker, Some(root_span_idx));
398        }
399
400        // Read once here, where the samplers below consume it; every path above returns without
401        // needing it.
402        let now = std::time::SystemTime::now();
403        let user_priority = self.get_user_priority(trace, root_span_idx);
404        if let Some(priority) = user_priority {
405            if priority < PRIORITY_AUTO_DROP {
406                // Manual drop: short-circuit and skip other samplers.
407                return (false, priority, "", Some(root_span_idx));
408            }
409
410            if rare {
411                return (true, priority, "", Some(root_span_idx));
412            }
413
414            if self.priority_sampler.sample(now, trace, root_span_idx, priority, 0.0) {
415                return (true, priority, "", Some(root_span_idx));
416            }
417        } else if self.is_otlp_trace(trace, root_span_idx) {
418            // Rare check mirrors agent behavior: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/agent/agent.go#L1129-L1140
419            if rare {
420                return (true, PRIORITY_AUTO_KEEP, "", Some(root_span_idx));
421            }
422
423            // some sampling happens upstream in the otlp receiver in the agent: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/api/otlp.go#L572
424            let root_trace_id = trace.trace_id_low;
425            if sample_by_rate(root_trace_id, self.otlp_sampling_rate) {
426                if let Some(root_span) = trace.spans_mut().get_mut(root_span_idx) {
427                    root_span.attributes.remove(PROB_RATE_KEY);
428                }
429                return (
430                    true,
431                    PRIORITY_AUTO_KEEP,
432                    DECISION_MAKER_PROBABILISTIC,
433                    Some(root_span_idx),
434                );
435            }
436        } else {
437            if rare {
438                return (true, PRIORITY_AUTO_KEEP, "", Some(root_span_idx));
439            }
440            if self.no_priority_sampler.sample(now, trace, root_span_idx) {
441                return (true, PRIORITY_AUTO_KEEP, "", Some(root_span_idx));
442            }
443        }
444
445        if self.error_sampling_enabled && self.trace_contains_error(trace, false) {
446            let keep = self.error_sampler.sample_error(now, trace, root_span_idx);
447            if keep {
448                return (true, PRIORITY_AUTO_KEEP, "", Some(root_span_idx));
449            }
450        }
451
452        // Default: drop the trace
453        (false, PRIORITY_AUTO_DROP, "", Some(root_span_idx))
454    }

456    /// Fills in trace-level metadata from the spans.
457    ///
458    /// Origin comes from the root span's `_dd.origin` tag, and the decision maker is the first
459    /// span in chunk order carrying `_dd.p.dm` - both only when not already set. This runs here,
460    /// downstream of span filtering, so the metadata anchors to the final span population rather
461    /// than the one that arrived at decode time.
462    fn backfill_trace_metadata(&self, trace: &mut Trace, root_span_idx: usize) {
463        if trace.origin.is_empty() {
464            if let Some(origin) = trace.spans()[root_span_idx]
465                .attributes
466                .get(TAG_ORIGIN)
467                .and_then(AttributeValue::as_string)
468            {
469                trace.origin = origin.clone();
470            }
471        }
472
473        if trace.decision_maker.is_none() {
474            let first_decision_maker = trace.spans().iter().find_map(|span| {
475                span.attributes
476                    .get(TAG_DECISION_MAKER)
477                    .and_then(AttributeValue::as_string)
478            });
479            if let Some(dm) = first_decision_maker {
480                trace.decision_maker = Some(dm.clone());
481            }
482        }
483
484        if !self.compute_top_level_by_span_kind {
485            compute_top_level(trace.spans_mut());
486        }
487    }

489    /// Apply sampling metadata to the trace in-place.
490    ///
491    /// The `root_span_id` parameter identifies which span should receive the sampling metadata.
492    /// This avoids recalculating the root span since it was already found in `run_samplers`.
493    fn apply_sampling_metadata(
494        &self, trace: &mut Trace, keep: bool, priority: i32, decision_maker: &'static str, root_span_idx: usize,
495    ) {
496        let is_otlp = self.is_otlp_trace(trace, root_span_idx);
497        // Chunk-level decision maker: the sampler's value when it decided, otherwise the first
498        // carrier promoted during backfill (falling back to the root's own tag).
499        let promoted_decision_maker = if decision_maker.is_empty() {
500            trace.decision_maker.clone()
501        } else {
502            None
503        };
504        let root_span_value = match trace.spans_mut().get_mut(root_span_idx) {
505            Some(span) => span,
506            None => return,
507        };
508
509        let decision_maker_meta = if decision_maker.is_empty() {
510            promoted_decision_maker.or_else(|| {
511                root_span_value
512                    .attributes
513                    .get(TAG_DECISION_MAKER)
514                    .and_then(AttributeValue::as_string)
515                    .cloned()
516            })
517        } else {
518            Some(MetaString::from_static(decision_maker))
519        };
520
521        // The span-level decision maker is stamped only when a sampler made the call: promotion
522        // targets trace chunk metadata, and each span's own tag is left in place because
523        // downstream systems rely on the span keeping its original value.
524        //
525        // With the APM-level probabilistic sampler on OTLP traces, `_dd.p.dm` is written to
526        // trace chunk tags only (not span meta); on the legacy sampling path it is written to
527        // both. The span meta write is skipped only when both conditions hold; the DM value
528        // still flows through trace fields to the encoder.
529        if priority > 0 && !decision_maker.is_empty() && !(is_otlp && self.probabilistic_sampler_enabled) {
530            root_span_value.attributes.insert(
531                MetaString::from_static(TAG_DECISION_MAKER),
532                AttributeValue::String(MetaString::from_static(decision_maker)),
533            );
534        }
535
536        // Now set sampling metadata directly on the trace.
537        trace.dropped_trace = !keep;
538        trace.priority = Some(priority);
539        trace.decision_maker = if priority > 0 { decision_maker_meta } else { None };
540        trace.otlp_sampling_rate = Some(if is_otlp {
541            self.otlp_sampling_rate
542        } else {
543            self.sampling_rate
544        });
545    }

547    fn process_trace(&mut self, trace: &mut Trace) -> bool {
548        // keep is a boolean that indicates if the trace should be kept or dropped
549        // priority is the sampling priority
550        // decision_maker is the tag that indicates the decision maker (probabilistic, error, etc.)
551        // root_span_idx is the index of the root span of the trace
552        let (keep, priority, decision_maker, root_span_idx) = self.run_samplers(trace);
553
554        // Backfill trace-derived metadata before any forwarding path (keep, ETS, or
555        // single-span/analytics sampling), so every forwarded trace carries metadata derived
556        // from its final span population.
557        if let Some(root_idx) = root_span_idx {
558            self.backfill_trace_metadata(trace, root_idx);
559        }
560
561        // Apply sampling metadata and forward if kept, or if ETS (dropped non-error traces are
562        // forwarded with DroppedTrace=true, suppressing SSS/analytics).
563        if keep || self.error_tracking_standalone {
564            if let Some(root_idx) = root_span_idx {
565                self.apply_sampling_metadata(trace, keep, priority, decision_maker, root_idx);
566            }
567            return true;
568        }
569
570        // logic taken from here: https://github.com/DataDog/datadog-agent/blob/main/pkg/trace/agent/agent.go#L980-L990
571        // try single span sampling (keeps spans marked for sampling when trace would be dropped)
572        let modified = self.single_span_sampling(trace);
573        if !modified {
574            // Fall back to analytics events if no SSS spans
575            if self.analyzed_span_sampling(trace) {
576                return true;
577            }
578        } else if self.has_analyzed_spans(trace) {
579            // Warn about both SSS and analytics events
580            debug!(
581                "Detected both analytics events AND single span sampling in the same trace. Single span sampling wins because App Analytics is deprecated."
582            );
583            return true;
584        }
585
586        // If we modified the trace with SSS, send it
587        if modified {
588            return true;
589        }
590
591        // Neither SSS nor analytics events found, drop the trace
592        debug!("Dropping trace with priority {}", priority);
593        false
594    }
595}

996        // buckets and keeps the trace when `bucket < (rate * 0x4000)`. These cases pin the exact keep/drop decision
997        // for known trace IDs at known rates, so a regression in the hash, the bucket mask, or the comparison is
998        // caught (a determinism-only check would not catch any of those).
999        //
1000        // Expected values were computed directly from the FNV-1a bucket math in `ProbabilisticSampler::sample`.
1001        // For reference, the trace IDs below hash to these buckets (out of 0x4000 = 16384):
1002        // 0x1234567890ABCDEF -> 1764, 0x0 -> 9301, u64::MAX -> 12365.
1003        struct Case {
1004            trace_id: u64,
1005            rate: f64,
1006            expected_keep: bool,
1007        }
1008
1009        let cases = [
1010            // rate 1.0 keeps every trace (the maximum bucket, 16383, is always below 16384).
1011            Case {
1012                trace_id: 0x1234567890ABCDEF,
1013                rate: 1.0,
1014                expected_keep: true,
1015            },
1016            // rate 0.0 drops every trace (no bucket is below 0).
1017            Case {
1018                trace_id: 0x1234567890ABCDEF,
1019                rate: 0.0,
1020                expected_keep: false,
1021            },
1022            // Same trace ID (bucket 1764) flips from drop to keep as the rate crosses its bucket ratio (~0.108).
1023            Case {
1024                trace_id: 0x1234567890ABCDEF,
1025                rate: 0.10,
1026                expected_keep: false,
1027            },
1028            Case {
1029                trace_id: 0x1234567890ABCDEF,
1030                rate: 0.20,
1031                expected_keep: true,
1032            },
1033            // Trace ID 0 (bucket 9301) straddles rate 0.5 (scaled bucket 8192) vs 0.6 (scaled bucket 9830).
1034            Case {
1035                trace_id: 0,
1036                rate: 0.50,
1037                expected_keep: false,
1038            },
1039            Case {
1040                trace_id: 0,
1041                rate: 0.60,
1042                expected_keep: true,

1048                expected_keep: false,
1049            },
1050            Case {
1051                trace_id: u64::MAX,

1053                expected_keep: true,
1054            },
