1//! Rare sampler for traces.
2//!
3//! Samples traces for span signature combinations (env, service, name, resource, error type, HTTP status)
4//! that aren't caught by the priority sampler. This ensures that rare or low-traffic trace shapes
5//! are still represented in the sampled data.
6//!
7//! The sampler works by:
8//! 1. Iterating top-level and measured spans in the trace.
9//! 2. Computing a per-span signature (hashed from service, name, resource, error, etc.).
10//! 3. Keeping the trace if any span's signature hasn't been seen within the cooldown TTL.
11//! 4. Using a token bucket to cap the overall rate of rare traces kept.
12//!
13//! Mirrors `datadog-agent/pkg/trace/sampler/rare_sampler.go`.
14

25/// The burst size for the token bucket rate limiter. Matches the Go agent default.
26const RARE_SAMPLER_BURST: usize = 50;
27
28/// Minimum time between TTL updates for an already-seen span signature.
29/// Avoids churning the map when the same signature is seen repeatedly within the TTL window.
30const TTL_RENEWAL_PERIOD: Duration = Duration::from_secs(60);
31
32/// Metric key set on the root span of a rare-sampled trace.
33pub(super) const RARE_KEY: &str = "_dd.rare";
34
35/// Metric key indicating a span is a top-level span (legacy, set by the agent).
36const KEY_TOP_LEVEL: &str = "_top_level";
37
38/// Metric key indicating a span is a top-level span, set directly by tracers/SDKs.
39///
40/// Modern Datadog tracers set this key instead of (or in addition to) `_top_level`. The Go agent
41/// normalizes it into `_top_level` via `UpdateTracerTopLevel` when `ClientComputedTopLevel` is
42/// true, but ADP doesn't run that normalization pass, so we check both keys directly.
43const KEY_TRACER_TOP_LEVEL: &str = "_dd.top_level";
44
45/// Metric key indicating a span is explicitly marked for stats computation.
46const KEY_MEASURED: &str = "_dd.measured";
47
48/// Tracks the set of span signatures seen within a single (env, service) shard.
49struct SeenSpans {
50    /// Maps span hash to the expiry `Instant` for that signature.
51    expires: FastHashMap<u32, Instant>,
52    /// Whether the map has been shrunk due to cardinality overflow.
53    shrunk: bool,
54    /// Maximum number of entries before triggering a shrink.
55    cardinality: usize,
56    telemetry: SamplerCounters,
57}
58
59impl SeenSpans {
60    fn new(cardinality: usize, telemetry: SamplerCounters) -> Self {
61        Self {
62            expires: FastHashMap::default(),
63            shrunk: false,
64            cardinality,
65            telemetry,
66        }
67    }
68
69    /// Compute the effective signature for a raw span hash, applying the shrink modulus if needed.
70    fn sign(&self, span_hash: u32) -> u32 {
71        if self.shrunk {
72            span_hash % self.cardinality as u32
73        } else {
74            span_hash
75        }
76    }
77
78    /// Record an expiry for a span signature.
79    ///
80    /// Skips the update if the stored entry is still live and the new expiry isn't meaningfully
81    /// later (within `TTL_RENEWAL_PERIOD`). If the stored entry is already expired, always updates.
82    /// This mirrors the Go agent's `ttlRenewalPeriod` check, which assumes `TTL > TTL_RENEWAL_PERIOD`.
83    fn add(&mut self, now: Instant, expire: Instant, span_hash: u32) {
84        let sig = self.sign(span_hash);
85        if let Some(&stored) = self.expires.get(&sig) {
86            // Only skip if the stored entry is still live and the new expiry isn't meaningfully later.
87            if stored > now && expire.duration_since(stored) < TTL_RENEWAL_PERIOD {
88                return;
89            }
90        }
91        self.expires.insert(sig, expire);
92        if self.expires.len() > self.cardinality {
93            self.shrink();
94        }
95    }
96
97    /// Returns the stored expiry for a span signature, if any.
98    fn get_expire(&self, sig: u32) -> Option<&Instant> {
99        self.expires.get(&sig)
100    }
101
102    /// Shrink the map to cap cardinality. Signatures are collapsed into `cardinality` buckets via
103    /// modular hashing. Matches the Go agent's shrink behavior.
104    fn shrink(&mut self) {
105        self.telemetry.record_rare_shrink();
106        let cardinality = self.cardinality;
107        let old = std::mem::take(&mut self.expires);
108        self.expires.reserve(cardinality);
109        for (h, expire) in old {
110            self.expires.insert(h % cardinality as u32, expire);
111        }
112        self.shrunk = true;
113    }
114}

116/// Rare sampler: keeps traces whose span signatures haven't been seen within the cooldown TTL.
117///
118/// Traces that pass through the rare sampler have `_dd.rare = 1` set on the first matching span.
119pub(super) struct RareSampler {
120    enabled: bool,
121    token_bucket: TokenBucket,
122    ttl: Duration,
123    cardinality: usize,
124    /// Keyed by (env, service) shard signature.
125    seen: FastHashMap<Signature, SeenSpans>,
126    telemetry: SamplerCounters,
127}
128
129/// The result of running the rare sampler over a trace.
130enum RareOutcome {
131    /// A new or expired signature was kept by the limiter.
132    Kept,
133    /// A new or expired signature was denied by the limiter.
134    Missed,
135    /// No span carried a new or expired signature: steady-state traffic.
136    NoCandidate,
137}
138
139impl RareSampler {
140    pub(super) fn new(enabled: bool, tps: f64, ttl: Duration, cardinality: usize, telemetry: SamplerCounters) -> Self {
141        Self {
142            enabled,
143            token_bucket: TokenBucket::new(tps, RARE_SAMPLER_BURST),
144            ttl,
145            cardinality,
146            seen: FastHashMap::default(),
147            telemetry,
148        }
149    }
150
151    /// Sample a trace. Returns `true` if the trace should be kept by the rare sampler.
152    ///
153    /// Iterates top-level and measured spans. If any span has a signature that hasn't been seen
154    /// within the TTL, the sampler attempts to consume a token and keep the trace. Only a genuine
155    /// limiter denial counts as a miss: traces with no new or expired signature are steady-state
156    /// traffic and move neither counter.
157    pub(super) fn sample(&mut self, trace: &mut Trace, root_span_idx: usize) -> bool {
158        if !self.enabled {
159            return false;
160        }
161        match self.handle_trace(trace, root_span_idx) {
162            RareOutcome::Kept => {
163                self.telemetry.record_rare_hit();
164                true
165            }
166            RareOutcome::Missed => {
167                self.telemetry.record_rare_miss();
168                false
169            }
170            RareOutcome::NoCandidate => false,
171        }
172    }
173
174    fn handle_trace(&mut self, trace: &mut Trace, root_span_idx: usize) -> RareOutcome {
175        let now = Instant::now();
176        let env = get_trace_env(trace, root_span_idx).map(|e| e.as_ref()).unwrap_or("");
177
178        let Some(sampled_idx) = self.find_rare_span(trace, env, now) else {
179            return RareOutcome::NoCandidate;
180        };
181
182        if !self.token_bucket.allow() {
183            return RareOutcome::Missed;
184        }
185
186        // Update TTLs first (last use of env — NLL ends the borrow of trace here).
187        self.record_all_top_level_spans(trace, env, now, now + self.ttl);
188
189        // Now safe to mutably borrow trace.
190        if let Some(span) = trace.spans_mut().get_mut(sampled_idx) {
191            span.attributes
192                .insert(MetaString::from_static(RARE_KEY), AttributeValue::Float(1.0));
193        }
194
195        RareOutcome::Kept
196    }
197
198    /// Find the index of the first top-level or measured span whose signature has expired or is new.
199    fn find_rare_span(&mut self, trace: &Trace, env: &str, now: Instant) -> Option<usize> {
200        let spans = trace.spans();
201        for (i, span) in spans.iter().enumerate() {
202            if !is_top_level_or_measured(span) {
203                continue;
204            }
205            let shard_sig = ServiceSignature::new(span.service(), env).hash();
206            let span_hash = span_hash_for_rare(span);
207            let seen = self
208                .seen
209                .entry(shard_sig)
210                .or_insert_with(|| SeenSpans::new(self.cardinality, self.telemetry.clone()));
211            let sig = seen.sign(span_hash);
212            let expired = seen.get_expire(sig).is_none_or(|expire| now > *expire);
213            if expired {
214                return Some(i);
215            }
216        }
217        None
218    }
219
220    fn record_all_top_level_spans(&mut self, trace: &Trace, env: &str, now: Instant, expire: Instant) {
221        for span in trace.spans() {
222            if !is_top_level_or_measured(span) {
223                continue;
224            }
225            let shard_sig = ServiceSignature::new(span.service(), env).hash();
226            let span_hash = span_hash_for_rare(span);
227            let seen = self
228                .seen
229                .entry(shard_sig)
230                .or_insert_with(|| SeenSpans::new(self.cardinality, self.telemetry.clone()));
231            seen.add(now, expire, span_hash);
232        }
233    }
234}

236/// Returns `true` if the span is top-level or measured.
237///
238/// Checks `_top_level` (agent-set), `_dd.top_level` (tracer-set), and `_dd.measured`, mirroring
239/// `HasTopLevel` + `IsMeasured` in the Go agent's `traceutil` package.
240fn is_top_level_or_measured(span: &Span) -> bool {
241    span.attributes
242        .get(KEY_TOP_LEVEL)
243        .and_then(AttributeValue::as_num)
244        .is_some_and(|v| v == 1.0)
245        || span
246            .attributes
247            .get(KEY_TRACER_TOP_LEVEL)
248            .and_then(AttributeValue::as_num)
249            .is_some_and(|v| v == 1.0)
250        || span
251            .attributes
252            .get(KEY_MEASURED)
253            .and_then(AttributeValue::as_num)
254            .is_some_and(|v| v == 1.0)
255}

268    fn make_span_with_metrics(
269        service: &str, name: &str, resource: &str, metrics: FastHashMap<MetaString, f64>,
270    ) -> DdSpan {
271        let attrs: FastHashMap<MetaString, AttributeValue> = metrics
272            .into_iter()
273            .map(|(k, v)| (k, AttributeValue::Float(v)))
274            .collect();
275        DdSpan::new(
276            MetaString::from(service),
277            MetaString::from(name),
278            MetaString::from(resource),
279            MetaString::from("web"),
280            1,
281            0,
282            0,
283            1000,
284            0,
285        )
286        .with_attributes(attrs)
287    }

290        let mut metrics = FastHashMap::default();

292        make_span_with_metrics(service, name, resource, metrics)
293    }

375    #[test]
376    fn rate_limit_drops_excess_rare_traces() {
377        // Use a very low TPS so the bucket doesn't refill during the loop, then create 60 distinct
378        // signatures to exceed the burst of 50. We're testing the burst cap, not the refill rate.
379        let mut sampler = RareSampler::new(
380            true,
381            0.000_000_001,
382            Duration::from_secs(300),
383            200,
384            SamplerCounters::new(),
385        );
386
387        let mut kept = 0usize;
388        for i in 0..60usize {
389            // Each trace has a unique resource so they're all distinct signatures.
390            let mut trace = make_trace(vec![make_top_level_span("svc", "op", &format!("res-{}", i))]);
391            if sampler.sample(&mut trace, 0) {
392                kept += 1;
393            }
394        }
395
396        // We have a burst of 50, so exactly 50 should be kept.
397        assert_eq!(kept, 50);
398    }

464    fn multiple_top_level_spans_rare_flag_on_first_new_signature() {
465        // Mirrors TestMultipleTopeLevels from the Go agent exactly.
466        // r1 and r2 are in the same service shard.

471        // Trace 3 [r1]:       r1 TTL was refreshed by trace 2 → dropped.
472        let mut sampler = RareSampler::new(true, 100.0, Duration::from_secs(300), 200, SamplerCounters::new());

478                .attributes
479                .get(RARE_KEY)

481            Some(1.0)
482        );
483
484        let mut trace2 = make_trace(vec![
485            make_top_level_span("s1", "op", "r1"),
486            make_top_level_span("s1", "op", "r2"),
