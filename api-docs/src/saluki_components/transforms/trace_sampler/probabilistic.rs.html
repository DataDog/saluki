1//! Probabilistic sampling.
2
3use super::signature::{fnv1a_32_continue, fnv1a_32_start};
4
5// The trace ID is hashed into one of 16384 buckets, and a trace is kept when its bucket number is
6// below the sampling rate scaled to the same range.
7const NUM_PROBABILISTIC_BUCKETS: u32 = 0x4000;
8const BITMASK_HASH_BUCKETS: u32 = NUM_PROBABILISTIC_BUCKETS - 1;
9
10/// Sampling-rate span attribute written to the root span of kept traces.
11pub(super) const PROB_RATE_KEY: &str = "_dd.prob_sr";
12
13/// Bucketed probabilistic sampler.
14pub(super) struct ProbabilisticSampler {
15    /// FNV-1a-32 state with the seed folded in. The seed prefix is constant per instance, so it is
16    /// hashed once at construction rather than on every trace.
17    hash_state: u32,
18    full_trace_id_mode: bool,
19}
20
21impl ProbabilisticSampler {
22    pub(super) fn new(hash_seed: u32, full_trace_id_mode: bool) -> Self {
23        Self {
24            hash_state: fnv1a_32_start(&hash_seed.to_le_bytes()),
25            full_trace_id_mode,
26        }
27    }
28
29    /// Deterministically samples a trace from its trace ID.
30    pub(super) fn sample(&self, trace_id_high: u64, trace_id_low: u64, sampling_rate: f64) -> bool {
31        // The rate bounds decide without hashing: the highest bucket, 16383, clears the 16384
32        // cutoff a rate of 1.0 scales to, and no bucket clears a cutoff of 0.
33        if sampling_rate >= 1.0 {
34            return true;
35        }
36        if sampling_rate <= 0.0 {
37            return false;
38        }
39
40        // The hash input is the 16-byte big-endian trace ID. Legacy mode hashes only the low
41        // half, zero-padded, so a trace keeps its bucket when the upper bits are lost in
42        // propagation; full mode hashes both halves, staying aligned with samplers that always
43        // decide on 128-bit IDs.
44        let trace_id = if self.full_trace_id_mode {
45            (u128::from(trace_id_high) << 64) | u128::from(trace_id_low)
46        } else {
47            u128::from(trace_id_low) << 64
48        };
49
50        let hash = fnv1a_32_continue(self.hash_state, &trace_id.to_be_bytes());
51        let scaled_sampling_percentage = (sampling_rate * NUM_PROBABILISTIC_BUCKETS as f64) as u32;
52        (hash & BITMASK_HASH_BUCKETS) < scaled_sampling_percentage
53    }
54}
