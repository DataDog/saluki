1//! APM event extraction.
2//!
3//! Promotes spans of a trace to standalone events the backend indexes even when the trace itself
4//! is dropped. A chain of extractors nominates spans, a per-event rate flip thins the nominees,
5//! and a decaying events-per-second budget caps the survivors. Survivors are marked with the
6//! analyzed key and their sampling-rate metadata whether the trace is kept or dropped; callers
7//! forward survivors as standalone events only when the trace is dropped.
8

25/// Sampling-rate metadata stamped on extracted events. Values below 1 are written; values of 1
26/// or above are omitted because the backend assumes 1.0 for absent keys.
27const KEY_EVENT_EXTRACTION_RATE: &str = "_dd1.sr.eausr";
28const KEY_MAX_EPS_RATE: &str = "_dd1.sr.eamax";
29const KEY_CLIENT_RATE: &str = "_dd1.sr.rcusr";
30const KEY_PRE_SAMPLE_RATE: &str = "_dd1.sr.rapre";
31
32const METRIC_EVENTS_EXTRACTED: &str = "datadog.trace_agent.receiver.events_extracted";
33const METRIC_EVENTS_SAMPLED: &str = "datadog.trace_agent.receiver.events_sampled";
34const METRIC_MAX_EPS_MAX_RATE: &str = "datadog.trace_agent.events.max_eps.max_rate";
35const METRIC_MAX_EPS_CURRENT_RATE: &str = "datadog.trace_agent.events.max_eps.current_rate";
36const METRIC_MAX_EPS_SAMPLE_RATE: &str = "datadog.trace_agent.events.max_eps.sample_rate";
37const METRIC_MAX_EPS_REACHED_MAX: &str = "datadog.trace_agent.events.max_eps.reached_max";

39/// Fraction of the rate-estimate score kept at each decay step.
40///
41/// Each second the score is divided by this factor, so an event's contribution fades by 12.5%
42/// per second and totals 9 points of score over its fading lifetime. That total makes
43/// [`COUNT_SCALE_FACTOR`] the score-to-events-per-second divisor, and the decay gives the
44/// estimate a ~6 second half-life: long enough to smooth bursts, short enough to follow real
45/// rate changes within one report window. A fixed behavior constant: changing it changes the
46/// budget's response curve and its verdict sequence.
47const DECAY_FACTOR: f64 = 1.125;
48
49/// Score-to-events-per-second divisor.
50///
51/// The largest lifetime score contribution of a single event, so dividing the score by this
52/// yields a lower bound on the rate; multiplying by [`DECAY_FACTOR`] yields the upper bound the
53/// budget decides against, erring toward sampling less.
54const COUNT_SCALE_FACTOR: f64 = 9.0;
55
56/// Extracted and sampled event counts for one trace.
57#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
58pub(super) struct EventCounts {
59    /// Events that passed the extraction rate flip.
60    pub(super) extracted: u64,
61    /// Events that also passed the events-per-second budget.
62    pub(super) sampled: u64,
63}

65/// Extracts and samples events from traces.
66///
67/// Cloning shares the extractor chain, the budget, and the event metrics, so the transform and
68/// its window reports observe the same state.

70pub(super) struct EventProcessor {

76    limiter: MaxEpsLimiter,

80impl EventProcessor {
81    /// Creates an event processor from the resolved analytics configuration.
82    ///
83    /// Spans carrying their own extraction rate are always nominated first; the configured maps
84    /// follow, with per-(service, operation) rates taking precedence over the legacy per-service
85    /// rates. At most one configured extractor is active.
86    pub(super) fn new(
87        analyzed_spans_by_service: &HashMap<String, HashMap<String, f64>>,
88        analyzed_rate_by_service: &HashMap<String, f64>, max_events_per_second: f64,
89    ) -> Self {
90        let mut extractors = vec![Extractor::Metric];
91        if !analyzed_spans_by_service.is_empty() {
92            // Keys are lowercased once here, so mixed-case configuration still matches the
93            // lowercased span lookups.
94            let rates_by_service = analyzed_spans_by_service
95                .iter()
96                .map(|(service, operations)| {
97                    let operations = operations
98                        .iter()
99                        .map(|(operation, rate)| (operation.to_lowercase(), *rate))
100                        .collect::<HashMap<String, f64>>();
101                    (service.to_lowercase(), operations)
102                })
103                .collect::<HashMap<String, HashMap<String, f64>>>();
104            extractors.push(Extractor::FixedRate(rates_by_service));
105        } else if !analyzed_rate_by_service.is_empty() {
106            warn!("analyzed_rate_by_service is deprecated, please use analyzed_spans instead");
107            let rates_by_service = analyzed_rate_by_service
108                .iter()
109                .map(|(service, rate)| (service.to_lowercase(), *rate))
110                .collect::<HashMap<String, f64>>();
111            extractors.push(Extractor::Legacy(rates_by_service));
112        }
113
114        Self {
115            inner: Arc::new(EventProcessorInner {
116                extractors,
117                limiter: MaxEpsLimiter::new(max_events_per_second),
118                telemetry: EventTelemetry::default(),
119            }),
120        }
121    }
122
123    /// Creates an event processor for tests, with the given rates and budget.
124    #[cfg(test)]
125    pub(super) fn for_tests(
126        analyzed_spans_by_service: &HashMap<String, HashMap<String, f64>>,
127        analyzed_rate_by_service: &HashMap<String, f64>, max_events_per_second: f64,
128    ) -> Self {
129        Self::new(
130            analyzed_spans_by_service,
131            analyzed_rate_by_service,
132            max_events_per_second,
133        )
134    }

136    /// Extracts and samples events from a trace, marking the surviving spans.
137    ///
138    /// Survivors carry the analyzed marker and their sampling-rate metadata whether the trace is
139    /// kept or dropped, so kept traces forward the marks with their full span payload. The
140    /// returned counts report how many events survived each flip.
141    pub(super) fn process(&self, trace: &mut Trace, root_span_idx: Option<usize>, priority: i32) -> EventCounts {
142        let trace_id = trace.trace_id_low;
143
144        // Rate metadata is copied from the root span; an absent rate reads as 1.0 and is not
145        // written back.
146        let (client_rate, pre_sample_rate) = root_span_idx
147            .and_then(|idx| trace.spans().get(idx))
148            .map(|root| {
149                (
150                    rate_attribute(root, KEY_CLIENT_RATE),
151                    rate_attribute(root, KEY_PRE_SAMPLE_RATE),
152                )
153            })
154            .unwrap_or((1.0, 1.0));
155
156        let mut counts = EventCounts::default();
157        for span in trace.spans_mut() {
158            // The first extractor to nominate a span decides its extraction rate.
159            let Some(extraction_rate) = self.inner.extractors.iter().find_map(|e| e.extract(span, priority)) else {
160                continue;
161            };
162            if !sample_by_rate(trace_id, extraction_rate) {
163                continue;
164            }
165            counts.extracted += 1;
166
167            // The budget flip hashes the same trace ID as the extraction flip, so the two
168            // compose to the stricter of their rates rather than multiplying: every event of a
169            // trace shares one coherent verdict.
170            let (sampled, eps_rate) = self.inner.limiter.sample(trace_id, priority);
171            if !sampled {
172                continue;
173            }
174            counts.sampled += 1;
175            mark_event_span(span, extraction_rate, eps_rate, client_rate, pre_sample_rate);
176        }
177
178        self.inner.telemetry.record_extracted(counts.extracted);
179        self.inner.telemetry.record_sampled(counts.sampled);
180        counts
181    }
182
183    /// Reports the event telemetry window as metric events.
184    ///
185    /// The transform's window tick calls this, so the event metrics share the sampler's report
186    /// cadence and flow through the same metrics output.
187    pub(super) fn take_window_events(&self) -> Vec<Event> {
188        self.inner.telemetry.take_window_events(&self.inner.limiter)
189    }
190}

192/// Reads a sampling-rate attribute, defaulting to 1.0.
193fn rate_attribute(span: &Span, key: &'static str) -> f64 {
194    span.attributes.get(key).and_then(AttributeValue::as_num).unwrap_or(1.0)
195}
196
197/// Marks a span as a sampled event and stamps its sampling-rate metadata.
198fn mark_event_span(span: &mut Span, extraction_rate: f64, eps_rate: f64, client_rate: f64, pre_sample_rate: f64) {
199    span.attributes
200        .insert(MetaString::from_static(KEY_ANALYZED_SPANS), AttributeValue::Float(1.0));
201    set_rate_below_one(&mut span.attributes, KEY_EVENT_EXTRACTION_RATE, extraction_rate);
202    set_rate_below_one(&mut span.attributes, KEY_MAX_EPS_RATE, eps_rate);
203    set_rate_below_one(&mut span.attributes, KEY_CLIENT_RATE, client_rate);
204    set_rate_below_one(&mut span.attributes, KEY_PRE_SAMPLE_RATE, pre_sample_rate);
205}
206
207/// Writes a rate below 1 and removes a rate of 1 or above, so absent keys read as 1.0.
208fn set_rate_below_one(attributes: &mut FastHashMap<MetaString, AttributeValue>, key: &'static str, rate: f64) {
209    if rate < 1.0 {
210        attributes.insert(MetaString::from_static(key), AttributeValue::Float(rate));
211    } else {
212        attributes.remove(key);
213    }
214}

216/// Nominates spans for event extraction.
217enum Extractor {
218    /// Nominates spans carrying their own extraction rate, whatever their service or operation.
219    Metric,
220    /// Nominates spans by configured (service, operation) extraction rates.
221    FixedRate(HashMap<String, HashMap<String, f64>>),
222    /// Nominates top-level spans by configured per-service extraction rates.
223    Legacy(HashMap<String, f64>),
224}
225
226impl Extractor {
227    /// Returns the extraction rate for a span this extractor nominates, or `None`.
228    fn extract(&self, span: &Span, priority: i32) -> Option<f64> {
229        match self {
230            Self::Metric => span
231                .attributes
232                .get(KEY_EVENT_EXTRACTION_RATE)
233                .and_then(AttributeValue::as_num)
234                .map(|rate| upscale_user_keep(rate, priority)),
235            Self::FixedRate(rates_by_service) => {
236                let service = span.service().to_lowercase();
237                let operation = span.name().to_lowercase();
238                rates_by_service
239                    .get(service.as_str())?
240                    .get(operation.as_str())
241                    .map(|rate| upscale_user_keep(*rate, priority))
242            }
243            Self::Legacy(rates_by_service) => {
244                // Only top-level spans are nominated, and legacy rates apply verbatim: this path
245                // predates user-keep handling and takes no upscaling.
246                let top_level = span
247                    .attributes
248                    .get(KEY_TOP_LEVEL)
249                    .and_then(AttributeValue::as_num)
250                    .is_some_and(|value| value == 1.0);
251                if !top_level {
252                    return None;
253                }
254                let service = span.service().to_lowercase();
255                rates_by_service.get(service.as_str()).copied()
256            }
257        }
258    }
259}
260
261/// Raises a positive extraction rate to 1 for manually kept traces, whose events the customer
262/// explicitly asked to keep.
263fn upscale_user_keep(rate: f64, priority: i32) -> f64 {
264    if rate > 0.0 && priority >= PRIORITY_USER_KEEP {
265        1.0
266    } else {
267        rate
268    }
269}

271/// The decaying events-per-second budget.
272///
273/// The budget keeps a score that gains one point per event decided and loses 12.5% per second,
274/// making it a smoothed estimate of the recent event rate rather than a windowed count. Events of
275/// manually kept traces bypass the budget and do not count against it, so they cannot squeeze
276/// other events out of the budget.
277struct MaxEpsLimiter {
278    max_eps: f64,
279    state: Mutex<LimiterState>,
280}
281
282struct LimiterState {
283    score: f64,
284    last_decay: Instant,
285}
286
287impl MaxEpsLimiter {
288    fn new(max_eps: f64) -> Self {
289        Self {
290            max_eps,
291            state: Mutex::new(LimiterState {
292                score: 0.0,
293                last_decay: Instant::now(),
294            }),
295        }
296    }
297
298    /// Decides whether an event fits the budget, returning the applied rate.
299    ///
300    /// The event counts against the budget before the rate is computed, so the estimate includes
301    /// the event being decided.
302    fn sample(&self, trace_id: u64, priority: i32) -> (bool, f64) {
303        if priority >= PRIORITY_USER_KEEP {
304            return (true, 1.0);
305        }
306
307        let mut state = self.state.lock().unwrap();
308        state.decay();
309        state.score += 1.0;
310        let rate = self.sample_rate_for(state.score);
311        (sample_by_rate(trace_id, rate), rate)
312    }
313
314    /// Returns the rates reported by the budget gauges.
315    fn snapshot(&self) -> BudgetSnapshot {
316        let mut state = self.state.lock().unwrap();
317        state.decay();
318        let sample_rate = self.sample_rate_for(state.score);
319        BudgetSnapshot {
320            max_rate: self.max_eps,
321            current_rate: (state.score / COUNT_SCALE_FACTOR) * DECAY_FACTOR,
322            sample_rate,
323        }
324    }
325
326    /// Rate applied at the current score, capped by the budget.
327    fn sample_rate_for(&self, score: f64) -> f64 {
328        let current_rate = (score / COUNT_SCALE_FACTOR) * DECAY_FACTOR;
329        if current_rate > self.max_eps {
330            self.max_eps / current_rate
331        } else {
332            1.0
333        }
334    }
335}
336
337impl LimiterState {
338    /// Decays the score by the whole seconds elapsed since the last step.
339    fn decay(&mut self) {
340        let elapsed = self.last_decay.elapsed();
341        let ticks = elapsed.as_secs();
342        if ticks > 0 {
343            self.score /= DECAY_FACTOR.powi(ticks.min(u32::MAX as u64) as i32);
344            self.last_decay += Duration::from_secs(ticks);
345        }
346    }
347}
348
349/// The budget rates at report time.
350struct BudgetSnapshot {
351    max_rate: f64,
352    current_rate: f64,
353    sample_rate: f64,
354}

356/// The event metrics.
357///
358/// Extracted and sampled counts are window deltas, accumulated between reports and reset on each
359/// one; the budget gauges report the current state every window.
360#[derive(Default)]
361struct EventTelemetry {
362    extracted_delta: AtomicU64,
363    sampled_delta: AtomicU64,
364}
365
366impl EventTelemetry {
367    fn record_extracted(&self, extracted: u64) {
368        self.extracted_delta.fetch_add(extracted, Ordering::Relaxed);
369    }
370
371    fn record_sampled(&self, sampled: u64) {
372        self.sampled_delta.fetch_add(sampled, Ordering::Relaxed);
373    }
374
375    /// Reports the window deltas and the budget gauges as metric events.
376    fn take_window_events(&self, limiter: &MaxEpsLimiter) -> Vec<Event> {
377        let mut events = Vec::new();
378
379        let extracted = self.extracted_delta.swap(0, Ordering::Relaxed);
380        if extracted > 0 {
381            events.push(counter_event(METRIC_EVENTS_EXTRACTED, extracted));
382        }
383        let sampled = self.sampled_delta.swap(0, Ordering::Relaxed);
384        if sampled > 0 {
385            events.push(counter_event(METRIC_EVENTS_SAMPLED, sampled));
386        }
387
388        let snapshot = limiter.snapshot();
389        events.push(gauge_event(METRIC_MAX_EPS_MAX_RATE, snapshot.max_rate));
390        events.push(gauge_event(METRIC_MAX_EPS_CURRENT_RATE, snapshot.current_rate));
391        events.push(gauge_event(METRIC_MAX_EPS_SAMPLE_RATE, snapshot.sample_rate));
392        if snapshot.sample_rate < 1.0 {
393            events.push(gauge_event(METRIC_MAX_EPS_REACHED_MAX, 1.0));
394            warn!(
395                "Max events per second reached (current={:.2}/s, max={:.2}/s). Some events are now being dropped \
396                 (sample rate={:.2}). Consider adjusting event sampling rates.",
397                snapshot.current_rate, snapshot.max_rate, snapshot.sample_rate
398            );
399        } else {
400            events.push(gauge_event(METRIC_MAX_EPS_REACHED_MAX, 0.0));
401        }
402
403        events
404    }
405}

407fn counter_event(name: &str, value: u64) -> Event {
408    Event::Metric(Metric::counter(
409        Context::from_parts(name, TagSet::default()),
410        value as f64,
411    ))
412}
413
414fn gauge_event(name: &str, value: f64) -> Event {
415    Event::Metric(Metric::gauge(Context::from_parts(name, TagSet::default()), value))
416}

567            let (sampled, rate) = limiter.sample(id, 0);
568            assert!(sampled);
569            assert_eq!(rate, 1.0);
570        }
571
572        // From the ninth event on, the estimate exceeds the budget and the rate drops below 1.
573        for id in 9..=11 {
574            let (_, rate) = limiter.sample(id, 0);
575            assert!(rate < 1.0, "expected a capped rate, got {rate}");
576        }
577    }
