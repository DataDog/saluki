1//! HTTP URL obfuscation.
2//!
3//! The algorithm lives in `libdd_trace_obfuscation::http`, a shared crate that tracks the reference implementation's
4//! URL handling: its escaping rules, its acceptance of relative references, and its decoding of percent-encoded path
5//! characters before the digit scan.
6
7use libdd_trace_obfuscation::http;
8use stringtheory::MetaString;
9
10use super::obfuscator::HttpObfuscationConfig;
11
12/// Obfuscates a URL string by removing userinfo, query strings, and/or path digits.
13///
14/// Returns `Some(obfuscated)` if the URL changed, `None` if it did not. Userinfo is always removed; the query string
15/// and path digits are removed per `config`. A URL that cannot be parsed loses its userinfo when both `config` options
16/// are off, and is replaced with `?` when either option is on.
17pub fn obfuscate_url(val: &str, config: &HttpObfuscationConfig) -> Option<MetaString> {
18    let obfuscated = http::obfuscate_url_string(val, config.remove_query_string, config.remove_paths_with_digits);
19
20    // The algorithm normalizes escaping even where nothing sensitive was found, so it can hand back a string equal to
21    // its input. Report that as unchanged so the caller keeps the URL it already has instead of replacing it with an
22    // identical copy.
23    if obfuscated == val {
24        return None;
25    }
26
27    Some(obfuscated.into())
28}
29
30#[cfg(test)]
31mod tests {
32    use super::*;
33
34    fn obfuscate(val: &str, remove_query_string: bool, remove_paths_with_digits: bool) -> Option<String> {
35        let config = HttpObfuscationConfig {
36            remove_query_string,
37            remove_paths_with_digits,
38        };
39        obfuscate_url(val, &config).map(|s| s.as_ref().to_owned())
40    }
41
42    // The algorithm normalizes escaping without finding anything sensitive, so it can hand back a string equal to
43    // its input. Report that as no change so the caller keeps the URL it already has.
44    #[test]
45    fn unchanged_url_reports_no_change() {
46        assert_eq!(obfuscate("http://foo.com/foo%20bar/", false, false), None);
47        assert_eq!(obfuscate("http://foo.com/foo%20bar/", true, true), None);
48    }
49}
