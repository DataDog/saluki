1//! [`SourceTree`]: configuration values kept together with the reason each one is present.
2//!
3//! A configuration layer is a tree of sections ending in leaves, where a leaf is one schema setting's
4//! entire value. Each leaf records the [`Provenance`] of its value, so a layer can say not just what
5//! a setting is but whether it has been set by a user or by default. This allows us to determine
6//! whether an Agent value should overwrite a locally set value. There are also cases where it is
7//! important for to know whether a held value is a default or was provided explicitly by
8//! configuration (`dd_url` over `site`, for example).
9//!
10//! Provenance crosses the crate boundary here: [`saluki_config::dynamic::Provenance`] describes a
11//! setting in transit on a configuration stream, and [`Provenance`] describes a value in the ADP
12//! model. [`SourceTree::set`] is where the two meet.
13

15use std::collections::HashSet;

24/// One configuration layer: every value it supplies, with the provenance of each.
25#[derive(Clone, Debug, PartialEq)]
26pub(crate) struct SourceTree {
27    root: Node,
28}
29
30/// A node in a [`SourceTree`].
31///
32/// The section/leaf split follows the source models' own leaf paths (see [`section_paths`]), so a
33/// map- or array-valued setting such as `additional_endpoints` is one leaf holding its whole value
34/// rather than a section of entries. A key the models do not know is a leaf for the same reason: with
35/// no schema to say otherwise, its value is indivisible.
36#[derive(Clone, Debug, PartialEq)]
37enum Node {
38    /// An intermediate path segment that a modeled leaf lives under.
39    Section(BTreeMap<String, Node>),
40
41    /// One setting's entire value, and whether an input set it explicitly.
42    Leaf { value: Value, provenance: Provenance },
43}
44
45impl SourceTree {
46    /// Creates a layer that supplies nothing.
47    pub(crate) fn empty() -> Self {
48        Self {
49            root: Node::Section(BTreeMap::new()),
50        }
51    }
52
53    /// Creates a layer in which every value present was set explicitly.
54    ///
55    /// This is the local file and environment. A value is in `value` only because the file set it or
56    /// an environment variable supplied it: absent keys are absent and null-valued keys were dropped
57    /// while the base was built. So presence here *is* the record that an input set the value, and no
58    /// provenance is being inferred after the fact.
59    pub(crate) fn all_explicit(value: Value) -> Self {
60        Self {
61            root: Node::from_value(value, &mut Vec::new(), Provenance::Explicit),
62        }
63    }
64
65    /// Creates a layer from a configuration producer's complete set of settings.
66    pub(crate) fn from_settings(settings: &[ConfigSetting]) -> Self {
67        let mut tree = Self::empty();
68        for setting in settings {
69            tree.set(setting);
70        }
71
72        tree
73    }
74
75    /// Applies one setting to this layer, replacing whatever it held for that key.
76    ///
77    /// The key is in the producer's own flat, possibly dotted form and expands into sections, while
78    /// the value is inserted whole: an object-valued setting keeps entry keys that themselves contain
79    /// dots (intake URLs, for example) intact.
80    pub(crate) fn set(&mut self, setting: &ConfigSetting) {
81        let provenance = match setting.provenance {
82            StreamProvenance::Explicit => Provenance::Explicit,
83            StreamProvenance::Default => Provenance::Default,
84        };
85
86        let mut node = &mut self.root;
87        let mut segments = setting.key.split('.').peekable();
88        while let Some(segment) = segments.next() {
89            // A leaf standing where a section is needed is replaced, so a later, more specific key
90            // wins over an earlier, shallower one.
91            if matches!(node, Node::Leaf { .. }) {
92                *node = Node::Section(BTreeMap::new());
93            }
94            let Node::Section(children) = node else {
95                unreachable!("a leaf was just replaced with a section")
96            };
97
98            if segments.peek().is_none() {
99                children.insert(
100                    segment.to_string(),
101                    Node::Leaf {
102                        value: setting.value.clone(),
103                        provenance,
104                    },
105                );
106                return;
107            }
108
109            node = children
110                .entry(segment.to_string())
111                .or_insert_with(|| Node::Section(BTreeMap::new()));
112        }
113    }
114
115    /// Layers `overlay` over this layer and returns the result.
116    ///
117    /// A value `overlay` set explicitly wins. A value `overlay` merely defaulted does not: it carries
118    /// no intent, so it must not shadow a value this layer set explicitly. When neither layer set a
119    /// value explicitly, `overlay` still wins, because a later default is a better answer than an
120    /// earlier one.
121    ///
122    /// This is what makes the Agent's configuration stream safe to treat as authoritative. The Agent
123    /// publishes every setting it knows about, including the ones nobody configured, so an
124    /// unqualified overlay would let its schema defaults erase the local file.
125    pub(crate) fn overlay(&self, overlay: &SourceTree) -> SourceTree {
126        SourceTree {
127            root: Node::merge(&self.root, &overlay.root),
128        }
129    }
130
131    /// Returns the values in this layer, discarding provenance.
132    ///
133    /// This is the shape the source models deserialize from.
134    pub(crate) fn to_value(&self) -> Value {
135        self.root.to_value()
136    }
137
138    /// Returns whether an input set `key`'s value explicitly.
139    ///
140    /// A key this layer does not supply reports [`Provenance::Default`]: a setting nobody supplied
141    /// takes the schema default, which is the same thing nothing set.
142    pub(crate) fn provenance(&self, key: &str) -> Provenance {
143        match self.root.get(key.split('.')) {
144            Some(Node::Leaf { provenance, .. }) => *provenance,
145            // A section is not a setting, so nothing set a value here.
146            Some(Node::Section(_)) | None => Provenance::Default,
147        }
148    }
149}
150
151impl Node {
152    /// Splits `value` into sections and leaves according to the source models' leaf paths.
153    ///
154    /// Every value that survives becomes a leaf with `provenance`.
155    fn from_value(value: Value, path: &mut Vec<String>, provenance: Provenance) -> Self {
156        // Recurse only through a section that is actually shaped like one; anywhere else the value is
157        // one setting, whole.
158        let Value::Object(fields) = value else {
159            return Node::Leaf { value, provenance };
160        };
161        if !path.is_empty() && !section_paths().contains(path.as_slice()) {
162            return Node::Leaf {
163                value: Value::Object(fields),
164                provenance,
165            };
166        }
167
168        let mut children = BTreeMap::new();
169        for (key, field) in fields {
170            path.push(key.clone());
171            children.insert(key, Node::from_value(field, path, provenance));
172            path.pop();
173        }
174
175        Node::Section(children)
176    }
177
178    /// Layers `overlay` over `base`. See [`SourceTree::overlay`].
179    fn merge(base: &Node, overlay: &Node) -> Node {
180        match (base, overlay) {
181            (Node::Section(base_children), Node::Section(overlay_children)) => {
182                let mut children = base_children.clone();
183                for (key, overlay_child) in overlay_children {
184                    let merged = match children.remove(key) {
185                        Some(base_child) => Node::merge(&base_child, overlay_child),
186                        None => overlay_child.clone(),
187                    };
188                    children.insert(key.clone(), merged);
189                }
190
191                Node::Section(children)
192            }
193            (
194                _,
195                Node::Leaf {
196                    provenance: Provenance::Default,
197                    ..
198                },
199            ) if base.has_explicit() => base.clone(),
200            // An explicit leaf, or a default with nothing explicit beneath it, replaces the base.
201            // A shape mismatch between the layers lands here too, and the overlay wins: the models
202            // agree on which paths are sections, so a mismatch means one layer holds a value the
203            // models do not describe, and there is no meaningful way to merge it.
204            _ => overlay.clone(),
205        }
206    }
207
208    /// Returns whether any input explicitly set a value at or beneath this node.
209    fn has_explicit(&self) -> bool {
210        match self {
211            Node::Leaf { provenance, .. } => *provenance == Provenance::Explicit,
212            Node::Section(children) => children.values().any(Node::has_explicit),
213        }
214    }
215
216    /// Returns the node at `path`, if this node has one.
217    fn get<'a>(&self, mut path: impl Iterator<Item = &'a str>) -> Option<&Node> {
218        match path.next() {
219            None => Some(self),
220            Some(segment) => match self {
221                Node::Section(children) => children.get(segment)?.get(path),
222                Node::Leaf { .. } => None,
223            },
224        }
225    }
226
227    /// Returns the values at and beneath this node, discarding provenance.
228    fn to_value(&self) -> Value {
229        match self {
230            Node::Leaf { value, .. } => value.clone(),
231            Node::Section(children) => Value::Object(
232                children
233                    .iter()
234                    .map(|(key, child)| (key.clone(), child.to_value()))
235                    .collect(),
236            ),
237        }
238    }
239}
240
241/// Every proper prefix of a modeled leaf path, across both source models.
242///
243/// A path in this set is a section that a tree descends through; a path that is a leaf, or that no model
244/// describes, is absent. Built once from the schema's own leaf tables, so it cannot drift from the
245/// models.
246fn section_paths() -> &'static HashSet<Vec<String>> {
247    static SECTIONS: OnceLock<HashSet<Vec<String>>> = OnceLock::new();
248    SECTIONS.get_or_init(|| {
249        let mut sections = HashSet::new();
250        let mut add_prefixes = |segments: Vec<String>| {
251            for end in 1..segments.len() {
252                sections.insert(segments[..end].to_vec());
253            }
254        };
255        for path in datadog_agent_config::datadog_leaf_paths() {
256            add_prefixes(path.iter().map(|segment| (*segment).to_string()).collect());
257        }
258        for path in saluki_env_overlay::leaf_paths() {
259            add_prefixes(path.to_vec());
260        }
261        sections
262    })
263}

273    /// Builds an Agent layer from `(key, value, provenance)` triples.
274    fn agent_layer(settings: &[(&str, serde_json::Value, StreamProvenance)]) -> SourceTree {
275        let settings: Vec<_> = settings
276            .iter()
277            .map(|(key, value, provenance)| ConfigSetting::new(*key, value.clone(), *provenance))
278            .collect();
279
280        SourceTree::from_settings(&settings)
281    }

301    #[test]
302    fn settings_expand_dotted_keys_and_keep_values_whole() {
303        let tree = agent_layer(&[
304            ("dogstatsd_port", json!(8125), StreamProvenance::Explicit),
305            ("otlp_config.traces.enabled", json!(true), StreamProvenance::Explicit),
306            // An object-valued setting is one leaf: these entry keys contain dots and must survive.
307            (
308                "additional_endpoints",
309                json!({ "https://app.datadoghq.eu": ["deadbeef"] }),
310                StreamProvenance::Explicit,
311            ),
312        ]);
313
314        assert_eq!(
315            tree.to_value(),
316            json!({
317                "dogstatsd_port": 8125,
318                "otlp_config": { "traces": { "enabled": true } },
319                "additional_endpoints": { "https://app.datadoghq.eu": ["deadbeef"] },
320            })
321        );
322        assert_eq!(tree.provenance("otlp_config.traces.enabled"), Provenance::Explicit);
323        // The dotted entry key inside the object value is not a path.
324        assert_eq!(
325            tree.provenance("additional_endpoints.https://app.datadoghq.eu"),
326            Provenance::Default
327        );
328    }
