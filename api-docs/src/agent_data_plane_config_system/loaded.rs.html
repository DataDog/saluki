1//! Loads local configuration before selecting its runtime authority.
2//!
3//! [`LoadedConfiguration::load`] prepares a typed snapshot of the local file and environment, which
4//! [`LoadedConfiguration::local`] exposes before values from the Datadog Agent stream are applied.
5//! [`LoadedConfiguration::run`] layers the Agent's configuration stream over the local sources,
6//! while [`LoadedConfiguration::standalone`] keeps the local sources authoritative. Both methods
7//! consume the loaded sources and return a [`ConfigurationSystem`].
8

21/// Where environment variables sit relative to the configuration file.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub enum EnvPrecedence {
24    /// Environment variables are read below the file: the file wins.
25    ///
26    /// The file takes precedence over environment variables.
27    BeforeFile,
28    /// Environment variables are read after the file: the environment wins. Matches the Datadog
29    /// Agent's precedence and is the value ADP uses.
30    AfterFile,
31    /// Environment variables are not read.
32    Disabled,
33}
34
35/// Local configuration prepared before a runtime authority is selected.
36pub struct LoadedConfiguration {
37    // Nested local base used for typed translation and Agent-layer merges.
38    base: SourceTree,
39    // Strictly translated local snapshot exposed before authority selection and used by standalone
40    // mode.
41    local: SalukiConfiguration,
42}
43
44impl LoadedConfiguration {
45    /// Loads and strictly translates the local file and environment using the requested precedence.
46    ///
47    /// The snapshot is translated but not validated: it is not yet authoritative. Under the Datadog
48    /// Agent the stream still has settings to contribute, so a local snapshot that could not run on
49    /// its own is normal here. Whichever authority [`run`](Self::run) or
50    /// [`standalone`](Self::standalone) selects applies validation.
51    ///
52    /// # Errors
53    ///
54    /// Returns an error if a local source cannot be read, decoded, deserialized, or translated.
55    pub async fn load(path: impl AsRef<Path>, env: EnvPrecedence) -> Result<Self, Error> {
56        // Every value the local sources supply was set explicitly: the file set it or an environment
57        // variable supplied it.
58        let base = SourceTree::all_explicit(build_base(path.as_ref(), env)?);
59        let local = translate_strict(&base)?;
60        Ok(Self { base, local })
61    }
62
63    /// Returns the typed snapshot of the local file and environment.
64    ///
65    /// Values from the Datadog Agent stream have not been applied to this snapshot.
66    pub fn local(&self) -> &SalukiConfiguration {
67        &self.local
68    }
69
70    /// Uses the Datadog Agent's configuration stream as the runtime authority.
71    ///
72    /// Waits for the initial Agent snapshot, layers it over the local sources, and strictly
73    /// translates the result. Returns the system together with the [`ConfigurationUpdates`] that
74    /// apply each later update. The caller must run them.
75    ///
76    /// # Errors
77    ///
78    /// Returns an error if the stream closes before its initial snapshot, or the merged configuration
79    /// cannot be deserialized or translated.
80    pub async fn run(
81        self, config_stream: mpsc::Receiver<ConfigUpdate>,
82    ) -> Result<(ConfigurationSystem, ConfigurationUpdates), Error> {
83        ConfigurationSystem::connected(config_stream, self.base).await
84    }
85
86    /// Uses the translated local configuration as the runtime authority.
87    ///
88    /// No configuration stream or update task is created. Because nothing further will be layered on,
89    /// the local snapshot is validated here, where [`run`](Self::run) instead validates the merged
90    /// result of the Agent's initial snapshot.
91    ///
92    /// # Errors
93    ///
94    /// Returns an error if the local configuration fails validation.
95    pub async fn standalone(self) -> Result<ConfigurationSystem, Error> {
96        validate(&self.local)?;
97        Ok(ConfigurationSystem::standalone(self.local, self.base))
98    }
99}

101/// Builds the typed base: the configuration file parsed to its nested shape, with environment
102/// variables read directly and decoded into the schema's shapes on top.
103///
104/// A null-valued key is dropped and an empty file is an empty object. The environment is then
105/// overlaid via the generated Datadog reader, the Saluki-only reader, and the canonical proxy
106/// variables. `env` sets whether the environment overwrites the file (`AfterFile`) or only fills
107/// absent keys (`BeforeFile`); `Disabled` skips the environment entirely.
108fn build_base(path: &Path, env: EnvPrecedence) -> Result<Value, Error> {
109    let text = std::fs::read_to_string(path).map_err(|e| Error::Base {
110        message: format!("read `{}`: {e}", path.display()),
111    })?;
112    let mut base: Value = saluki_config::yaml::from_str(&text).map_err(|e| Error::Base {
113        message: format!("parse `{}`: {e}", path.display()),
114    })?;
115    drop_nulls(&mut base);
116    if base.is_null() {
117        base = Value::Object(serde_json::Map::new());
118    }
119
120    let overwrite = match env {
121        EnvPrecedence::Disabled => return Ok(base),
122        EnvPrecedence::AfterFile => true,
123        EnvPrecedence::BeforeFile => false,
124    };
125    apply_datadog_env(&mut base, overwrite).map_err(|message| Error::Base { message })?;
126    saluki_env_overlay::apply_env(&mut base, overwrite).map_err(|message| Error::Base { message })?;
127    Ok(base)
128}
129
130/// Recursively removes object entries whose value is JSON null: an explicitly null YAML key must not
131/// override a model default with null.
132fn drop_nulls(value: &mut Value) {
133    if let Value::Object(map) = value {
134        map.retain(|_, v| !v.is_null());
135        for v in map.values_mut() {
136            drop_nulls(v);
137        }
138    }
139}

152        let _guard = test_env_lock();
153        let path = std::env::temp_dir().join(format!("adp_build_base_{}.yaml", std::process::id()));
154        std::fs::write(
155            &path,

158        .unwrap();
159        std::env::set_var("DD_DOGSTATSD_PORT", "9125");

170        assert_eq!(base.get("dogstatsd_port"), Some(&json!(8125)));
171
172        // Disabled: the environment is ignored entirely.
173        let base = build_base(&path, EnvPrecedence::Disabled).expect("base builds");

242    /// Waits until `system` reports `level` as the log level.
243    async fn await_log_level(system: &ConfigurationSystem, level: &str) {
244        tokio::time::timeout(std::time::Duration::from_secs(2), async {
245            while system.config().control.logging.level != level {
246                tokio::time::sleep(std::time::Duration::from_millis(5)).await;
247            }
248        })
249        .await
250        .unwrap_or_else(|_| panic!("the configuration should report the log level `{level}`"));
251    }

324    // `LoadedConfiguration::load` is `async` only for symmetry with the rest of the API; it awaits
325    // nothing. The environment tests below drive it on a local runtime rather than with
326    // `#[tokio::test]`, so the blocking environment guard is never held across an await point.
327    fn block_on<F: std::future::Future>(future: F) -> F::Output {
328        tokio::runtime::Builder::new_current_thread()
329            .enable_time()
330            .build()
331            .expect("runtime builds")
332            .block_on(future)
333    }

493    /// Sets process environment variables for the duration of a scope, removing them when it drops.
494    ///
495    /// The caller holds [`test_env_lock`] for at least as long.
496    struct ScopedEnv(Vec<&'static str>);
497
498    impl ScopedEnv {
499        fn set(vars: &[(&'static str, &str)]) -> Self {
500            for (name, value) in vars {
501                std::env::set_var(name, value);
502            }
503            Self(vars.iter().map(|(name, _)| *name).collect())
504        }
505    }
506
507    impl Drop for ScopedEnv {
508        fn drop(&mut self) {
509            for name in &self.0 {
510                std::env::remove_var(name);
511            }
512        }
513    }

575    /// Waits until `system` reports `port` as the DogStatsD port.
576    async fn await_dogstatsd_port(system: &ConfigurationSystem, port: u16) {
577        tokio::time::timeout(std::time::Duration::from_secs(2), async {
578            while system.config().domains.dogstatsd.listeners.port != port {
579                tokio::time::sleep(std::time::Duration::from_millis(5)).await;
580            }
581        })
582        .await
583        .unwrap_or_else(|_| panic!("the configuration should report the DogStatsD port {port}"));
584    }

586    #[test]
587    fn an_invalid_octal_integer_reads_as_zero_from_every_source() {
588        // The Agent starts with these values and its integer getters read them as 0, so ADP must
589        // neither stop at startup nor reject the stream's raw string.
590        let loaded = load_with_environment(
591            "invalid_octal",
592            "api_key: test-api-key\ndogstatsd_buffer_size: \"0128\"\n",
593            &[
594                ("DD_DOGSTATSD_PORT", "08"),
595                ("DD_CMD_PORT", "08.0"),
596                ("DD_AGENT_IPC_GRPC_MAX_MESSAGE_SIZE", "-09"),
597            ],
598        )
599        .expect("an invalid octal integer loads");
600
601        let local = loaded.local();
602        assert_eq!(local.domains.dogstatsd.listeners.port, 0);
603        assert_eq!(local.control.ipc.cmd_port, 0);
604        assert_eq!(local.control.ipc.grpc_max_message_size, 0);
605        assert_eq!(local.domains.dogstatsd.listeners.buffer_size, 0);
606
607        block_on(async {
608            // The stream forwards the Agent's raw strings, which connected startup accepts.
609            let (agent_tx, agent_rx) = mpsc::channel(8);
610            agent_tx
611                .send(ConfigUpdate::snapshot([
612                    ConfigSetting::explicit("dogstatsd_port", json!("010")),
613                    ConfigSetting::explicit("dogstatsd_buffer_size", json!("09")),
614                ]))
615                .await
616                .unwrap();
617            let (system, updates) = loaded.run(agent_rx).await.expect("the initial snapshot is accepted");
618            assert_eq!(system.config().domains.dogstatsd.listeners.port, 8);
619            assert_eq!(system.config().domains.dogstatsd.listeners.buffer_size, 0);
620
621            // A later update applies too. A rejected update would leave the last value in place.
622            tokio::spawn(updates.run());
623            agent_tx
624                .send(ConfigUpdate::Partial(ConfigSetting::explicit(
625                    "dogstatsd_port",
626                    json!("0x10"),
627                )))
628                .await
629                .unwrap();
630            await_dogstatsd_port(&system, 16).await;
631            agent_tx
632                .send(ConfigUpdate::Partial(ConfigSetting::explicit(
633                    "dogstatsd_port",
634                    json!("-08"),
635                )))
636                .await
637                .unwrap();
638            await_dogstatsd_port(&system, 0).await;
639        });
640    }
