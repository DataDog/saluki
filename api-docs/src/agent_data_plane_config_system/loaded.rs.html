1//! Loads local configuration before selecting its runtime authority.
2//!
3//! [`LoadedConfiguration::load`] prepares a typed snapshot of the local file and environment, which
4//! [`LoadedConfiguration::local`] exposes before values from the Datadog Agent stream are applied.
5//! [`LoadedConfiguration::run`] layers the Agent's configuration stream over the local sources,
6//! while [`LoadedConfiguration::standalone`] keeps the local sources authoritative. Both methods
7//! consume the loaded sources and return a [`ConfigurationSystem`].
8

21/// Where environment variables sit relative to the configuration file.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub enum EnvPrecedence {
24    /// Environment variables are read below the file: the file wins.
25    ///
26    /// The file takes precedence over environment variables.
27    BeforeFile,
28    /// Environment variables are read after the file: the environment wins. Matches the Datadog
29    /// Agent's precedence and is the value ADP uses.
30    AfterFile,
31    /// Environment variables are not read.
32    Disabled,
33}
34
35/// Local configuration prepared before a runtime authority is selected.
36pub struct LoadedConfiguration {
37    // Nested local base used for typed translation and Agent-layer merges.
38    base: SourceTree,
39    // Strictly translated local snapshot exposed before authority selection and used by standalone
40    // mode.
41    local: SalukiConfiguration,
42}
43
44impl LoadedConfiguration {
45    /// Loads and strictly translates the local file and environment using the requested precedence.
46    ///
47    /// The snapshot is translated but not validated: it is not yet authoritative. Under the Datadog
48    /// Agent the stream still has settings to contribute, so a local snapshot that could not run on
49    /// its own is normal here. Whichever authority [`run`](Self::run) or
50    /// [`standalone`](Self::standalone) selects applies validation.
51    ///
52    /// # Errors
53    ///
54    /// Returns an error if a local source cannot be read, decoded, deserialized, or translated.
55    pub async fn load(path: impl AsRef<Path>, env: EnvPrecedence) -> Result<Self, Error> {
56        // Every value the local sources supply was set explicitly: the file set it or an environment
57        // variable supplied it.
58        let base = SourceTree::all_explicit(build_base(path.as_ref(), env)?);
59        let local = translate_strict(&base)?;
60        Ok(Self { base, local })
61    }
62
63    /// Returns the typed snapshot of the local file and environment.
64    ///
65    /// Values from the Datadog Agent stream have not been applied to this snapshot.
66    pub fn local(&self) -> &SalukiConfiguration {
67        &self.local
68    }
69
70    /// Uses the Datadog Agent's configuration stream as the runtime authority.
71    ///
72    /// Waits for the initial Agent snapshot, layers it over the local sources, and strictly
73    /// translates the result. Returns the system together with the [`ConfigurationUpdates`] that
74    /// apply each later update. The caller must run them.
75    ///
76    /// # Errors
77    ///
78    /// Returns an error if the stream closes before its initial snapshot, or the merged configuration
79    /// cannot be deserialized or translated.
80    pub async fn run(
81        self, config_stream: mpsc::Receiver<ConfigUpdate>,
82    ) -> Result<(ConfigurationSystem, ConfigurationUpdates), Error> {
83        ConfigurationSystem::connected(config_stream, self.base).await
84    }
85
86    /// Uses the translated local configuration as the runtime authority.
87    ///
88    /// No configuration stream or update task is created. Because nothing further will be layered on,
89    /// the local snapshot is validated here, where [`run`](Self::run) instead validates the merged
90    /// result of the Agent's initial snapshot.
91    ///
92    /// # Errors
93    ///
94    /// Returns an error if the local configuration fails validation.
95    pub async fn standalone(self) -> Result<ConfigurationSystem, Error> {
96        validate(&self.local)?;
97        Ok(ConfigurationSystem::standalone(self.local, self.base))
98    }
99}

101/// Builds the typed base: the configuration file parsed to its nested shape, with environment
102/// variables read directly and decoded into the schema's shapes on top.
103///
104/// A null-valued key is dropped and an empty file is an empty object. The environment is then
105/// overlaid via the generated Datadog reader, the Saluki-only reader, and the canonical proxy
106/// variables. `env` sets whether the environment overwrites the file (`AfterFile`) or only fills
107/// absent keys (`BeforeFile`); `Disabled` skips the environment entirely.
108fn build_base(path: &Path, env: EnvPrecedence) -> Result<Value, Error> {
109    let text = std::fs::read_to_string(path).map_err(|e| Error::Base {
110        message: format!("read `{}`: {e}", path.display()),
111    })?;
112    // `Disabled` never reads the process environment, so it is not captured either.
113    let vars: Vec<(String, String)> = match env {
114        EnvPrecedence::Disabled => Vec::new(),
115        EnvPrecedence::AfterFile | EnvPrecedence::BeforeFile => std::env::vars().collect(),
116    };
117    build_base_from(&text, path, &vars, env)
118}

120/// Builds the typed base from the configuration file's text and explicit environment variables.
121///
122/// The same composition as [`build_base`], without reading the filesystem or the process
123/// environment: `text` is the file's content, `origin` is the path named in a parse error, and
124/// `vars` are the environment variable name/value pairs. `vars` is ignored when `env` is
125/// `Disabled`.
126///
127/// # Errors
128///
129/// Returns [`Error::Base`] when `text` is not valid YAML or an environment value is malformed for
130/// its key's shape.
131pub(crate) fn build_base_from(
132    text: &str, origin: &Path, vars: &[(String, String)], env: EnvPrecedence,
133) -> Result<Value, Error> {
134    let mut base: Value = saluki_config::yaml::from_str(text).map_err(|e| Error::Base {
135        message: format!("parse `{}`: {e}", origin.display()),
136    })?;
137    drop_nulls(&mut base);
138    if base.is_null() {
139        base = Value::Object(serde_json::Map::new());
140    }
141
142    let overwrite = match env {
143        EnvPrecedence::Disabled => return Ok(base),
144        EnvPrecedence::AfterFile => true,
145        EnvPrecedence::BeforeFile => false,
146    };
147    apply_datadog_env_vars(&mut base, vars.iter().cloned(), overwrite).map_err(|message| Error::Base { message })?;
148    saluki_env_overlay::apply_env_vars(&mut base, vars, overwrite).map_err(|message| Error::Base { message })?;
149    Ok(base)
150}
151
152/// Recursively removes object entries whose value is JSON null: an explicitly null YAML key must not
153/// override a model default with null.
154fn drop_nulls(value: &mut Value) {
155    if let Value::Object(map) = value {
156        map.retain(|_, v| !v.is_null());
157        for v in map.values_mut() {
158            drop_nulls(v);
159        }
160    }
161}

174        let _guard = test_env_lock();
175        let path = std::env::temp_dir().join(format!("adp_build_base_{}.yaml", std::process::id()));
176        std::fs::write(

193
194        // Disabled: the environment is ignored entirely.
195        let base = build_base(&path, EnvPrecedence::Disabled).expect("base builds");
196        assert_eq!(base.get("dogstatsd_port"), Some(&json!(8125)));

264    /// Waits until `system` reports `level` as the log level.
265    async fn await_log_level(system: &ConfigurationSystem, level: &str) {
266        tokio::time::timeout(std::time::Duration::from_secs(2), async {
267            while system.config().control.logging.level != level {
268                tokio::time::sleep(std::time::Duration::from_millis(5)).await;
269            }
270        })
271        .await
272        .unwrap_or_else(|_| panic!("the configuration should report the log level `{level}`"));
273    }

346    // `LoadedConfiguration::load` is `async` only for symmetry with the rest of the API; it awaits
347    // nothing. The environment tests below drive it on a local runtime rather than with
348    // `#[tokio::test]`, so the blocking environment guard is never held across an await point.
349    fn block_on<F: std::future::Future>(future: F) -> F::Output {
350        tokio::runtime::Builder::new_current_thread()
351            .enable_time()
352            .build()
353            .expect("runtime builds")
354            .block_on(future)
355    }

596    /// Sets process environment variables for the duration of a scope, removing them when it drops.
597    ///
598    /// The caller holds [`test_env_lock`] for at least as long.
599    struct ScopedEnv(Vec<&'static str>);
600
601    impl ScopedEnv {
602        fn set(vars: &[(&'static str, &str)]) -> Self {
603            for (name, value) in vars {
604                std::env::set_var(name, value);
605            }
606            Self(vars.iter().map(|(name, _)| *name).collect())
607        }
608    }
609
610    impl Drop for ScopedEnv {
611        fn drop(&mut self) {
612            for name in &self.0 {
613                std::env::remove_var(name);
614            }
615        }
616    }

618    /// Loads a temporary configuration file with the environment applied after it.
619    fn load_with_environment(
620        test: &str, contents: &str, vars: &[(&'static str, &str)],
621    ) -> Result<LoadedConfiguration, Error> {
622        let _guard = test_env_lock();
623        let path = std::env::temp_dir().join(format!("adp_{test}_{}.yaml", std::process::id()));
624        std::fs::write(&path, contents).unwrap();
625        let _env = ScopedEnv::set(vars);
626
627        let loaded = block_on(LoadedConfiguration::load(&path, EnvPrecedence::AfterFile));
628
629        std::fs::remove_file(&path).ok();
630        loaded
631    }

679    async fn await_dogstatsd_port(system: &ConfigurationSystem, port: u16) {
680        tokio::time::timeout(std::time::Duration::from_secs(2), async {

684        })
685        .await
686        .unwrap_or_else(|_| panic!("the configuration should report the DogStatsD port {port}"));

691        // The Agent starts with these values and its integer getters read them as 0, so ADP must
692        // neither stop at startup nor reject the stream's raw string.
693        let loaded = load_with_environment(
694            "invalid_octal",
695            "api_key: test-api-key\ndogstatsd_buffer_size: \"0128\"\n",
696            &[
697                ("DD_DOGSTATSD_PORT", "08"),
698                ("DD_CMD_PORT", "08.0"),
699                ("DD_AGENT_IPC_GRPC_MAX_MESSAGE_SIZE", "-09"),
700            ],
701        )
702        .expect("an invalid octal integer loads");
703
704        let local = loaded.local();
705        assert_eq!(local.domains.dogstatsd.listeners.port, 0);
706        assert_eq!(local.control.ipc.cmd_port, 0);
707        assert_eq!(local.control.ipc.grpc_max_message_size, 0);
708        assert_eq!(local.domains.dogstatsd.listeners.buffer_size, 0);
709
710        block_on(async {
711            // The stream forwards the Agent's raw strings, which connected startup accepts.
712            let (agent_tx, agent_rx) = mpsc::channel(8);
713            agent_tx
714                .send(ConfigUpdate::snapshot([
715                    ConfigSetting::explicit("dogstatsd_port", json!("010")),
716                    ConfigSetting::explicit("dogstatsd_buffer_size", json!("09")),
717                ]))
718                .await
719                .unwrap();
720            let (system, updates) = loaded.run(agent_rx).await.expect("the initial snapshot is accepted");
721            assert_eq!(system.config().domains.dogstatsd.listeners.port, 8);
722            assert_eq!(system.config().domains.dogstatsd.listeners.buffer_size, 0);
723
724            // A later update applies too. A rejected update would leave the last value in place.
725            tokio::spawn(updates.run());

727                .send(ConfigUpdate::Partial(ConfigSetting::explicit(
728                    "dogstatsd_port",

733            await_dogstatsd_port(&system, 16).await;
734            agent_tx
735                .send(ConfigUpdate::Partial(ConfigSetting::explicit(
736                    "dogstatsd_port",
737                    json!("-08"),
738                )))
739                .await
740                .unwrap();
741            await_dogstatsd_port(&system, 0).await;
742        });
743    }
