1//! Compatibility checks against the merged configuration sources.
2//!
3//! The typed model is the Agent schema pruned to the keys ADP supports, so the unsupported keys this
4//! check exists to catch are absent from it by construction. It reads the by-key view of the merged
5//! sources instead, which holds every key any input supplied along with the provenance of each.
6
7use std::collections::HashSet;
8
9use agent_data_plane_config::Provenance;
10use datadog_agent_config::classifier::{ConfigClassifier, Pipeline, PipelineAffinity, Severity, SupportLevel};
11use saluki_error::{generic_error, GenericError};
12use tracing::{debug, error, trace, warn};
13
14use crate::ConfigurationSystem;
15
16impl ConfigurationSystem {
17    /// Checks settings that an input set explicitly and that affect active pipelines, logging the
18    /// severity of each.
19    ///
20    /// # Errors
21    ///
22    /// Returns an error if high-severity incompatibilities exist. All keys are checked before
23    /// returning, so the error includes the total count.
24    pub fn check_compatibility(&self, active_pipelines: &HashSet<Pipeline>) -> Result<(), GenericError> {
25        let classifier = ConfigClassifier::new();
26        let mut high_severity_incompatibilities = 0u32;
27        debug!("Analyzing configuration.");
28        for (key, value, provenance) in self.sources.load().flattened_keys() {
29            let Some(classification) = classifier.classify(&key, value) else {
30                continue;
31            };
32
33            let pipeline_is_active = match &classification.pipeline_affinity {
34                PipelineAffinity::Pipelines(affected) => affected.iter().any(|p| active_pipelines.contains(p)),
35                PipelineAffinity::CrossCutting => true,
36            };
37            if !pipeline_is_active {
38                continue;
39            }
40
41            // A producer publishes every key it knows about, the ones nobody configured included, so
42            // only a key an input set explicitly says anything about what the operator asked for.
43            if provenance == Provenance::Default {
44                trace!(key = %key, "Configuration key is not set by any input.");
45                continue;
46            }
47
48            match classification.support_level {
49                SupportLevel::Incompatible(Severity::Low) => {
50                    debug!("Low-severity incompatible key detected. Proceeding.")
51                }
52                SupportLevel::Partial => {
53                    warn!(key = %key, "Partially supported configuration key. See documentation for details. Proceeding.")
54                }
55                SupportLevel::Incompatible(Severity::Medium) => {
56                    warn!(key = %key, "Unsupported configuration key. Proceeding.")
57                }
58                SupportLevel::Incompatible(Severity::High) => {
59                    error!(key = %key, "Unsupported configuration key with non-default value. ADP cannot run safely with \
60                    this setting.");
61                    high_severity_incompatibilities += 1;
62                }
63                SupportLevel::Ignored | SupportLevel::Unrecognized => {
64                    trace!(key = %key, "Configuration key not-applicable. Silently ignoring.")
65                }
66            }
67        }
68
69        if high_severity_incompatibilities > 0 {
70            return Err(generic_error!(
71                "{high_severity_incompatibilities} incompatible configuration detected. ADP cannot start. Review error \
72                logs for details."
73            ));
74        }
75
76        Ok(())
77    }
78}

80#[cfg(test)]
81mod tests {
82    use std::collections::HashSet;
83
84    use datadog_agent_config::classifier::Pipeline;
85    use saluki_config::dynamic::{ConfigSetting, Provenance as StreamProvenance};
86    use saluki_config::ConfigurationLoader;
87    use serde_json::{json, Value};
88
89    use crate::system::translate_strict;
90    use crate::{source::SourceTree, ConfigurationSystem};
91
92    /// Builds a system whose sources are a local configuration file, so every key present was set
93    /// explicitly.
94    async fn system_with(file: Value) -> ConfigurationSystem {
95        system_from(SourceTree::all_explicit(file)).await
96    }
97
98    /// Builds a system whose sources are the settings a configuration producer published, each
99    /// carrying its own provenance.
100    async fn system_from_settings(settings: &[(&str, Value, StreamProvenance)]) -> ConfigurationSystem {
101        let settings: Vec<_> = settings
102            .iter()
103            .map(|(key, value, provenance)| ConfigSetting::new(*key, value.clone(), *provenance))
104            .collect();
105
106        system_from(SourceTree::from_settings(&settings)).await
107    }
108
109    async fn system_from(sources: SourceTree) -> ConfigurationSystem {
110        let (raw_map, _) = ConfigurationLoader::for_tests(None, None, false).await;
111        let config = translate_strict(&sources).expect("sources translate");
112        ConfigurationSystem::standalone(raw_map, config, sources)
113    }
114
115    fn pipelines(active: &[Pipeline]) -> HashSet<Pipeline> {
116        active.iter().copied().collect()
117    }
118
119    fn otlp_tls_settings(cert_pem: &str, key_pem: &str) -> Value {
120        json!({
121            "otlp_config": {
122                "receiver": {
123                    "protocols": {
124                        "http": {
125                            "tls": { "cert_pem": cert_pem, "key_pem": key_pem }
126                        }
127                    }
128                }
129            }
130        })
131    }
132
133    #[tokio::test]
134    async fn high_severity_keys_fail_the_check_and_are_all_counted() {
135        let system = system_with(otlp_tls_settings("/etc/adp/cert.pem", "/etc/adp/key.pem")).await;
136
137        let error = system
138            .check_compatibility(&pipelines(&[Pipeline::Otlp]))
139            .expect_err("a high-severity incompatible key should fail the check");
140
141        assert!(error.to_string().contains("2 incompatible configuration detected"));
142    }
143
144    #[tokio::test]
145    async fn a_high_severity_key_nobody_set_is_skipped() {
146        // The Agent publishes every key it knows about, so a key it reports at its own default is a
147        // key nobody configured, whatever value it holds.
148        let system = system_from_settings(&[
149            (
150                "otlp_config.receiver.protocols.http.tls.cert_pem",
151                json!("/etc/adp/cert.pem"),
152                StreamProvenance::Default,
153            ),
154            (
155                "otlp_config.receiver.protocols.http.tls.key_pem",
156                json!("/etc/adp/key.pem"),
157                StreamProvenance::Default,
158            ),
159        ])
160        .await;
161
162        system
163            .check_compatibility(&pipelines(&[Pipeline::Otlp]))
164            .expect("a key nobody set is not an incompatibility");
165    }
166
167    #[tokio::test]
168    async fn a_high_severity_key_set_to_its_default_value_is_still_checked() {
169        // Writing an unsupported key is a request ADP cannot honor, so it is reported even when the
170        // value written happens to be the one the schema would have supplied.
171        let system = system_with(otlp_tls_settings("", "")).await;
172
173        let error = system
174            .check_compatibility(&pipelines(&[Pipeline::Otlp]))
175            .expect_err("an explicitly set key should fail the check");
176
177        assert!(error.to_string().contains("2 incompatible configuration detected"));
178    }
179
180    #[tokio::test]
181    async fn a_high_severity_key_affecting_no_active_pipeline_is_skipped() {
182        let system = system_with(otlp_tls_settings("/etc/adp/cert.pem", "/etc/adp/key.pem")).await;
183
184        system
185            .check_compatibility(&pipelines(&[Pipeline::DogStatsD]))
186            .expect("an inactive pipeline's keys are not incompatibilities");
187    }
188
189    #[tokio::test]
190    async fn lower_severity_keys_pass_and_cross_cutting_keys_ignore_active_pipelines() {
191        let system = system_with(json!({ "dogstatsd_queue_size": 2048, "min_tls_version": "tlsv1.3" })).await;
192        system
193            .check_compatibility(&pipelines(&[Pipeline::DogStatsD]))
194            .expect("only high-severity incompatibilities fail the check");
195
196        let cross_cutting = system_with(json!({ "heroku_dyno": true })).await;
197        cross_cutting
198            .check_compatibility(&pipelines(&[]))
199            .expect_err("a cross-cutting high-severity key fails the check with no pipeline active");
200    }
