1//! Compatibility checks against the merged configuration sources.
2//!
3//! The typed model is the Agent schema pruned to the keys ADP supports, so the unsupported keys this
4//! check exists to catch are absent from it by construction. It reads the by-key view of the merged
5//! sources instead, which holds every key any input supplied along with the provenance of each.
6
7use std::collections::HashSet;
8use std::sync::OnceLock;

13use tracing::{debug, error, trace, warn};

17/// Returns the compatibility classifier, built once for the process.
18pub(crate) fn classifier() -> &'static ConfigClassifier {
19    static CLASSIFIER: OnceLock<ConfigClassifier> = OnceLock::new();
20    CLASSIFIER.get_or_init(ConfigClassifier::new)
21}
22
23impl ConfigurationSystem {
24    /// Checks settings that an input set explicitly and that affect active pipelines, logging the
25    /// severity of each. A setting at its default value is skipped, unless the Agent's behavior
26    /// depends on whether the key is set.
27    ///
28    /// # Errors
29    ///
30    /// Returns an error if high-severity incompatibilities exist. All keys are checked before
31    /// returning, so the error includes the total count.
32    pub fn check_compatibility(&self, active_pipelines: &HashSet<Pipeline>) -> Result<(), GenericError> {
33        let mut high_severity_incompatibilities = 0u32;
34        debug!("Analyzing configuration.");
35        for (key, support_level) in self.compatibility_findings(active_pipelines) {
36            match support_level {
37                SupportLevel::Incompatible(Severity::Low) => {
38                    debug!("Low-severity incompatible key detected. Proceeding.")
39                }
40                SupportLevel::Partial => {
41                    warn!(key = %key, "Partially supported configuration key. See documentation for details. Proceeding.")
42                }
43                SupportLevel::Incompatible(Severity::Medium) => {
44                    warn!(key = %key, "Unsupported configuration key. Proceeding.")
45                }
46                SupportLevel::Incompatible(Severity::High) => {
47                    error!(key = %key, "Unsupported configuration key is set to a value other than its default. ADP \
48                    cannot run safely with this setting.");
49                    high_severity_incompatibilities += 1;
50                }
51                SupportLevel::Ignored | SupportLevel::Unrecognized => {
52                    trace!(key = %key, "Configuration key not-applicable. Silently ignoring.")
53                }
54            }
55        }
56
57        if high_severity_incompatibilities > 0 {
58            return Err(generic_error!(
59                "{high_severity_incompatibilities} incompatible configuration detected. ADP cannot start. Review error \
60                logs for details."
61            ));
62        }
63
64        Ok(())
65    }
66
67    /// Returns each classified key that an input set explicitly and that affects an active pipeline,
68    /// with its support level.
69    ///
70    /// A key set to its default value is left out, unless the Agent's behavior depends on whether the
71    /// key is set.
72    fn compatibility_findings(&self, active_pipelines: &HashSet<Pipeline>) -> Vec<(String, SupportLevel)> {
73        let classifier = classifier();
74        let mut findings = Vec::new();
75        for (key, value, provenance) in self.sources().flattened_keys() {
76            let Some(classification) = classifier.classify(&key, value) else {
77                continue;
78            };
79
80            let pipeline_is_active = match &classification.pipeline_affinity {
81                PipelineAffinity::Pipelines(affected) => affected.iter().any(|p| active_pipelines.contains(p)),
82                PipelineAffinity::CrossCutting => true,
83            };
84            if !pipeline_is_active {
85                continue;
86            }
87
88            // A producer publishes every key it knows about, the ones nobody configured included, so
89            // only a key an input set explicitly says anything about what the operator asked for.
90            if provenance == Provenance::Default {
91                trace!(key = %key, "Configuration key is not set by any input.");
92                continue;
93            }
94
95            // A key set explicitly to its schema default asks for the behavior ADP has when nobody
96            // sets it. The comparison can miss a default written in another form, such as a duration
97            // or a coerced scalar. A miss reports the key, so the check fails safe. A key whose
98            // presence alone changes the Agent's behavior is reported at its default too.
99            if classification.is_default && !classification.presence_matters {
100                trace!(key = %key, "Configuration key is set to its default value.");
101                continue;
102            }
103
104            findings.push((key, classification.support_level));
105        }
106
107        findings
108    }
109}

123    /// Builds a system whose sources are a local configuration file, so every key present was set
124    /// explicitly.
125    fn system_with(file: Value) -> ConfigurationSystem {
126        system_from(SourceTree::all_explicit(file))
127    }
128
129    /// Builds a system whose sources are the settings a configuration producer published, each
130    /// carrying its own provenance.
131    fn system_from_settings(settings: &[(&str, Value, StreamProvenance)]) -> ConfigurationSystem {
132        let settings: Vec<_> = settings
133            .iter()
134            .map(|(key, value, provenance)| ConfigSetting::new(*key, value.clone(), *provenance))
135            .collect();
136
137        system_from(SourceTree::from_settings(&settings))
138    }
139
140    fn system_from(sources: SourceTree) -> ConfigurationSystem {
141        let config = translate_strict(&sources).expect("sources translate");
142        ConfigurationSystem::standalone(config, sources)
143    }
144
145    /// Classifies `key` as the compatibility check sees it among the merged sources.
146    fn classification_of(system: &ConfigurationSystem, key: &str) -> Option<Classification> {
147        system
148            .sources()
149            .flattened_keys()
150            .into_iter()
151            .find(|(flattened, _, _)| flattened == key)
152            .and_then(|(flattened, value, _)| classifier().classify(&flattened, value))
153    }
154
155    fn pipelines(active: &[Pipeline]) -> HashSet<Pipeline> {
156        active.iter().copied().collect()
157    }

159    fn otlp_tls_settings(cert_pem: &str, key_pem: &str) -> Value {
160        json!({
161            "otlp_config": {
162                "receiver": {
163                    "protocols": {
164                        "http": {
165                            "tls": { "cert_pem": cert_pem, "key_pem": key_pem }
166                        }
167                    }
168                }
169            }
170        })
171    }

312    #[test]
313    fn an_unsupported_object_setting_is_classified_under_its_own_key() {
314        // `use_v3_api.distributions.endpoints` holds an object and no model describes it. The check must see
315        // it as the setting it is, whether it comes from a file or from the configuration stream.
316        let key = "use_v3_api.distributions.endpoints";
317        let endpoints = json!({ "https://example.test": ["api-key"] });
318        let from_file = system_with(json!({ "use_v3_api": { "distributions": { "endpoints": endpoints } } }));
319        let from_stream = system_from_settings(&[(key, endpoints.clone(), StreamProvenance::Explicit)]);
320
321        for system in [from_file, from_stream] {
322            let classification = classification_of(&system, key).expect("the setting is classified under its own key");
323            assert_eq!(
324                classification.support_level,
325                SupportLevel::Incompatible(Severity::Medium)
326            );
327            assert_eq!(
328                classification.pipeline_affinity,
329                PipelineAffinity::Pipelines(&[Pipeline::DogStatsD])
330            );
331        }
332    }
