40/// A handle for updating the health of a component.
41pub struct Health {
42    shared: Arc<SharedComponentState>,
43    liveness: Option<LivenessChannel>,
44    readiness_notify: Arc<Notify>,
45}
46
47impl Health {
48    /// Marks the component as ready.
49    pub fn mark_ready(&mut self) {
50        self.update_readiness(true);
51    }
52
53    /// Marks the component as not ready.
54    pub fn mark_not_ready(&mut self) {
55        self.update_readiness(false);
56    }
57
58    fn update_readiness(&self, ready: bool) {
59        self.shared.ready.store(ready, Relaxed);
60        self.shared.telemetry.update_readiness(ready);
61
62        // Wake any tasks waiting in `HealthRegistry::all_ready` so they can re-check whether all components are ready.
63        if ready {
64            self.readiness_notify.notify_waiters();
65        }
66    }
67
68    /// Waits for a liveness probe to be sent to the component, and then responds to it.
69    ///
70    /// This should generally be polled as part of a `select!` block to ensure it's checked alongside other
71    /// asynchronous operations.
72    ///
73    /// For a component that runs in a topology, this method never resolves, because the topology answers liveness
74    /// probes around the run future of the component.
75    pub async fn live(&mut self) {
76        match self.liveness.as_mut() {
77            Some(channel) => {
78                channel.respond().await;
79            }
80            None => std::future::pending().await,
81        }
82    }
83
84    /// Takes the liveness responder out of this handle.
85    ///
86    /// After this call, the returned [`LivenessResponder`] answers the liveness probes for the component, and
87    /// [`live`][Self::live] never resolves. The handle still controls readiness.
88    ///
89    /// Returns `None` if the responder was already taken.
90    pub(crate) fn take_liveness_responder(&mut self) -> Option<LivenessResponder> {
91        self.liveness.take().map(|channel| LivenessResponder { channel })
92    }
93}

95/// The channel pair that carries liveness probes to a component and its responses back to the registry.
96struct LivenessChannel {
97    request_rx: mpsc::Receiver<LivenessRequest>,
98    response_tx: mpsc::Sender<LivenessResponse>,
99}
100
101impl LivenessChannel {
102    /// Waits for a liveness probe and responds to it.
103    ///
104    /// Returns `false` if the registry closed the probe channel, and `true` otherwise.
105    async fn respond(&mut self) -> bool {
106        match self.request_rx.recv().await {
107            Some(request) => {
108                let response = request.into_response();
109                let _ = self.response_tx.send(response).await;
110                true
111            }
112            None => false,
113        }
114    }
115}

117/// Answers liveness probes for a component while it runs a future.
118///
119/// A component that waits on an `.await` outside its `select!` loop can't answer probes. One example is a send into a
120/// full downstream channel. The component is not stuck. It waits for backpressure to clear, but the registry marks it
121/// as not live. `LivenessResponder` moves probe handling out of the component: [`drive`][Self::drive] answers probes
122/// whenever the task of the component runs, whatever the component is waiting on.
123///
124/// The component is live while its task is scheduled and polled. A probe still fails if the task blocks the thread, or
125/// if the runtime can't poll the task within the probe timeout.
126///
127/// # Missing
128///
129/// - Detection of a run future that waits forever. Such a component stays live, because its task still gets polled.
130///   Examples are two components that each wait on a full channel that the other one drains, and a forwarder that
131///   waits on a request that never completes. Catching these cases needs a progress-based signal, such as a check that
132///   a component with buffered input still sends events.
133pub(crate) struct LivenessResponder {
134    channel: LivenessChannel,
135}
136
137impl LivenessResponder {
138    /// Runs `fut` to completion and answers liveness probes while it runs.
139    ///
140    /// The probes and `fut` run in the same task. When `fut` completes, the responder is dropped, and the registry
141    /// marks the component as dead at the next probe.
142    pub(crate) async fn drive<F: Future>(mut self, fut: F) -> F::Output {
143        pin!(fut);
144
145        loop {
146            select! {
147                // Check probes first, so that each time the task wakes, a waiting probe is seen, even when `fut` uses
148                // up the cooperative budget of the task.
149                biased;
150
151                open = self.channel.respond() => {
152                    if !open {
153                        // The registry is gone, so there are no more probes to answer.
154                        return fut.await;
155                    }
156                },
157                output = &mut fut => return output,
158            }
159        }
160    }
161}

172struct Telemetry {
173    component_ready: Gauge,
174    component_live: Gauge,
175    #[metric(level = trace)]
176    component_liveness_latency_seconds: Histogram,
177}
178
179impl Telemetry {
180    fn from_name(name: &SubsystemIdentifier) -> Self {
181        Self::new(name)
182    }
183
184    fn update_readiness(&self, ready: bool) {
185        self.component_ready().set(if ready { 1.0 } else { 0.0 });
186    }
187
188    fn update_liveness(&self, state: HealthState, response_latency: Duration) {
189        let live = match state {
190            HealthState::Live => 1.0,
191            HealthState::Unknown => 0.0,
192            HealthState::Dead => -1.0,
193        };
194
195        self.component_live().set(live);
196        self.component_liveness_latency_seconds()
197            .record(response_latency.as_secs_f64());
198    }

206struct ComponentState {
207    name: SubsystemIdentifier,
208    health: HealthState,
209    shared: Arc<SharedComponentState>,
210    request_tx: mpsc::Sender<LivenessRequest>,
211    last_response: Instant,
212    last_response_latency: Duration,
213}
214
215impl ComponentState {
216    fn new(
217        name: SubsystemIdentifier, response_tx: mpsc::Sender<LivenessResponse>, readiness_notify: Arc<Notify>,
218    ) -> (Self, Health) {
219        let shared = Arc::new(SharedComponentState {
220            ready: AtomicBool::new(false),
221            telemetry: Telemetry::from_name(&name),
222        });
223        let (request_tx, request_rx) = mpsc::channel(1);
224
225        let state = Self {
226            name,
227            health: HealthState::Unknown,
228            shared: Arc::clone(&shared),
229            request_tx,
230            last_response: Instant::now(),
231            last_response_latency: Duration::from_secs(0),
232        };
233
234        let handle = Health {
235            shared,
236            liveness: Some(LivenessChannel {
237                request_rx,
238                response_tx,
239            }),
240            readiness_notify,
241        };
242
243        (state, handle)
244    }
245
246    fn is_ready(&self) -> bool {
247        // We consider a component ready if it's marked as ready (duh) and it's not dead.
248        //
249        // Being "dead" is a special case as it means the component is very likely not even running at all, not just
250        // responding slowly or deadlocked. In these cases, it can't possibly be ready since it's not even running.
251        self.shared.ready.load(Relaxed) && self.health != HealthState::Dead
252    }
253
254    fn is_live(&self) -> bool {
255        self.health == HealthState::Live
256    }
257
258    fn mark_live(&mut self, response_sent: Instant, response_latency: Duration) {
259        self.health = HealthState::Live;
260        self.last_response = response_sent;
261        self.last_response_latency = response_latency;
262        self.shared.telemetry.update_liveness(self.health, response_latency);
263    }
264
265    fn mark_not_live(&mut self) {
266        self.health = HealthState::Unknown;
267
268        // We use the default timeout as the latency for when the component is not considered alive.
269        self.shared
270            .telemetry
271            .update_liveness(self.health, DEFAULT_PROBE_TIMEOUT_DUR);
272    }
273
274    fn mark_dead(&mut self) {
275        self.health = HealthState::Dead;
276
277        // We use the default timeout as the latency for when the component is not considered alive.
278        self.shared
279            .telemetry
280            .update_liveness(self.health, DEFAULT_PROBE_TIMEOUT_DUR);
281    }
282}

357/// A registry of components and their health.
358///
359/// `HealthRegistry` is responsible for tracking the health of all registered components, by storing both their
360/// readiness, which indicates whether or not they're initialized and generally ready to process data, as well as
361/// probing their liveness, which indicates if they're currently responding, or able to respond, to requests.
362///
363/// # Telemetry
364///
365/// The health registry emits some internal telemetry about the status of registered components. In particular, three
366/// metrics are emitted:
367///
368/// - `health_component_ready`: whether or not a component is ready (`gauge`, `0` for not ready, `1` for ready)
369/// - `health_component_live`: whether or not a component is alive (`gauge`, `0` for not alive/unknown, `1` for alive, `-1` for dead)
370/// - `health_component_liveness_latency_seconds`: the response latency of the component for liveness probes (`histogram`,
371///   in seconds)
372///
373/// All metrics have a `component_id` tag that corresponds to the name of the component that was given when registering it.
374#[derive(Clone)]
375pub struct HealthRegistry {
376    inner: Arc<Mutex<RegistryState>>,
377}
378
379impl HealthRegistry {
380    /// Creates an empty registry.
381    pub fn new() -> Self {
382        Self {
383            inner: Arc::new(Mutex::new(RegistryState::new())),
384        }
385    }

387    #[cfg(test)]
388    fn state(&self) -> Arc<Mutex<RegistryState>> {
389        Arc::clone(&self.inner)
390    }

392    /// Registers a component with the registry, keyed by its canonical [`SubsystemIdentifier`].
393    ///
394    /// Returns `None` if a component with the same identifier is already registered. Otherwise, a handle is returned
395    /// that must be used by the component to set its readiness as well as respond to liveness probes. See
396    /// [`Health::mark_ready`], [`Health::mark_not_ready`], and [`Health::live`] for more information.
397    pub fn register_component(&self, id: &SubsystemIdentifier) -> Option<Health> {
398        let mut inner = self.inner.lock().unwrap();
399
400        // Make sure we don't already have this component registered.
401        if !inner.registered_components.insert(id.clone()) {
402            return None;
403        }
404
405        // Add the component state.
406        let readiness_notify = Arc::clone(&inner.readiness_notify);
407        let (state, handle) = ComponentState::new(id.clone(), inner.responses_tx.clone(), readiness_notify);
408        let component_id = inner.component_state.len();
409        inner.component_state.push(state);
410
411        debug!(component_id, "Registered component '{}'.", id);
412
413        // Mark ourselves as having a pending component that needs to be scheduled.
414        inner.pending_components.push(component_id);
415        inner.pending_components_notify.notify_one();
416
417        Some(handle)
418    }

420    /// Gets an API handler for reporting the health of all components.
421    ///
422    /// This handler exposes routes for querying the readiness and liveness of all registered components. See
423    /// [`HealthAPIHandler`] for more information about routes and responses.
424    pub fn api_handler(&self) -> HealthAPIHandler {
425        HealthAPIHandler::from_state(Arc::clone(&self.inner))
426    }

428    /// Waits until all registered components are ready.
429    ///
430    /// If no components are registered, or all currently registered components are ready, the method returns immediately. Otherwise,
431    /// the method will return as soon as all registered components transition to ready.
432    ///
433    /// Note that components can be registered while this method is waiting, which will influence how long this method
434    /// takes to return. Callers should ensure that all components have been registered before calling this method.
435    pub async fn all_ready(&self) {
436        self.all_ready_matching(|_| true).await
437    }
438
439    /// Waits until all currently registered components whose name matches `predicate` are ready.
440    ///
441    /// This is a scoped variant of [`all_ready`][Self::all_ready] that only considers components whose name satisfies
442    /// the given predicate, which is useful for waiting on a specific subsystem's components to become ready without
443    /// waiting on every other component in the registry.
444    ///
445    /// If no registered component matches the predicate, the method returns immediately. As with
446    /// [`all_ready`][Self::all_ready], components can be registered while this method is waiting, so callers should
447    /// ensure all components they care about have been registered before calling this method.
448    pub async fn all_ready_matching<F>(&self, predicate: F)
449    where
450        F: Fn(&SubsystemIdentifier) -> bool,
451    {
452        let readiness_notify = {
453            let inner = self.inner.lock().unwrap();
454            Arc::clone(&inner.readiness_notify)
455        };
456
457        loop {
458            // Register as a waiter _before_ checking to avoid missing notifications during the check.
459            let notified = readiness_notify.notified();
460
461            if self.check_ready_matching(&predicate) {
462                return;
463            }
464
465            notified.await;
466        }
467    }
468
469    /// Waits until all currently registered components that are strict descendants of `root` are ready.
470    ///
471    /// This is a convenience wrapper over [`all_ready_matching`][Self::all_ready_matching] for the common case of
472    /// waiting on a single subsystem: it matches exactly the components whose identifier is a strict descendant of
473    /// `root` (see [`SubsystemIdentifier::is_ancestor_of`]). The same registration caveats as
474    /// [`all_ready`][Self::all_ready] apply.
475    pub async fn all_ready_under(&self, root: SubsystemIdentifier) {
476        self.all_ready_matching(move |id| root.is_ancestor_of(id)).await
477    }
478
479    fn check_ready_matching<F>(&self, predicate: &F) -> bool
480    where
481        F: Fn(&SubsystemIdentifier) -> bool,
482    {
483        let inner = self.inner.lock().unwrap();
484        inner
485            .component_state
486            .iter()
487            .filter(|component| predicate(&component.name))
488            .all(|component| component.is_ready())
489    }

491    /// Returns a JSON snapshot of the current readiness and liveness state of all registered components.
492    ///
493    /// Each component appears as a key in the returned JSON object, with its `live` and `ready` boolean fields
494    /// reflecting the state at the time of the call. This is the same data exposed by the `/ready` and `/live`
495    /// HTTP endpoints, but collected in a single pass for use outside of the HTTP handler path (for
496    /// example, when building a diagnostic artifact).
497    pub fn snapshot_json(&self) -> String {
498        #[derive(serde::Serialize)]
499        struct ComponentSnapshot {
500            live: bool,
501            ready: bool,
502        }
503
504        let inner = self.inner.lock().unwrap();
505        let mut state: std::collections::HashMap<String, ComponentSnapshot> = std::collections::HashMap::new();
506        for component in &inner.component_state {
507            state.insert(
508                component.name.to_string(),
509                ComponentSnapshot {
510                    live: component.is_live(),
511                    ready: component.is_ready(),
512                },
513            );
514        }
515        serde_json::to_string_pretty(&state).unwrap_or_else(|e| format!("{{\"error\": \"{e}\"}}"))
516    }

526    pub(crate) fn into_runner(self) -> Result<Runner, GenericError> {
527        // Make sure the runner hasn't already been spawned.
528        let (responses_rx, pending_components_notify) = {
529            let mut inner = self.inner.lock().unwrap();
530            let responses_rx = match inner.responses_rx.take() {
531                Some(rx) => rx,
532                None => return Err(generic_error!("health registry already spawned")),
533            };
534
535            let pending_components_notify = Arc::clone(&inner.pending_components_notify);
536            (responses_rx, pending_components_notify)
537        };
538
539        Ok(Runner::new(self.inner, responses_rx, pending_components_notify))
540    }

543/// A guard that returns the response receiver back to the registry when dropped.
544///
545/// This allows the health registry runner to be restarted gracefully: whenever the runner task
546/// finishes and this guard is dropped (for example, after a shutdown or task cancellation), the
547/// receiver is returned to the registry state so that a subsequent call to `spawn()` can succeed.
548struct RunnerGuard {
549    registry: Arc<Mutex<RegistryState>>,
550    responses_rx: Option<mpsc::Receiver<LivenessResponse>>,
551}
552
553impl Drop for RunnerGuard {
554    fn drop(&mut self) {
555        if let Some(rx) = self.responses_rx.take() {
556            let mut inner = self.registry.lock().expect("registry state poisoned");
557            inner.responses_rx = Some(rx);
558            debug!("Returned response receiver to registry state.");
559        }
560    }
561}

603pub(super) struct Runner {
604    registry: Arc<Mutex<RegistryState>>,
605    pending_probes: DelayQueue<usize>,
606    pending_timeouts: DelayQueue<usize>,
607    guard: RunnerGuard,
608    pending_components_notify: Arc<Notify>,
609    #[cfg(test)]
610    state: Arc<RunnerState>,
611}
612
613impl Runner {
614    fn new(
615        registry: Arc<Mutex<RegistryState>>, responses_rx: mpsc::Receiver<LivenessResponse>,
616        pending_components_notify: Arc<Notify>,
617    ) -> Self {
618        #[cfg(test)]
619        let state = Arc::new(RunnerState::new());
620
621        let guard = RunnerGuard {
622            registry: Arc::clone(&registry),
623            responses_rx: Some(responses_rx),
624        };
625
626        Self {
627            registry,
628            pending_probes: DelayQueue::new(),
629            pending_timeouts: DelayQueue::new(),
630            guard,
631            pending_components_notify,
632            #[cfg(test)]
633            state,
634        }
635    }
636
637    #[cfg(test)]
638    fn state(&self) -> Arc<RunnerState> {
639        Arc::clone(&self.state)
640    }
641
642    fn drain_pending_components(&mut self) -> Vec<usize> {
643        // Drain all pending components.
644        let mut registry = self.registry.lock().unwrap();
645        registry.pending_components.drain(..).collect()
646    }
647
648    fn send_component_probe_request(&mut self, component_id: usize) -> Option<HealthUpdate> {
649        let mut registry = self.registry.lock().unwrap();
650        let component_state = &mut registry.component_state[component_id];
651
652        // Check if our component is already dead, in which case we don't need to send a liveness probe.
653        if component_state.request_tx.is_closed() {
654            debug!(component_name = %component_state.name, "Component is dead, skipping liveness probe.");
655            return Some(HealthUpdate::Dead);
656        }
657
658        trace!(component_name = %component_state.name, probe_timeout = ?DEFAULT_PROBE_TIMEOUT_DUR, "Sending liveness probe to component.");
659
660        // Our component _isn't_ dead, so try to send a liveness probe to it.
661        //
662        // We'll register an entry in `pending_timeouts` that automatically marks the component as not live if we don't
663        // receive a response to the liveness probe within the timeout duration.
664        let timeout_key = self.pending_timeouts.insert(component_id, DEFAULT_PROBE_TIMEOUT_DUR);
665
666        #[cfg(test)]
667        self.state.increment_pending_probe_timeouts();
668
669        let request = LivenessRequest::new(component_id, timeout_key);
670        if let Err(TrySendError::Closed(request)) = component_state.request_tx.try_send(request) {
671            debug!(component_name = %component_state.name, "Component is dead, removing pending timeout.");
672
673            // We failed to send the probe to the component due to the component being dead. We'll drop our pending
674            // timeout as we're going to mark this component dead right now.
675            //
676            // When our send fails due to the channel being full, that's OK: it means it's going to be handled by an
677            // existing timeout and will be probed again later.
678            self.pending_timeouts.remove(&request.timeout_key);
679
680            #[cfg(test)]
681            self.state.decrement_pending_probe_timeouts();
682
683            return Some(HealthUpdate::Dead);
684        }
685
686        None
687    }
688
689    fn schedule_probe_for_component(&mut self, component_id: usize, duration: Duration) {
690        #[cfg(test)]
691        self.state.increment_pending_scheduled_probes();
692
693        self.pending_probes.insert(component_id, duration);
694    }

696    fn schedule_all_existing_components(&mut self, responses_rx: &mut mpsc::Receiver<LivenessResponse>) {
697        // First, drain any pending components to avoid scheduling them twice.
698        // This handles the case where components were registered before the runner started.
699        let _pending = self.drain_pending_components();
700
701        // Drain any queued probe responses from the previous runner. These responses were sent by
702        // components before the runner shut down but weren't processed. Processing them now updates
703        // `last_response` timestamps, which affects the staleness check below — a response that
704        // arrived just before shutdown should count as fresh.
705        while let Ok(response) = responses_rx.try_recv() {
706            self.handle_component_probe_response(response);
707        }
708
709        // Determine which components have stale probe results. Components whose last response is
710        // within the probe timeout are considered fresh and their health state is preserved,
711        // avoiding unnecessary bursts of failed liveness/readiness probes on runner restart.
712        let (component_count, stale_component_ids) = {
713            let registry = self.registry.lock().unwrap();
714            let now = Instant::now();
715            let stale_ids: Vec<usize> = (0..registry.component_state.len())
716                .filter(|&id| {
717                    let last = registry.component_state[id].last_response;
718                    saluki_antithesis::always_or_unreachable!(
719                        now >= last,
720                        "health probe last-response clock did not move backward"
721                    );
722                    now.saturating_duration_since(last) >= DEFAULT_PROBE_TIMEOUT_DUR
723                })
724                .collect();
725            (registry.component_state.len(), stale_ids)
726        };
727
728        // Only reset health to Unknown for components with stale probe results.
729        for &component_id in &stale_component_ids {
730            self.process_component_health_update(component_id, HealthUpdate::Unknown);
731        }
732
733        // Schedule immediate probes for all components regardless of staleness.
734        for component_id in 0..component_count {
735            self.schedule_probe_for_component(component_id, Duration::ZERO);
736        }
737
738        if component_count > 0 {
739            let fresh_count = component_count - stale_component_ids.len();
740            debug!(
741                component_count,
742                fresh_count,
743                stale_count = stale_component_ids.len(),
744                "Scheduled probes for all existing components."
745            );
746        }
747    }
748
749    fn handle_component_probe_response(&mut self, response: LivenessResponse) {
750        let component_id = response.request.component_id;
751        let timeout_key = response.request.timeout_key;
752        let request_sent = response.request.request_sent;
753        let response_sent = response.response_sent;
754        let response_latency = response_sent.checked_duration_since(request_sent).unwrap_or_default();
755
756        // Clear any pending timeouts for this component and schedule the next probe.
757        let timeout_was_pending = self.pending_timeouts.try_remove(&timeout_key).is_some();
758        if !timeout_was_pending {
759            let mut registry = self.registry.lock().unwrap();
760            let component_state = &mut registry.component_state[component_id];
761
762            debug!(component_name = %component_state.name, "Received probe response for component that already timed out.");
763        }
764
765        // Update the component's health to show as alive.
766        let update = HealthUpdate::Alive {
767            last_response: response_sent,
768            last_response_latency: response_latency,
769        };
770        self.process_component_health_update(component_id, update);
771
772        // Only schedule the next probe if we successfully removed the timeout, meaning it hadn't fired yet.
773        // This prevents duplicate probe scheduling when a response arrives after a timeout.
774        if timeout_was_pending {
775            #[cfg(test)]
776            self.state.decrement_pending_probe_timeouts();
777
778            self.schedule_probe_for_component(component_id, DEFAULT_PROBE_BACKOFF_DUR);
779        }
780    }
781
782    fn handle_component_timeout(&mut self, component_id: usize) {
783        // Update the component's health to show as not alive.
784        self.process_component_health_update(component_id, HealthUpdate::Unknown);
785
786        // Schedule the next probe for this component.
787        self.schedule_probe_for_component(component_id, DEFAULT_PROBE_BACKOFF_DUR);
788    }
789
790    fn process_component_health_update(&mut self, component_id: usize, update: HealthUpdate) {
791        // Update the component's health state based on the given update.
792        let mut registry = self.registry.lock().unwrap();
793        let component_state = &mut registry.component_state[component_id];
794        trace!(component_name = %component_state.name, status = update.as_str(), "Updating component health status.");
795
796        match update {
797            HealthUpdate::Alive {
798                last_response,
799                last_response_latency,
800            } => component_state.mark_live(last_response, last_response_latency),
801            HealthUpdate::Unknown => component_state.mark_not_live(),
802            HealthUpdate::Dead => component_state.mark_dead(),
803        }
804    }
805
806    async fn run<F: Future<Output = ()>>(mut self, shutdown: F) {
807        info!("Health checker running.");
808
809        // Take the response receiver out of the guard so we can use it in the select loop.
810        // It will be put back when the guard is dropped.
811        let mut responses_rx = self
812            .guard
813            .responses_rx
814            .take()
815            .expect("responses_rx should always be Some when Runner is created");
816
817        // Schedule probes for all existing components. This allows the runner to "pick up where it
818        // left off" after a restart - any components that were registered before the runner was
819        // restarted will be immediately probed.
820        self.schedule_all_existing_components(&mut responses_rx);
821
822        // Pin the shutdown future so we can poll it in the select loop.
823        pin!(shutdown);
824
825        loop {
826            select! {
827                // Shutdown signal received - exit the run loop gracefully.
828                _ = &mut shutdown => {
829                    info!("Health checker shutting down.");
830                    break;
831                },
832
833                // A component has been scheduled to have a liveness probe sent to it.
834                Some(entry) = self.pending_probes.next() => {
835                    #[cfg(test)]
836                    self.state.decrement_pending_scheduled_probes();
837
838                    let component_id = entry.into_inner();
839                    if let Some(health_update) = self.send_component_probe_request(component_id) {
840                        // If we got a health update for this component, that means we detected that it's dead, so we need
841                        // to do an out-of-band update to its health.
842                        self.process_component_health_update(component_id, health_update);
843                    }
844                },
845
846                // A component's outstanding liveness probe has expired.
847                Some(entry) = self.pending_timeouts.next() => {
848                    #[cfg(test)]
849                    self.state.decrement_pending_probe_timeouts();
850
851                    let component_id = entry.into_inner();
852                    self.handle_component_timeout(component_id);
853                },
854
855                // A probe response has been received.
856                Some(response) = responses_rx.recv() => {
857                    self.handle_component_probe_response(response);
858                },
859
860                // A component is pending finalization of their registration.
861                _ = self.pending_components_notify.notified() => {
862                    // Drain all pending components, give them a clean initial state of "unknown", and immediately schedule a probe for them.
863                    let pending_component_ids = self.drain_pending_components();
864                    for pending_component_id in pending_component_ids {
865                        self.process_component_health_update(pending_component_id, HealthUpdate::Unknown);
866                        self.schedule_probe_for_component(pending_component_id, Duration::ZERO);
867                    }
868                },
869            }
870        }
871
872        // Put the receiver back in the guard so it can be returned to the registry state when dropped.
873        self.guard.responses_rx = Some(responses_rx);
874
875        // When we exit the loop, the RunnerGuard will be dropped, returning the response receiver
876        // back to the registry state so that a subsequent spawn() can succeed.
877    }

1294        let registry = HealthRegistry {

1333    #[tokio::test(start_paused = true)]
1334    async fn responder_answers_probes_while_driven_future_waits() {
1335        let (mut handle, mut registry, registry_state, _runner_state) =
1336            initialize_registry_with_component(COMPONENT_ID);
1337        let responder = handle.take_liveness_responder().expect("responder should be present");
1338
1339        // The driven future waits on a channel that nothing sends to yet. This models a component that waits for room
1340        // in a full downstream channel.
1341        let (unblock_tx, unblock_rx) = oneshot::channel::<u32>();
1342        let mut driven = spawn(responder.drive(async move { unblock_rx.await.unwrap() }));
1343        assert_pending!(driven.poll());
1344
1345        // Send a probe. The responder must answer it, even though the driven future is still waiting.
1346        drive_until_quiesced(&mut registry);
1347        assert!(!component_live(&registry_state, COMPONENT_ID));
1348
1349        assert!(driven.is_woken());
1350        assert_pending!(driven.poll());
1351
1352        assert!(registry.is_woken());
1353        drive_until_quiesced(&mut registry);
1354        assert!(component_live(&registry_state, COMPONENT_ID));
1355
1356        // Once the driven future completes, `drive` returns its output.
1357        unblock_tx.send(42).unwrap();
1358        assert!(driven.is_woken());
1359        assert_eq!(assert_ready!(driven.poll()), 42);
1360    }
