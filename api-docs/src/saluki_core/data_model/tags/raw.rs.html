1use std::borrow::Cow;

3/// A wrapper over raw tags in their unprocessed form.
4///
5/// This type is meant to handle raw tags that have been extracted from network payloads, such as DogStatsD, where the
6/// input byte slice contains the tags -- whether bare or key/value -- packed together and separated by commas (`,`,
7/// 0x2C) character.
8///
9/// `RawTags` supports iteration over these tags in an efficient, zero-copy fashion and returns string references to
10/// each individual tag. It supports configuration to control how many tags can be returned, and the maximum allowable
11/// length for a tag. This allows easy usage where limits must be enforced, without having to write additional code to
12/// filter the resulting iterator.
13///
14/// ## Cloning
15///
16/// `RawTags` can be cloned to create a new iterator with its own iteration state. Borrowed tags retain their input
17/// slice, while normalized tags retain their owned backing string.
18#[derive(Clone)]
19pub struct RawTags<'a> {
20    raw_tags: Cow<'a, str>,
21    max_tag_count: usize,
22    max_tag_len: usize,
23}
24
25impl<'a> RawTags<'a> {
26    /// Creates `RawTags` from a borrowed tag block.
27    ///
28    /// The maximum tag count and maximum tag length control how many tags are returned from the iterator and their
29    /// length. If the iterator encounters more tags than the maximum count, it will simply stop returning tags. If the
30    /// iterator encounters any tag that's longer than the maximum length, it will truncate the tag to configured
31    /// length, or to a smaller length, whichever is closer to a valid UTF-8 character boundary.
32    pub const fn new(raw_tags: &'a str, max_tag_count: usize, max_tag_len: usize) -> Self {
33        Self {
34            raw_tags: Cow::Borrowed(raw_tags),
35            max_tag_count,
36            max_tag_len,
37        }
38    }
39
40    /// Creates `RawTags` from an owned, normalized tag block.
41    pub fn from_owned(raw_tags: String, max_tag_count: usize, max_tag_len: usize) -> Self {
42        Self {
43            raw_tags: Cow::Owned(raw_tags),
44            max_tag_count,
45            max_tag_len,
46        }
47    }
48
49    /// Creates an empty `RawTags`.
50    pub const fn empty() -> Self {
51        Self {
52            raw_tags: Cow::Borrowed(""),
53            max_tag_count: 0,
54            max_tag_len: 0,
55        }
56    }
57
58    /// Returns an iterator over the individual tags.
59    pub fn iter(&self) -> RawTagsIter<'_> {
60        RawTagsIter {
61            raw_tags: &self.raw_tags,
62            parsed_tags: 0,
63            max_tag_len: self.max_tag_len,
64            max_tag_count: self.max_tag_count,
65        }
66    }
67}
68
69#[derive(Clone)]
70pub struct RawTagsIter<'a> {
71    raw_tags: &'a str,
72    parsed_tags: usize,
73    max_tag_len: usize,
74    max_tag_count: usize,
75}
76
77impl<'a> Iterator for RawTagsIter<'a> {
78    type Item = &'a str;
79
80    fn next(&mut self) -> Option<Self::Item> {
81        let (raw_tag, tail) = split_at_delimiter(self.raw_tags, b',')?;
82        self.raw_tags = tail;
83
84        if self.parsed_tags >= self.max_tag_count {
85            // We've reached the maximum number of tags, so we just skip the rest.
86            return None;
87        }
88
89        let tag = limit_str_to_len(raw_tag, self.max_tag_len);
90
91        self.parsed_tags += 1;
92
93        Some(tag)
94    }
95}

97#[inline]
98fn limit_str_to_len(s: &str, limit: usize) -> &str {
99    if limit >= s.len() {
100        s
101    } else {
102        let sb = s.as_bytes();
103
104        // Search through the last four bytes of the string, ending at the index `limit`, and look for the byte that
105        // defines the boundary of a full UTF-8 character.
106        let start = limit.saturating_sub(3);
107        let new_index = sb[start..=limit]
108            .iter()
109            // Bit twiddling magic for checking if `b` is < 128 or >= 192.
110            .rposition(|b| (*b as i8) >= -0x40);
111
112        // SAFETY: UTF-8 characters are a maximum of four bytes, so we know we will have found a valid character
113        // boundary by searching over four bytes, regardless of where the slice started.
114        //
115        // Similarly we know that taking everything from index 0 to the detected character boundary index will be a
116        // valid UTF-8 string.
117        unsafe {
118            let safe_end = start + new_index.unwrap_unchecked();
119            std::str::from_utf8_unchecked(&sb[..safe_end])
120        }
121    }
122}

124#[inline]
125fn split_at_delimiter(input: &str, delimiter: u8) -> Option<(&str, &str)> {
126    match memchr::memchr(delimiter, input.as_bytes()) {
127        Some(index) => Some((&input[0..index], &input[index + 1..input.len()])),
128        None => {
129            if input.is_empty() {
130                None
131            } else {
132                Some((input, ""))
133            }
134        }
135    }
