1//! State that tracks workers for supervisors and scopes.
2//!
3//! `WorkerState` owns a set of child tasks that run for a [`Supervisor`](super::Supervisor) or a
4//! [`Scope`](super::Scope). It gives the operations that both of them need:
5//!
6//! - spawn a child
7//! - wait for the next child to finish
8//! - shut down all children concurrently
9//!
10//! It is intentionally independent of the restart policy. The owner decides what to do when a worker exits.
11

35/// Per-worker bookkeeping held by a [`WorkerState`].
36struct ProcessState {
37    /// Caller-assigned identifier for the worker.
38    ///
39    /// `WorkerState` does not interpret the value. The owner gives each child a stable id from a monotonic counter.
40    /// [`WorkerState::wait_for_next_worker`] returns it, so that the caller can match the exit to its own records.
41    worker_id: u64,
42    /// Fully qualified process name, retained so shutdown can name precisely which worker had to be forcefully
43    /// aborted.
44    worker_name: Arc<str>,
45    shutdown_strategy: ShutdownStrategy,
46    /// Whether the shutdown budget of the owner applies to this child.
47    ///
48    /// False for a nested supervisor, which bounds itself through its own children. Aborting one would cut its subtree
49    /// off mid-drain, and -- for a supervisor on a dedicated runtime, whose work lives on another OS thread -- would
50    /// not even stop it, while still reporting it as stopped.
51    budget_applies: bool,
52    /// Coordinator that signals this child.
53    ///
54    /// This coordinator is present also for a worker that reports
55    /// [`wants_shutdown_signal`][super::Supervisable::wants_shutdown_signal] as false. That worker gets a
56    /// [`ShutdownHandle::noop`] and never sees the signal that we fire. But the scope of the worker observes the
57    /// signal, so that it can tell a requested exit of one of its children from an exit that nobody requested.
58    shutdown_coordinator: ShutdownCoordinator,
59    abort_handle: AbortHandle,
60}

62/// Tracks the set of running child tasks for a supervisor or scope.
63pub(super) struct WorkerState {
64    process: Process,
65    /// Handle to the supervisor these workers belong to.
66    ///
67    /// Installed as the ambient supervisor for every worker task, so anything a worker spawns through
68    /// [`spawn`][crate::runtime::spawn] becomes a sibling of that worker rather than needing a handle threaded to it.
69    handle: SupervisorHandle,
70    /// Ceiling on the whole shutdown, if the supervisor configured one.
71    ///
72    /// Applied on top of each child's own strategy, so a child with no finite deadline of its own is still bounded,
73    /// and one that has a shorter deadline still exits first.
74    shutdown_budget: Option<Duration>,
75    /// Supervision-tree bookkeeping for the supervisor these workers belong to.
76    ///
77    /// Named as each worker's parent so that a supervisor a worker drives internally can attach itself to the tree.
78    node: Arc<SupervisorNode>,
79    /// The stop signals of the process that owns these workers and of the processes above it, if a scope owns them.
80    ///
81    /// Each worker gets this chain, so that the worker can see when a process above it is told to stop.
82    ancestors: Option<Arc<StopLink>>,
83    worker_tasks: JoinSet<Result<(), WorkerError>>,
84    worker_map: FastIndexMap<Id, ProcessState>,
85
86    /// The workers that a shutdown in progress signalled and still waits for.
87    ///
88    /// This is a field rather than a local of the shutdown, so that [`Drop`] can name these workers if the task that
89    /// owns this state is aborted in the middle of the shutdown.
90    draining: FastIndexMap<Id, Draining>,
91}

94    pub(super) fn new(
95        process: Process, handle: SupervisorHandle, shutdown_budget: Option<Duration>, node: Arc<SupervisorNode>,
96    ) -> Self {
97        Self {
98            process,
99            handle,
100            shutdown_budget,
101            node,
102            ancestors: None,
103            worker_tasks: JoinSet::new(),
104            worker_map: FastIndexMap::default(),
105            draining: FastIndexMap::default(),
106        }
107    }

109    /// Gives each worker `ancestors` as the stop signals of the processes above it.
110    ///
111    /// A scope uses this, so that its children can see when its owner, or a process above its owner, is told to stop.
112    pub(super) fn with_ancestors(mut self, ancestors: Arc<StopLink>) -> Self {
113        self.ancestors = Some(ancestors);
114        self
115    }

117    /// Spawns the child described by `child_spec`, tracking it under the given `worker_id`.
118    ///
119    /// `config` supplies the per-child overrides chosen at registration time: which runtime to spawn the child's task
120    /// on, and how the child's shutdown strategy is determined.
121    ///
122    /// Returns the identity of the process the child was started under, which is the only point at which that process
123    /// exists and so the only point at which it can be recorded.
124    pub(super) fn add_worker(
125        &mut self, worker_id: u64, child_spec: &SupervisedChild, config: &ChildConfig,
126    ) -> Result<StartedChild, SupervisorError> {
127        let process = child_spec.create_process(&self.process);
128        let worker_name: Arc<str> = process.name().into();
129
130        let started = StartedChild::new(&process, Arc::clone(&worker_name));
131
132        // Every child gets a coordinator, whether or not it observes the signal itself. A worker that does not observe
133        // it gets a no-op handle. But the scope of the worker still watches the signal, to know whether an exit of one
134        // of its children was requested.
135        let (mut shutdown_coordinator, shutdown_handle) = ShutdownHandle::paired();
136        let shutdown_handle = if child_spec.wants_shutdown_signal() {
137            shutdown_handle
138        } else {
139            drop(shutdown_handle);
140            ShutdownHandle::noop()
141        };
142
143        let mut worker_future = child_spec.create_worker_future(process.clone(), shutdown_handle)?;
144        let shutdown_strategy = match config.shutdown() {
145            ChildShutdown::Worker => child_spec.shutdown_strategy(),
146            ChildShutdown::Explicit(strategy) => strategy,
147            // The child has no deadline of its own, so the supervisor's budget is what bounds it. If nothing bounds
148            // it after all, it would be free to stall the drain forever, so fall back to whatever the worker asks
149            // for. Note that this asks whether the budget yields a *deadline*, not merely whether one was set: a
150            // budget too large to represent as an instant bounds nothing, and is no better than having none. A scope
151            // has no budget, so its children always fall back to their own strategies.
152            ChildShutdown::BudgetBounded => {
153                match resolve_budget_deadline(tokio::time::Instant::now(), self.shutdown_budget) {
154                    Some(_) => ShutdownStrategy::Graceful(Duration::MAX),
155                    None => child_spec.shutdown_strategy(),
156                }
157            }
158        };
159
160        // A worker owns a scope for the children that it spawns, and it does not finish until they finish. A nested
161        // supervisor already has its own children, and it has no body to drive with them.
162        if !child_spec.is_supervisor() {
163            worker_future = drive(
164                worker_future,
165                DriveContext {
166                    owner_name: Arc::clone(&worker_name),
167                    process: process.clone(),
168                    supervisor: self.handle.clone(),
169                    owner_signal: shutdown_coordinator.register().view(),
170                    ancestors: self.ancestors.clone(),
171                    adopt: config.adopt().cloned(),
172                },
173            );
174        }
175
176        // Every worker's task is timed, keyed on its fully qualified process name -- the same name
177        // `spawn_traced_named` would have recorded for an equivalent standalone task. A worker is a top-level task, so
178        // it is polled once per wake-up rather than once per unit of work, which keeps the two clock reads per poll well
179        // amortized against whatever the poll actually does.
180        let task = worker_future
181            .into_process_future(process)
182            .with_task_instrumentation(worker_name.to_string());
183
184        // Make ourselves the ambient supervisor for the worker's whole task, initialization included, so the worker
185        // can spawn siblings without being handed a handle.
186        let task = CURRENT_SUPERVISOR.scope(self.handle.clone(), task);
187
188        // Name the child slot this worker occupies for the same span, so a supervisor it builds and runs inside its
189        // own future -- rather than handing it to us as a child -- can attach itself to the tree there.
190        let task = CURRENT_TREE_PARENT.scope(TreeParent::new(Arc::clone(&self.node), worker_id), task);
191
192        let abort_handle = match config.runtime() {
193            Some(handle) => self.worker_tasks.spawn_on(task, handle),
194            None => self.worker_tasks.spawn(task),
195        };
196        self.worker_map.insert(
197            abort_handle.id(),
198            ProcessState {
199                worker_id,
200                worker_name,
201                shutdown_strategy,
202                budget_applies: !child_spec.is_supervisor(),
203                shutdown_coordinator,
204                abort_handle,
205            },
206        );
207        Ok(started)
208    }

210    /// Awaits the next worker to finish, returning its `worker_id` and result.
211    pub(super) async fn wait_for_next_worker(&mut self) -> (u64, Result<(), WorkerError>) {
212        debug!("Waiting for next process to complete.");
213
214        // If there are no workers to wait on, park indefinitely so the supervisor's select loop only proceeds via its
215        // other arms (shutdown, or a newly-added dynamic child). Without this guard, `join_next_with_id` would return
216        // `None` immediately on an empty set and the supervisor would busy-loop. The set legitimately empties when all
217        // children are non-restartable (e.g. `RestartType::Temporary`) and have exited.
218        if self.worker_tasks.is_empty() {
219            pending::<()>().await;
220        }
221
222        match self.worker_tasks.join_next_with_id().await {
223            Some(joined) => self.complete(joined),
224            None => unreachable!(
225                "join set is non-empty here: we park above while empty, and only this method removes workers"
226            ),
227        }
228    }
229
230    /// Polls for the next worker to finish, and returns its `worker_id` and result.
231    ///
232    /// Resolves to `None` if no worker runs, and does not park. The caller is a poll loop with other sources of
233    /// wake-ups, and it decides what an empty set means.
234    pub(super) fn poll_next_worker(&mut self, cx: &mut Context<'_>) -> Poll<Option<(u64, Result<(), WorkerError>)>> {
235        match self.worker_tasks.poll_join_next_with_id(cx) {
236            Poll::Ready(Some(joined)) => Poll::Ready(Some(self.complete(joined))),
237            Poll::Ready(None) => Poll::Ready(None),
238            Poll::Pending => Poll::Pending,
239        }
240    }
241
242    /// Removes the records of a finished worker, and converts the end of its task into a worker result.
243    fn complete(&mut self, joined: Result<(Id, Result<(), WorkerError>), JoinError>) -> (u64, Result<(), WorkerError>) {
244        match joined {
245            Ok((worker_task_id, worker_result)) => {
246                let process_state = self
247                    .worker_map
248                    .shift_remove(&worker_task_id)
249                    .expect("worker task ID not found");
250                (process_state.worker_id, worker_result)
251            }
252            Err(e) => {
253                let worker_task_id = e.id();
254                let process_state = self
255                    .worker_map
256                    .shift_remove(&worker_task_id)
257                    .expect("worker task ID not found");
258                let e = if e.is_cancelled() {
259                    ProcessError::Aborted
260                } else {
261                    ProcessError::Panicked
262                };
263                (process_state.worker_id, Err(WorkerError::Runtime(e.into())))
264            }
265        }
266    }

268    /// Shuts down all workers, honoring each worker's shutdown strategy.
269    ///
270    /// Every worker is signalled up front and then awaited concurrently under its **own** graceful deadline, so a
271    /// worker that ignores shutdown is aborted at its configured timeout regardless of its siblings. Total shutdown
272    /// time is therefore bounded by the slowest individual worker rather than the sum of all timeouts.
273    ///
274    /// Ordering between workers is deliberately not the supervisor's concern. A supervisor has no way to know what
275    /// depends on what, so workers that must stop in a particular order arrange it themselves through the ordinary
276    /// means -- an input channel closing, a barrier, a notification -- which is also what makes the order visible in
277    /// the code that establishes the dependency rather than in a distant list of registrations.
278    ///
279    /// A worker whose graceful timeout is effectively unbounded (such as a nested supervisor, which uses
280    /// `Duration::MAX` because it bounds itself via its own children's deadlines) is waited on indefinitely and is
281    /// never aborted here.
282    ///
283    /// Returns the number of workers that had to be forcefully aborted because they exceeded their graceful shutdown
284    /// timeout. The count includes aborts reported by nested child supervisors (which surface their own tally via
285    /// [`WorkerError::ShutdownTimedOut`]), so the value reflects the entire supervision tree rooted at this
286    /// supervisor.
287    pub(super) async fn shutdown_workers(&mut self) -> usize {
288        debug!("Shutting down all processes.");
289
290        let aborted = self.shutdown_workers_inner().await;
291
292        debug_assert!(self.worker_map.is_empty(), "worker map should be empty after shutdown");
293        debug_assert!(self.draining.is_empty(), "draining map should be empty after shutdown");
294        debug_assert!(
295            self.worker_tasks.is_empty(),
296            "worker tasks should be empty after shutdown"
297        );
298
299        aborted
300    }
301
302    async fn shutdown_workers_inner(&mut self) -> usize {
303        // Take ownership of all worker bookkeeping so we can consume each worker's shutdown coordinator. Signal every
304        // graceful worker and immediately abort brutal ones, recording a per-worker abort deadline so each is held to
305        // its own timeout rather than a single shared one.
306        let now = tokio::time::Instant::now();
307        let budget_deadline = resolve_budget_deadline(now, self.shutdown_budget);
308        for (task_id, process_state) in std::mem::take(&mut self.worker_map) {
309            let ProcessState {
310                worker_id,
311                worker_name,
312                shutdown_strategy,
313                budget_applies,
314                shutdown_coordinator,
315                abort_handle,
316            } = process_state;
317
318            match shutdown_strategy {
319                ShutdownStrategy::Graceful(timeout) => {
320                    debug!(worker_id, shutdown_timeout = ?timeout, "Gracefully shutting down process.");
321                    shutdown_coordinator.shutdown();
322                    let budget_deadline = budget_applies.then_some(budget_deadline).flatten();
323                    let deadline = resolve_abort_deadline(now, timeout, budget_deadline);
324                    self.draining.insert(
325                        task_id,
326                        Draining {
327                            worker_id,
328                            worker_name,
329                            abort_handle,
330                            deadline,
331                        },
332                    );
333                }
334                ShutdownStrategy::Brutal => {
335                    debug!(worker_id, "Forcefully aborting process.");
336                    abort_handle.abort();
337                }
338            }
339        }
340
341        // Wait for every task to exit. Each iteration sleeps until the earliest still-pending abort deadline; when it
342        // fires we abort exactly those workers whose own deadline has passed (their tasks are then reaped by a later
343        // `join_next`). Brutal workers were aborted above and aren't tracked here.
344        //
345        // If only workers with no finite deadline remain (e.g. nested supervisors), we wait for them to exit on their
346        // own. This is the path the topology takes -- each per-component supervisor is graceful-with-`MAX`, so its own
347        // forced-abort tally is reported here and merged into ours.
348        let mut aborted_total = 0;
349        while !self.worker_tasks.is_empty() {
350            let next_abort = self.draining.values().filter_map(|worker| worker.deadline).min();
351            let abort_due = async move {
352                match next_abort {
353                    Some(deadline) => tokio::time::sleep_until(deadline).await,
354                    None => pending::<()>().await,
355                }
356            };
357
358            select! {
359                joined = self.worker_tasks.join_next_with_id() => {
360                    let task_id = match joined {
361                        Some(Ok((task_id, output))) => {
362                            // A nested child supervisor that timed out reports its abort tally here; merge it.
363                            aborted_total += reported_abort_count(&output);
364                            Some(task_id)
365                        }
366                        Some(Err(e)) => Some(e.id()),
367                        None => None,
368                    };
369                    if let Some(task_id) = task_id {
370                        self.draining.swap_remove(&task_id);
371                    }
372                }
373                _ = abort_due => {
374                    let now = tokio::time::Instant::now();
375                    self.draining.retain(|_, worker| {
376                        if worker.deadline.is_some_and(|deadline| deadline <= now) {
377                            warn!(worker_id = worker.worker_id, worker_name = %worker.worker_name, "Worker ignored graceful shutdown; forcefully aborting after timeout.");
378                            worker.abort_handle.abort();
379                            aborted_total += 1;
380                            false
381                        } else {
382                            true
383                        }
384                    });
385                }
386            }
387        }
388
389        aborted_total
390    }
391}

393impl Drop for WorkerState {
394    fn drop(&mut self) {
395        // Workers that are still tracked here were never shut down: the task that owns this state was aborted, for
396        // example because its own shutdown deadline passed. The join set aborts them as it drops, without a signal and
397        // without anything counting them. Name them, so that an overrun still says which task was responsible.
398        let running: Vec<&str> = self
399            .worker_map
400            .values()
401            .filter(|worker| !worker.abort_handle.is_finished())
402            .map(|worker| &*worker.worker_name)
403            .chain(
404                self.draining
405                    .values()
406                    .filter(|worker| !worker.abort_handle.is_finished())
407                    .map(|worker| &*worker.worker_name),
408            )
409            .collect();
410
411        if !running.is_empty() {
412            warn!(
413                process_name = self.process.name(),
414                children = ?running,
415                "Process was aborted while its children were still running; aborting them with it."
416            );
417        }
418    }
419}

421/// A worker that was told to stop, and that the shutdown still waits for.
422struct Draining {
423    worker_id: u64,
424    worker_name: Arc<str>,
425    abort_handle: AbortHandle,
426
427    /// When the worker is forcefully aborted, if it has a deadline.
428    deadline: Option<tokio::time::Instant>,
429}
430
431/// Resolves the instant at which a worker must be forcefully aborted, if it must be at all.
432///
433/// A timeout of `Duration::MAX` means the worker carries no deadline of its own. That's correct for a nested
434/// supervisor, which bounds itself through its own children, and for the children of a supervisor that holds a
435/// shutdown budget on their behalf. When both a worker deadline and a budget apply, whichever elapses first wins, so
436/// the budget acts as a ceiling rather than an override.
437fn resolve_abort_deadline(
438    now: tokio::time::Instant, timeout: Duration, budget_deadline: Option<tokio::time::Instant>,
439) -> Option<tokio::time::Instant> {
440    // `checked_add` rather than `+`: adding a large duration to an instant panics on overflow, and both the sentinel
441    // (`Duration::MAX`) and any duration near it are reachable from caller-supplied configuration. A deadline too far
442    // out to represent is indistinguishable from no deadline at all, so both become `None`.
443    let own_deadline = now.checked_add(timeout);
444    match (own_deadline, budget_deadline) {
445        (Some(own), Some(budget)) => Some(own.min(budget)),
446        (deadline, None) | (None, deadline) => deadline,
447    }
448}
449
450/// Resolves the instant at which a supervisor's whole shutdown must be cut off, if it configured a budget.
451///
452/// Overflows the same way as [`resolve_abort_deadline`]: a budget too large to represent is treated as no budget.
453fn resolve_budget_deadline(now: tokio::time::Instant, budget: Option<Duration>) -> Option<tokio::time::Instant> {
454    budget.and_then(|budget| now.checked_add(budget))
455}

457/// Extracts the number of force-aborts a reaped child reported.
458///
459/// A nested child supervisor that completes a requested shutdown after forcefully aborting one or more of its own
460/// workers returns [`WorkerError::ShutdownTimedOut`]; its tally is merged into the parent's so the count aggregates
461/// across the whole supervision tree. Any other completion (clean exit, our own abort surfacing as a cancellation,
462/// panic) contributes nothing here.
463pub(super) fn reported_abort_count(output: &Result<(), WorkerError>) -> usize {
464    match output {
465        Err(WorkerError::ShutdownTimedOut { aborted }) => *aborted,
466        // A `Supervisable` worker that internally drives a supervisor (such as a topology blueprint) flattens that
467        // supervisor's `SupervisorError` into a `GenericError` at its boundary, so it surfaces here as `Runtime`
468        // rather than `ShutdownTimedOut`. Recover the structured count via downcast so it still aggregates upward; the
469        // concrete error type is preserved because the boundary converts with a plain `Into` (no added context).
470        Err(WorkerError::Runtime(e)) => match e.downcast_ref::<SupervisorError>() {
471            Some(SupervisorError::ShutdownTimedOut { aborted }) => *aborted,
472            _ => 0,
473        },
474        _ => 0,
475    }
476}

478#[cfg(test)]
479mod tests {
480    use super::*;
481
482    fn instant() -> tokio::time::Instant {
483        tokio::time::Instant::now()
484    }
485
486    #[tokio::test]
487    async fn abort_deadline_uses_the_workers_own_timeout_when_unbudgeted() {
488        let now = instant();
489        assert_eq!(
490            resolve_abort_deadline(now, Duration::from_secs(3), None),
491            Some(now + Duration::from_secs(3))
492        );
493    }
494
495    #[tokio::test]
496    async fn abort_deadline_is_none_for_an_unbounded_worker_without_a_budget() {
497        // `Duration::MAX` is the "no deadline of its own" sentinel: a nested supervisor, or a child whose supervisor
498        // holds the budget. With no budget in play there is nothing to bound it.
499        assert_eq!(resolve_abort_deadline(instant(), Duration::MAX, None), None);
500    }
501
502    #[tokio::test]
503    async fn abort_deadline_falls_back_to_the_budget_for_an_unbounded_worker() {
504        let now = instant();
505        let budget = now + Duration::from_secs(10);
506        assert_eq!(resolve_abort_deadline(now, Duration::MAX, Some(budget)), Some(budget));
507    }
508
509    #[tokio::test]
510    async fn abort_deadline_takes_whichever_of_worker_and_budget_is_sooner() {
511        let now = instant();
512        let budget = now + Duration::from_secs(10);
513
514        // Worker sooner than the budget.
515        assert_eq!(
516            resolve_abort_deadline(now, Duration::from_secs(2), Some(budget)),
517            Some(now + Duration::from_secs(2))
518        );
519
520        // Budget sooner than the worker: a worker cannot buy itself more time than its supervisor allows.
521        assert_eq!(
522            resolve_abort_deadline(now, Duration::from_secs(30), Some(budget)),
523            Some(budget)
524        );
525    }
526
527    #[tokio::test]
528    async fn abort_deadline_does_not_overflow_on_a_near_max_timeout() {
529        // Only `Duration::MAX` exactly used to be special-cased, so anything just under it panicked when added to an
530        // instant. A deadline too far out to represent is treated as no deadline.
531        let now = instant();
532        assert_eq!(
533            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), None),
534            None
535        );
536
537        // Even then, a budget still bounds the worker.
538        let budget = now + Duration::from_secs(5);
539        assert_eq!(
540            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), Some(budget)),
541            Some(budget)
542        );
543    }
544
545    #[tokio::test]
546    async fn budget_deadline_is_overflow_safe() {
547        let now = instant();
548        assert_eq!(resolve_budget_deadline(now, None), None);
549        assert_eq!(
550            resolve_budget_deadline(now, Some(Duration::from_secs(7))),
551            Some(now + Duration::from_secs(7))
552        );
553
554        // `Duration::MAX` is the natural spelling of "no ceiling", and used to panic the supervisor task.
555        assert_eq!(resolve_budget_deadline(now, Some(Duration::MAX)), None);
556    }
