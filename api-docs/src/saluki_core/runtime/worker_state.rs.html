1//! Worker-tracking state for the supervisor.
2//!
3//! `WorkerState` owns the set of running child tasks for a [`Supervisor`](super::Supervisor) and provides the common
4//! operations it needs: spawning a child, awaiting the next child to finish, and shutting all children down (either in
5//! order or concurrently). It is deliberately agnostic about restart policy -- the supervisor decides what to do when a
6//! worker exits.
7

27/// Per-worker bookkeeping held by a [`WorkerState`].
28struct ProcessState {
29    /// Caller-assigned identifier for the worker.
30    ///
31    /// Opaque to `WorkerState`: the supervisor assigns each child a stable id from a monotonic counter. It is returned
32    /// from [`WorkerState::wait_for_next_worker`] so the caller can correlate the exit with its own bookkeeping.
33    worker_id: u64,
34    /// Fully qualified process name, retained so shutdown can name precisely which worker had to be forcefully
35    /// aborted.
36    worker_name: String,
37    shutdown_strategy: ShutdownStrategy,
38    /// Whether this child is subject to the supervisor's shutdown budget.
39    ///
40    /// False for a nested supervisor, which bounds itself through its own children. Aborting one would cut its subtree
41    /// off mid-drain, and -- for a supervisor on a dedicated runtime, whose work lives on another OS thread -- would
42    /// not even stop it, while still reporting it as stopped.
43    budget_applies: bool,
44    /// Coordinator for signalling this worker, if it observes the signal at all.
45    ///
46    /// `None` for a worker that reports [`wants_shutdown_signal`][super::Supervisable::wants_shutdown_signal] as
47    /// false -- a closure-based worker, typically -- which was handed a [`ShutdownHandle::noop`] and would never see
48    /// anything we fired. Only the `Graceful` shutdown paths read this, and only a worker that wanted the signal can
49    /// reach them with anything to signal.
50    shutdown_coordinator: Option<ShutdownCoordinator>,
51    abort_handle: AbortHandle,
52}
53
54/// Tracks the set of running child tasks for a supervisor.
55pub(super) struct WorkerState {
56    process: Process,
57    /// Handle to the supervisor these workers belong to.
58    ///
59    /// Installed as the ambient supervisor for every worker task, so anything a worker spawns through
60    /// [`spawn`][crate::runtime::spawn] becomes a sibling of that worker rather than needing a handle threaded to it.
61    handle: SupervisorHandle,
62    /// Ceiling on the whole shutdown, if the supervisor configured one.
63    ///
64    /// Applied on top of each child's own strategy, so a child with no finite deadline of its own is still bounded,
65    /// and one that has a shorter deadline still exits first.
66    shutdown_budget: Option<Duration>,
67    worker_tasks: JoinSet<Result<(), WorkerError>>,
68    worker_map: FastIndexMap<Id, ProcessState>,
69}
70
71impl WorkerState {
72    pub(super) fn new(process: Process, handle: SupervisorHandle, shutdown_budget: Option<Duration>) -> Self {
73        Self {
74            process,
75            handle,
76            shutdown_budget,
77            worker_tasks: JoinSet::new(),
78            worker_map: FastIndexMap::default(),
79        }
80    }
81
82    /// Spawns the child described by `child_spec`, tracking it under the given `worker_id`.
83    ///
84    /// `config` supplies the per-child overrides chosen at registration time: which runtime to spawn the child's task
85    /// on, and how the child's shutdown strategy is determined.
86    pub(super) fn add_worker(
87        &mut self, worker_id: u64, child_spec: &SupervisedChild, config: &ChildConfig,
88    ) -> Result<(), SupervisorError> {
89        let process = child_spec.create_process(&self.process);
90        let worker_name = process.name().to_string();
91
92        // Only create a coordinator for a child that actually observes the signal. Most workers don't: they run until
93        // their own terminal condition and ignore whatever we fire at them, so a coordinator for them is an
94        // allocation and a wake-up that buy nothing.
95        let (shutdown_coordinator, shutdown_handle) = if child_spec.wants_shutdown_signal() {
96            let (coordinator, handle) = ShutdownHandle::paired();
97            (Some(coordinator), handle)
98        } else {
99            (None, ShutdownHandle::noop())
100        };
101
102        let worker_future = child_spec.create_worker_future(process.clone(), shutdown_handle)?;
103        let shutdown_strategy = match config.shutdown() {
104            ChildShutdown::Worker => child_spec.shutdown_strategy(),
105            ChildShutdown::Explicit(strategy) => strategy,
106            // The child has no deadline of its own, so the supervisor's budget is what bounds it. If nothing bounds
107            // it after all, it would be free to stall the drain forever, so fall back to whatever the worker asks
108            // for. Note that this asks whether the budget yields a *deadline*, not merely whether one was set: a
109            // budget too large to represent as an instant bounds nothing, and is no better than having none.
110            ChildShutdown::BudgetBounded => {
111                match resolve_budget_deadline(tokio::time::Instant::now(), self.shutdown_budget) {
112                    Some(_) => ShutdownStrategy::Graceful(Duration::MAX),
113                    None => child_spec.shutdown_strategy(),
114                }
115            }
116        };
117
118        // Every worker's task is timed, keyed on its fully qualified process name -- the same name
119        // `spawn_traced_named` would have recorded for an equivalent standalone task. A worker is a top-level task, so
120        // it is polled once per wake-up rather than once per unit of work, which keeps the two clock reads per poll well
121        // amortized against whatever the poll actually does.
122        let task = worker_future
123            .into_process_future(process)
124            .with_task_instrumentation(worker_name.clone());
125
126        // Make ourselves the ambient supervisor for the worker's whole task, initialization included, so the worker
127        // can spawn siblings without being handed a handle.
128        let task = CURRENT_SUPERVISOR.scope(self.handle.clone(), task);
129
130        let abort_handle = match config.runtime() {
131            Some(handle) => self.worker_tasks.spawn_on(task, handle),
132            None => self.worker_tasks.spawn(task),
133        };
134        self.worker_map.insert(
135            abort_handle.id(),
136            ProcessState {
137                worker_id,
138                worker_name,
139                shutdown_strategy,
140                budget_applies: !child_spec.is_supervisor(),
141                shutdown_coordinator,
142                abort_handle,
143            },
144        );
145        Ok(())
146    }
147
148    /// Awaits the next worker to finish, returning its `worker_id` and result.
149    pub(super) async fn wait_for_next_worker(&mut self) -> (u64, Result<(), WorkerError>) {
150        debug!("Waiting for next process to complete.");
151
152        // If there are no workers to wait on, park indefinitely so the supervisor's select loop only proceeds via its
153        // other arms (shutdown, or a newly-added dynamic child). Without this guard, `join_next_with_id` would return
154        // `None` immediately on an empty set and the supervisor would busy-loop. The set legitimately empties when all
155        // children are non-restartable (e.g. `RestartType::Temporary`) and have exited.
156        if self.worker_tasks.is_empty() {
157            pending::<()>().await;
158        }
159
160        match self.worker_tasks.join_next_with_id().await {
161            Some(Ok((worker_task_id, worker_result))) => {
162                let process_state = self
163                    .worker_map
164                    .shift_remove(&worker_task_id)
165                    .expect("worker task ID not found");
166                (process_state.worker_id, worker_result)
167            }
168            Some(Err(e)) => {
169                let worker_task_id = e.id();
170                let process_state = self
171                    .worker_map
172                    .shift_remove(&worker_task_id)
173                    .expect("worker task ID not found");
174                let e = if e.is_cancelled() {
175                    ProcessError::Aborted
176                } else {
177                    ProcessError::Panicked
178                };
179                (process_state.worker_id, Err(WorkerError::Runtime(e.into())))
180            }
181            None => unreachable!(
182                "join set is non-empty here: we park above while empty, and only this method removes workers"
183            ),
184        }
185    }
186
187    /// Shuts down all workers, honoring each worker's shutdown strategy.
188    ///
189    /// Every worker is signalled up front and then awaited concurrently under its **own** graceful deadline, so a
190    /// worker that ignores shutdown is aborted at its configured timeout regardless of its siblings. Total shutdown
191    /// time is therefore bounded by the slowest individual worker rather than the sum of all timeouts.
192    ///
193    /// Ordering between workers is deliberately not the supervisor's concern. A supervisor has no way to know what
194    /// depends on what, so workers that must stop in a particular order arrange it themselves through the ordinary
195    /// means -- an input channel closing, a barrier, a notification -- which is also what makes the order visible in
196    /// the code that establishes the dependency rather than in a distant list of registrations.
197    ///
198    /// A worker whose graceful timeout is effectively unbounded (such as a nested supervisor, which uses
199    /// `Duration::MAX` because it bounds itself via its own children's deadlines) is waited on indefinitely and is
200    /// never aborted here.
201    ///
202    /// Returns the number of workers that had to be forcefully aborted because they exceeded their graceful shutdown
203    /// timeout. The count includes aborts reported by nested child supervisors (which surface their own tally via
204    /// [`WorkerError::ShutdownTimedOut`]), so the value reflects the entire supervision tree rooted at this
205    /// supervisor.
206    pub(super) async fn shutdown_workers(&mut self) -> usize {
207        debug!("Shutting down all processes.");
208
209        let aborted = self.shutdown_workers_inner().await;
210
211        debug_assert!(self.worker_map.is_empty(), "worker map should be empty after shutdown");
212        debug_assert!(
213            self.worker_tasks.is_empty(),
214            "worker tasks should be empty after shutdown"
215        );
216
217        aborted
218    }
219
220    async fn shutdown_workers_inner(&mut self) -> usize {
221        // Take ownership of all worker bookkeeping so we can consume each worker's shutdown coordinator. Signal every
222        // graceful worker and immediately abort brutal ones, recording a per-worker abort deadline so each is held to
223        // its own timeout rather than a single shared one.
224        let now = tokio::time::Instant::now();
225        let budget_deadline = resolve_budget_deadline(now, self.shutdown_budget);
226        let mut pending: FastIndexMap<Id, (u64, String, AbortHandle, Option<tokio::time::Instant>)> =
227            FastIndexMap::default();
228        for (task_id, process_state) in std::mem::take(&mut self.worker_map) {
229            let ProcessState {
230                worker_id,
231                worker_name,
232                shutdown_strategy,
233                budget_applies,
234                shutdown_coordinator,
235                abort_handle,
236            } = process_state;
237
238            match shutdown_strategy {
239                ShutdownStrategy::Graceful(timeout) => {
240                    debug!(worker_id, shutdown_timeout = ?timeout, "Gracefully shutting down process.");
241                    // Absent for a worker that never observes the signal; we still wait for it to finish on its own.
242                    if let Some(shutdown_coordinator) = shutdown_coordinator {
243                        shutdown_coordinator.shutdown();
244                    }
245                    let budget_deadline = budget_applies.then_some(budget_deadline).flatten();
246                    let deadline = resolve_abort_deadline(now, timeout, budget_deadline);
247                    pending.insert(task_id, (worker_id, worker_name, abort_handle, deadline));
248                }
249                ShutdownStrategy::Brutal => {
250                    debug!(worker_id, "Forcefully aborting process.");
251                    abort_handle.abort();
252                }
253            }
254        }
255
256        // Wait for every task to exit. Each iteration sleeps until the earliest still-pending abort deadline; when it
257        // fires we abort exactly those workers whose own deadline has passed (their tasks are then reaped by a later
258        // `join_next`). Brutal workers were aborted above and aren't tracked here.
259        let mut aborted_total = 0;
260        while !self.worker_tasks.is_empty() {
261            match pending.values().filter_map(|(_, _, _, deadline)| *deadline).min() {
262                Some(deadline) => {
263                    select! {
264                        joined = self.worker_tasks.join_next_with_id() => {
265                            let task_id = match joined {
266                                Some(Ok((task_id, output))) => {
267                                    // A nested child supervisor that timed out reports its abort tally here; merge it.
268                                    aborted_total += reported_abort_count(&output);
269                                    Some(task_id)
270                                }
271                                Some(Err(e)) => Some(e.id()),
272                                None => None,
273                            };
274                            if let Some(task_id) = task_id {
275                                pending.swap_remove(&task_id);
276                            }
277                        }
278                        _ = tokio::time::sleep_until(deadline) => {
279                            let now = tokio::time::Instant::now();
280                            pending.retain(|_, (worker_id, worker_name, abort_handle, deadline)| {
281                                if deadline.is_some_and(|deadline| deadline <= now) {
282                                    warn!(worker_id = *worker_id, worker_name = %worker_name, "Worker ignored graceful shutdown; forcefully aborting after timeout.");
283                                    abort_handle.abort();
284                                    aborted_total += 1;
285                                    false
286                                } else {
287                                    true
288                                }
289                            });
290                        }
291                    }
292                }
293                // Only workers with no finite deadline remain (e.g. nested supervisors); wait for them to exit on their
294                // own. This is the path the topology takes -- each per-component supervisor is graceful-with-`MAX`, so
295                // its own forced-abort tally is reported here and merged into ours.
296                None => {
297                    if let Some(Ok((_, output))) = self.worker_tasks.join_next_with_id().await {
298                        aborted_total += reported_abort_count(&output);
299                    }
300                }
301            }
302        }
303
304        aborted_total
305    }
306}

308/// Resolves the instant at which a worker must be forcefully aborted, if it must be at all.
309///
310/// A timeout of `Duration::MAX` means the worker carries no deadline of its own. That's correct for a nested
311/// supervisor, which bounds itself through its own children, and for the children of a supervisor that holds a
312/// shutdown budget on their behalf. When both a worker deadline and a budget apply, whichever elapses first wins, so
313/// the budget acts as a ceiling rather than an override.
314fn resolve_abort_deadline(
315    now: tokio::time::Instant, timeout: Duration, budget_deadline: Option<tokio::time::Instant>,
316) -> Option<tokio::time::Instant> {
317    // `checked_add` rather than `+`: adding a large duration to an instant panics on overflow, and both the sentinel
318    // (`Duration::MAX`) and any duration near it are reachable from caller-supplied configuration. A deadline too far
319    // out to represent is indistinguishable from no deadline at all, so both become `None`.
320    let own_deadline = now.checked_add(timeout);
321    match (own_deadline, budget_deadline) {
322        (Some(own), Some(budget)) => Some(own.min(budget)),
323        (deadline, None) | (None, deadline) => deadline,
324    }
325}
326
327/// Resolves the instant at which a supervisor's whole shutdown must be cut off, if it configured a budget.
328///
329/// Overflows the same way as [`resolve_abort_deadline`]: a budget too large to represent is treated as no budget.
330fn resolve_budget_deadline(now: tokio::time::Instant, budget: Option<Duration>) -> Option<tokio::time::Instant> {
331    budget.and_then(|budget| now.checked_add(budget))
332}

334/// Extracts the number of force-aborts a reaped child reported.
335///
336/// A nested child supervisor that completes a requested shutdown after forcefully aborting one or more of its own
337/// workers returns [`WorkerError::ShutdownTimedOut`]; its tally is merged into the parent's so the count aggregates
338/// across the whole supervision tree. Any other completion (clean exit, our own abort surfacing as a cancellation,
339/// panic) contributes nothing here.
340fn reported_abort_count(output: &Result<(), WorkerError>) -> usize {
341    match output {
342        Err(WorkerError::ShutdownTimedOut { aborted }) => *aborted,
343        // A `Supervisable` worker that internally drives a supervisor (such as a topology blueprint) flattens that
344        // supervisor's `SupervisorError` into a `GenericError` at its boundary, so it surfaces here as `Runtime`
345        // rather than `ShutdownTimedOut`. Recover the structured count via downcast so it still aggregates upward; the
346        // concrete error type is preserved because the boundary converts with a plain `Into` (no added context).
347        Err(WorkerError::Runtime(e)) => match e.downcast_ref::<SupervisorError>() {
348            Some(SupervisorError::ShutdownTimedOut { aborted }) => *aborted,
349            _ => 0,
350        },
351        _ => 0,
352    }

355#[cfg(test)]
356mod tests {
357    use super::*;
358
359    fn instant() -> tokio::time::Instant {
360        tokio::time::Instant::now()
361    }
362
363    #[tokio::test]
364    async fn abort_deadline_uses_the_workers_own_timeout_when_unbudgeted() {
365        let now = instant();
366        assert_eq!(
367            resolve_abort_deadline(now, Duration::from_secs(3), None),
368            Some(now + Duration::from_secs(3))
369        );
370    }
371
372    #[tokio::test]
373    async fn abort_deadline_is_none_for_an_unbounded_worker_without_a_budget() {
374        // `Duration::MAX` is the "no deadline of its own" sentinel: a nested supervisor, or a child whose supervisor
375        // holds the budget. With no budget in play there is nothing to bound it.
376        assert_eq!(resolve_abort_deadline(instant(), Duration::MAX, None), None);
377    }
378
379    #[tokio::test]
380    async fn abort_deadline_falls_back_to_the_budget_for_an_unbounded_worker() {
381        let now = instant();
382        let budget = now + Duration::from_secs(10);
383        assert_eq!(resolve_abort_deadline(now, Duration::MAX, Some(budget)), Some(budget));
384    }
385
386    #[tokio::test]
387    async fn abort_deadline_takes_whichever_of_worker_and_budget_is_sooner() {
388        let now = instant();
389        let budget = now + Duration::from_secs(10);
390
391        // Worker sooner than the budget.
392        assert_eq!(
393            resolve_abort_deadline(now, Duration::from_secs(2), Some(budget)),
394            Some(now + Duration::from_secs(2))
395        );
396
397        // Budget sooner than the worker: a worker cannot buy itself more time than its supervisor allows.
398        assert_eq!(
399            resolve_abort_deadline(now, Duration::from_secs(30), Some(budget)),
400            Some(budget)
401        );
402    }
403
404    #[tokio::test]
405    async fn abort_deadline_does_not_overflow_on_a_near_max_timeout() {
406        // Only `Duration::MAX` exactly used to be special-cased, so anything just under it panicked when added to an
407        // instant. A deadline too far out to represent is treated as no deadline.
408        let now = instant();
409        assert_eq!(
410            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), None),
411            None
412        );
413
414        // Even then, a budget still bounds the worker.
415        let budget = now + Duration::from_secs(5);
416        assert_eq!(
417            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), Some(budget)),
418            Some(budget)
419        );
420    }
421
422    #[tokio::test]
423    async fn budget_deadline_is_overflow_safe() {
424        let now = instant();
425        assert_eq!(resolve_budget_deadline(now, None), None);
426        assert_eq!(
427            resolve_budget_deadline(now, Some(Duration::from_secs(7))),
428            Some(now + Duration::from_secs(7))
429        );
430
431        // `Duration::MAX` is the natural spelling of "no ceiling", and used to panic the supervisor task.
432        assert_eq!(resolve_budget_deadline(now, Some(Duration::MAX)), None);
433    }
