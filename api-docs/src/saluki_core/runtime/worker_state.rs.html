1//! Worker-tracking state for the supervisor.
2//!
3//! `WorkerState` owns the set of running child tasks for a [`Supervisor`](super::Supervisor) and provides the common
4//! operations it needs: spawning a child, awaiting the next child to finish, and shutting all children down (either in
5//! order or concurrently). It is deliberately agnostic about restart policy -- the supervisor decides what to do when a
6//! worker exits.
7

29/// Per-worker bookkeeping held by a [`WorkerState`].
30struct ProcessState {
31    /// Caller-assigned identifier for the worker.
32    ///
33    /// Opaque to `WorkerState`: the supervisor assigns each child a stable id from a monotonic counter. It is returned
34    /// from [`WorkerState::wait_for_next_worker`] so the caller can correlate the exit with its own bookkeeping.
35    worker_id: u64,
36    /// Fully qualified process name, retained so shutdown can name precisely which worker had to be forcefully
37    /// aborted.
38    worker_name: Arc<str>,
39    shutdown_strategy: ShutdownStrategy,
40    /// Whether this child is subject to the supervisor's shutdown budget.
41    ///
42    /// False for a nested supervisor, which bounds itself through its own children. Aborting one would cut its subtree
43    /// off mid-drain, and -- for a supervisor on a dedicated runtime, whose work lives on another OS thread -- would
44    /// not even stop it, while still reporting it as stopped.
45    budget_applies: bool,
46    /// Coordinator for signalling this worker, if it observes the signal at all.
47    ///
48    /// `None` for a worker that reports [`wants_shutdown_signal`][super::Supervisable::wants_shutdown_signal] as
49    /// false -- a closure-based worker, typically -- which was handed a [`ShutdownHandle::noop`] and would never see
50    /// anything we fired. Only the `Graceful` shutdown paths read this, and only a worker that wanted the signal can
51    /// reach them with anything to signal.
52    shutdown_coordinator: Option<ShutdownCoordinator>,
53    abort_handle: AbortHandle,
54}
55
56/// Tracks the set of running child tasks for a supervisor.
57pub(super) struct WorkerState {
58    process: Process,
59    /// Handle to the supervisor these workers belong to.
60    ///
61    /// Installed as the ambient supervisor for every worker task, so anything a worker spawns through
62    /// [`spawn`][crate::runtime::spawn] becomes a sibling of that worker rather than needing a handle threaded to it.
63    handle: SupervisorHandle,
64    /// Ceiling on the whole shutdown, if the supervisor configured one.
65    ///
66    /// Applied on top of each child's own strategy, so a child with no finite deadline of its own is still bounded,
67    /// and one that has a shorter deadline still exits first.
68    shutdown_budget: Option<Duration>,
69    /// Supervision-tree bookkeeping for the supervisor these workers belong to.
70    ///
71    /// Named as each worker's parent so that a supervisor a worker drives internally can attach itself to the tree.
72    node: Arc<SupervisorNode>,
73    worker_tasks: JoinSet<Result<(), WorkerError>>,
74    worker_map: FastIndexMap<Id, ProcessState>,
75}
76
77impl WorkerState {
78    pub(super) fn new(
79        process: Process, handle: SupervisorHandle, shutdown_budget: Option<Duration>, node: Arc<SupervisorNode>,
80    ) -> Self {
81        Self {
82            process,
83            handle,
84            shutdown_budget,
85            node,
86            worker_tasks: JoinSet::new(),
87            worker_map: FastIndexMap::default(),
88        }
89    }
90
91    /// Spawns the child described by `child_spec`, tracking it under the given `worker_id`.
92    ///
93    /// `config` supplies the per-child overrides chosen at registration time: which runtime to spawn the child's task
94    /// on, and how the child's shutdown strategy is determined.
95    ///
96    /// Returns the identity of the process the child was started under, which is the only point at which that process
97    /// exists and so the only point at which it can be recorded.
98    pub(super) fn add_worker(
99        &mut self, worker_id: u64, child_spec: &SupervisedChild, config: &ChildConfig,
100    ) -> Result<StartedChild, SupervisorError> {
101        let process = child_spec.create_process(&self.process);
102        let worker_name: Arc<str> = process.name().into();
103
104        let started = StartedChild::new(&process, Arc::clone(&worker_name));
105
106        // Only create a coordinator for a child that actually observes the signal. Most workers don't: they run until
107        // their own terminal condition and ignore whatever we fire at them, so a coordinator for them is an
108        // allocation and a wake-up that buy nothing.
109        let (shutdown_coordinator, shutdown_handle) = if child_spec.wants_shutdown_signal() {
110            let (coordinator, handle) = ShutdownHandle::paired();
111            (Some(coordinator), handle)
112        } else {
113            (None, ShutdownHandle::noop())
114        };
115
116        let worker_future = child_spec.create_worker_future(process.clone(), shutdown_handle)?;
117        let shutdown_strategy = match config.shutdown() {
118            ChildShutdown::Worker => child_spec.shutdown_strategy(),
119            ChildShutdown::Explicit(strategy) => strategy,
120            // The child has no deadline of its own, so the supervisor's budget is what bounds it. If nothing bounds
121            // it after all, it would be free to stall the drain forever, so fall back to whatever the worker asks
122            // for. Note that this asks whether the budget yields a *deadline*, not merely whether one was set: a
123            // budget too large to represent as an instant bounds nothing, and is no better than having none.
124            ChildShutdown::BudgetBounded => {
125                match resolve_budget_deadline(tokio::time::Instant::now(), self.shutdown_budget) {
126                    Some(_) => ShutdownStrategy::Graceful(Duration::MAX),
127                    None => child_spec.shutdown_strategy(),
128                }
129            }
130        };
131
132        // Every worker's task is timed, keyed on its fully qualified process name -- the same name
133        // `spawn_traced_named` would have recorded for an equivalent standalone task. A worker is a top-level task, so
134        // it is polled once per wake-up rather than once per unit of work, which keeps the two clock reads per poll well
135        // amortized against whatever the poll actually does.
136        let task = worker_future
137            .into_process_future(process)
138            .with_task_instrumentation(worker_name.to_string());
139
140        // Make ourselves the ambient supervisor for the worker's whole task, initialization included, so the worker
141        // can spawn siblings without being handed a handle.
142        let task = CURRENT_SUPERVISOR.scope(self.handle.clone(), task);
143
144        // Name the child slot this worker occupies for the same span, so a supervisor it builds and runs inside its
145        // own future -- rather than handing it to us as a child -- can attach itself to the tree there.
146        let task = CURRENT_TREE_PARENT.scope(TreeParent::new(Arc::clone(&self.node), worker_id), task);
147
148        let abort_handle = match config.runtime() {
149            Some(handle) => self.worker_tasks.spawn_on(task, handle),
150            None => self.worker_tasks.spawn(task),
151        };
152        self.worker_map.insert(
153            abort_handle.id(),
154            ProcessState {
155                worker_id,
156                worker_name,
157                shutdown_strategy,
158                budget_applies: !child_spec.is_supervisor(),
159                shutdown_coordinator,
160                abort_handle,
161            },
162        );
163        Ok(started)
164    }
165
166    /// Awaits the next worker to finish, returning its `worker_id` and result.
167    pub(super) async fn wait_for_next_worker(&mut self) -> (u64, Result<(), WorkerError>) {
168        debug!("Waiting for next process to complete.");
169
170        // If there are no workers to wait on, park indefinitely so the supervisor's select loop only proceeds via its
171        // other arms (shutdown, or a newly-added dynamic child). Without this guard, `join_next_with_id` would return
172        // `None` immediately on an empty set and the supervisor would busy-loop. The set legitimately empties when all
173        // children are non-restartable (e.g. `RestartType::Temporary`) and have exited.
174        if self.worker_tasks.is_empty() {
175            pending::<()>().await;
176        }
177
178        match self.worker_tasks.join_next_with_id().await {
179            Some(Ok((worker_task_id, worker_result))) => {
180                let process_state = self
181                    .worker_map
182                    .shift_remove(&worker_task_id)
183                    .expect("worker task ID not found");
184                (process_state.worker_id, worker_result)
185            }
186            Some(Err(e)) => {
187                let worker_task_id = e.id();
188                let process_state = self
189                    .worker_map
190                    .shift_remove(&worker_task_id)
191                    .expect("worker task ID not found");
192                let e = if e.is_cancelled() {
193                    ProcessError::Aborted
194                } else {
195                    ProcessError::Panicked
196                };
197                (process_state.worker_id, Err(WorkerError::Runtime(e.into())))
198            }
199            None => unreachable!(
200                "join set is non-empty here: we park above while empty, and only this method removes workers"
201            ),
202        }
203    }
204
205    /// Shuts down all workers, honoring each worker's shutdown strategy.
206    ///
207    /// Every worker is signalled up front and then awaited concurrently under its **own** graceful deadline, so a
208    /// worker that ignores shutdown is aborted at its configured timeout regardless of its siblings. Total shutdown
209    /// time is therefore bounded by the slowest individual worker rather than the sum of all timeouts.
210    ///
211    /// Ordering between workers is deliberately not the supervisor's concern. A supervisor has no way to know what
212    /// depends on what, so workers that must stop in a particular order arrange it themselves through the ordinary
213    /// means -- an input channel closing, a barrier, a notification -- which is also what makes the order visible in
214    /// the code that establishes the dependency rather than in a distant list of registrations.
215    ///
216    /// A worker whose graceful timeout is effectively unbounded (such as a nested supervisor, which uses
217    /// `Duration::MAX` because it bounds itself via its own children's deadlines) is waited on indefinitely and is
218    /// never aborted here.
219    ///
220    /// Returns the number of workers that had to be forcefully aborted because they exceeded their graceful shutdown
221    /// timeout. The count includes aborts reported by nested child supervisors (which surface their own tally via
222    /// [`WorkerError::ShutdownTimedOut`]), so the value reflects the entire supervision tree rooted at this
223    /// supervisor.
224    pub(super) async fn shutdown_workers(&mut self) -> usize {
225        debug!("Shutting down all processes.");
226
227        let aborted = self.shutdown_workers_inner().await;
228
229        debug_assert!(self.worker_map.is_empty(), "worker map should be empty after shutdown");
230        debug_assert!(
231            self.worker_tasks.is_empty(),
232            "worker tasks should be empty after shutdown"
233        );
234
235        aborted
236    }
237
238    async fn shutdown_workers_inner(&mut self) -> usize {
239        // Take ownership of all worker bookkeeping so we can consume each worker's shutdown coordinator. Signal every
240        // graceful worker and immediately abort brutal ones, recording a per-worker abort deadline so each is held to
241        // its own timeout rather than a single shared one.
242        let now = tokio::time::Instant::now();
243        let budget_deadline = resolve_budget_deadline(now, self.shutdown_budget);
244        let mut pending: FastIndexMap<Id, (u64, Arc<str>, AbortHandle, Option<tokio::time::Instant>)> =
245            FastIndexMap::default();
246        for (task_id, process_state) in std::mem::take(&mut self.worker_map) {
247            let ProcessState {
248                worker_id,
249                worker_name,
250                shutdown_strategy,
251                budget_applies,
252                shutdown_coordinator,
253                abort_handle,
254            } = process_state;
255
256            match shutdown_strategy {
257                ShutdownStrategy::Graceful(timeout) => {
258                    debug!(worker_id, shutdown_timeout = ?timeout, "Gracefully shutting down process.");
259                    // Absent for a worker that never observes the signal; we still wait for it to finish on its own.
260                    if let Some(shutdown_coordinator) = shutdown_coordinator {
261                        shutdown_coordinator.shutdown();
262                    }
263                    let budget_deadline = budget_applies.then_some(budget_deadline).flatten();
264                    let deadline = resolve_abort_deadline(now, timeout, budget_deadline);
265                    pending.insert(task_id, (worker_id, worker_name, abort_handle, deadline));
266                }
267                ShutdownStrategy::Brutal => {
268                    debug!(worker_id, "Forcefully aborting process.");
269                    abort_handle.abort();
270                }
271            }
272        }
273
274        // Wait for every task to exit. Each iteration sleeps until the earliest still-pending abort deadline; when it
275        // fires we abort exactly those workers whose own deadline has passed (their tasks are then reaped by a later
276        // `join_next`). Brutal workers were aborted above and aren't tracked here.
277        let mut aborted_total = 0;
278        while !self.worker_tasks.is_empty() {
279            match pending.values().filter_map(|(_, _, _, deadline)| *deadline).min() {
280                Some(deadline) => {
281                    select! {
282                        joined = self.worker_tasks.join_next_with_id() => {
283                            let task_id = match joined {
284                                Some(Ok((task_id, output))) => {
285                                    // A nested child supervisor that timed out reports its abort tally here; merge it.
286                                    aborted_total += reported_abort_count(&output);
287                                    Some(task_id)
288                                }
289                                Some(Err(e)) => Some(e.id()),
290                                None => None,
291                            };
292                            if let Some(task_id) = task_id {
293                                pending.swap_remove(&task_id);
294                            }
295                        }
296                        _ = tokio::time::sleep_until(deadline) => {
297                            let now = tokio::time::Instant::now();
298                            pending.retain(|_, (worker_id, worker_name, abort_handle, deadline)| {
299                                if deadline.is_some_and(|deadline| deadline <= now) {
300                                    warn!(worker_id = *worker_id, worker_name = %worker_name, "Worker ignored graceful shutdown; forcefully aborting after timeout.");
301                                    abort_handle.abort();
302                                    aborted_total += 1;
303                                    false
304                                } else {
305                                    true
306                                }
307                            });
308                        }
309                    }
310                }
311                // Only workers with no finite deadline remain (e.g. nested supervisors); wait for them to exit on their
312                // own. This is the path the topology takes -- each per-component supervisor is graceful-with-`MAX`, so
313                // its own forced-abort tally is reported here and merged into ours.
314                None => {
315                    if let Some(Ok((_, output))) = self.worker_tasks.join_next_with_id().await {
316                        aborted_total += reported_abort_count(&output);
317                    }
318                }
319            }
320        }
321
322        aborted_total
323    }
324}

326/// Resolves the instant at which a worker must be forcefully aborted, if it must be at all.
327///
328/// A timeout of `Duration::MAX` means the worker carries no deadline of its own. That's correct for a nested
329/// supervisor, which bounds itself through its own children, and for the children of a supervisor that holds a
330/// shutdown budget on their behalf. When both a worker deadline and a budget apply, whichever elapses first wins, so
331/// the budget acts as a ceiling rather than an override.
332fn resolve_abort_deadline(
333    now: tokio::time::Instant, timeout: Duration, budget_deadline: Option<tokio::time::Instant>,
334) -> Option<tokio::time::Instant> {
335    // `checked_add` rather than `+`: adding a large duration to an instant panics on overflow, and both the sentinel
336    // (`Duration::MAX`) and any duration near it are reachable from caller-supplied configuration. A deadline too far
337    // out to represent is indistinguishable from no deadline at all, so both become `None`.
338    let own_deadline = now.checked_add(timeout);
339    match (own_deadline, budget_deadline) {
340        (Some(own), Some(budget)) => Some(own.min(budget)),
341        (deadline, None) | (None, deadline) => deadline,
342    }
343}
344
345/// Resolves the instant at which a supervisor's whole shutdown must be cut off, if it configured a budget.
346///
347/// Overflows the same way as [`resolve_abort_deadline`]: a budget too large to represent is treated as no budget.
348fn resolve_budget_deadline(now: tokio::time::Instant, budget: Option<Duration>) -> Option<tokio::time::Instant> {
349    budget.and_then(|budget| now.checked_add(budget))
350}

352/// Extracts the number of force-aborts a reaped child reported.
353///
354/// A nested child supervisor that completes a requested shutdown after forcefully aborting one or more of its own
355/// workers returns [`WorkerError::ShutdownTimedOut`]; its tally is merged into the parent's so the count aggregates
356/// across the whole supervision tree. Any other completion (clean exit, our own abort surfacing as a cancellation,
357/// panic) contributes nothing here.
358fn reported_abort_count(output: &Result<(), WorkerError>) -> usize {
359    match output {
360        Err(WorkerError::ShutdownTimedOut { aborted }) => *aborted,
361        // A `Supervisable` worker that internally drives a supervisor (such as a topology blueprint) flattens that
362        // supervisor's `SupervisorError` into a `GenericError` at its boundary, so it surfaces here as `Runtime`
363        // rather than `ShutdownTimedOut`. Recover the structured count via downcast so it still aggregates upward; the
364        // concrete error type is preserved because the boundary converts with a plain `Into` (no added context).
365        Err(WorkerError::Runtime(e)) => match e.downcast_ref::<SupervisorError>() {
366            Some(SupervisorError::ShutdownTimedOut { aborted }) => *aborted,
367            _ => 0,
368        },
369        _ => 0,
370    }

373#[cfg(test)]
374mod tests {
375    use super::*;
376
377    fn instant() -> tokio::time::Instant {
378        tokio::time::Instant::now()
379    }
380
381    #[tokio::test]
382    async fn abort_deadline_uses_the_workers_own_timeout_when_unbudgeted() {
383        let now = instant();
384        assert_eq!(
385            resolve_abort_deadline(now, Duration::from_secs(3), None),
386            Some(now + Duration::from_secs(3))
387        );
388    }
389
390    #[tokio::test]
391    async fn abort_deadline_is_none_for_an_unbounded_worker_without_a_budget() {
392        // `Duration::MAX` is the "no deadline of its own" sentinel: a nested supervisor, or a child whose supervisor
393        // holds the budget. With no budget in play there is nothing to bound it.
394        assert_eq!(resolve_abort_deadline(instant(), Duration::MAX, None), None);
395    }
396
397    #[tokio::test]
398    async fn abort_deadline_falls_back_to_the_budget_for_an_unbounded_worker() {
399        let now = instant();
400        let budget = now + Duration::from_secs(10);
401        assert_eq!(resolve_abort_deadline(now, Duration::MAX, Some(budget)), Some(budget));
402    }
403
404    #[tokio::test]
405    async fn abort_deadline_takes_whichever_of_worker_and_budget_is_sooner() {
406        let now = instant();
407        let budget = now + Duration::from_secs(10);
408
409        // Worker sooner than the budget.
410        assert_eq!(
411            resolve_abort_deadline(now, Duration::from_secs(2), Some(budget)),
412            Some(now + Duration::from_secs(2))
413        );
414
415        // Budget sooner than the worker: a worker cannot buy itself more time than its supervisor allows.
416        assert_eq!(
417            resolve_abort_deadline(now, Duration::from_secs(30), Some(budget)),
418            Some(budget)
419        );
420    }
421
422    #[tokio::test]
423    async fn abort_deadline_does_not_overflow_on_a_near_max_timeout() {
424        // Only `Duration::MAX` exactly used to be special-cased, so anything just under it panicked when added to an
425        // instant. A deadline too far out to represent is treated as no deadline.
426        let now = instant();
427        assert_eq!(
428            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), None),
429            None
430        );
431
432        // Even then, a budget still bounds the worker.
433        let budget = now + Duration::from_secs(5);
434        assert_eq!(
435            resolve_abort_deadline(now, Duration::MAX - Duration::from_nanos(1), Some(budget)),
436            Some(budget)
437        );
438    }
439
440    #[tokio::test]
441    async fn budget_deadline_is_overflow_safe() {
442        let now = instant();
443        assert_eq!(resolve_budget_deadline(now, None), None);
444        assert_eq!(
445            resolve_budget_deadline(now, Some(Duration::from_secs(7))),
446            Some(now + Duration::from_secs(7))
447        );
448
449        // `Duration::MAX` is the natural spelling of "no ceiling", and used to panic the supervisor task.
450        assert_eq!(resolve_budget_deadline(now, Some(Duration::MAX)), None);
451    }
