1//! Remote agent registration and the services that the Datadog Agent queries.
2//!
3//! [`RemoteAgentBootstrap`] registers the process with the Datadog Agent over the Remote Agent Registry, keeps that
4//! registration alive, and creates everything that depends on it: the configuration stream, and the workers that serve
5//! the status, flare, and telemetry services and report diagnostic events back to the Datadog Agent.
6//!
7//! # Bootstrapping prior to supervision
8//!
9//! Registration and the initial configuration are generally required before any decisions can be made about continuing
10//! to start up, and this means that a supervision tree is not likely to yet be running where we can spawn our
11//! background tasks for managing registration.
12//!
13//! The registration loop and the configuration stream loop therefore each start early, in a [`ResumableWorker`], and
14//! run in a task until their worker takes them over in the supervision tree. The worker continues with the same
15//! session, refresh schedule, and open stream, and so does each restart of the worker.
16//!
17//! The registration task finishes a registration attempt in flight before it hands over, so a handoff never registers
18//! twice. The registration worker shuts down brutally, even in the middle of an attempt, and a restarted worker retries
19//! an interrupted attempt right away. Dropping a loop's worker before it starts cancels the loop.
20

100impl RemoteAgentBootstrap {
101    /// Creates a new `RemoteAgentBootstrap` from the given configurations.
102    ///
103    /// A remote agent client is created and immediately attempts to register with the Core Agent, advertising
104    /// `secure_api_listen_address` as the address where the Core Agent reaches the status, flare, and telemetry
105    /// services. This function does not return until the first registration attempt finishes, whether successful or
106    /// not. `additional_service_names` advertises services the binary adds through
107    /// [`create_services_worker`](Self::create_services_worker). Once registered, a task keeps the registration alive
108    /// until the worker from [`into_registration_worker`][Self::into_registration_worker] takes over.
109    ///
110    /// # Errors
111    ///
112    /// Returns an error if `secure_api_listen_address` isn't a valid gRPC target address, if the client can't connect
113    /// to the Core Agent, or if the first registration attempt fails. Nothing is left running when this returns an
114    /// error.
115    pub async fn new(
116        client_config: &RemoteAgentClientConfiguration, secure_api_listen_address: &ListenAddress,
117        additional_service_names: &[&str],
118    ) -> Result<Self, GenericError> {
119        let api_listen_addr = GrpcTargetAddress::try_from_listen_addr(secure_api_listen_address)
120            .ok_or_else(|| generic_error!("Failed to get valid gRPC target address from secure API listen address."))?;
121
122        // Generate our remote agent state, which is mostly fixed but has a few dynamic bits.
123        let mut service_names = vec![
124            <StatusProviderServer<()> as NamedService>::NAME.to_string(),
125            <FlareProviderServer<()> as NamedService>::NAME.to_string(),
126            <TelemetryProviderServer<()> as NamedService>::NAME.to_string(),
127        ];
128        service_names.extend(additional_service_names.iter().map(|name| (*name).to_owned()));
129        let details = RegistrationDetails::new(api_listen_addr, service_names);
130        let session_id = SessionIdHandle::empty();
131
132        let client = RemoteAgentClient::connect(client_config).await?;
133        let registration = Registration::new(Box::new(client.clone()), details, session_id.clone())
134            .register_and_spawn()
135            .await?;
136
137        Ok(Self {
138            client,
139            session_id,
140            registration,
141            internal_metrics: get_shared_metrics_state(),
142            dataspace: Arc::new(OnceLock::new()),
143        })
144    }
145
146    /// Returns the worker that keeps the registration with the Core Agent alive.
147    ///
148    /// The worker is named `remote-agent-session`, and must be added to the supervision tree. This consumes the
149    /// bootstrap, so create the other workers first. Dropping the bootstrap without taking the worker cancels the
150    /// registration loop.
151    pub fn into_registration_worker(self) -> RemoteAgentRegistrationWorker {
152        self.registration
153    }

189    /// Creates a worker that serves the status, flare, and telemetry services on the privileged API endpoint.
190    ///
191    /// The status service always reports the built-in main section fields (version, Git commit, architecture, start
192    /// time, and the Agent version this build was compiled against), followed by the fields written by each of the
193    /// given `status_sections`, in order.
194    ///
195    /// The telemetry service reports the internal metrics that match the given `telemetry_rules`, renamed to the names
196    /// the Datadog Agent expects. Metrics that match no rule are not reported, so if `telemetry_rules` is empty, the
197    /// service reports no metrics at all.
198    ///
199    /// `extend_services` adds binary-specific services using the live registration session handle. Their names must
200    /// be included in the `additional_service_names` passed to [`new`](Self::new).
201    pub fn create_services_worker(
202        &self, status_sections: Vec<Arc<dyn StatusSectionProvider>>, telemetry_rules: Vec<RemapperRule>,
203        extend_services: impl FnOnce(Routes, SessionIdHandle) -> Routes,
204    ) -> RemoteAgentServicesWorker {
205        let routes = Routes::new(StatusProviderServer::new(self.build_impl(status_sections, Vec::new())))
206            .add_service(FlareProviderServer::new(self.build_impl(Vec::new(), Vec::new())))
207            .add_service(TelemetryProviderServer::new(
208                self.build_impl(Vec::new(), telemetry_rules),
209            ));
210
211        RemoteAgentServicesWorker {
212            routes: extend_services(routes, self.session_id.clone()),
213        }
214    }

219    /// A task feeds the stream until the worker takes over, so that the initial configuration can arrive before the

222        ConfigStreamWorker::spawn(Box::new(self.client.clone()), self.session_id.clone())

282/// The registration loop, which refreshes the registration periodically and registers again whenever a refresh fails.
283struct Registration {
284    client: Box<dyn RegistryClient>,
285    details: RegistrationDetails,
286    session_id: SessionIdHandle,
287    timer: Interval,
288    attempt_in_flight: bool,
289}
290
291impl Registration {
292    fn new(client: Box<dyn RegistryClient>, details: RegistrationDetails, session_id: SessionIdHandle) -> Self {
293        let mut timer = interval(DEFAULT_REFRESH_INTERVAL);
294        timer.set_missed_tick_behavior(MissedTickBehavior::Delay);
295
296        Self {
297            client,
298            details,
299            session_id,
300            timer,
301            attempt_in_flight: false,
302        }
303    }
304
305    /// Registers, and then starts the registration loop in a task until the returned worker takes it over.
306    ///
307    /// # Errors
308    ///
309    /// If a registration attempt fails before one succeeds, an error is returned, and no task is started.
310    async fn register_and_spawn(mut self) -> Result<RemoteAgentRegistrationWorker, GenericError> {
311        debug!("Remote Agent registration task started.");
312
313        // An invalid session ID leaves us unregistered without failing the attempt, so we try again on the next tick.
314        while self.session_id.get().is_none() {
315            self.timer.tick().await;
316            self.register().await?;
317        }
318
319        Ok(RemoteAgentRegistrationWorker {
320            registration: ResumableWorker::start_early(REMOTE_AGENT_SESSION_WORKER_NAME, self),
321        })
322    }
323
324    async fn refresh_or_register(&mut self) {
325        match self.session_id.get() {
326            Some(session_id) => {
327                debug!(%session_id, "Refreshing registration with Datadog Agent.");
328
329                if self.client.refresh(&session_id).await.is_err() {
330                    self.timer.reset_after(REFRESH_FAILED_RETRY_INTERVAL);
331                    self.session_id.update(None);
332                    warn!("Failed to refresh registration with the Datadog Agent. Resetting session ID and attempting to re-register shortly.");
333                }
334            }
335            None => {
336                // `register` logs its own failures.
337                let _ = self.register().await;
338            }
339        }
340    }
341
342    async fn register(&mut self) -> Result<(), GenericError> {
343        let resp = match self.client.register(&self.details).await {
344            Ok(resp) => resp,
345            Err(e) => {
346                warn!(error = %e, "Failed to register with the Datadog Agent. Registration will be retried periodically in the background.");
347                self.timer.reset_after(DEFAULT_REFRESH_INTERVAL);
348                return Err(e);
349            }
350        };
351
352        match SessionId::new(&resp.session_id) {
353            Ok(new_session_id) => {
354                let new_refresh_interval = resp.recommended_refresh_interval_secs;
355                info!(
356                    session_id = %new_session_id,
357                    process_instance_id = %self.details.process_instance_id,
358                    "Successfully registered with the Datadog Agent. Refreshing every {} seconds.",
359                    new_refresh_interval
360                );
361
362                self.session_id.update(Some(new_session_id));
363                self.timer.reset_after(Duration::from_secs(new_refresh_interval as u64));
364            }
365            Err(e) => {
366                warn!(error = %e, "Received invalid session ID from Datadog Agent after registation. Registration will be retried periodically in the background.");
367                self.timer.reset_after(DEFAULT_REFRESH_INTERVAL);
368            }
369        }
370
371        Ok(())
372    }
373}

375impl Resumable for Registration {
376    fn shutdown_strategy(&self) -> ShutdownStrategy {
377        // There is nothing to finish on shutdown, and a restart retries an interrupted attempt.
378        ShutdownStrategy::Brutal
379    }
380
381    async fn run(&mut self, stop: ShutdownHandle) -> Result<(), GenericError> {
382        pin!(stop);
383
384        loop {
385            // An attempt that a dropped call interrupted is retried right away.
386            if !self.attempt_in_flight {
387                select! {
388                    biased;
389                    _ = &mut stop => return Ok(()),
390                    _ = self.timer.tick() => {}
391                }
392                self.attempt_in_flight = true;
393            }
394
395            // Not raced with `stop`, so that a handoff never registers twice.
396            self.refresh_or_register().await;
397            self.attempt_in_flight = false;
398        }
399    }
400}

415/// The configuration stream loop, which forwards configuration events from the Core Agent to a channel.
416struct ConfigStream {
417    source: Box<dyn ConfigEventSource>,
418    sender: mpsc::Sender<ConfigUpdate>,
419    session_id: SessionIdHandle,
420    events: Option<ConfigEventStream>,
421    reconnect_at: Option<Instant>,
422    pending: Option<ConfigUpdate>,
423}
424
425impl ConfigStream {
426    fn new(
427        source: Box<dyn ConfigEventSource>, sender: mpsc::Sender<ConfigUpdate>, session_id: SessionIdHandle,
428    ) -> Self {
429        Self {
430            source,
431            sender,
432            session_id,
433            events: None,
434            reconnect_at: None,
435            pending: None,
436        }
437    }
438
439    /// Forwards configuration events until the receiver is dropped.
440    ///
441    /// Every await point is cancel-safe: the open stream, the reconnect deadline, and an update waiting for room in the
442    /// channel are all kept in `self`.
443    async fn forward(&mut self) {
444        if self.sender.is_closed() {
445            return;
446        }
447
448        loop {
449            if self.pending.is_some() {
450                let Ok(permit) = self.sender.reserve().await else {
451                    warn!("Dynamic configuration channel closed. Config stream shutting down.");
452                    self.events = None;
453                    self.pending = None;
454                    return;
455                };
456                permit.send(self.pending.take().expect("pending update should be present"));
457            }
458
459            let events = match &mut self.events {
460                Some(events) => events,
461                None => {
462                    if let Some(reconnect_at) = self.reconnect_at {
463                        sleep_until(reconnect_at).await;
464                        self.reconnect_at = None;
465                    }
466
467                    debug!("Establishing a new config stream connection to the Core Agent.");
468
469                    // Read the session ID on every reconnect, since it can change in the background due to
470                    // re-registration.
471                    let session_id = self.session_id.wait_for_update().await;
472                    self.events.insert(self.source.open(&session_id))
473                }
474            };
475
476            match events.next().await {
477                Some(Ok(event)) => self.pending = config_event_to_update(event),
478                Some(Err(e)) => error!("Error while reading config event stream: {}.", e),
479                None => {
480                    debug!("Config stream ended, retrying in 5 seconds...");
481                    self.events = None;
482                    self.reconnect_at = Some(Instant::now() + CONFIG_STREAM_RETRY_INTERVAL);
483                }
484            }
485        }
486    }
487}

489impl Resumable for ConfigStream {
490    async fn run(&mut self, stop: ShutdownHandle) -> Result<(), GenericError> {
491        pin!(stop);
492
493        select! {
494            biased;
495            _ = &mut stop => return Ok(()),
496            _ = self.forward() => {}
497        }
498
499        // The receiver is gone. Wait to be stopped rather than return, which would make a worker restart.
500        stop.await;
501        Ok(())
502    }

533/// Creates the processor behind the telemetry service, or `None` if there are no remapper rules.
534///
535/// A `TelemetryProcessor` without rules renders every metric under its raw name, so we skip it entirely rather than
536/// report metrics that no rule selected.
537fn telemetry_processor(remapper_rules: Vec<RemapperRule>) -> Option<Mutex<TelemetryProcessor>> {
538    (!remapper_rules.is_empty()).then(|| Mutex::new(TelemetryProcessor::new().with_remapper_rules(remapper_rules)))
539}

542///
543/// Without a processor, the payload is empty.
544async fn render_telemetry(processor: Option<&Mutex<TelemetryProcessor>>, state: &AggregatedMetricsState) -> String {
545    match processor {
546        Some(processor) => processor.lock().await.process(state),
547        None => String::new(),
548    }

631        target_arch = "aarch64",
632        target_arch = "x86",
633        target_arch = "x86_64",
634        target_arch = "s390x"
635    )
636))]
637const TASK_DUMP_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
638
639/// Timeout for collecting all component-owned diagnostic artifacts.
640///
641/// Each `collect()` closure is run on a blocking thread via `spawn_blocking`. If the overall
642/// budget is exhausted, collection stops and whatever artifacts were gathered are returned.
643/// This ensures that a slow or deadlocked component cannot stall the entire response.
644const DIAGNOSTIC_COLLECT_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(4);
645
646/// Maximum size in bytes for a single diagnostic artifact.
647///
648/// Artifacts that exceed this limit are truncated and a truncation marker is appended so that
649/// readers know the content is incomplete. High-cardinality hosts can produce very large workload
650/// tag dumps; this cap keeps the overall archive size predictable.
651const DIAGNOSTIC_ARTIFACT_MAX_BYTES: usize = 5 * 1024 * 1024; // 5 MiB
652
653/// Truncation marker appended to diagnostic artifacts that exceed [`DIAGNOSTIC_ARTIFACT_MAX_BYTES`] in size.
654const DIAGNOSTIC_TRUNCATION_MARKER: &[u8] = b"\n\n[... truncated: artifact exceeded the 5 MiB per-file limit ...]\n";
655
656/// Caps `data` to [`DIAGNOSTIC_ARTIFACT_MAX_BYTES`], appending [`DIAGNOSTIC_TRUNCATION_MARKER`] if truncated.
657fn cap_artifact_data(mut data: Vec<u8>) -> Vec<u8> {
658    if data.len() > DIAGNOSTIC_ARTIFACT_MAX_BYTES {
659        data.truncate(DIAGNOSTIC_ARTIFACT_MAX_BYTES);
660        data.extend_from_slice(DIAGNOSTIC_TRUNCATION_MARKER);
661    }
662    data
663}

701#[async_trait]
702impl FlareProvider for RemoteAgentImpl {
703    async fn get_flare_files(
704        &self, _request: tonic::Request<GetFlareFilesRequest>,
705    ) -> Result<tonic::Response<GetFlareFilesResponse>, Status> {
706        return self
707            .session_id_middleware(async || {
708                let mut files: HashMap<String, Vec<u8>> = HashMap::new();
709
710                // Grab and run all asserted diagnostic collectors.
711                if let Some(dataspace) = self.dataspace.get() {
712                    let collectors = dataspace.current_values::<DiagnosticCollector>(IdentifierFilter::all());
713                    let total_collectors = collectors.len();
714                    let deadline = Instant::now() + DIAGNOSTIC_COLLECT_TIMEOUT;
715
716                    for collector in collectors {
717                        if Instant::now() >= deadline {
718                            break;
719                        }
720
721                        let artifact_name = collector.artifact_name().to_string();
722                        let remaining = deadline.saturating_duration_since(Instant::now());
723
724                        let collect = timeout(
725                            remaining,
726                            spawn_blocking(move || collector.collect()),
727                        );
728                        let artifact_data = match collect.await {
729                            Ok(Ok(data)) => data,
730                            Ok(Err(e)) => format!("collection panicked for '{artifact_name}': {e}").into_bytes(),
731                            Err(_) => format!("collection timed out for '{artifact_name}'").into_bytes(),
732                        };
733
734                        files.insert(artifact_name, cap_artifact_data(artifact_data));
735                    }
736
737                    if files.len() < total_collectors {
738                        warn!("Diagnostic artifact collection timed out; collected {}/{total_collectors} artifacts.", files.len());
739                    }
740                }
741
742                // Process-level artifacts.
743                //
744                // These don't belong to any individual component — they describe the process
745                // itself. They stay hardwired here since there is no natural component owner to
746                // assert a handle for them.
747
748                // Process info: pid, uptime, RSS, args, fd count, thread count.
749                let pid = std::process::id();
750                let uptime = format_uptime(self.started, Timestamp::now().ok());
751                let args: Vec<String> = std::env::args().collect();
752                let rss_display = MemoryQuerier::default()
753                    .resident_set_size()
754                    .map(|b| b.to_string())
755                    .unwrap_or_else(|| "unavailable".to_string());
756                let process_info = format!(
757                    "pid: {pid}\nuptime_seconds: {uptime}\nrss_bytes: {rss_display}\nargs: {args:?}\nfd_count: {fd_count}\nthread_count: {thread_count}\n",
758                    uptime = uptime,
759                    fd_count = fd_count(),
760                    thread_count = thread_count(),
761                );
762                files.insert("runtime_debug_info.log".to_string(), cap_artifact_data(process_info.into_bytes()));
763
764                // Tokio task dump (only with the `taskdump` feature enabled, on targets where Tokio supports task dumps).
765                //
766                // Wrapped in an explicit timeout because `Handle::dump()`
767                // may never resolve if a runtime worker is blocked for
768                // more than 250ms
769                #[cfg(all(
770                    feature = "taskdump",
771                    target_os = "linux",
772                    any(
773                        target_arch = "aarch64",
774                        target_arch = "x86",
775                        target_arch = "x86_64",
776                        target_arch = "s390x"
777                    )
778                ))]
779                {
780                    let task_dump = match timeout(
781                        TASK_DUMP_TIMEOUT,
782                        tokio::runtime::Handle::current().dump(),
783                    )
784                    .await
785                    {
786                        Ok(dump) => {
787                            let mut output = String::new();
788                            for (i, task) in dump.tasks().iter().enumerate() {
789                                output.push_str(&format!("task {i}:\n{}\n", task.trace()));
790                            }
791                            output
792                        }
793                        Err(_) => format!(
794                            "task dump timed out after {}ms — a runtime worker may be blocked",
795                            TASK_DUMP_TIMEOUT.as_millis()
796                        ),
797                    };
798                    files.insert("runtime-dump.txt".to_string(), cap_artifact_data(task_dump.into_bytes()));
799                }
800
801                Ok(tonic::Response::new(GetFlareFilesResponse { files }))
802            })
803            .await;
804    }
805}

970async fn run_event_reporter(
971    mut client: RemoteAgentClient, session_id: SessionIdHandle, mut subscription: Subscription<DiagnosticEvent>,
972) -> Result<(), GenericError> {
973    debug!("Remote Agent event reporter started.");
974
975    let mut invalid_api_key_reports = InvalidApiKeyReportThrottle::default();
976
977    while let Some(update) = subscription.recv().await {
978        let DataspaceUpdate::Message(_, event) = update else {
979            continue;
980        };
981
982        let Some(remote_agent_event) = diagnostic_to_remote_agent_event(&event) else {
983            debug!("Received diagnostic event with no remote agent event mapping; skipping.");
984            continue;
985        };
986
987        let Some(session_id) = session_id.get() else {
988            debug!("No active remote agent session; dropping diagnostic event.");
989            continue;
990        };
991
992        let invalid_api_key = matches!(event.details(), &DiagnosticDetails::InvalidApiKey);
993        if invalid_api_key && !invalid_api_key_reports.is_allowed() {
994            debug!("Throttling an InvalidApiKey event report.");
995            continue;
996        }
997
998        let request = ReportRemoteAgentEventRequest {
999            session_id: session_id.as_str().to_string(),
1000            events: vec![remote_agent_event],
1001        };
1002
1003        match client.report_remote_agent_event(request).await {
1004            Ok(_) => {
1005                // The window opens only on delivery, so a failed report never holds back the next one.
1006                if invalid_api_key {
1007                    invalid_api_key_reports.record();
1008                }
1009            }
1010            Err(e) => {
1011                warn!(error = %e, "Failed to report remote agent event to the Datadog Agent.");
1012            }
1013        }
1014    }
1015
1016    debug!("Remote Agent event reporter stopped.");
1017    Ok(())
1018}

1020/// Minimum time between `InvalidApiKey` reports.
1021const INVALID_API_KEY_REPORT_INTERVAL: Duration = Duration::from_secs(30);
1022
1023/// Rate limits `InvalidApiKey` reports.
1024///
1025/// Checking and recording are separate so that only a delivered report opens the window: a report that never
1026/// reaches the Core Agent leaves the next one free to send immediately.
1027#[derive(Debug, Default)]
1028struct InvalidApiKeyReportThrottle {
1029    last_reported: Option<Instant>,
1030}
1031
1032impl InvalidApiKeyReportThrottle {
1033    fn is_allowed(&self) -> bool {
1034        self.last_reported
1035            .is_none_or(|last| last.elapsed() >= INVALID_API_KEY_REPORT_INTERVAL)
1036    }
1037
1038    fn record(&mut self) {
1039        self.last_reported = Some(Instant::now());
1040    }
1041}

1043/// Converts a [`DiagnosticEvent`] into the Datadog Agent's remote agent event representation.
1044///
1045/// Returns `None` for diagnostic details that have no corresponding remote agent event variant, in which case the
1046/// event is not reported.
1047fn diagnostic_to_remote_agent_event(event: &DiagnosticEvent) -> Option<RemoteAgentEvent> {
1048    let details = match event.details() {
1049        DiagnosticDetails::InvalidApiKey => RemoteAgentEventDetails::InvalidApiKey(InvalidApiKeyEvent {}),
1050        // `DiagnosticDetails` is `#[non_exhaustive]`; any future variant without a remote agent mapping is skipped.
1051        _ => return None,
1052    };
1053
1054    Some(RemoteAgentEvent {
1055        message: event.message().to_string(),
1056        details: Some(details),
1057    })
1058}

1257        calls: Vec<(u64, Call)>,
1258        registrations: VecDeque<RegisterRemoteAgentResponse>,
1259        rejected_registrations: usize,
1260        failed_refreshes: usize,
1261        hung_refreshes: usize,
1262        registration_delay: Option<Duration>,
1263        streams: VecDeque<mpsc::UnboundedReceiver<ConfigEvent>>,
1264    }

1273    impl FakeAgent {
1274        fn new() -> Self {
1275            Self {
1276                state: Arc::new(std::sync::Mutex::new(FakeAgentState {
1277                    created: Instant::now(),
1278                    calls: Vec::new(),
1279                    registrations: VecDeque::new(),
1280                    rejected_registrations: 0,
1281                    failed_refreshes: 0,
1282                    hung_refreshes: 0,
1283                    registration_delay: None,
1284                    streams: VecDeque::new(),
1285                })),
1286            }
1287        }
1288
1289        fn lock(&self) -> std::sync::MutexGuard<'_, FakeAgentState> {
1290            self.state.lock().unwrap()
1291        }
1292
1293        fn accept_registration(&self, session_id: &str, recommended_refresh_interval_secs: u32) {
1294            self.lock().registrations.push_back(RegisterRemoteAgentResponse {
1295                session_id: session_id.to_string(),
1296                recommended_refresh_interval_secs,
1297            });
1298        }

1300        /// Rejects the next registration, ahead of any scripted response.
1301        fn reject_next_registration(&self) {
1302            self.lock().rejected_registrations += 1;
1303        }
1304
1305        fn fail_next_refresh(&self) {
1306            self.lock().failed_refreshes += 1;
1307        }
1308
1309        /// Never answers the next refresh, ahead of any failed one.
1310        fn hang_next_refresh(&self) {
1311            self.lock().hung_refreshes += 1;
1312        }
1313
1314        /// Answers the next registration only after `delay`.
1315        fn delay_next_registration(&self, delay: Duration) {
1316            self.lock().registration_delay = Some(delay);
1317        }

1319        /// Queues a stream for the next `open`, and returns the sender that feeds it.
1320        fn add_stream(&self) -> mpsc::UnboundedSender<ConfigEvent> {
1321            let (events_tx, events_rx) = mpsc::unbounded_channel();
1322            self.lock().streams.push_back(events_rx);
1323            events_tx
1324        }

1326        fn calls(&self) -> Vec<(u64, Call)> {
1327            self.lock().calls.clone()
1328        }

1330        /// Returns `true` if nothing other than this handle holds the fake.
1331        fn is_released(&self) -> bool {
1332            Arc::strong_count(&self.state) == 1
1333        }

1343    #[async_trait]
1344    impl RegistryClient for FakeAgent {
1345        async fn register(&mut self, _: &RegistrationDetails) -> Result<RegisterRemoteAgentResponse, GenericError> {
1346            let delay = {
1347                let mut state = self.lock();
1348                state.record(Call::Register);
1349                state.registration_delay.take()
1350            };
1351            if let Some(delay) = delay {
1352                tokio::time::sleep(delay).await;
1353            }
1354
1355            let mut state = self.lock();
1356            if state.rejected_registrations > 0 {
1357                state.rejected_registrations -= 1;
1358                return Err(generic_error!("registration rejected"));
1359            }
1360            state
1361                .registrations
1362                .pop_front()
1363                .ok_or_else(|| generic_error!("registration rejected"))
1364        }
1365
1366        async fn refresh(&mut self, session_id: &SessionId) -> Result<(), GenericError> {
1367            let hang = {
1368                let mut state = self.lock();
1369                state.record(Call::Refresh(session_id.to_string()));
1370                if state.hung_refreshes > 0 {
1371                    state.hung_refreshes -= 1;
1372                    true
1373                } else if state.failed_refreshes > 0 {
1374                    state.failed_refreshes -= 1;
1375                    return Err(generic_error!("unknown session"));
1376                } else {
1377                    false
1378                }
1379            };
1380            if hang {
1381                std::future::pending::<()>().await;
1382            }
1383            Ok(())
1384        }
1385    }

1433    /// Stops `worker` the way its owner does: a brutal worker is halted, and a graceful one is signalled.
1434    async fn stop(worker: &dyn Supervisable, coordinator: ShutdownCoordinator, task: tokio::task::JoinHandle<()>) {
1435        match worker.shutdown_strategy() {
1436            ShutdownStrategy::Brutal => {
1437                task.abort();
1438                if let Err(e) = task.await {
1439                    assert!(e.is_cancelled(), "worker task should not panic");
1440                }
1441            }
1442            ShutdownStrategy::Graceful(_) => {
1443                coordinator.shutdown();
1444                tokio::time::timeout(Duration::from_secs(1), task)
1445                    .await
1446                    .expect("worker should stop promptly")
1447                    .expect("worker task should not panic");
1448            }
1449        }
