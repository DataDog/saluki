68///
69/// This helper type is used to coordinate the initialization of remote agent state by registering to the Core Agent and
70/// acquiring the necessary information to allow initialization of the subagent itself to proceed.
71pub struct RemoteAgentBootstrap {
72    client: RemoteAgentClient,
73    session_id: SessionIdHandle,
74    internal_metrics: Reflector<AggregatedMetricsProcessor>,
75    dataspace: Arc<OnceLock<DataspaceRegistry>>,
76}

78impl RemoteAgentBootstrap {
79    /// Creates a new `RemoteAgentBootstrap` from the given configurations.
80    ///
81    /// A remote agent client is created and immediately attempts to register with the Core Agent, advertising
82    /// `secure_api_listen_address` as the address where the Core Agent reaches the status, flare, and telemetry
83    /// services. This function does not return until registration finishes, whether successful or not.
84    ///
85    /// # Errors
86    ///
87    /// If the configuration is invalid, an error is returned.
88    pub async fn new(
89        client_config: &RemoteAgentClientConfiguration, secure_api_listen_address: &ListenAddress,
90    ) -> Result<Self, GenericError> {
91        let api_listen_addr = GrpcTargetAddress::try_from_listen_addr(secure_api_listen_address)
92            .ok_or_else(|| generic_error!("Failed to get valid gRPC target address from secure API listen address."))?;
93
94        // Generate our remote agent state, which is mostly fixed but has a few dynamic bits.
95        let service_names = vec![
96            <StatusProviderServer<()> as NamedService>::NAME.to_string(),
97            <FlareProviderServer<()> as NamedService>::NAME.to_string(),
98            <TelemetryProviderServer<()> as NamedService>::NAME.to_string(),
99        ];
100
101        let (state, init_reg_rx) = RemoteAgentState::new(api_listen_addr, service_names);
102        let session_id = state.session_id.clone();
103
104        // Create our client, and then immediately start the registration loop.
105        //
106        // Wait for the result of the initial registration attempt before proceeding.
107        let client = RemoteAgentClient::connect(client_config).await?;
108        let task_name = format!("{}-remote-agent-task", saluki_metadata::get_app_details().identifier());
109        spawn_traced_named(task_name, run_remote_agent_registration_loop(client.clone(), state));
110
111        match init_reg_rx.await {
112            Ok(Ok(())) => (),
113            Ok(Err(e)) => return Err(e),
114            Err(_) => {
115                return Err(generic_error!(
116                    "Failed to initialize remote agent state. Registration task failed unexpectedly."
117                ))
118            }
119        }
120
121        Ok(Self {
122            client,
123            session_id,
124            internal_metrics: get_shared_metrics_state(),
125            dataspace: Arc::new(OnceLock::new()),
126        })
127    }

164    ///

213impl RemoteAgentState {
214    fn new(
215        api_listen_addr: GrpcTargetAddress, service_names: Vec<String>,
216    ) -> (Self, oneshot::Receiver<Result<(), GenericError>>) {
217        let app_details = saluki_metadata::get_app_details();
218        let display_name = app_details.full_name().to_string();
219        let flavor = app_details.full_name().replace(" ", "_").to_lowercase();
220
221        let (init_reg_tx, init_reg_rx) = oneshot::channel();
222
223        let state = Self {
224            pid: std::process::id(),
225            display_name,
226            flavor,
227            api_listen_addr: api_listen_addr.to_string(),
228            session_id: SessionIdHandle::empty(),
229            service_names,
230            initial_registration_tx: Some(init_reg_tx),
231        };
232
233        (state, init_reg_rx)
234    }
235}

237async fn run_remote_agent_registration_loop(mut client: RemoteAgentClient, mut state: RemoteAgentState) {
238    let mut loop_timer = interval(DEFAULT_REFRESH_INTERVAL);
239    loop_timer.set_missed_tick_behavior(MissedTickBehavior::Delay);
240
241    debug!("Remote Agent registration task started.");
242
243    loop {
244        loop_timer.tick().await;
245
246        match state.session_id.get() {
247            Some(session_id) => {
248                debug!(%session_id, "Refreshing registration with Datadog Agent.");
249
250                if client.refresh_remote_agent(&session_id).await.is_err() {
251                    loop_timer.reset_after(REFRESH_FAILED_RETRY_INTERVAL);
252                    state.session_id.update(None);
253                    warn!("Failed to refresh registration with the Datadog Agent. Resetting session ID and attempting to re-register shortly.");
254
255                    continue;
256                }
257            }
258            None => {
259                match client
260                    .register_remote_agent(
261                        state.pid,
262                        &state.display_name,
263                        &state.flavor,
264                        &state.api_listen_addr,
265                        state.service_names.clone(),
266                    )
267                    .await
268                {
269                    Ok(resp) => {
270                        let resp = resp.into_inner();
271                        let new_session_id = match SessionId::new(&resp.session_id) {
272                            Ok(session_id) => session_id,
273                            Err(e) => {
274                                warn!(error = %e, "Received invalid session ID from Datadog Agent after registation. Registration will be retried periodically in the background.");
275                                loop_timer.reset_after(DEFAULT_REFRESH_INTERVAL);
276                                continue;
277                            }
278                        };
279                        let new_refresh_interval = resp.recommended_refresh_interval_secs;
280                        info!(session_id = %new_session_id, "Successfully registered with the Datadog Agent. Refreshing every {} seconds.", new_refresh_interval);
281
282                        state.session_id.update(Some(new_session_id));
283                        loop_timer.reset_after(Duration::from_secs(new_refresh_interval as u64));
284
285                        if let Some(tx) = state.initial_registration_tx.take() {
286                            let _ = tx.send(Ok(()));
287                        }
288                    }
289                    Err(e) => {
290                        warn!(error = %e, "Failed to register with the Datadog Agent. Registration will be retried periodically in the background.");
291                        loop_timer.reset_after(DEFAULT_REFRESH_INTERVAL);
292
293                        if let Some(tx) = state.initial_registration_tx.take() {
294                            let _ = tx.send(Err(e));
295                        }
296                    }
297                }
298            }
299        }
300    }
301}

303async fn run_config_stream_event_loop(
304    mut client: RemoteAgentClient, sender: mpsc::Sender<ConfigUpdate>, session_id: SessionIdHandle,
305) {
306    loop {
307        debug!("Establishing a new config stream connection to the Core Agent.");
308
309        // Read the current session ID.
310        //
311        // We do this every loop since it can change in the background due to re-registration.
312        let current_session_id = session_id.wait_for_update().await;
313
314        let mut stream = client.stream_config_events(&current_session_id);
315        while let Some(result) = stream.next().await {
316            match result {
317                Ok(event) => {
318                    let update = match event.event {
319                        Some(config_event::Event::Snapshot(snapshot)) => {
320                            Some(ConfigUpdate::Snapshot(snapshot_to_settings(&snapshot)))
321                        }
322                        Some(config_event::Event::Update(update)) => update
323                            .setting
324                            .as_ref()
325                            .map(|setting| ConfigUpdate::Partial(setting_to_config_setting(setting))),
326                        None => {
327                            error!("Received a configuration update event with no data.");
328                            None
329                        }
330                    };
331
332                    if let Some(update) = update {
333                        if sender.send(update).await.is_err() {
334                            warn!("Dynamic configuration channel closed. Config stream shutting down.");
335                            return;
336                        }
337                    }
338                }
339                Err(e) => {
340                    error!("Error while reading config event stream: {}.", e);
341                }
342            }
343        }
344
345        debug!("Config stream ended, retrying in 5 seconds...");
346        tokio::time::sleep(Duration::from_secs(5)).await;
347    }
348}

350/// Sources that indicate the Agent supplied the value rather than an operator.
351const AGENT_DEFAULT_SOURCE: &str = "default";
352/// A value that was not set by the user nor does the schema define default value for.
353const AGENT_DECLARED_ONLY_SOURCE: &str = "schema";
354const AGENT_UNSET_SOURCES: [&str; 2] = [AGENT_DEFAULT_SOURCE, AGENT_DECLARED_ONLY_SOURCE];
355
356/// Converts a setting from the Agent's RPC wire protocol to our `ConfigSetting` type.
357fn setting_to_config_setting(setting: &AgentConfigSetting) -> ConfigSetting {
358    let provenance = if AGENT_UNSET_SOURCES.contains(&setting.source.as_str()) {
359        Provenance::Default
360    } else {
361        Provenance::Explicit
362    };
363
364    ConfigSetting::new(
365        setting.key.clone(),
366        proto_value_to_serde_value(&setting.value),
367        provenance,
368    )
369}
370
371/// Converts a `ConfigSnapshot` into the settings it carries.
372fn snapshot_to_settings(snapshot: &ConfigSnapshot) -> Vec<ConfigSetting> {
373    snapshot.settings.iter().map(setting_to_config_setting).collect()
374}

376/// Recursively converts a `google::protobuf::Value` into a `serde_json::Value`.
377fn proto_value_to_serde_value(proto_val: &Option<prost_types::Value>) -> Value {
378    let Some(kind) = proto_val.as_ref().and_then(|v| v.kind.as_ref()) else {
379        return Value::Null;
380    };
381
382    match kind {
383        Kind::NullValue(_) => Value::Null,
384        Kind::NumberValue(n) => {
385            if n.fract() == 0.0 && *n >= i64::MIN as f64 && *n <= i64::MAX as f64 {
386                Value::from(*n as i64)
387            } else {
388                Value::from(*n)
389            }
390        }
391        Kind::StringValue(s) => Value::String(s.clone()),
392        Kind::BoolValue(b) => Value::Bool(*b),
393        Kind::StructValue(s) => {
394            let json_map: Map<String, Value> = s
395                .fields
396                .iter()
397                .map(|(k, v)| (k.clone(), proto_value_to_serde_value(&Some(v.clone()))))
398                .collect();
399            Value::Object(json_map)
400        }
401        Kind::ListValue(l) => {
402            let json_list: Vec<Value> = l
403                .values
404                .iter()
405                .map(|v| proto_value_to_serde_value(&Some(v.clone())))
406                .collect();
407            Value::Array(json_list)
408        }
409    }
410}

412/// The implementation behind the status, flare, and telemetry services that the Datadog Agent queries.
413///
414/// Created through [`RemoteAgentBootstrap::create_status_service`], [`RemoteAgentBootstrap::create_flare_service`],
415/// and [`RemoteAgentBootstrap::create_telemetry_service`].
416pub struct RemoteAgentImpl {
417    started: DateTime<Utc>,
418    internal_metrics: Reflector<AggregatedMetricsProcessor>,
419    processor: Option<Mutex<TelemetryProcessor>>,
420    status_sections: Vec<Arc<dyn StatusSectionProvider>>,
421    session_id: SessionIdHandle,
422    dataspace: Arc<OnceLock<DataspaceRegistry>>,
423}

425/// Creates the processor behind the telemetry service, or `None` if there are no remapper rules.
426///
427/// A `TelemetryProcessor` without rules renders every metric under its raw name, so we skip it entirely rather than
428/// report metrics that no rule selected.
429fn telemetry_processor(remapper_rules: Vec<RemapperRule>) -> Option<Mutex<TelemetryProcessor>> {
430    (!remapper_rules.is_empty()).then(|| Mutex::new(TelemetryProcessor::new().with_remapper_rules(remapper_rules)))
431}
432
433/// Renders the telemetry payload for the given internal metrics.
434///
435/// Without a processor, the payload is empty.
436async fn render_telemetry(processor: Option<&Mutex<TelemetryProcessor>>, state: &AggregatedMetricsState) -> String {
437    match processor {
438        Some(processor) => processor.lock().await.process(state),
439        None => String::new(),
440    }
441}

443impl RemoteAgentImpl {
444    async fn session_id_middleware<Resp, Next>(&self, next: Next) -> Result<tonic::Response<Resp>, Status>
445    where
446        Next: AsyncFnOnce() -> Result<tonic::Response<Resp>, Status>,
447    {
448        let metadata_session_id = self
449            .session_id
450            .get()
451            .ok_or(Status::failed_precondition(
452                "session ID not set; must be registered with Core Agent",
453            ))?
454            .to_grpc_header_value();
455
456        next().await.map(|mut resp| {
457            resp.metadata_mut().append(SESSION_ID_METADATA_KEY, metadata_session_id);
458            resp
459        })
460    }
461}

514/// Timeout for the tokio task dump step within [`FlareProvider::get_flare_files`].
515///
516/// `Handle::dump()` may never resolve if a runtime worker is blocked for more than 250 ms. We use
517/// a conservative 5-second budget so that a hung runtime still allows the other diagnostic artifacts
518/// to be collected and returned.
519#[cfg(all(
520    feature = "taskdump",
521    target_os = "linux",
522    any(
523        target_arch = "aarch64",
524        target_arch = "x86",
525        target_arch = "x86_64",
526        target_arch = "s390x"
527    )
528))]
529const TASK_DUMP_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
530
531/// Timeout for collecting all component-owned diagnostic artifacts.
532///
533/// Each `collect()` closure is run on a blocking thread via `spawn_blocking`. If the overall
534/// budget is exhausted, collection stops and whatever artifacts were gathered are returned.
535/// This ensures that a slow or deadlocked component cannot stall the entire response.
536const DIAGNOSTIC_COLLECT_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(4);
537
538/// Maximum size in bytes for a single diagnostic artifact.
539///
540/// Artifacts that exceed this limit are truncated and a truncation marker is appended so that
541/// readers know the content is incomplete. High-cardinality hosts can produce very large workload
542/// tag dumps; this cap keeps the overall archive size predictable.
543const DIAGNOSTIC_ARTIFACT_MAX_BYTES: usize = 5 * 1024 * 1024; // 5 MiB
544
545/// Truncation marker appended to diagnostic artifacts that exceed [`DIAGNOSTIC_ARTIFACT_MAX_BYTES`] in size.
546const DIAGNOSTIC_TRUNCATION_MARKER: &[u8] = b"\n\n[... truncated: artifact exceeded the 5 MiB per-file limit ...]\n";
547
548/// Caps `data` to [`DIAGNOSTIC_ARTIFACT_MAX_BYTES`], appending [`DIAGNOSTIC_TRUNCATION_MARKER`] if truncated.
549fn cap_artifact_data(mut data: Vec<u8>) -> Vec<u8> {
550    if data.len() > DIAGNOSTIC_ARTIFACT_MAX_BYTES {
551        data.truncate(DIAGNOSTIC_ARTIFACT_MAX_BYTES);
552        data.extend_from_slice(DIAGNOSTIC_TRUNCATION_MARKER);
553    }
554    data
555}

557/// Returns the number of open file descriptors for the current process.
558///
559/// Reads from `/proc/self/fd` on Linux. Returns `"unavailable"` on other platforms.
560#[cfg(target_os = "linux")]
561fn fd_count() -> String {
562    std::fs::read_dir("/proc/self/fd")
563        .map(|entries| entries.count().to_string())
564        .unwrap_or_else(|_| "unavailable".to_string())
565}

568fn fd_count() -> String {
569    "unavailable".to_string()
570}

572/// Returns the number of threads in the current process.
573///
574/// Reads the `Threads:` field from `/proc/self/status` on Linux. Returns `"unavailable"` on other
575/// platforms.
576#[cfg(target_os = "linux")]
577fn thread_count() -> String {
578    std::fs::read_to_string("/proc/self/status")
579        .ok()
580        .and_then(|s| {
581            s.lines()
582                .find(|l| l.starts_with("Threads:"))
583                .and_then(|l| l.split_whitespace().nth(1).map(str::to_string))
584        })
585        .unwrap_or_else(|| "unavailable".to_string())
586}

588#[cfg(not(target_os = "linux"))]
589fn thread_count() -> String {
590    "unavailable".to_string()

593#[async_trait]
594impl FlareProvider for RemoteAgentImpl {
595    async fn get_flare_files(
596        &self, _request: tonic::Request<GetFlareFilesRequest>,
597    ) -> Result<tonic::Response<GetFlareFilesResponse>, Status> {
598        return self
599            .session_id_middleware(async || {
600                let mut files: HashMap<String, Vec<u8>> = HashMap::new();
601
602                // Grab and run all asserted diagnostic collectors.
603                if let Some(dataspace) = self.dataspace.get() {
604                    let collectors = dataspace.current_values::<DiagnosticCollector>(IdentifierFilter::all());
605                    let total_collectors = collectors.len();
606                    let deadline = Instant::now() + DIAGNOSTIC_COLLECT_TIMEOUT;
607
608                    for collector in collectors {
609                        if Instant::now() >= deadline {
610                            break;
611                        }
612
613                        let artifact_name = collector.artifact_name().to_string();
614                        let remaining = deadline.saturating_duration_since(Instant::now());
615
616                        let collect = timeout(
617                            remaining,
618                            spawn_blocking(move || collector.collect()),
619                        );
620                        let artifact_data = match collect.await {
621                            Ok(Ok(data)) => data,
622                            Ok(Err(e)) => format!("collection panicked for '{artifact_name}': {e}").into_bytes(),
623                            Err(_) => format!("collection timed out for '{artifact_name}'").into_bytes(),
624                        };
625
626                        files.insert(artifact_name, cap_artifact_data(artifact_data));
627                    }
628
629                    if files.len() < total_collectors {
630                        warn!("Diagnostic artifact collection timed out; collected {}/{total_collectors} artifacts.", files.len());
631                    }
632                }
633
634                // Process-level artifacts.
635                //
636                // These don't belong to any individual component — they describe the process
637                // itself. They stay hardwired here since there is no natural component owner to
638                // assert a handle for them.
639
640                // Process info: pid, uptime, RSS, args, fd count, thread count.
641                let pid = std::process::id();
642                let uptime = Utc::now().signed_duration_since(self.started);
643                let args: Vec<String> = std::env::args().collect();
644                let rss_display = MemoryQuerier::default()
645                    .resident_set_size()
646                    .map(|b| b.to_string())
647                    .unwrap_or_else(|| "unavailable".to_string());
648                let process_info = format!(
649                    "pid: {pid}\nuptime_seconds: {uptime}\nrss_bytes: {rss_display}\nargs: {args:?}\nfd_count: {fd_count}\nthread_count: {thread_count}\n",
650                    uptime = uptime.num_seconds(),
651                    fd_count = fd_count(),
652                    thread_count = thread_count(),
653                );
654                files.insert("runtime_debug_info.log".to_string(), cap_artifact_data(process_info.into_bytes()));
655
656                // Tokio task dump (only with the `taskdump` feature enabled, on targets where Tokio supports task dumps).
657                //
658                // Wrapped in an explicit timeout because `Handle::dump()`
659                // may never resolve if a runtime worker is blocked for
660                // more than 250ms
661                #[cfg(all(
662                    feature = "taskdump",
663                    target_os = "linux",
664                    any(
665                        target_arch = "aarch64",
666                        target_arch = "x86",
667                        target_arch = "x86_64",
668                        target_arch = "s390x"
669                    )
670                ))]
671                {
672                    let task_dump = match timeout(
673                        TASK_DUMP_TIMEOUT,
674                        tokio::runtime::Handle::current().dump(),
675                    )
676                    .await
677                    {
678                        Ok(dump) => {
679                            let mut output = String::new();
680                            for (i, task) in dump.tasks().iter().enumerate() {
681                                output.push_str(&format!("task {i}:\n{}\n", task.trace()));
682                            }
683                            output
684                        }
685                        Err(_) => format!(
686                            "task dump timed out after {}ms — a runtime worker may be blocked",
687                            TASK_DUMP_TIMEOUT.as_millis()
688                        ),
689                    };
690                    files.insert("runtime-dump.txt".to_string(), cap_artifact_data(task_dump.into_bytes()));
691                }
692
693                Ok(tonic::Response::new(GetFlareFilesResponse { files }))
694            })
695            .await;
696    }
697}

699/// A worker that captures the dataspace from the supervisor context and stores it in a shared
700/// [`OnceLock`] for use by the diagnostic artifact collection service.
701///
702/// The gRPC handler runs in tasks spawned by tonic, which do not inherit tokio task-locals.
703/// This worker bridges that gap by capturing the dataspace during its own `initialize()` call,
704/// which runs inside the supervisor process where the dataspace is available.
705pub struct DataspaceAnchorWorker {
706    dataspace: Arc<OnceLock<DataspaceRegistry>>,
707}
708
709#[async_trait]
710impl Supervisable for DataspaceAnchorWorker {
711    fn name(&self) -> &str {
712        "flare-dataspace-anchor"
713    }
714
715    async fn initialize(&self, process_shutdown: ShutdownHandle) -> Result<SupervisorFuture, InitializationError> {
716        if let Some(ds) = DataspaceRegistry::try_current() {
717            let _ = self.dataspace.set(ds);
718        }
719
720        Ok(Box::pin(async move {
721            process_shutdown.await;
722            Ok(())
723        }))
724    }
725}

727/// A worker that reports diagnostic events to the Core Agent as remote agent events.
728///
729/// It subscribes to [`DiagnosticEvent`]s emitted anywhere in the process converts the ones it recognizes into the
730/// Datadog Agent's remote agent event representation, and reports them to the Core Agent over the remote agent gRPC
731/// API.
732pub struct RemoteAgentEventReporter {
733    client: RemoteAgentClient,
734    session_id: SessionIdHandle,
735}
736
737#[async_trait]
738impl Supervisable for RemoteAgentEventReporter {
739    fn name(&self) -> &str {
740        "remote_agent_event_reporter"
741    }
742
743    async fn initialize(&self, process_shutdown: ShutdownHandle) -> Result<SupervisorFuture, InitializationError> {
744        let client = self.client.clone();
745        let session_id = self.session_id.clone();
746
747        Ok(Box::pin(async move {
748            let subscription = subscribe_events(IdentifierFilter::all())
749                .map_err(|e| generic_error!("Failed to subscribe to diagnostic events: {}", e))?;
750
751            select! {
752                _ = process_shutdown => Ok(()),
753                result = run_event_reporter(client, session_id, subscription) => result,
754            }
755        }))
756    }
757}
758
759async fn run_event_reporter(
760    mut client: RemoteAgentClient, session_id: SessionIdHandle, mut subscription: Subscription<DiagnosticEvent>,
761) -> Result<(), GenericError> {
762    debug!("Remote Agent event reporter started.");
763
764    while let Some(update) = subscription.recv().await {
765        let DataspaceUpdate::Message(_, event) = update else {
766            continue;
767        };
768
769        let Some(remote_agent_event) = diagnostic_to_remote_agent_event(&event) else {
770            debug!("Received diagnostic event with no remote agent event mapping; skipping.");
771            continue;
772        };
773
774        let Some(session_id) = session_id.get() else {
775            debug!("No active remote agent session; dropping diagnostic event.");
776            continue;
777        };
778
779        let request = ReportRemoteAgentEventRequest {
780            session_id: session_id.as_str().to_string(),
781            events: vec![remote_agent_event],
782        };
783
784        if let Err(e) = client.report_remote_agent_event(request).await {
785            warn!(error = %e, "Failed to report remote agent event to the Datadog Agent.");
786        }
787    }
788
789    debug!("Remote Agent event reporter stopped.");
790    Ok(())
791}

793/// Converts a [`DiagnosticEvent`] into the Datadog Agent's remote agent event representation.
794///
795/// Returns `None` for diagnostic details that have no corresponding remote agent event variant, in which case the
796/// event is not reported.
797fn diagnostic_to_remote_agent_event(event: &DiagnosticEvent) -> Option<RemoteAgentEvent> {
798    let details = match event.details() {
799        DiagnosticDetails::InvalidApiKey => RemoteAgentEventDetails::InvalidApiKey(InvalidApiKeyEvent {}),
800        // `DiagnosticDetails` is `#[non_exhaustive]`; any future variant without a remote agent mapping is skipped.
801        _ => return None,
802    };
803
804    Some(RemoteAgentEvent {
805        message: event.message().to_string(),
806        details: Some(details),
807    })
808}
