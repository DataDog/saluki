18/// Creates the identity a remote agent sends when polling the Datadog Agent for Remote Configuration.
19///
20/// Uses the same name as the remote agent's separate configuration stream, plus this build's version.
21fn rc_identity(app_details: &AppDetails) -> ClientKind {
22    ClientKind::Agent(AgentIdentity::new(
23        client_name(app_details),
24        app_details.version().raw(),
25    ))
26}
27
28/// Connects to the Datadog Agent, and creates a Remote Configuration client and its polling worker.
29///
30/// # Errors
31///
32/// If the Datadog Agent can't be reached, or the client can't be created, an error is returned.
33pub(super) async fn connect(
34    client_config: &RemoteAgentClientConfiguration,
35) -> Result<(RemoteConfigurationClient, RemoteConfigurationWorker), GenericError> {
36    // Remote Configuration has its own Agent connection, separate from other remote agent services.
37    let agent = RemoteAgentClient::connect(client_config)
38        .await
39        .error_context("Failed to connect to the Datadog Agent for Remote Configuration.")?;
40    let config = RcClientConfiguration::new(rc_identity(saluki_metadata::get_app_details()));
41    Ok(RemoteConfigurationClient::new(agent, config)?)
42}

44/// Runs the Remote Configuration polling worker under its own supervisor.
45///
46/// The worker can restart five times within 60 seconds. If it keeps failing, polling stops but the rest of the remote
47/// agent keeps running with the last accepted settings. The worker's `initialize` must not fail: initialization errors
48/// stop the remote agent rather than triggering a restart.
49pub(super) fn remote_config_child(
50    worker: impl Supervisable + 'static,
51) -> Result<ChildSpecification<SupervisorSpec>, SupervisorError> {
52    let mut supervisor = Supervisor::new("remote-config")?
53        .with_restart_strategy(RestartStrategy::one_to_one().with_intensity_and_period(5, Duration::from_secs(60)));
54    supervisor.add_worker(worker);
55    // `hosted()` would propagate this subtree's restart exhaustion to the root, which would stop the remote agent.
56    // `temporary()` keeps the finite RC-only restart budget isolated.
57    #[allow(deprecated)]
58    let child = nested_supervisor(supervisor).temporary().build();
59    Ok(child)
60}

114    async fn wait_until(what: &str, condition: impl Fn() -> bool) {
115        tokio::time::timeout(Duration::from_secs(10), async {
116            while !condition() {
117                tokio::time::sleep(Duration::from_millis(10)).await;
118            }
119        })
120        .await
121        .unwrap_or_else(|_| panic!("timed out waiting until {what}"));
122    }

124    #[tokio::test]
125    async fn exhausted_remote_config_child_stops_without_stopping_root() {
126        let sibling_starts = Arc::new(AtomicUsize::new(0));
127        let worker_starts = Arc::new(AtomicUsize::new(0));
128
129        // Match the facade root's no-restart strategy; the sibling stands in for the topology.
130        let mut root = Supervisor::new("root")
131            .unwrap()
132            .with_restart_strategy(RestartStrategy::new(RestartMode::OneForOne, 0, Duration::from_secs(5)));
133        root.add_worker(LongRunning {
134            starts: Arc::clone(&sibling_starts),
135        });
136        root.add_worker(
137            remote_config_child(AlwaysPanics {
138                starts: Arc::clone(&worker_starts),
139            })
140            .unwrap(),
141        );
142
143        let (shutdown_tx, shutdown_rx) = oneshot::channel::<()>();
144        let run = tokio::spawn(async move { root.run_with_shutdown(shutdown_rx).await });
145
146        // The first run and five restarts, then the sixth failure stops the child without stopping root.
147        wait_until("the worker has started six times", || {
148            worker_starts.load(Ordering::SeqCst) >= 6
149        })
150        .await;
151        tokio::time::sleep(Duration::from_millis(500)).await;
152        assert_eq!(worker_starts.load(Ordering::SeqCst), 6);
153        assert_eq!(sibling_starts.load(Ordering::SeqCst), 1);
154        assert!(!run.is_finished(), "root stopped with the remote-config child");
155
156        shutdown_tx.send(()).unwrap();
157        let result = tokio::time::timeout(Duration::from_secs(5), run)
158            .await
159            .expect("root should shut down promptly")
160            .expect("root task should not panic");
161        assert!(result.is_ok(), "root shut down with {result:?}");
162    }
