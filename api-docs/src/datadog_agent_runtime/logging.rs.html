1//! Translation between a remote agent's logging settings and the logging stack's own configuration.
2//!
3//! Remote agent logging must follow the Datadog Agent's logging configuration for the settings that are sensibly
4//! shared (level, format, console output, rotation), but each remote agent must use its own destination so it doesn't
5//! collide with the Core Agent's own log file. This module owns those rules in one place.
6
7use std::{future, path::PathBuf};
8
9use bytesize::ByteSize;
10use datadog_agent_commons::platform::PlatformSettings;
11use futures::{Stream, StreamExt as _};
12use saluki_app::logging::{LogLevel, LoggingConfiguration};
13use saluki_error::{generic_error, ErrorContext as _, GenericError};
14use tracing::warn;
15
16/// Log targets that a plain log level applies to, regardless of which remote agent is running.
17///
18/// These cover the Saluki and Datadog crates that remote agents are built from. `tracing` targets use Rust crate and
19/// module names, so Cargo package names with hyphens appear with underscores. A remote agent adds its own crates with
20/// [`LoggingTranslator::with_first_party_targets`].
21///
22/// Each entry is a target prefix, so `saluki` covers every `saluki_*` crate and `datadog_agent` covers every
23/// `datadog_agent_*` crate. A prefix also covers any third-party crate whose name starts with it, which is why the list
24/// uses `datadog_agent` rather than `datadog`.
25pub const FIRST_PARTY_LOG_TARGETS: &[&str] = &["datadog_agent", "saluki"];
26
27/// Logging settings shared with the Datadog Agent.
28///
29/// These mirror the Datadog Agent's own logging settings. A remote agent fills them in from its configuration system
30/// and hands them to a [`LoggingTranslator`].
31#[derive(Clone, Debug)]
32#[non_exhaustive]
33pub struct LoggingSettings {
34    /// Log level: either a plain level name (`trace`, `debug`, `info`, `warn`, `error`, or `off`) or a comma-separated
35    /// list of filter directives.
36    ///
37    /// A plain level name only applies to first-party log targets. Defaults to `info`.
38    pub level: String,
39
40    /// Whether to emit log records as JSON. Defaults to `false`.
41    pub format_json: bool,
42
43    /// Whether to use RFC 3339 timestamps in log output. Defaults to `false`.
44    pub format_rfc3339: bool,
45
46    /// Whether to write log records to standard output. Defaults to `true`.
47    pub to_console: bool,
48
49    /// Whether to write log records to syslog. Defaults to `false`.
50    pub to_syslog: bool,
51
52    /// Whether to use the Agent's RFC-style syslog header. Only applies when `to_syslog` is `true`. Defaults to
53    /// `false`.
54    pub syslog_rfc: bool,
55
56    /// URI of the syslog destination. Only applies when `to_syslog` is `true`.
57    ///
58    /// Defaults to empty, which selects the platform's default local syslog URI.
59    pub syslog_uri: String,
60
61    /// Path to the log file, if one was configured explicitly.
62    ///
63    /// Defaults to `None`, which selects the remote agent's default log file. An explicitly empty path also selects
64    /// the default log file.
65    pub file: Option<String>,
66
67    /// Whether to disable logging to a file entirely. Takes precedence over `file`. Defaults to `false`.
68    pub disable_file_logging: bool,
69
70    /// Maximum number of rolled-over log files to retain. Defaults to `1`.
71    pub file_max_rolls: usize,
72
73    /// Maximum size, in bytes, of a log file before it's rolled over, if one was configured explicitly.
74    ///
75    /// Defaults to `None`, which keeps the logging stack's own default of 10 MiB.
76    pub file_max_size: Option<u64>,
77}
78
79impl Default for LoggingSettings {
80    fn default() -> Self {
81        Self {
82            level: "info".to_string(),
83            format_json: false,
84            format_rfc3339: false,
85            to_console: true,
86            to_syslog: false,
87            syslog_rfc: false,
88            syslog_uri: String::new(),
89            file: None,
90            disable_file_logging: false,
91            file_max_rolls: 1,
92            file_max_size: None,
93        }
94    }
95}

97/// Logging configuration translator for matching the Datadog Agent's logging behavior.
98///
99/// In the Datadog Agent, all processes generally follow the same logging configuration, paying attention to the same
100/// settings for determining log level, log format, and so on. They differ in some ways, such as determining what file
101/// to write to when logging to file is enabled. Remote agents follow the same pattern.
102///
103/// `LoggingTranslator` takes [`LoggingSettings`] and generates a Saluki-oriented [`LoggingConfiguration`] from it,
104/// applying the remote agent's own default log file and first-party log targets. This ensures that a remote agent obeys
105/// all the logging configuration rules set by the Datadog Agent but logs to the right location for its own process.
106#[derive(Clone, Debug)]
107pub struct LoggingTranslator {
108    default_log_file: PathBuf,
109    first_party_targets: Vec<&'static str>,
110}
111
112impl LoggingTranslator {
113    /// Creates a new `LoggingTranslator` that logs to `default_log_file` unless a log file is configured explicitly.
114    ///
115    /// A plain log level applies to [`FIRST_PARTY_LOG_TARGETS`]. Add the remote agent's own crates with
116    /// [`with_first_party_targets`](Self::with_first_party_targets).
117    pub fn new(default_log_file: impl Into<PathBuf>) -> Self {
118        Self {
119            default_log_file: default_log_file.into(),
120            first_party_targets: FIRST_PARTY_LOG_TARGETS.to_vec(),
121        }
122    }
123
124    /// Adds the given log targets to the ones that a plain log level applies to.
125    ///
126    /// Like the entries in [`FIRST_PARTY_LOG_TARGETS`], each target is a prefix, so `my_remote_agent` also covers a
127    /// `my_remote_agent_config` crate.
128    pub fn with_first_party_targets(mut self, targets: &[&'static str]) -> Self {
129        self.first_party_targets.extend_from_slice(targets);
130        self
131    }
132
133    /// Builds a [`LoggingConfiguration`] from the given settings, applying this remote agent's rules.
134    ///
135    /// # Errors
136    ///
137    /// Returns an error if the configured log level is not a level name or a valid set of filter directives.
138    pub fn translate(&self, settings: LoggingSettings) -> Result<LoggingConfiguration, GenericError> {
139        let mut config = LoggingConfiguration::simple();
140
141        config.log_level = self.parse_log_level(&settings.level)?;
142        config.log_format_json = settings.format_json;
143        config.log_format_rfc3339 = settings.format_rfc3339;
144        config.log_to_console = settings.to_console;
145        config.log_to_syslog = settings.to_syslog;
146
147        if settings.to_syslog {
148            config.syslog_rfc = settings.syslog_rfc;
149            config.syslog_uri = if settings.syslog_uri.is_empty() {
150                PlatformSettings::get_default_syslog_uri().to_string()
151            } else {
152                settings.syslog_uri
153            };
154        }
155
156        // Preserve the logging stack's binary default when no maximum size was configured.
157        if let Some(file_max_size) = settings.file_max_size {
158            config.log_file_max_size = ByteSize::b(file_max_size);
159        }
160        config.log_file_max_rolls = settings.file_max_rolls;
161
162        // Use the remote agent's default unless its log file was set explicitly.
163        config.log_file = if settings.disable_file_logging {
164            String::new()
165        } else {
166            match &settings.file {
167                Some(file) if !file.is_empty() => file.clone(),
168                _ => self.default_log_file.to_string_lossy().into_owned(),
169            }
170        };
171
172        Ok(config)
173    }
174
175    /// Translates each log level from `levels` into the form that
176    /// [`DynamicLogLevelWorker`][saluki_app::logging::DynamicLogLevelWorker] applies.
177    ///
178    /// Each level is parsed the same way as [`LoggingSettings::level`] in [`translate`](Self::translate), so a plain
179    /// level applies to the same first-party log targets as at startup. A level that can't be parsed is logged and
180    /// skipped, which leaves the current log level in place.
181    pub fn translate_log_levels<S>(&self, levels: S) -> impl Stream<Item = LogLevel> + Send + 'static
182    where
183        S: Stream<Item = String> + Send + 'static,
184    {
185        let translator = self.clone();
186        levels.filter_map(move |level| {
187            future::ready(match translator.parse_log_level(&level) {
188                Ok(log_level) => Some(log_level),
189                Err(e) => {
190                    warn!(error = %e, "Failed to parse updated log level.");
191                    None
192                }
193            })
194        })
195    }
196
197    /// Parses a configured log level, expanding a plain level name into per-target directives.
198    ///
199    /// Plain levels apply to first-party log targets; other values are `target=level` filter directives.
200    fn parse_log_level(&self, value: &str) -> Result<LogLevel, GenericError> {
201        let trimmed = value.trim();
202        if let Some(level) = plain_log_level(trimmed) {
203            self.first_party_log_level_filter(level)
204        } else if !trimmed.contains('=') {
205            // We want to check here to make sure there's a target directive (for example, `foo=info` or
206            // `foo=trace,info`) as if there's not, it means we got a bare level directive (equivalent to getting only
207            // `info`, etc), but with an _invalid_ level.
208            Err(generic_error!("Failed to parse invalid log level '{trimmed}'."))
209        } else {
210            LogLevel::try_from(value.to_string()).error_context("Failed to parse log filter directives.")
211        }
212    }
213
214    fn first_party_log_level_filter(&self, level: &str) -> Result<LogLevel, GenericError> {
215        let filter = self
216            .first_party_targets
217            .iter()
218            .map(|target| format!("{target}={level}"))
219            .collect::<Vec<_>>()
220            .join(",");
221
222        LogLevel::try_from(filter).error_context("Failed to parse first-party log filter directives.")
223    }
224}
225
226fn plain_log_level(value: &str) -> Option<&'static str> {
227    match value.to_ascii_lowercase().as_str() {
228        "trace" => Some("trace"),
229        "debug" => Some("debug"),
230        "info" => Some("info"),
231        "warn" | "warning" => Some("warn"),
232        "error" | "critical" => Some("error"),
233        "off" => Some("off"),
234        _ => None,
235    }
236}

245    const DEFAULT_LOG_FILE: &str = "/var/log/datadog/example-remote-agent.log";

247    fn translator() -> LoggingTranslator {
248        LoggingTranslator::new(DEFAULT_LOG_FILE).with_first_party_targets(&["example_remote_agent"])
249    }
250
251    fn translate(settings: LoggingSettings) -> LoggingConfiguration {
252        translator().translate(settings).expect("translate logging config")
253    }
254
255    fn translate_level(level: &str) -> Result<Vec<String>, GenericError> {
256        let settings = LoggingSettings {
257            level: level.to_string(),
258            ..LoggingSettings::default()
259        };
260
261        translator()
262            .translate(settings)
263            .map(|config| config.log_level.as_targets().to_string())
264            .map(|filter| filter.split(',').map(str::to_string).collect())
265    }
266
267    fn translate_log_level(level: &str) -> LogLevel {
268        translate(LoggingSettings {
269            level: level.to_string(),
270            ..LoggingSettings::default()
271        })
272        .log_level
273    }

354    async fn translate_levels(levels: &[&str]) -> Vec<Vec<String>> {
355        let levels = futures::stream::iter(levels.iter().map(|level| level.to_string()).collect::<Vec<_>>());
356        translator()
357            .translate_log_levels(levels)
358            .map(|log_level| {
359                log_level
360                    .as_targets()
361                    .to_string()
362                    .split(',')
363                    .map(str::to_string)
364                    .collect()
365            })
366            .collect()
367            .await
368    }
