112impl Endpoints {
113    /// Returns the primary intake endpoint, as configured and without normalization.
114    ///
115    /// An explicitly configured [`dd_url`](Self::dd_url) overrides [`site`](Self::site), even when
116    /// its value equals the schema default: the operator asked for that URL. Otherwise the endpoint
117    /// is derived from `site`. An empty `site` cannot produce an endpoint, so the effective `dd_url`
118    /// value is used instead; it already carries the source schema's default URL.
119    pub fn primary_endpoint(&self) -> String {
120        if self.dd_url.is_explicit() || self.site.value.is_empty() {
121            self.dd_url.value.clone()
122        } else {
123            format!("https://app.{}", self.site.value)
124        }
125    }

288impl Forwarder {
289    /// Returns the effective maximum size, in bytes, of the in-memory retry queue.
290    ///
291    /// An explicit [`retry_queue_payloads_max_size`](Self::retry_queue_payloads_max_size) wins, then
292    /// an explicit [`retry_queue_max_size`](Self::retry_queue_max_size), and otherwise the effective
293    /// payload-size value, which carries the source schema's default. Selection cannot look at the
294    /// values themselves: `0` is both the deprecated setting's schema default and a value an
295    /// operator can mean.
296    pub fn effective_retry_queue_max_size_bytes(&self) -> u64 {
297        if !self.retry_queue_payloads_max_size.is_explicit() && self.retry_queue_max_size.is_explicit() {
298            self.retry_queue_max_size.value
299        } else {
300            self.retry_queue_payloads_max_size.value
301        }
302    }

452/// Whether series are routed to the V3 metrics intake (`use_v3_api.series.*`).
453///
454/// Each variant serializes to the spelling [`FromStr`] reads, so a serialized mode round-trips.
455#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize)]
456pub enum V3SeriesMode {
457    /// Route series to the V3 intake.
458    #[serde(rename = "true")]
459    Enabled,
460
461    /// Route series to the older intake.
462    #[serde(rename = "false")]
463    Disabled,
464
465    /// Route series to the V3 intake only for endpoints that are Datadog intake URLs.
466    #[default]
467    #[serde(rename = "datadog_only")]
468    DatadogOnly,
469}
470
471impl FromStr for V3SeriesMode {
472    type Err = Error;
473
474    // The Agent reads this setting as a string and then interprets it, accepting more spellings than
475    // `strconv.ParseBool` does, so the accepted set is wider than that of a `boolean` leaf.
476    fn from_str(value: &str) -> Result<Self, Self::Err> {
477        match value.trim().to_ascii_lowercase().as_str() {
478            "true" | "1" | "t" | "yes" | "on" => Ok(Self::Enabled),
479            "false" | "0" | "f" | "no" | "off" | "" => Ok(Self::Disabled),
480            "datadog_only" => Ok(Self::DatadogOnly),
481            other => Err(Error::new_without_source(format!(
482                "unknown V3 series mode `{other}`; expected a boolean or `datadog_only`"
483            ))),
484        }
485    }
486}
