10/// Resolved logs configuration.
11#[derive(Clone, Debug, Default, PartialEq, Serialize)]
12pub struct Domain {
13    /// Processing rules applied, in order, to every log: `logs_config.processing_rules`.
14    ///
15    /// Empty unless the logs agent is enabled, since the Agent only reads the rules then. The rules aren't
16    /// validated: like the Agent, ADP validates them when it starts the logs pipeline. See
17    /// [`Domain::validate_processing_rules`].
18    pub processing_rules: ProcessingRules,
19}
20
21impl Domain {
22    /// Decodes and validates the processing rules as the Agent's `GlobalProcessingRules` does
23    /// (`comp/logs/agent/config/config.go`), and returns them in order.
24    ///
25    /// A JSON string is decoded first. Then the rules are validated in order, as the Agent's `ValidateProcessingRules`
26    /// does (`comp/logs/agent/config/processing_rules.go`), and the error is the Agent's message for the first problem
27    /// found. The patterns aren't compiled here: the component that applies the rules compiles them.
28    ///
29    /// # Errors
30    ///
31    /// Returns an error if a JSON string isn't a JSON array of rules or has a field of the wrong type, a rule has no
32    /// name, no type or an unknown type, no pattern where its type needs one, or a `remap_source` rule has no matching
33    /// entries or an entry with an empty field.
34    pub fn validate_processing_rules(&self) -> Result<Vec<ProcessingRule>, InvalidProcessingRules> {
35        let decoded;
36        let entries = match &self.processing_rules {
37            ProcessingRules::Entries(entries) => entries,
38            // Like the Agent, read an empty string as no rules.
39            ProcessingRules::Json(json) if json.is_empty() => return Ok(Vec::new()),
40            ProcessingRules::Json(json) => {
41                let invalid = |error: &dyn fmt::Display| InvalidProcessingRules(format!("invalid JSON: {error}"));
42                let rules = match serde_json::from_str::<Value>(json).map_err(|error| invalid(&error))? {
43                    // Like `encoding/json`, read `null` as no rules.
44                    Value::Null => Vec::new(),
45                    Value::Array(rules) => rules,
46                    other => return Err(invalid(&unexpected("rules", "a list", &other))),
47                };
48                decoded = rules
49                    .iter()
50                    .enumerate()
51                    .map(|(index, rule)| {
52                        ProcessingRuleEntry::decode(rule, Decoder::Json)
53                            .map_err(|error| invalid(&format_args!("processing rule {index}: {error}")))
54                    })
55                    .collect::<Result<Vec<_>, _>>()?;
56                &decoded
57            }
58        };
59        entries.iter().map(ProcessingRuleEntry::validate).collect()
60    }
61}

63/// Logs processing rules as configured.
64///
65/// Serializes as the configured value: a list of rules, or the JSON string.
66#[derive(Clone, Debug, PartialEq, Serialize)]
67#[serde(untagged)]
68pub enum ProcessingRules {
69    /// Rules configured as a sequence.
70    Entries(Vec<ProcessingRuleEntry>),
71
72    /// Rules configured as a JSON-encoded string, such as `'[{"type": "exclude_at_match", ...}]'`.
73    ///
74    /// Like the Agent, ADP doesn't decode the string until it validates the rules.
75    Json(String),
76}
77
78impl Default for ProcessingRules {
79    fn default() -> Self {
80        Self::Entries(Vec::new())
81    }
82}

84/// A logs processing rule as configured, before validation.
85///
86/// Every field is empty when the configuration doesn't set it, or sets it to null.
87#[derive(Clone, Debug, Default, PartialEq, Serialize)]
88pub struct ProcessingRuleEntry {
89    /// Type of the rule, such as `exclude_at_match`.
90    #[serde(rename = "type")]
91    pub rule_type: String,
92    /// Name of the rule.
93    pub name: String,
94    /// Regular expression of the rule.
95    pub pattern: String,
96    /// Replacement for each match of a `mask_sequences` rule.
97    pub replace_placeholder: String,
98    /// Attribute values that a `remap_source` rule matches.
99    pub matching: Vec<SourceMatch>,
100}
101
102impl ProcessingRuleEntry {
103    /// Decodes one rule of a sequence as the Agent's `structure.UnmarshalKey` does: with `mapstructure`, and weakly
104    /// typed input.
105    ///
106    /// # Errors
107    ///
108    /// Returns an error if the rule isn't a map, or a field holds a value that doesn't convert: a list or a map where a
109    /// string is expected, or a `matching` entry that isn't a map.
110    pub fn from_sequence_value(value: &Value) -> Result<Self, InvalidProcessingRules> {
111        Self::decode(value, Decoder::Mapstructure)
112    }
113
114    fn decode(value: &Value, decoder: Decoder) -> Result<Self, InvalidProcessingRules> {
115        let entry = map("rule", value)?;
116        Ok(Self {
117            // The Agent's `Type`, `Name`, and `Pattern` fields have no tag, so the exact keys are capitalized.
118            rule_type: string("type", field(entry, "Type"), decoder)?,
119            name: string("name", field(entry, "Name"), decoder)?,
120            pattern: string("pattern", field(entry, "Pattern"), decoder)?,
121            replace_placeholder: string("replace_placeholder", field(entry, "replace_placeholder"), decoder)?,
122            matching: list("matching", field(entry, "matching"), decoder)?
123                .into_iter()
124                .enumerate()
125                .map(|(index, value)| SourceMatch::decode(&format!("matching[{index}]"), value, decoder))
126                .collect::<Result<_, _>>()?,
127        })
128    }
129
130    fn validate(&self) -> Result<ProcessingRule, InvalidProcessingRules> {
131        let invalid = |message: String| Err(InvalidProcessingRules(message));
132        let Self {
133            rule_type,
134            name,
135            pattern,
136            replace_placeholder,
137            matching,
138        } = self.clone();
139
140        if name.is_empty() {
141            return invalid("all processing rules must have a name".to_string());
142        }
143        if matches!(
144            rule_type.as_str(),
145            "exclude_at_match" | "include_at_match" | "mask_sequences" | "multi_line"
146        ) && pattern.is_empty()
147        {
148            return invalid(format!("no pattern provided for processing rule: {name}"));
149        }
150        Ok(match rule_type.as_str() {
151            "exclude_at_match" => ProcessingRule::ExcludeAtMatch { name, pattern },
152            "include_at_match" => ProcessingRule::IncludeAtMatch { name, pattern },
153            "mask_sequences" => ProcessingRule::MaskSequences {
154                name,
155                pattern,
156                replace_placeholder,
157            },
158            "multi_line" => ProcessingRule::MultiLine { name, pattern },
159            "exclude_truncated" => ProcessingRule::ExcludeTruncated { name },
160            "remap_source" => {
161                if matching.is_empty() {
162                    return invalid(format!("no matching entries provided for processing rule: {name}"));
163                }
164                for (index, entry) in matching.iter().enumerate() {
165                    for (field, value) in [
166                        ("attribute", &entry.attribute),
167                        ("value", &entry.value),
168                        ("new_source", &entry.new_source),
169                    ] {
170                        if value.is_empty() {
171                            return invalid(format!("match {index} has empty {field} in processing rule: {name}"));
172                        }
173                    }
174                }
175                ProcessingRule::RemapSource { name, matching }
176            }
177            "" => return invalid(format!("type must be set for processing rule `{name}`")),
178            other => return invalid(format!("type {other} is not supported for processing rule `{name}`")),
179        })
180    }
181}

183/// A validated logs processing rule.
184#[derive(Clone, Debug, PartialEq)]
185pub enum ProcessingRule {
186    /// Drops logs that match `pattern`.
187    ExcludeAtMatch {
188        /// Name of the rule.
189        name: String,
190        /// Regular expression matched against the content of each log.
191        pattern: String,
192    },
193
194    /// Drops logs that don't match `pattern`.
195    IncludeAtMatch {
196        /// Name of the rule.
197        name: String,
198        /// Regular expression matched against the content of each log.
199        pattern: String,
200    },
201
202    /// Replaces each match of `pattern` with `replace_placeholder`.
203    MaskSequences {
204        /// Name of the rule.
205        name: String,
206        /// Regular expression matched against the content of each log.
207        pattern: String,
208        /// Replacement for each match, where `$1` and `${name}` refer to capture groups.
209        replace_placeholder: String,
210    },
211
212    /// Drops truncated logs.
213    ExcludeTruncated {
214        /// Name of the rule.
215        name: String,
216    },
217
218    /// Changes the source of structured logs whose attribute has a given value.
219    RemapSource {
220        /// Name of the rule.
221        name: String,
222        /// Attribute values to match, and the source each one maps to.
223        matching: Vec<SourceMatch>,
224    },
225
226    /// Aggregates multi-line logs.
227    MultiLine {
228        /// Name of the rule.
229        name: String,
230        /// Regular expression matching the start of a log.
231        pattern: String,
232    },
233}
234
235/// An attribute value that a `remap_source` rule maps to a new source.
236#[derive(Clone, Debug, Default, PartialEq, Serialize)]
237pub struct SourceMatch {
238    /// Path of the attribute.
239    pub attribute: String,
240    /// Value the attribute must have.
241    pub value: String,
242    /// Source given to a log that matches.
243    pub new_source: String,
244}
245
246impl SourceMatch {
247    fn decode(path: &str, value: &Value, decoder: Decoder) -> Result<Self, InvalidProcessingRules> {
248        let entry = map(path, value)?;
249        Ok(Self {
250            attribute: string(&format!("{path}.attribute"), field(entry, "attribute"), decoder)?,
251            value: string(&format!("{path}.value"), field(entry, "value"), decoder)?,
252            new_source: string(&format!("{path}.new_source"), field(entry, "new_source"), decoder)?,
253        })
254    }
255}

257/// Reports why the logs processing rules are invalid, with the Agent's message.
258#[derive(Clone, Debug, PartialEq, Eq)]
259pub struct InvalidProcessingRules(String);
260
261impl fmt::Display for InvalidProcessingRules {
262    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
263        f.write_str(&self.0)
264    }
265}
266
267impl std::error::Error for InvalidProcessingRules {}
268
269impl From<DecodeError> for InvalidProcessingRules {
270    fn from(error: DecodeError) -> Self {
271        Self(error.to_string())
272    }
