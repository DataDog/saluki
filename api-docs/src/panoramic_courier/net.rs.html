8/// Runs `operation` on `target`, failing with [`Error::TimedOut`] once `limit` passes.
9pub(crate) async fn within<T>(
10    operation: &'static str, target: &(dyn std::fmt::Display + Sync), limit: Duration,
11    work: impl Future<Output = Result<T, Error>>,
12) -> Result<T, Error> {
13    match tokio::time::timeout(limit, work).await {
14        Ok(result) => result,
15        Err(_) => TimedOutSnafu {
16            operation,
17            target: target.to_string(),
18            limit,
19        }
20        .fail(),
21    }
22}

24pub(crate) async fn resolve(host_port: &HostPort) -> Result<Vec<SocketAddr>, Error> {
25    let addrs: Vec<_> = tokio::net::lookup_host((host_port.host.as_str(), host_port.port))
26        .await
27        .context(ResolveSnafu { host: &host_port.host })?
28        .collect();
29    if addrs.is_empty() {
30        return Err(io::Error::new(io::ErrorKind::NotFound, "no addresses"))
31            .context(ResolveSnafu { host: &host_port.host });
32    }
33    Ok(addrs)
34}

36/// Connects to the first resolved address that accepts.
37pub(crate) async fn connect_tcp(host_port: &HostPort) -> Result<TcpStream, Error> {
38    let mut last_error = None;
39    for addr in resolve(host_port).await? {
40        match TcpStream::connect(addr).await {
41            Ok(stream) => return Ok(stream),
42            Err(e) => last_error = Some(e),
43        }
44    }
45    let error = last_error.expect("resolve returns at least one address");
46    Err(error).context(TargetSnafu {
47        action: "connect to",
48        target: host_port.to_string(),
49    })
50}

52/// Opens the named pipe `\\.\pipe\<name>` as a client that only writes, so a server that accepts inbound data only
53/// still admits it.
54#[cfg(windows)]
55pub(crate) fn open_pipe(name: &str) -> io::Result<tokio::net::windows::named_pipe::NamedPipeClient> {
56    tokio::net::windows::named_pipe::ClientOptions::new()
57        .read(false)
58        .open(format!(r"\\.\pipe\{name}"))
