17/// An error reading or parsing sources.
18#[derive(Debug, Snafu)]
19#[snafu(context(suffix(Snafu)))]
20pub enum Error {
21    /// A directory or file couldn't be read.
22    #[snafu(display("failed to read {}: {source}", path.display()))]
23    Read {
24        /// The path being read.
25        path: PathBuf,
26        /// The underlying error.
27        source: io::Error,
28    },
29
30    /// A file couldn't be parsed as Rust.
31    #[snafu(display("failed to parse {}:{line}: {message}", path.display()))]
32    Parse {
33        /// The file.
34        path: PathBuf,
35        /// The 1-based line of the error.
36        line: usize,
37        /// The parse error message.
38        message: String,
39    },
40
41    /// The scanned directories hold no `.rs` file.
42    #[snafu(display("no `.rs` files under {}", display_paths(dirs)))]
43    NoSources {
44        /// The scanned directories.
45        dirs: Vec<PathBuf>,
46    },
47
48    /// The crate root given for alias derivation isn't among the scanned files.
49    #[snafu(display("crate root {} isn't a scanned file", path.display()))]
50    MissingCrateRoot {
51        /// The crate root.
52        path: PathBuf,
53    },
54
55    /// A glob import of a scanned name binds names that alias derivation can't see.
56    #[snafu(display(
57        "{}:{line}: glob import of a guarded name; import the names explicitly",
58        path.display()
59    ))]
60    GlobImport {
61        /// The file.
62        path: PathBuf,
63        /// The 1-based line of the glob.
64        line: usize,
65    },
66}

68fn display_paths(paths: &[PathBuf]) -> String {
69    let paths: Vec<_> = paths.iter().map(|path| path.display().to_string()).collect();
70    paths.join(", ")
71}

73/// The syntax containing a matched identifier.
74#[derive(Clone, Copy, Debug, Eq, PartialEq)]
75pub enum Form {
76    /// A path segment, with or without a leading `::`.
77    Path,
78    /// A name in a `use` tree, including the target of an `as` rename.
79    Use,
80    /// The crate or rename of an `extern crate` item.
81    ExternCrate,
82    /// An identifier token in a macro body or attribute argument.
83    Token,
84}
85
86impl fmt::Display for Form {
87    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
88        f.write_str(match self {
89            Self::Path => "path",
90            Self::Use => "use",
91            Self::ExternCrate => "extern crate",
92            Self::Token => "token",
93        })
94    }
95}

97/// A matched identifier and its source location.
98#[derive(Clone, Debug, Eq, PartialEq)]
99pub struct Mention {
100    /// The file.
101    pub path: PathBuf,
102    /// The 1-based line.
103    pub line: usize,
104    /// The 1-based column.
105    pub column: usize,
106    /// The name.
107    pub name: String,
108    /// The syntax that names it.
109    pub form: Form,
110}
111
112impl fmt::Display for Mention {
113    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
114        write!(
115            f,
116            "{}:{}:{}: {} `{}`",
117            self.path.display(),
118            self.line,
119            self.column,
120            self.form,
121            self.name
122        )
123    }
124}

126struct SourceFile {
127    path: PathBuf,
128    syntax: syn::File,
129}
130
131impl SourceFile {
132    fn parse(path: PathBuf, text: &str) -> Result<Self, Error> {
133        match syn::parse_file(text) {
134            Ok(syntax) => Ok(Self { path, syntax }),
135            Err(error) => ParseSnafu {
136                path,
137                line: error.span().start().line,
138                message: error.to_string(),
139            }
140            .fail(),
141        }
142    }
143
144    fn mentions(&self, names: &BTreeSet<String>) -> Vec<Mention> {
145        let mut finder = MentionFinder::new(&self.path, names);
146        finder.visit_file(&self.syntax);
147        finder.found
148    }
149
150    /// Adds cross-file aliases to `out`, following private alias chains within this file.
151    /// [`SourceTree::aliases`] defines which bindings count across files.
152    fn exported_aliases(
153        &self, crate_root: bool, known: &BTreeSet<String>, out: &mut BTreeSet<String>,
154    ) -> Result<(), Error> {
155        let mut scope = known.clone();
156        loop {
157            let mut bindings = Bindings {
158                path: &self.path,
159                names: &scope,
160                exported: BTreeSet::new(),
161                private: BTreeSet::new(),
162            };
163            bindings.items(&self.syntax.items, crate_root)?;
164            let Bindings { exported, private, .. } = bindings;
165            let before = scope.len();
166            scope.extend(exported.iter().chain(&private).cloned());
167            if scope.len() == before {
168                out.extend(exported);
169                return Ok(());
170            }
171        }
172    }
173}

175/// The parsed `.rs` files under a set of directories.
176pub struct SourceTree {
177    files: Vec<SourceFile>,
178}
179
180impl SourceTree {
181    /// Reads and parses every `.rs` file under `dirs`, recursively.
182    ///
183    /// # Errors
184    ///
185    /// Returns an error if a directory or file can't be read, a file doesn't parse, or `dirs` hold no `.rs` file.
186    pub fn read(dirs: &[PathBuf]) -> Result<Self, Error> {
187        let mut paths = Vec::new();
188        for dir in dirs {
189            collect_rust_files(dir, &mut paths)?;
190        }
191        if paths.is_empty() {
192            return NoSourcesSnafu { dirs: dirs.to_vec() }.fail();
193        }
194        let files = paths
195            .into_iter()
196            .map(|path| {
197                let text = fs::read_to_string(&path).context(ReadSnafu { path: &path })?;
198                SourceFile::parse(path, &text)
199            })
200            .collect::<Result<_, _>>()?;
201        Ok(Self { files })
202    }
203
204    /// Returns every place in these files that names one of `names`.
205    ///
206    /// Comments, doc text, and string literals never count.
207    pub fn mentions(&self, names: &BTreeSet<String>) -> Vec<Mention> {
208        self.files.iter().flat_map(|file| file.mentions(names)).collect()
209    }
210
211    /// Collects aliases of `names` from this tree.
212    ///
213    /// An alias binds a name through an item mentioning a guarded name or another alias. At file and inline-module
214    /// scope, the scan includes:
215    ///
216    /// - `use`, `type`, and `extern crate` items with visibility other than private or `pub(self)`
217    /// - private bindings of those kinds at the top level of `crate_root`, accessible to descendant modules
218    /// - `macro_rules!` definitions, regardless of visibility
219    ///
220    /// Private bindings elsewhere contribute only to same-file alias chains. Bindings inside blocks or macro
221    /// invocations aren't collected. Matches ignore Rust scope, so unrelated names can match.
222    ///
223    /// # Errors
224    ///
225    /// Returns an error if `crate_root` isn't scanned or an included `use` item glob-imports a path containing a
226    /// guarded name or alias. Glob-bound names can't be enumerated from the import syntax.
227    pub fn aliases(&self, crate_root: &Path, names: &BTreeSet<String>) -> Result<BTreeSet<String>, Error> {
228        if !self.files.iter().any(|file| file.path == crate_root) {
229            return MissingCrateRootSnafu { path: crate_root }.fail();
230        }
231        let mut known = names.clone();
232        loop {
233            let mut bound = BTreeSet::new();
234            for file in &self.files {
235                file.exported_aliases(file.path == crate_root, &known, &mut bound)?;
236            }
237            let before = known.len();
238            known.extend(bound);
239            if known.len() == before {
240                break;
241            }
242        }
243        Ok(known.difference(names).cloned().collect())
244    }
245}

247fn collect_rust_files(dir: &Path, out: &mut Vec<PathBuf>) -> Result<(), Error> {
248    let mut entries = fs::read_dir(dir)
249        .context(ReadSnafu { path: dir })?
250        .map(|entry| entry.map(|entry| entry.path()))
251        .collect::<Result<Vec<_>, _>>()
252        .context(ReadSnafu { path: dir })?;
253    entries.sort();
254    for path in entries {
255        if path.is_dir() {
256            collect_rust_files(&path, out)?;
257        } else if path.extension().is_some_and(|extension| extension == "rs") {
258            out.push(path);
259        }
260    }
261    Ok(())
262}

264struct MentionFinder<'a> {
265    path: &'a Path,
266    names: &'a BTreeSet<String>,
267    found: Vec<Mention>,
268}
269
270impl<'a> MentionFinder<'a> {
271    fn new(path: &'a Path, names: &'a BTreeSet<String>) -> Self {
272        Self {
273            path,
274            names,
275            found: Vec::new(),
276        }
277    }
278
279    fn check(&mut self, ident: &Ident, form: Form) {
280        let name = ident.unraw().to_string();
281        if self.names.contains(&name) {
282            let start = ident.span().start();
283            self.found.push(Mention {
284                path: self.path.to_path_buf(),
285                line: start.line,
286                column: start.column + 1,
287                name,
288                form,
289            });
290        }
291    }
292
293    fn tokens(&mut self, tokens: &TokenStream) {
294        for token in tokens.clone() {
295            match token {
296                TokenTree::Ident(ident) => self.check(&ident, Form::Token),
297                TokenTree::Group(group) => self.tokens(&group.stream()),
298                TokenTree::Punct(_) | TokenTree::Literal(_) => {}
299            }
300        }
301    }
302}
303
304impl<'ast> Visit<'ast> for MentionFinder<'_> {
305    fn visit_item_extern_crate(&mut self, i: &'ast syn::ItemExternCrate) {
306        self.check(&i.ident, Form::ExternCrate);
307        if let Some((_, rename)) = &i.rename {
308            self.check(rename, Form::ExternCrate);
309        }
310        visit::visit_item_extern_crate(self, i);
311    }
312
313    fn visit_path_segment(&mut self, i: &'ast syn::PathSegment) {
314        self.check(&i.ident, Form::Path);
315        visit::visit_path_segment(self, i);
316    }
317
318    // Macro bodies, attribute arguments, and unparsed syntax.
319    fn visit_token_stream(&mut self, i: &'ast TokenStream) {
320        self.tokens(i);
321    }
322
323    fn visit_use_name(&mut self, i: &'ast syn::UseName) {
324        self.check(&i.ident, Form::Use);
325    }
326
327    fn visit_use_path(&mut self, i: &'ast syn::UsePath) {
328        self.check(&i.ident, Form::Use);
329        visit::visit_use_path(self, i);
330    }
331
332    fn visit_use_rename(&mut self, i: &'ast syn::UseRename) {
333        self.check(&i.ident, Form::Use);
334        self.check(&i.rename, Form::Use);
335    }
336}

338/// Alias bindings split by whether the scan propagates them across files.
339struct Bindings<'a> {
340    path: &'a Path,
341    names: &'a BTreeSet<String>,
342    exported: BTreeSet<String>,
343    private: BTreeSet<String>,
344}
345
346impl Bindings<'_> {
347    fn items(&mut self, items: &[Item], export_private: bool) -> Result<(), Error> {
348        for item in items {
349            match item {
350                Item::Use(item) => {
351                    let exported = export_private || is_visible(&item.vis);
352                    self.use_tree(&item.tree, None, exported)?;
353                }
354                Item::Type(item) => {
355                    let mut finder = MentionFinder::new(self.path, self.names);
356                    finder.visit_item_type(item);
357                    if !finder.found.is_empty() {
358                        self.bind(&item.ident, export_private || is_visible(&item.vis));
359                    }
360                }
361                Item::ExternCrate(item) => {
362                    if self.names_ident(&item.ident) {
363                        let ident = item.rename.as_ref().map_or(&item.ident, |(_, rename)| rename);
364                        self.bind(ident, export_private || is_visible(&item.vis));
365                    }
366                }
367                Item::Macro(item) => {
368                    if let Some(ident) = &item.ident {
369                        let mut finder = MentionFinder::new(self.path, self.names);
370                        finder.visit_macro(&item.mac);
371                        if !finder.found.is_empty() {
372                            self.bind(ident, true);
373                        }
374                    }
375                }
376                Item::Mod(item) => {
377                    if let Some((_, items)) = &item.content {
378                        self.items(items, false)?;
379                    }
380                }
381                _ => {}
382            }
383        }
384        Ok(())
385    }
386
387    /// Collects bindings imported through guarded paths. `prefix` holds the preceding segment (for `self` imports)
388    /// and whether any segment so far names a guarded identifier.
389    fn use_tree(&mut self, tree: &UseTree, prefix: Option<(&Ident, bool)>, exported: bool) -> Result<(), Error> {
390        let named = prefix.is_some_and(|(_, named)| named);
391        match tree {
392            UseTree::Path(path) => {
393                let named = named || self.names_ident(&path.ident);
394                self.use_tree(&path.tree, Some((&path.ident, named)), exported)?;
395            }
396            UseTree::Name(name) => {
397                if named || self.names_ident(&name.ident) {
398                    match prefix {
399                        Some((parent, _)) if name.ident == "self" => self.bind(parent, exported),
400                        _ => self.bind(&name.ident, exported),
401                    }
402                }
403            }
404            UseTree::Rename(rename) => {
405                if (named || self.names_ident(&rename.ident)) && rename.rename != "_" {
406                    self.bind(&rename.rename, exported);
407                }
408            }
409            UseTree::Glob(glob) => {
410                if named && exported {
411                    return GlobImportSnafu {
412                        path: self.path,
413                        line: glob.star_token.spans[0].start().line,
414                    }
415                    .fail();
416                }
417            }
418            UseTree::Group(group) => {
419                for tree in &group.items {
420                    self.use_tree(tree, prefix, exported)?;
421                }
422            }
423        }
424        Ok(())
425    }
426
427    fn names_ident(&self, ident: &Ident) -> bool {
428        self.names.contains(&ident.unraw().to_string())
429    }
430
431    fn bind(&mut self, ident: &Ident, exported: bool) {
432        let name = ident.unraw().to_string();
433        if exported {
434            self.exported.insert(name);
435        } else {
436            self.private.insert(name);
437        }
438    }
439}

441/// Whether the scan treats visibility as non-private: neither inherited nor `pub(self)`.
442fn is_visible(vis: &Visibility) -> bool {
443    match vis {
444        Visibility::Public(_) => true,
445        Visibility::Restricted(restricted) => !restricted.path.is_ident("self"),
446        Visibility::Inherited => false,
447    }

686        let dir = tempfile::tempdir().unwrap();
687        fs::write(dir.path().join("good.rs"), "fn f() {}").unwrap();
688        fs::write(dir.path().join("bad.rs"), "fn f() {}\nfn g( {}\n").unwrap();
689
690        match SourceTree::read(&[dir.path().to_path_buf()]) {
691            Err(Error::Parse { path, line, .. }) => {
692                assert_eq!(path, dir.path().join("bad.rs"));
693                assert_eq!(line, 2);
694            }
695            Err(error) => panic!("unexpected error: {error}"),
696            Ok(_) => panic!("a file that doesn't parse must fail the read"),
697        }
698    }
