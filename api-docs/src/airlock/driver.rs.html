390    fn port_publishing_options(&self) -> (Option<bool>, Option<Vec<String>>) {
391        if self.exposed_ports.is_empty() {
392            return (None, None);
393        }
394
395        let exposed_ports = self
396            .exposed_ports
397            .iter()
398            .map(|(protocol, internal_port)| format!("{}/{}", internal_port, protocol))
399            .collect();
400        let publish_all_ports = match self.container_os {
401            ContainerOs::Linux => Some(true),
402            ContainerOs::Windows => None,
403        };
404
405        (publish_all_ports, Some(exposed_ports))
406    }

408    /// Returns the full set of bind mounts to apply to this container, including OS-specific
409    /// defaults and any additional named volume mounts.
410    ///
411    /// Linux containers receive the shared `/airlock` volume plus read-only mounts of host
412    /// paths needed for origin detection (`/proc`, `/sys/fs/cgroup`, the Docker socket).
413    /// Windows containers receive only the shared `C:\airlock` volume; the host-resource
414    /// mounts have no Windows-container equivalent and the `:z` shared-relabel mount option is
415    /// Linux-specific.
416    fn container_binds_from(&self, isolation_group_name: &str, mut binds: Vec<String>) -> Vec<String> {
417        match self.container_os {
418            ContainerOs::Linux => {
419                binds.push(format!("{}:/airlock:z", isolation_group_name));
420                binds.push("/proc:/host/proc:ro".to_string());
421                binds.push("/sys/fs/cgroup:/host/sys/fs/cgroup:ro".to_string());
422                binds.push("/var/run/docker.sock:/var/run/docker.sock:ro".to_string());
423            }
424            ContainerOs::Windows => {
425                binds.push(format!("{}:C:\\airlock", isolation_group_name));
426            }
427        }
428
429        binds.extend(self.additional_volume_mounts.clone());
430        binds
431    }

442/// Inserts an `internal_port` -> host port mapping when `host_port` parses as a valid `u16`.
443///
444/// Docker reports each binding's host port as a string, and we treat values that don't parse as
445/// "no mapping available" rather than failing the whole inspect call. `internal_port` is the
446/// existing key (already including the protocol suffix, for example `"58125/udp"`).
447fn insert_port_mapping_if_parseable(
448    port_mappings: &mut HashMap<String, u16>, internal_port: impl Into<String>, host_port: Option<&str>,
449) {
450    if let Some(host_port) = host_port.and_then(|value| value.parse::<u16>().ok()) {
451        port_mappings.insert(internal_port.into(), host_port);
452    }
453}

491    /// # Isolation group
492    ///
493    /// The isolation group ID serves as a unique identifier to be used for both the name of the container as well as
494    /// the shared resources that are created and attached to the container. If two drivers share the same isolation
495    /// group ID, the containers they spawn will be located in the same network namespace, have access to the same
496    /// shared Airlock volume, etc.
497    ///
498    /// # Shared volume
499    ///
500    /// The container will have a volume bind-mounted at `/airlock` that's shared between all containers in the same
501    /// isolation group. This volume is mounted as world writeable (777) so all containers can freely read and write to
502    /// it. This makes it easier for containers to share data between one another, but also means that care should be
503    /// taken to avoid conflicts between trying to write to the same file, etc.
504    ///
505    /// # Errors

507    /// If the Docker client can't be created/configured, an error will be returned.

513            container_name: format!("airlock-{}-{}", isolation_group_id, config.driver_id),
514            isolation_group_id,

551    async fn remove_isolation_group(docker: &Docker, isolation_group_id: &str) -> Result<(), GenericError> {
552        let isolation_group_name = format!("airlock-{}", isolation_group_id);
553        let isolation_group_label = format!("airlock-isolation-group={}", isolation_group_id);
554        let mut failures = Vec::new();
555
556        let list_filters: HashMap<&str, Vec<&str>> =
557            [("label", vec!["created_by=airlock", isolation_group_label.as_str()])]
558                .into_iter()
559                .collect();
560        let list_options = ListContainersOptionsBuilder::default()
561            .all(true)
562            .filters(&list_filters)
563            .build();
564        match docker.list_containers(Some(list_options)).await {
565            Ok(containers) => {
566                for container_id in containers.into_iter().filter_map(|container| container.id) {
567                    let remove_options = RemoveContainerOptionsBuilder::default().force(true).build();
568                    match docker.remove_container(&container_id, Some(remove_options)).await {
569                        Err(e) if !is_not_found(&e) => failures.push(format!("container '{}': {}", container_id, e)),
570                        _ => debug!("Removed container '{}'.", container_id),
571                    }
572                }
573            }
574            Err(e) => failures.push(format!("container list: {}", e)),
575        }
576
577        match docker
578            .remove_volume(
579                &isolation_group_name,
580                None::<bollard::query_parameters::RemoveVolumeOptions>,
581            )
582            .await
583        {
584            Err(e) if !is_not_found(&e) => failures.push(format!("volume '{}': {}", isolation_group_name, e)),
585            _ => debug!("Removed shared volume '{}'.", isolation_group_name),
586        }
587
588        match docker.remove_network(&isolation_group_name).await {
589            Err(e) if !is_not_found(&e) => failures.push(format!("network '{}': {}", isolation_group_name, e)),
590            _ => debug!("Removed shared network '{}'.", isolation_group_name),
591        }
592
593        if failures.is_empty() {
594            Ok(())
595        } else {
596            Err(generic_error!(

761    async fn create_container_inner(
762        &self, container_name: String, image: String, entrypoint: Option<Vec<String>>, cmd: Option<Vec<String>>,
763        binds: Vec<String>, env: Option<Vec<String>>,
764    ) -> Result<String, GenericError> {
765        let binds = self.config.container_binds_from(&self.isolation_group_name, binds);
766
767        // Set up NetworkingConfig to apply aliases on the primary network, if any are configured.

773                    aliases: Some(self.config.network_aliases.clone()),

803            image: Some(image),
804            entrypoint,
805            cmd,
806            host_config: Some(HostConfig {
807                binds: Some(binds),
808                network_mode: Some(self.isolation_group_name.clone()),
809                publish_all_ports,
810                pid_mode,
811                cgroupns_mode,
812                ..Default::default()

814            healthcheck: self.config.healthcheck.clone(),
815            exposed_ports,
816            labels: Some(get_default_airlock_labels(self.isolation_group_id.as_str())),

861    async fn start_container_inner(&self, container_name: &str) -> Result<DriverDetails, GenericError> {
862        // Retrying is safe: the daemon treats starting an already-running container as a no-op rather than an error.
863        with_transient_error_retry("container start", &self.isolation_group_id, move || async move {
864            self.docker
865                .start_container(container_name, None)
866                .await
867                .map_err(GenericError::from)
868        })
869        .await?;
870
871        let mut details = DriverDetails {
872            container_name: container_name.to_string(),
873            ..Default::default()
874        };
875
876        let response = self.docker.inspect_container(container_name, None).await?;
877        if let Some(network_settings) = response.network_settings {
878            // Look up the IP only on the primary isolation-group network. Falling back to
879            // "any other network's IP" would be non-deterministic and effectively wrong for
880            // assertion targeting.
881            if let Some(networks) = network_settings.networks.as_ref() {
882                details.container_ip = networks
883                    .get(&self.isolation_group_name)
884                    .and_then(|settings| settings.ip_address.clone())
885                    .filter(|address| !address.is_empty());
886            }
887
888            if let Some(ports) = network_settings.ports {
889                let port_mappings = details.port_mappings.get_or_insert_with(HashMap::new);
890                for (internal_port, bindings) in ports {
891                    if let Some(bindings) = bindings {
892                        for binding in bindings {
893                            insert_port_mapping_if_parseable(
894                                port_mappings,
895                                internal_port.clone(),
896                                binding.host_port.as_deref(),
897                            );
898                            if port_mappings.contains_key(&internal_port) {
899                                break;
900                            }
901                        }
902                    }
903                }
904            }
905        }
906
907        Ok(details)
908    }

1006    /// Waits until the container is marked as healthy.
1007    ///
1008    /// If the container has no health checks defined, this returns early and does no waiting.
1009    ///
1010    /// # Errors
1011    ///
1012    /// If there is an error while inspecting the container, it will be returned.
1013    pub async fn wait_for_container_healthy(&mut self) -> Result<(), GenericError> {
1014        loop {
1015            // Inspect the container, and see if it even has any health checks defined. If not, then we can return early.
1016            let response = self.docker.inspect_container(&self.container_name, None).await?;
1017            let state = response
1018                .state
1019                .ok_or_else(|| generic_error!("Container state should be present."))?;
1020
1021            // Make sure the container is actually running.
1022            let status = state
1023                .status
1024                .ok_or_else(|| generic_error!("Container status should be present."))?;
1025            if status != ContainerStateStatusEnum::RUNNING {
1026                return Err(generic_error!(
1027                    "Container exited unexpectedly (driver_id: {}, container: {}). Check logs in the test run directory.",
1028                    self.config.driver_id,
1029                    self.container_name
1030                ));
1031            }
1032
1033            if let Some(health_status) = state.health.and_then(|h| h.status) {
1034                match health_status {
1035                    // No healthcheck defined, or healthy, so we're good to go.
1036                    HealthStatusEnum::EMPTY | HealthStatusEnum::NONE | HealthStatusEnum::HEALTHY => {
1037                        debug!(
1038                            driver_id = self.config.driver_id,
1039                            "Container '{}' healthy or no healthcheck defined. Proceeding.", &self.container_name
1040                        );
1041                        return Ok(());
1042                    }
1043
1044                    // Not healthy yet, so we'll keep waiting.
1045                    HealthStatusEnum::STARTING => {
1046                        debug!(
1047                            driver_id = self.config.driver_id,
1048                            "Container '{}' not yet healthy. Waiting...", &self.container_name
1049                        );
1050                    }
1051
1052                    HealthStatusEnum::UNHEALTHY => {
1053                        return Err(generic_error!(
1054                            "Container became unhealthy (driver_id: {}, container: {}). Check logs in the test run directory.",
1055                            self.config.driver_id,
1056                            self.container_name
1057                        ));
1058                    }
1059                }
1060            } else {
1061                debug!(
1062                    driver_id = self.config.driver_id,
1063                    "Container '{}' has no healthcheck defined. Proceeding.", &self.container_name
1064                );
1065                return Ok(());
1066            }
1067
1068            // Wait for a second and then check again.
1069            sleep(Duration::from_secs(1)).await;
1070        }
1071    }

1073    async fn wait_for_container_exit_inner(&self, container_name: &str) -> Result<ExitStatus, GenericError> {
1074        let mut wait_stream = self.docker.wait_container(container_name, None);
1075        match wait_stream.next().await {
1076            Some(result) => match result {
1077                Ok(response) => {
1078                    // When the exit code is non-zero, `bollard` transforms the normal `ContainerWaitResponse` into
1079                    // `Error::DockerContainerWaitError`, which is why we have these asserts here to catch any scenario
1080                    // where there's _somehow_ an error condition being indicated without it having been transformed into
1081                    // `Error::DockerContainerWaitError`.
1082                    //
1083                    // Essentially, getting to this point should imply successfully exiting, but the API isn't very
1084                    // ergonomic in that regard, so we're just making sure.
1085                    assert_eq!(response.error, None);
1086                    assert_eq!(response.status_code, 0);
1087
1088                    Ok(ExitStatus::Success)
1089                }
1090
1091                Err(Error::DockerContainerWaitError { error, code }) => {
1092                    let error = if error.is_empty() {
1093                        String::from("<no error message provided>")
1094                    } else {
1095                        error
1096                    };
1097                    Ok(ExitStatus::Failed { code, error })
1098                }
1099
1100                Err(e) => Err(generic_error!("Failed to wait for container to finish: {:?}", e)),
1101            },
1102            None => unreachable!("Docker wait stream ended unexpectedly."),
1103        }
1104    }

1134    /// Executes a command inside the running container and returns its stdout.
1135    ///
1136    /// The command runs as root with no TTY. Stderr is discarded, and only stdout is returned. If the command exits with a
1137    /// nonzero status, an error is returned.
1138    ///
1139    /// # Errors
1140    ///
1141    /// If the exec creation, start, output collection, or command exit code indicates failure, an error is returned.
1142    pub async fn exec_in_container(&self, cmd: Vec<String>) -> Result<String, GenericError> {
1143        let exec_opts = CreateExecOptions {
1144            attach_stdout: Some(true),
1145            attach_stderr: Some(false),
1146            cmd: Some(cmd.clone()),
1147            ..Default::default()
1148        };
1149
1150        let exec = self
1151            .docker
1152            .create_exec(&self.container_name, exec_opts)
1153            .await
1154            .with_error_context(|| format!("Failed to create exec instance for container {}.", self.container_name))?;
1155
1156        let exec_id = exec.id.clone();
1157
1158        let output = self
1159            .docker
1160            .start_exec(&exec.id, None)
1161            .await
1162            .with_error_context(|| format!("Failed to start exec for container {}.", self.container_name))?;
1163
1164        let mut stdout = String::new();
1165        if let StartExecResults::Attached { mut output, .. } = output {
1166            while let Some(chunk) = output.try_next().await? {
1167                if let LogOutput::StdOut { message } = chunk {
1168                    stdout.push_str(&String::from_utf8_lossy(&message));
1169                }
1170            }
1171        }
1172
1173        // Check the command's exit code.
1174        let inspect = self
1175            .docker
1176            .inspect_exec(&exec_id)
1177            .await
1178            .error_context("Failed to inspect exec result.")?;
1179
1180        if let Some(code) = inspect.exit_code {
1181            if code != 0 {
1182                return Err(generic_error!(
1183                    "Command {:?} exited with code {} in container {}.",
1184                    cmd,
1185                    code,
1186                    self.container_name
1187                ));
1188            }
1189        }
1190
1191        Ok(stdout)
1192    }

1230        &self, container_log_dir: PathBuf, log_name: &str, container_name: &str,
1231    ) -> Result<(), GenericError> {
1232        // Make sure the directories exist first and prepare the files, just to get any permissions issues out of the
1233        // way up front before we spawn our background task.
1234        tokio::fs::create_dir_all(&container_log_dir)
1235            .await
1236            .error_context("Failed to create logs directory. Possible permissions issue.")?;
1237
1238        let stdout_log_path = container_log_dir.join(format!("{}.stdout.log", log_name));
1239        let stderr_log_path = container_log_dir.join(format!("{}.stderr.log", log_name));
1240
1241        let mut stdout_file = tokio::fs::File::create(&stdout_log_path)
1242            .await
1243            .map(BufWriter::new)
1244            .error_context("Failed to create standard output log file. Possible permissions issue.")?;
1245        let mut stderr_file = tokio::fs::File::create(&stderr_log_path)
1246            .await
1247            .map(BufWriter::new)
1248            .error_context("Failed to create standard error log file. Possible permissions issue.")?;
1249
1250        // Spawn a background task to capture the logs.
1251        let logs_config = LogsOptions {
1252            follow: true,
1253            stdout: true,
1254            stderr: true,
1255            ..Default::default()
1256        };
1257        let mut log_stream = self.docker.logs(container_name, Some(logs_config));
1258
1259        tokio::spawn(async move {
1260            while let Some(log_result) = log_stream.next().await {
1261                match log_result {
1262                    Ok(log) => match log {
1263                        LogOutput::StdErr { message } => {
1264                            if let Err(e) = stderr_file.write_all(&strip_ansi_codes(&message)).await {
1265                                error!(error = %e, "Failed to write log line to standard error log file.");
1266                                break;
1267                            }
1268                            if let Err(e) = stderr_file.flush().await {
1269                                error!(error = %e, "Failed to flush standard error log file.");
1270                                break;
1271                            }
1272                        }
1273                        LogOutput::StdOut { message } => {
1274                            if let Err(e) = stdout_file.write_all(&strip_ansi_codes(&message)).await {
1275                                error!(error = %e, "Failed to write log line to standard output log file.");
1276                                break;
1277                            }
1278                            if let Err(e) = stdout_file.flush().await {
1279                                error!(error = %e, "Failed to flush standard output log file.");
1280                                break;
1281                            }
1282                        }
1283                        LogOutput::StdIn { .. } | LogOutput::Console { .. } => {}
1284                    },
1285                    Err(e) => {
1286                        error!(error = %e, "Failed to read log line from container.");
1287                        break;
1288                    }
1289                }
1290            }
1291
1292            // One final fsync to ensure the logs are fully written to disk.
1293            if let Err(e) = stdout_file.get_mut().sync_all().await {
1294                error!(error = %e, "Failed to fsync standard output log file.");
1295            }
1296
1297            if let Err(e) = stderr_file.get_mut().sync_all().await {
1298                error!(error = %e, "Failed to fsync standard error log file.");
1299            }
1300        });
1301
1302        Ok(())
1303    }
1304}

1306/// Removes ANSI escape sequences (`ESC[...letter`) from a byte slice.
1307fn strip_ansi_codes(input: &[u8]) -> Vec<u8> {
1308    let mut out = Vec::with_capacity(input.len());
1309    let mut i = 0;
1310    while i < input.len() {
1311        if input[i] == 0x1b && input.get(i + 1) == Some(&b'[') {
1312            i += 2;
1313            while i < input.len() && !input[i].is_ascii_alphabetic() {
1314                i += 1;
1315            }
1316            i += 1;
1317        } else {
1318            out.push(input[i]);
1319            i += 1;
1320        }
1321    }
1322    out
1323}

1335/// Returns `true` if `error` is a transient Windows HNS collision.
1336///
1337/// Windows containers are networked through the Host Network Service, and HNS serializes poorly: when several
1338/// containers create a NAT network or attach an endpoint at the same moment, the losing caller gets a 500 from the
1339/// daemon carrying `hnsCall failed in Win32: The object already exists. (0x1392)`. The object in question is internal
1340/// to HNS -- airlock names every network after a unique isolation group, so this is never a name conflict of ours --
1341/// and the conflict clears once the winning call completes.
1342///
1343/// The predicate matches on the message rather than on [`ContainerOs`] because `hnsCall` has no analogue on Linux,
1344/// which makes it inert there. It deliberately requires the `0x1392` code as well as `hnsCall`: other HNS failures
1345/// indicate real breakage and should fail fast rather than be papered over by a retry.
1346fn is_transient_hns_error(error: &GenericError) -> bool {
1347    error.downcast_ref::<Error>().is_some_and(|error| match error {
1348        Error::DockerResponseServerError { message, .. } => message.contains("hnsCall") && message.contains("0x1392"),
1349        _ => false,
1350    })
1351}

1353/// Returns how long to wait before the given transient error retry attempt.
1354///
1355/// `seed` keeps concurrent callers out of step with each other. Isolation group IDs are unique per
1356/// test, so two drivers that collided at the same instant draw different jitter and don't line
1357/// back up on the retry.
1358fn transient_error_retry_backoff(attempt: u32, seed: &str) -> Duration {
1359    let mut hasher = DefaultHasher::new();
1360    seed.hash(&mut hasher);
1361    attempt.hash(&mut hasher);
1362
1363    let jitter_ms = hasher.finish() % (TRANSIENT_ERROR_RETRY_MAX_JITTER.as_millis() as u64 + 1);
1364
1365    TRANSIENT_ERROR_RETRY_BASE_BACKOFF * attempt + Duration::from_millis(jitter_ms)
1366}

1368/// Runs `run_attempt`, retrying it when it fails with a known transient error.
1369///
1370/// Any other error, and the error from the final attempt, is returned to the caller untouched. "Known" transient errors
1371/// are hard-coded/categorized by hand, so this function mostly represents known areas where we've seen transient issues
1372/// that can generally be solved by simply retrying.
1373///
1374/// `run_attempt` must be safe to re-run: every call site either re-checks for the resource it is creating, or targets a
1375/// Docker endpoint that is idempotent.
1376///
1377/// Each retry is logged at warning level so that runs which only passed because of a retry stay visible in CI output,
1378/// rather than the mitigation quietly hiding how often the collision fires.
1379async fn with_transient_error_retry<F, Fut, T>(
1380    operation: &str, seed: &str, mut run_attempt: F,
1381) -> Result<T, GenericError>
1382where
1383    F: FnMut() -> Fut,
1384    Fut: Future<Output = Result<T, GenericError>>,
1385{
1386    let mut attempt = 1;
1387    loop {
1388        match run_attempt().await {
1389            Ok(value) => return Ok(value),
1390            Err(e) => {
1391                if attempt >= TRANSIENT_ERROR_RETRY_MAX_ATTEMPTS || !is_transient_error(&e) {
1392                    return Err(e);
1393                }
1394
1395                let backoff = transient_error_retry_backoff(attempt, seed);
1396                warn!(
1397                    isolation_group = seed,
1398                    error = %e,
1399                    "Transient error during {}. Retrying in {:?} (attempt {} of {}).",
1400                    operation,
1401                    backoff,
1402                    attempt + 1,
1403                    TRANSIENT_ERROR_RETRY_MAX_ATTEMPTS
1404                );
1405
1406                sleep(backoff).await;
1407                attempt += 1;
1408            }
1409        }
1410    }
1411}

1684    /// A fake Docker daemon with configurable responses and recorded requests.
1685    struct FakeDaemon {
1686        docker: Docker,
1687        /// Each request's method and path.
1688        requests: Arc<Mutex<Vec<String>>>,
1689        server: tokio::task::JoinHandle<()>,
1690    }

1692    impl FakeDaemon {
1693        /// Starts a daemon using `respond` to map request methods and paths to HTTP statuses and JSON bodies.
1694        async fn start(respond: fn(&str, &str) -> (u16, &'static str)) -> Self {
1695            let listener = tokio::net::TcpListener::bind("127.0.0.1:0")
1696                .await
1697                .expect("the fake daemon should bind");
1698            let address = listener.local_addr().expect("the listener should have an address");
1699            let requests = Arc::new(Mutex::new(Vec::new()));
1700            let recorded = requests.clone();
1701            let server = tokio::spawn(async move {
1702                loop {
1703                    let (mut stream, _) = listener.accept().await.expect("the fake daemon should accept");
1704                    let mut head = Vec::new();
1705                    let mut buf = [0; 1024];
1706                    while !head.windows(4).any(|window| window == b"\r\n\r\n") {
1707                        let read = stream.read(&mut buf).await.expect("the request should be readable");
1708                        if read == 0 {
1709                            break;
1710                        }
1711                        head.extend_from_slice(&buf[..read]);
1712                    }
1713                    let head = String::from_utf8_lossy(&head);
1714                    let mut request_line = head.lines().next().unwrap_or_default().split(' ');
1715                    let method = request_line.next().unwrap_or_default().to_string();
1716                    let path = request_line.next().unwrap_or_default().to_string();
1717                    let (status, body) = respond(&method, &path);
1718                    recorded.lock().unwrap().push(format!("{} {}", method, path));
1719                    let response = format!(
1720                        "HTTP/1.1 {} Fake\r\nContent-Type: application/json\r\nContent-Length: {}\r\n\
1721                         Connection: close\r\n\r\n{}",
1722                        status,
1723                        body.len(),
1724                        body
1725                    );
1726                    let _ = stream.write_all(response.as_bytes()).await;
1727                    let _ = stream.shutdown().await;
1728                }
1729            });
1730            let docker = Docker::connect_with_http(&format!("http://{}", address), 5, bollard::API_DEFAULT_VERSION)
1731                .expect("bollard should accept the fake daemon's address");
1732            Self {
1733                docker,
1734                requests,
1735                server,
1736            }
1737        }
1738
1739        /// Runs isolation-group cleanup and returns its result and recorded requests.
1740        async fn remove_isolation_group(&self) -> (Result<(), GenericError>, Vec<String>) {
1741            let removed = tokio::time::timeout(BOUND, Driver::remove_isolation_group(&self.docker, GROUP))
1742                .await
1743                .expect("removal should finish within the bound");
1744            (removed, self.requests.lock().unwrap().clone())
1745        }
1746    }

1751        }
