1//! Host programs as process groups, with streamed output and an exit that reports its signal.
2//!
3//! This is the host process primitive of the participant backends. [`unix`][crate::unix] serves the legacy runner and
4//! keeps its own shape: it pushes lines into a sink and loses the signal of an exit.
5//!
6//! A [`HostProcess`] leads a process group, so it and its descendants can be signalled together. Its output is an
7//! [`OutputStream`] of what the program writes to its standard output and error, and its exit is a watch that holds
8//! how the program exited once it has. The pipes stay read until they close, even when the stream is dropped, so a
9//! program is never blocked or signalled by its observer going away. Stopping sends SIGTERM to the group, then SIGKILL
10//! after a grace. Dropping a process sends SIGKILL to its group.
11

27/// How long [`HostProcess::stop`] waits after SIGKILL for the exit and for the group to empty.
28#[cfg(unix)]
29const KILL_WAIT: Duration = Duration::from_secs(2);
30
31/// How many chunks of output wait for whoever reads them, before the program's pipes fill.
32const OUTPUT_BUFFER: usize = 64;
33
34/// How often [`HostProcess::stop`] checks whether the group is empty after SIGKILL.
35#[cfg(unix)]
36const GROUP_POLL: Duration = Duration::from_millis(20);

38/// A program to run.
39#[derive(Clone, Debug)]
40pub struct HostCommand {
41    /// Names the process in messages.
42    pub name: String,
43    /// The file to execute.
44    pub program: PathBuf,
45    /// The arguments, not including the program itself.
46    pub args: Vec<String>,
47    /// The program's whole environment: nothing is inherited.
48    pub env: BTreeMap<String, String>,
49    /// The working directory.
50    pub dir: PathBuf,
51    /// Written to the program's standard input, which is then closed. Without it, the program has no standard input.
52    pub stdin: Option<Bytes>,
53}

55/// How a program exited.

64/// Holds `None` while the program runs, then how it exited, or why that cannot be told.
65pub type ExitWatch = watch::Receiver<Option<Result<HostExit, String>>>;

67/// A running program and its process group.
68///
69/// # Limits
70///
71/// - A descendant can leave the group, for example with `setsid`. Signals no longer reach it.
72/// - `Drop` does not run if the owning process is killed with SIGKILL, so a program may outlive it.
73/// - The OS may reuse the ID of an empty group before it is signalled, which would signal an unrelated group.
74#[derive(Debug)]
75pub struct HostProcess {
76    name: String,
77    #[cfg(unix)]
78    exit: ExitWatch,
79    /// The group to kill on drop, or `None` once it was sent SIGKILL.
80    #[cfg(unix)]
81    group: Option<i32>,
82}
83
84impl HostProcess {
85    /// Starts `command` in a new process group.
86    ///
87    /// Returns the process, its output, and the watch of its exit.
88    ///
89    /// # Errors
90    ///
91    /// Returns an error if the program cannot be started.
92    pub fn spawn(command: HostCommand) -> Result<(Self, OutputStream, ExitWatch), GenericError> {
93        let HostCommand {
94            name,
95            program,
96            args,
97            env,
98            dir,
99            stdin,
100        } = command;
101        let mut process = Command::new(&program);
102        process
103            .args(args)
104            .env_clear()
105            .envs(env)
106            .current_dir(dir)
107            .stdin(if stdin.is_some() { Stdio::piped() } else { Stdio::null() })
108            .stdout(Stdio::piped())
109            .stderr(Stdio::piped())
110            .kill_on_drop(true);
111        #[cfg(unix)]
112        process.process_group(0);
113        let mut child = process
114            .spawn()
115            .with_error_context(|| format!("Failed to start '{}' ({}).", name, program.display()))?;
116
117        #[cfg(unix)]
118        let group = child.id().and_then(|id| i32::try_from(id).ok());
119        let stdout = child
120            .stdout
121            .take()
122            .ok_or_else(|| generic_error!("No standard output."))?;
123        let stderr = child
124            .stderr
125            .take()
126            .ok_or_else(|| generic_error!("No standard error."))?;
127        if let (Some(payload), Some(mut input)) = (stdin, child.stdin.take()) {
128            // A program that stops reading early is its own business: the write simply ends.
129            tokio::spawn(async move {
130                let _ = input.write_all(&payload).await;
131                let _ = input.shutdown().await;
132            });
133        }
134
135        let stdout = output_of(stdout, &name, "output", Output::Stdout);
136        let stderr = output_of(stderr, &name, "error", Output::Stderr);
137        let output = keep_draining(stream::select(stdout, stderr).boxed());
138
139        let (sender, exit) = watch::channel(None);
140        let waited = name.clone();
141        // The task owns the child, so it is the only thing that reaps it.
142        tokio::spawn(async move {
143            let exited = match child.wait().await {
144                Ok(status) => Ok(HostExit {
145                    code: status.code(),
146                    signal: signal_of(&status),
147                }),
148                Err(e) => Err(format!("Failed to wait for '{}': {}", waited, e)),
149            };
150            sender.send_replace(Some(exited));
151        });
152
153        let process = Self {
154            name,
155            #[cfg(unix)]
156            exit: exit.clone(),
157            #[cfg(unix)]
158            group,
159        };
160        Ok((process, output, exit))
161    }

163    /// The name the process was given.
164    pub fn name(&self) -> &str {
165        &self.name
166    }

168    /// Stops the process and its group: SIGTERM, then SIGKILL after `grace`, then up to two seconds for the exit and
169    /// for the group to have no live member. A zombie left in the group does not count.
170    ///
171    /// Whatever is left in the group is killed even when the program itself exited. Dropping the future before the
172    /// SIGKILL drops the process, which sends SIGKILL.
173    ///
174    /// # Errors
175    ///
176    /// Returns an error if the process has not exited, or its group still has a live member, after the SIGKILL. A
177    /// process on a target with no process groups is not signalled, and stopping it does nothing.
178    #[cfg_attr(not(unix), allow(unused_mut, unused_variables))]
179    pub async fn stop(mut self, grace: Duration) -> Result<(), GenericError> {
180        #[cfg(unix)]
181        {
182            use tokio::time::timeout;
183
184            if self.exit.borrow().is_none() {
185                self.signal(libc::SIGTERM);
186                let _ = timeout(grace, self.exit.wait_for(Option::is_some)).await;
187            }
188            self.signal(libc::SIGKILL);
189            let group = self.group.take();
190            let gone = async {
191                let _ = self.exit.wait_for(Option::is_some).await;
192                if let Some(group) = group {
193                    while group_has_live_member(group) {
194                        tokio::time::sleep(GROUP_POLL).await;
195                    }
196                }
197            };
198            if timeout(KILL_WAIT, gone).await.is_err() {
199                return Err(generic_error!(
200                    "'{}' or something in its process group did not exit within {:?} of SIGKILL",
201                    self.name,
202                    KILL_WAIT
203                ));
204            }
205        }
206        Ok(())
207    }
208
209    /// Kills the group with SIGKILL without waiting for it to exit, for a caller that abandons the process on purpose.
210    /// Unlike a drop, it does not warn that the process leaked.
211    #[cfg_attr(not(unix), allow(unused_mut))]
212    pub fn kill(mut self) {
213        #[cfg(unix)]
214        {
215            self.signal(libc::SIGKILL);
216            self.group = None;
217        }
218    }
219
220    /// Sends `signal` to the group, ignoring failure: the group may already be empty.
221    #[cfg(unix)]
222    fn signal(&self, signal: libc::c_int) {
223        if let Some(group) = self.group {
224            // SAFETY: `killpg` takes no pointers and touches no memory of this process.
225            unsafe {
226                libc::killpg(group, signal);
227            }
228        }
229    }
230}

232impl Drop for HostProcess {
233    fn drop(&mut self) {
234        #[cfg(unix)]
235        if self.group.is_some() {
236            self.signal(libc::SIGKILL);
237            tracing::warn!(name = %self.name, "A host process was dropped before it was stopped. Killed its group.");
238        }
239    }
240}

242/// The chunks `reader` yields, as `tag` makes them, with `name`'s read errors.
243fn output_of<R>(reader: R, name: &str, stream: &'static str, tag: fn(Bytes) -> Output) -> OutputStream
244where
245    R: AsyncRead + Send + Unpin + 'static,
246{
247    let name = name.to_string();
248    ReaderStream::new(reader)
249        .map(move |chunk| match chunk {
250            Ok(bytes) => Ok(tag(bytes)),
251            Err(e) => Err(generic_error!(
252                "Failed to read the standard {} of '{}': {}",
253                stream,
254                name,
255                e
256            )),
257        })
258        .boxed()
259}

261/// Whether group `group` has a member that is not a zombie.
262///
263/// A zombie stays in its group until its parent reaps it, which may never happen when an orphan is adopted by a
264/// PID 1 that does not reap. A zombie is no longer running, so it is not a straggler.
265#[cfg(unix)]
266fn group_has_live_member(group: i32) -> bool {
267    // `killpg` cannot tell a zombie from a live process, so read the states from `/proc`.
268    #[cfg(target_os = "linux")]
269    if let Ok(entries) = std::fs::read_dir("/proc") {
270        return entries.flatten().any(|entry| {
271            // The process may exit, or be unreadable, during the scan. Such a process is not counted.
272            let Ok(stat) = std::fs::read_to_string(entry.path().join("stat")) else {
273                return false;
274            };
275            // The command name is in parentheses and may contain spaces or parentheses, so skip to the last `)`.
276            let Some((_, rest)) = stat.rsplit_once(')') else {
277                return false;
278            };
279            let mut fields = rest.split_whitespace();
280            let state = fields.next();
281            let pgrp = fields.nth(1).and_then(|pgrp| pgrp.parse::<i32>().ok());
282            pgrp == Some(group) && !matches!(state, Some("Z" | "X"))
283        });
284    }
285
286    // Other targets (macOS) hand orphans to an init that reaps them promptly, so no zombie lingers.
287    // SAFETY: `killpg` takes no pointers and touches no memory of this process. Signal 0 only checks.
288    let sent = unsafe { libc::killpg(group, 0) };
289    // A group that the process may not signal still exists.
290    sent == 0 || std::io::Error::last_os_error().raw_os_error() != Some(libc::ESRCH)
291}

293/// Passes on the chunks of `output`, and goes on reading it after the returned stream is dropped.
294///
295/// The stream owns the pipes only through the task, so whoever stops observing a program does not close its pipes,
296/// which would make the program's next write fail while it is still running.
297fn keep_draining(mut output: OutputStream) -> OutputStream {
298    let (sender, receiver) = mpsc::channel(OUTPUT_BUFFER);
299    tokio::spawn(async move {
300        let mut observed = true;
301        while let Some(chunk) = output.next().await {
302            if observed && sender.send(chunk).await.is_err() {
303                observed = false;
304            }
305        }
306    });
307    stream::unfold(receiver, |mut receiver| async move {
308        receiver.recv().await.map(|chunk| (chunk, receiver))
309    })
310    .boxed()
311}

313#[cfg(unix)]
314fn signal_of(status: &std::process::ExitStatus) -> Option<i32> {
315    std::os::unix::process::ExitStatusExt::signal(status)
316}

318#[cfg(not(unix))]
319fn signal_of(_status: &std::process::ExitStatus) -> Option<i32> {
320    None

331    fn text_of(chunk: Output) -> String {
332        match chunk {
333            Output::Stdout(bytes) | Output::Stderr(bytes) => String::from_utf8_lossy(&bytes).into_owned(),
334        }

337    #[tokio::test]
338    async fn stop_signals_the_whole_group_and_kills_what_ignores_sigterm() {
339        // The subshell and its `sleep` ignore SIGTERM, so only SIGKILL ends them.
340        let script = "trap 'echo got-term' TERM; (trap '' TERM; sleep 60) & echo ready; wait";
341        let command = HostCommand {
342            name: "test".to_string(),
343            program: PathBuf::from("/bin/sh"),
344            args: vec!["-c".to_string(), script.to_string()],
345            env: BTreeMap::from([("PATH".to_string(), std::env::var("PATH").unwrap_or_default())]),
346            dir: std::env::temp_dir(),
347            stdin: None,
348        };
349        let (process, mut output, _exit) = HostProcess::spawn(command).unwrap();
350
351        let mut text = String::new();
352        tokio::time::timeout(WAIT, async {
353            while !text.contains("ready") {
354                text.push_str(&text_of(
355                    output.next().await.expect("output ended before ready").unwrap(),
356                ));
357            }
358        })
359        .await
360        .expect("the program should say ready");
361
362        tokio::time::timeout(WAIT, process.stop(Duration::from_millis(500)))
363            .await
364            .expect("stop should return")
365            .expect("stop should confirm the exit");
366
367        // The output ends once nothing in the group holds its pipes.
368        tokio::time::timeout(WAIT, async {
369            while let Some(chunk) = output.next().await {
370                text.push_str(&text_of(chunk.unwrap()));
371            }
372        })
373        .await
374        .expect("a process still holds the output pipes");
375        assert!(text.contains("got-term"), "{text}");
376    }
