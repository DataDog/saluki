6use bytes::Bytes;

8use saluki_error::{generic_error, ErrorContext as _, GenericError};
9use tokio::{
10    io::AsyncWriteExt as _,
11    time::{sleep, timeout_at, Instant},
12};
13
14use crate::container::{Container, Output};

16/// How often to ask Docker whether an exec whose output has ended has exited.
17const EXIT_POLL_INTERVAL: Duration = Duration::from_millis(10);
18
19/// The result of a command that ran to completion inside a container.
20#[derive(Clone, Debug, Eq, PartialEq)]
21pub struct ExecOutput {
22    /// The command's exit code.
23    pub exit_code: i64,
24    /// Everything the command wrote to standard output.
25    pub stdout: Vec<u8>,
26    /// Everything the command wrote to standard error.
27    pub stderr: Vec<u8>,
28}
29
30impl Container {
31    /// Runs `command` in the running container and waits for it to exit.
32    ///
33    /// With `stdin`, the bytes are written to the command's standard input, which is then closed. The command may exit
34    /// without reading them all. Without it, the command has no standard input. A nonzero exit code is a result, not an
35    /// error.
36    ///
37    /// # Errors
38    ///
39    /// Returns an error if Docker cannot run the command, if its output cannot be read, if Docker does not report
40    /// its exit code, or if `deadline` passes before the command exits.
41    pub async fn exec(
42        &self, command: Vec<String>, stdin: Option<Bytes>, deadline: Instant,
43    ) -> Result<ExecOutput, GenericError> {
44        let described = format!("{:?}", command);
45        timeout_at(deadline, self.exec_unbounded(command, stdin))
46            .await
47            .map_err(|_| {
48                generic_error!(
49                    "The deadline passed before {} in container '{}' exited.",
50                    described,
51                    self.name
52                )
53            })?
54    }
55
56    async fn exec_unbounded(&self, command: Vec<String>, stdin: Option<Bytes>) -> Result<ExecOutput, GenericError> {
57        let options = CreateExecOptions {
58            attach_stdin: Some(stdin.is_some()),
59            attach_stdout: Some(true),
60            attach_stderr: Some(true),
61            cmd: Some(command.clone()),
62            ..Default::default()
63        };
64        let exec = self
65            .docker
66            .create_exec(&self.name, options)
67            .await
68            .with_error_context(|| {
69                format!(
70                    "Failed to create an exec of {:?} in container '{}'.",
71                    command, &self.name
72                )
73            })?;
74
75        let StartExecResults::Attached { mut output, mut input } = self
76            .docker
77            .start_exec(&exec.id, None)
78            .await
79            .with_error_context(|| format!("Failed to start {:?} in container '{}'.", command, &self.name))?
80        else {
81            return Err(generic_error!("Docker started {:?} detached from its output.", command));
82        };
83
84        // Write and read at once, so a command that answers before reading all of its input cannot stall either side.
85        // A command that exits without reading all of its input is not an error: the write fails with a broken pipe or
86        // stalls, as Docker stops reading, and it ends when the command exits.
87        let write = async {
88            if let Some(stdin) = stdin {
89                let written = async {
90                    input.write_all(&stdin).await?;
91                    input.shutdown().await
92                };
93                match written.await {
94                    Err(e) if e.kind() != std::io::ErrorKind::BrokenPipe => return e,
95                    _ => {}
96                }
97            }
98            std::future::pending().await
99        };
100        let finish = async {
101            let mut stdout = Vec::new();
102            let mut stderr = Vec::new();
103            while let Some(log) = output
104                .try_next()
105                .await
106                .with_error_context(|| format!("Failed to read the output of {:?}.", command))?
107            {
108                match Output::from_log(log) {
109                    Some(Output::Stdout(bytes)) => stdout.extend_from_slice(&bytes),
110                    Some(Output::Stderr(bytes)) => stderr.extend_from_slice(&bytes),
111                    None => {}
112                }
113            }
114
115            // Docker can still report the exec as running after its output ends.
116            let inspect = loop {
117                let inspect = self
118                    .docker
119                    .inspect_exec(&exec.id)
120                    .await
121                    .with_error_context(|| format!("Failed to inspect the exec of {:?}.", command))?;
122                if inspect.running != Some(true) {
123                    break inspect;
124                }
125                sleep(EXIT_POLL_INTERVAL).await;
126            };
127            let exit_code = inspect
128                .exit_code
129                .ok_or_else(|| generic_error!("Docker reported no exit code for {:?}.", command))?;
130
131            Ok(ExecOutput {
132                exit_code,
133                stdout,
134                stderr,
135            })
136        };
137        tokio::select! {
138            error = write => {
139                Err(error).with_error_context(|| format!("Failed to write the standard input of {:?}.", command))
140            }
141            output = finish => output,
142        }
143    }
144}
