11/// Pulls `image` unless the Docker daemon already has it.
12///
13/// # Errors
14///
15/// Returns an error if Docker cannot inspect the image, if the pull fails, or if `deadline` passes first.
16pub async fn ensure_image(docker: &Docker, image: &str, deadline: Instant) -> Result<(), GenericError> {
17    timeout_at(deadline, ensure_image_unbounded(docker, image))
18        .await
19        .map_err(|_| generic_error!("The deadline passed before image '{}' was available.", image))?
20}
21
22async fn ensure_image_unbounded(docker: &Docker, image: &str) -> Result<(), GenericError> {
23    match docker.inspect_image(image).await {
24        Ok(_) => return Ok(()),
25        Err(e) if is_not_found(&e) => {}
26        Err(e) => return Err(generic_error!("Failed to inspect image '{}': {}", image, e)),
27    }
28
29    debug!("Pulling image '{}'...", image);
30    let options = CreateImageOptions {
31        from_image: Some(image.to_string()),
32        // Without a tag, Docker pulls every tag.
33        tag: (!names_tag_or_digest(image)).then(|| "latest".to_string()),
34        ..Default::default()
35    };
36    docker
37        .create_image(Some(options), None, None)
38        .try_for_each(|_| async { Ok(()) })
39        .await
40        .map_err(|e| generic_error!("Failed to pull image '{}': {}", image, e))
41}

43/// Whether `image` names a tag or a digest. A colon before the last slash is a registry port, not a tag.
44fn names_tag_or_digest(image: &str) -> bool {
45    let name = image.rsplit_once('/').map_or(image, |(_, name)| name);
46    name.contains([':', '@'])
47}
