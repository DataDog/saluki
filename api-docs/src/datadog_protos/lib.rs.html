97#[cfg(test)]
98mod tests {
99    use protobuf::Message as _;
100
101    use super::traces::{self, AgentPayload};
102
103    #[test]
104    fn agent_payload_round_trips_classic_and_indexed_tracer_payloads() {
105        let mut classic_payload = traces::TracerPayload::new();
106        classic_payload.containerID = "classic-container".to_string();
107
108        let mut indexed_payload = traces::idx::TracerPayload::new();
109        indexed_payload.strings = vec!["".to_string(), "indexed-container".to_string()];
110        indexed_payload.containerIDRef = 1;
111
112        let mut payload = AgentPayload::new();
113        payload.tracerPayloads.push(classic_payload);
114        payload.idxTracerPayloads.push(indexed_payload);
115
116        let encoded = payload.write_to_bytes().expect("payload should encode");
117        let decoded = AgentPayload::parse_from_bytes(&encoded).expect("payload should decode");
118
119        let indexed_payloads: &[traces::idx::TracerPayload] = decoded.idxTracerPayloads();
120        assert_eq!(decoded.tracerPayloads()[0].containerID(), "classic-container");
121        assert_eq!(indexed_payloads[0].containerIDRef(), 1);
122        assert_eq!(indexed_payloads[0].strings(), &["", "indexed-container"]);
123    }
124}
